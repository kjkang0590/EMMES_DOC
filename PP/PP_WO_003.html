SQL 보기
------------ WO 플랜트 콤보박스 쿼리

SELECT DISTINCT

    EQ.PLANTID AS CODE,

    CF.FACILITYNAME AS NAME

  FROM CIM_PRODUCTDEFINITION PD

 INNER JOIN CIM_PRODUCTPROCESSREL PW 

    ON PW.SITEID = PD.SITEID 

   AND PW.ISUSABLE = PD.ISUSABLE

   AND PW.PRODUCTDEFINITIONID = PD.PRODUCTDEFINITIONID

 INNER JOIN CIM_PROCESSNODE PN 

    ON PN.SITEID = PW.SITEID 

   AND PN.ISUSABLE = PW.ISUSABLE

   AND PN.PROCESSDEFINITIONID = PW.PROCESSDEFINITIONID

   AND PN.ISSTARTNODE = 'Y'

 INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL EL 

    ON EL.SITEID = PN.SITEID 

   AND EL.ISUSABLE = PN.ISUSABLE

   AND EL.PROCESSSEGMENTID = PN.PROCESSSEGMENTID

 INNER JOIN CIM_EQUIPMENT EQ 

    ON EQ.SITEID = EL.SITEID 

   AND EQ.ISUSABLE = EL.ISUSABLE

   AND EQ.EQUIPMENTID = EL.EQUIPMENTID

  LEFT OUTER JOIN CIM_FACILITY CF 

    ON CF.SITEID = EQ.SITEID

   AND CF.FACILITYID = EQ.PLANTID

 WHERE 1=1

   AND PD.SITEID = :SITEID 

   AND PD.ISUSABLE = 'Usable'

   AND PD.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID
