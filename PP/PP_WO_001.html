SQL 보기
SELECT DISTINCT 

    PL.PRODUCTDEFINITIONID AS PRODUCTDEFINITIONID,

	ST.SHORTNAME AS SITECODE,

	FC.SHORTNAME AS LINECODE,

	PS.SHORTNAME AS OPERCODE,

	TO_CHAR(SYSDATE ,'YYMMDD') AS DAY,

	PD.SHORTNAME AS PRODCODE

FROM

	CIM_PRODUCTPROCESSREL PL

INNER JOIN CIM_SITE ST 

   ON ST.SITEID = PL.SITEID

  AND ST.ISUSABLE = 'Usable'

INNER JOIN CIM_PRODUCTDEFINITION PD

   ON PD.PRODUCTDEFINITIONID = PL.PRODUCTDEFINITIONID

  AND PD.SITEID = PL.SITEID

  AND PD.ISUSABLE = 'Usable'

INNER JOIN CIM_PROCESSNODE PN

   ON PN.PROCESSDEFINITIONID = PL.PROCESSDEFINITIONID

  AND PN.ISSTARTNODE = 'Y'

  AND PN.SITEID = PL.SITEID

  AND PN.ISUSABLE = 'Usable'

INNER JOIN CIM_PROCESSSEGMENT PS 

   ON PS.PROCESSSEGMENTID = PN.PROCESSSEGMENTID

  AND PS.SITEID = PN.SITEID

  AND PS.ISUSABLE = 'Usable'

INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL EL 

   ON EL.SITEID = PS.SITEID

  AND EL.PROCESSSEGMENTID = PS.PROCESSSEGMENTID

INNER JOIN CIM_EQUIPMENT EQ 

   ON EQ.SITEID = EL.SITEID

  AND EQ.EQUIPMENTID = EL.EQUIPMENTID

INNER JOIN CIM_FACILITY FC 

   ON FC.SITEID = EQ.SITEID

  AND FC.FACILITYID = EQ.LINEID

WHERE 1 = 1

  AND PL.SITEID = :SITEID

  AND PL.ISDEFAULT = 'Y'

  AND PL.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID
