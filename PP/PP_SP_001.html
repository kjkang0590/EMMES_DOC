SELECT 

	CD.DELIVERYPLANDATE,

	CD.DELIVERYORDERID ,

	CD.DELIVERYITEMNO ,

	CD.SITEID ,

    CD.SALESORDERID ,

	CD.SALESITEMNO , 

	CD.DELIVERYORDERTYPE  ,

	CCO.CODENAME  , 

	CD.CUSTOMERID , 

	CC.CUSTOMERNAME ,

	CD.PRODUCTDEFINITIONID ,

	CP.PRODUCTDEFINITIONNAME ,

	CD.DELIVERYDATE ,

	CD.QTY ,

	'0'  AS OUTQTY ,

	'0'  AS LEFTQTY ,

	CD.UNITID ,

	CD.STATE ,

	CS.STATENAME 

  FROM CIM_DELIVERYORDER CD

 INNER JOIN CIM_CUSTOMER CC

 		 ON CC.CUSTOMERID = CD.CUSTOMERID

        AND CC.SITEID = CD.SITEID  

        AND CC.ISUSABLE = 'Usable'

 INNER JOIN CIM_PRODUCTDEFINITION CP

         ON CP.PRODUCTDEFINITIONID = CD.PRODUCTDEFINITIONID  

        AND CP.SITEID = CD.SITEID

        AND CP.ISUSABLE = 'Usable'

 INNER JOIN CIM_STATE CS

         ON CS.STATEID   = CD.STATE 

        AND CS.STATEMODELID  = 'DeliveryOrderState'

        AND CS.SITEID = CD.SITEID 

        AND CS.ISUSABLE = 'Usable'   

 INNER JOIN CIM_CODE CCO

         ON CCO.CODEID  = CD.DELIVERYORDERTYPE  

        AND CCO.CODECLASSID  = 'DELIVERYORDERTYPE'

        AND CCO.SITEID = CD.SITEID 

        AND CCO.ISUSABLE = 'Usable'

 WHERE CD.DELIVERYPLANDATE BETWEEN to_date(:FROMDATE,'YYYY-MM-DD') AND to_date(:TODATE,'YYYY-MM-DD')

   AND CD.ISUSABLE = 'Usable'

   AND CD.SITEID = :SITEID 

   AND (:SALESORDERID IS NULL OR UPPER(CD.SALESORDERID) LIKE :SALESORDERID || '%')

   AND (:DELIVERYORDERID IS NULL OR UPPER(CD.DELIVERYORDERID) LIKE :DELIVERYORDERID || '%')

   AND (:CUSTOMERID IS NULL OR CD.CUSTOMERID = :CUSTOMERID)

   AND (:STATE IS NULL  OR CD.STATE = :STATE)

   AND (:PRODUCTDEFINITIONID IS NULL OR UPPER(CD.PRODUCTDEFINITIONID) LIKE :PRODUCTDEFINITIONID || '%')

   AND (:PRODUCTDEFINITIONNAME IS NULL  OR UPPER(CP.PRODUCTDEFINITIONNAME) LIKE :PRODUCTDEFINITIONNAME || '%')
