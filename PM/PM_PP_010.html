SELECT 

	FACILITYID AS CODE, -- LINEID,

	FACILITYNAME AS NAME, -- LINENAME,

	SHORTNAME, SHOP, LINEGROUP

FROM CIM_FACILITY LINE

	WHERE 1=1

	AND UPPER(LINE.FACILITYTYPE) = 'LINE'

	AND LINE.SITEID = :SITEID

		WHERE 1=1

			AND (:PLANTID IS NULL OR PLANTLINE.PARENTFACILITYID = :PLANTID)

테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

-- SelectProcessSegmentList

SELECT DISTINCT EQ.SITEID, EQ.PLANTID, EQ.LINEID
	, PC.PROCESSSEGMENTID AS CODE
	, PC.PROCESSSEGMENTNAME AS NAME
	, PC.PROCESSTYPE
	, PC.BIZRULEID
	, CASE WHEN (SELECT COUNT(*) FROM CIM_EQUIPMENT SUB WHERE SUB.PARENTID = EQ.EQUIPMENTID) > 0 THEN 'Y' ELSE 'N' END ISSUBEQP
FROM CIM_EQUIPMENT EQ
	INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER
	ON 1=1
	AND PER.EQUIPMENTID = EQ.EQUIPMENTID
	AND PER.SITEID = EQ.SITEID
	AND PER.ISUSABLE = 'Usable'
	INNER JOIN CIM_PROCESSSEGMENT PC
	ON 1=1
	AND PC.PROCESSSEGMENTID = PER.PROCESSSEGMENTID
	AND PC.SITEID = PER.SITEID
	AND PC.ISUSABLE = 'Usable'
WHERE 1=1
	AND (:PLANTID IS NULL OR EQ.PLANTID = :PLANTID)
	AND (:LINEID IS NULL OR  EQ.LINEID = :LINEID)
ORDER BY EQ.PLANTID, EQ.LINEID, PC.PROCESSSEGMENTID
