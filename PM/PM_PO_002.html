-- PM_PO_002_WO 현황
-- SelectWorkOrderCurrentList V00001
-- Param : SITEID, LINEID, FROMDATE, TODATE, PROCESSTYPE, MATERIALCLASSID, MATERIALDEFINITIONID, PRODUCTORDERID, LOTID, ORDERSTATE
----------------------------------------------------------------------------------------------------------------------------

SELECT CONVERT(DATE, CW.REQUESTTIME) AS REQUESTTIME
     , CW.WORKORDERID
     , CW.PRODUCTORDERID
     , CM.MATERIALCLASSID
     , MC.MATERIALCLASSNAME
     , CM.MATERIALDEFINITIONID
     , CM.MATERIALDEFINITIONNAME
     , CW.SHIFTID
     , CW.QTY
     , CW.RESULTQTY
     , CW.UNITID
     , CW.STATE
     , CW.DESCRIPTION
  FROM CIM_WORKORDER CW WITH(NOLOCK)
 INNER JOIN CIM_MATERIALDEFINITION CM WITH(NOLOCK) ON CM.SITEID = CW.SITEID AND CM.MATERIALDEFINITIONID = CW.PRODUCTDEFINITIONID
 INNER JOIN CIM_MATERIALCLASS MC WITH(NOLOCK) ON MC.SITEID = CM.SITEID AND MC.MATERIALCLASSID = CM.MATERIALCLASSID
 WHERE CW.SITEID = @SITEID
   AND (@LINEID = '' OR CW.LINEID = @LINEID)
   AND (@FROMDATE = '' OR CONVERT(DATE, CW.REQUESTTIME) >= CONVERT(DATE, @FROMDATE))
   AND (@TODATE = '' OR CONVERT(DATE, CW.REQUESTTIME) <= CONVERT(DATE, @TODATE))
   AND (@PROCESSTYPE = '' OR CW.PROCESSTYPE = @PROCESSTYPE)
   AND (@MATERIALCLASSID = '' OR MC.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALDEFINITIONID = '' OR CW.PRODUCTDEFINITIONID LIKE '%' + @MATERIALDEFINITIONID + '%')
   AND (@PRODUCTORDERID = '' OR CW.PRODUCTORDERID = @PRODUCTORDERID)
   AND (@LOTID = '' OR EXISTS (
         SELECT 1
           FROM CIM_LOT CL WITH(NOLOCK)
          WHERE CL.SITEID = CW.SITEID
            AND CL.WORKORDERID = CW.WORKORDERID
            AND CL.LOTID LIKE '%' + @LOTID + '%'
            AND CL.ISUSABLE = 'Usable'
       ))
   AND (@ORDERSTATE = '' OR CW.STATE = @ORDERSTATE)
 ORDER BY CW.REQUESTTIME DESC, CW.WORKORDERID
