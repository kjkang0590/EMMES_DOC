--PM_CS_002_원부자재 현황 조회
--SelectMaterialCurrentList
--Param : SITEID, LOCATION, MATERIALCLASSID, MATERIALDEFINITIONID, MATERIALDEFINITIONNAME
------------------------------------------
SELECT ML.LOCATION
   	 , CF.FACILITYNAME
     , ML.MATERIALCLASSID
     , MC.MATERIALCLASSNAME
     , ML.MATERIALDEFINITIONID
     , MD.MATERIALDEFINITIONNAME
     , SUM(ML.QTY) AS SUMQTY
     , ML.UNITID
     , CU.UNITNAME 
  FROM CST_MATERIALLOT ML
  LEFT OUTER JOIN CIM_FACILITY CF ON CF.SITEID = ML.SITEID AND CF.FACILITYID = ML.LOCATION
  LEFT OUTER JOIN CIM_MATERIALDEFINITION MD ON MD.SITEID = ML.SITEID AND MD.MATERIALDEFINITIONID = ML.MATERIALDEFINITIONID
  LEFT OUTER JOIN CIM_MATERIALCLASS MC ON MC.SITEID = MD.SITEID AND MC.MATERIALCLASSID = ML.MATERIALCLASSID
  LEFT OUTER JOIN CIM_UNIT CU ON CU.SITEID = ML.SITEID  AND CU.UNITID = ML.UNITID 
 WHERE ML.SITEID = @SITEID
   AND (@LOCATION = '' OR ML.LOCATION = @LOCATION)
   AND (@MATERIALCLASSID = '' OR ML.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALDEFINITIONID = '' OR ML.MATERIALDEFINITIONID LIKE '%' + @MATERIALDEFINITIONID + '%')
   AND (@MATERIALDEFINITIONNAME = '' OR MD.MATERIALDEFINITIONNAME LIKE '%' + @MATERIALDEFINITIONNAME + '%')
 GROUP BY ML.LOCATION
 	 , CF.FACILITYNAME
     , ML.MATERIALCLASSID
     , MC.MATERIALCLASSNAME
     , ML.MATERIALDEFINITIONID
     , MD.MATERIALDEFINITIONNAME
     , ML.UNITID
     , CU.UNITNAME

--PM_CS_002_원부자재 현황(LOT 별) 조회
--SelectMaterialLotCurrentList
--Param : SITEID, LOCATION, MATERIALDEFINITIONID
------------------------------------------
SELECT ML.LOCATION
     , CF.FACILITYNAME
     , ML.MATERIALCLASSID
     , MC.MATERIALCLASSNAME
     , ML.MATERIALDEFINITIONID
     , MD.MATERIALDEFINITIONNAME
     , ML.MATERIALLOTID
     , ML.KITTINGTIME
     , ML.QTY
     , ML.UNITID
     , CU.UNITNAME
  FROM CST_MATERIALLOT ML
  LEFT OUTER JOIN CIM_FACILITY CF ON CF.SITEID = ML.SITEID AND CF.FACILITYID = ML.LOCATION
  LEFT OUTER JOIN CIM_MATERIALDEFINITION MD ON MD.SITEID = ML.SITEID AND MD.MATERIALDEFINITIONID = ML.MATERIALDEFINITIONID
  LEFT OUTER JOIN CIM_MATERIALCLASS MC ON MC.SITEID = MD.SITEID AND MC.MATERIALCLASSID = MD.MATERIALCLASSID
  LEFT OUTER JOIN CIM_UNIT CU ON CU.SITEID = ML.SITEID  AND CU.UNITID = ML.UNITID 
 WHERE ML.SITEID = @SITEID
   AND ML.LOCATION = @LOCATION
   AND ML.MATERIALDEFINITIONID = @MATERIALDEFINITIONID
   AND ML.STATE <> 'Transfer'
 ORDER BY ML.KITTINGTIME DESC
