--PM_PB_003 공정 진행 (Li, Co, Ni) - Lot 목록 조회
--SelectMetalProcessList
--Param : SITEID, LINEID, FROMDATE, TODATE, STATE, MATERIALCLASSID, MATERIALDEFINITIONID, PRODUCTORDERID
------------------------------------------
SELECT WO.PRODUCTORDERID
     , WO.WORKORDERID
     , WO.PRODUCTDEFINITIONID AS ROOTMATERIALDEFINITIONID
     , WO.STATE AS WOSTATE
     , CC.CODENAME AS WOSTATENAME
     , CL.LOTID
     , CL.STATE AS LOTSTATUS
     , CS.STATENAME AS LOTSTATENAME
     , MD.MATERIALCLASSID
     , MD.MATERIALDEFINITIONID
     , MD.MATERIALDEFINITIONNAME
     , CL.PROCESSSEGMENTID
     , PC.PROCESSSEGMENTNAME
     , CL.EQUIPMENTID
     , CE.EQUIPMENTNAME
     , WO.REQUESTTIME
     , CL.TRACKINTIME
     , CL.TRACKOUTTIME
     , CASE WHEN CL.PROCESSSEGMENTID IS NULL THEN ROUND(CONVERT(FLOAT, WO.LITHIUMMETALQTY), 2)
            WHEN CL.PROCESSSEGMENTID = 'P2030' THEN CL.QTY
            ELSE 0
        END AS EXPECTLIQTY
     , CASE WHEN CL.PROCESSSEGMENTID IS NULL THEN ROUND(CONVERT(FLOAT, WO.COBALTMETALQTY), 2)
            WHEN CL.PROCESSSEGMENTID = 'P2040' THEN CL.QTY
            ELSE 0
        END AS EXPECTCOQTY
     , CASE WHEN CL.PROCESSSEGMENTID IS NULL THEN ROUND(CONVERT(FLOAT, WO.NICKELMETALQTY), 2)
            WHEN CL.PROCESSSEGMENTID = 'P2050' THEN CL.QTY
            ELSE 0
        END AS EXPECTNIQTY
     , WO.BEFORELOT
     , WO.RESULTQTY AS BEFOREOUTPUTQTY
  FROM (SELECT DISTINCT W.PRODUCTORDERID, W.WORKORDERID, W.STATE, W.REQUESTTIME, W.SITEID, W.LINEID, W.PRODUCTDEFINITIONID, W.PROCESSTYPE
             , L.LOTID AS BEFORELOT, L.RESULTQTY, L.PRODUCTDEFINITIONID AS MATERIALDEFINITIONID
             , CM.LOTID, L.LITHIUMMETALQTY, L.COBALTMETALQTY, L.NICKELMETALQTY
          FROM CIM_WORKORDER W
         INNER JOIN CIM_LOT L ON L.SITEID = W.SITEID AND L.WORKORDERID = W.WORKORDERID AND L.STATE IN ('Finished', 'Consumed') AND L.PROCESSSEGMENTID = 'P2020'
          LEFT JOIN CIM_LOTMATERIALLOTREL CM ON CM.SITEID = L.SITEID AND CM.MATERIALLOTID = L.LOTID
       ) AS WO
 INNER JOIN CIM_MATERIALDEFINITION MD ON MD.SITEID = WO.SITEID AND MD.MATERIALDEFINITIONID = WO.MATERIALDEFINITIONID
  LEFT JOIN CIM_LOT CL ON CL.SITEID = WO.SITEID AND CL.WORKORDERID = WO.WORKORDERID AND CL.LOTID = WO.LOTID AND CL.PROCESSSEGMENTID IN ('P2030', 'P2040', 'P2050')
  LEFT JOIN CIM_PROCESSSEGMENT PC ON PC.SITEID = CL.SITEID AND PC.PROCESSSEGMENTID = CL.PROCESSSEGMENTID
  LEFT JOIN CIM_EQUIPMENT CE ON CE.SITEID = CL.SITEID AND CE.EQUIPMENTID = CL.EQUIPMENTID 
  LEFT JOIN CIM_CODE CC ON CC.SITEID = WO.SITEID AND CC.CODEID = WO.STATE AND CC.CODECLASSID = 'ORDERSTATE'
  LEFT JOIN CIM_STATE CS ON CS.SITEID = CL.SITEID AND CS.STATEID = CL.STATE AND CS.STATEMODELID = 'LotState' and CS.ISUSABLE = 'Usable'
 WHERE WO.SITEID = @SITEID
   AND (@LINEID = '' OR WO.LINEID = @LINEID)
   AND (@WOSTATE = '' OR WO.STATE = @WOSTATE) 
   AND WO.PROCESSTYPE = 'P2000'
   AND WO.REQUESTTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@MATERIALCLASSID = '' OR MD.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALDEFINITIONID = '' OR MD.MATERIALDEFINITIONID = @MATERIALDEFINITIONID)
   AND (@PRODUCTORDERID = '' OR WO.PRODUCTORDERID LIKE '%' + @PRODUCTORDERID + '%')
 ORDER BY WO.PRODUCTORDERID, WO.WORKORDERID, WO.BEFORELOT, CL.PROCESSSEGMENTID, CL.LOTID
