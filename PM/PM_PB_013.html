--PM_PB_013_부자재 사용 현황 - 부자재별 사용량(합계) 조회 쿼리
--SelectUsedSubMaterialSumList
--Param : SITEID, FROMDATE, TODATE, SHIFTID, TYPECODE
------------------------------------------
SELECT UM.TYPECODE
     , UM.TYPENAME
--     , ROUND(SUM(UM.USEDQTY), 5) AS USEDQTY
     , ROUND(CASE WHEN UM.TYPECODE = 'H2SO4' THEN SUM(UM.USEDQTY) ELSE SUM(UM.FQIUSEDQTY) END, 5) AS USEDQTY
FROM CST_RPT_USEDSUBMATERIALQTY_SHIFT UM
INNER JOIN CIM_CODE C
ON  TYPECODE = C.CODEID
AND C.CODECLASSID = 'SUBMATERIALQTY'
WHERE UM.SITEID = @SITEID
AND   UM.WORKDATE BETWEEN REPLACE(@FROMDATE , '-', '') AND REPLACE(@TODATE, '-', '')
AND   UM.SHIFTID = CASE WHEN ISNULL(@SHIFTID, '') = '' THEN UM.SHIFTID ELSE @SHIFTID END
--AND   UM.TYPECODE = CASE WHEN ISNULL(@TYPECODE, '') = '' THEN UM.TYPECODE ELSE @TYPECODE END
AND   UM.TYPEID = CASE WHEN ISNULL(@TYPECODE, '') = '' THEN UM.TYPEID ELSE @TYPECODE END
GROUP BY UM.TYPECODE, UM.TYPENAME, C.UISEQUENCE
ORDER BY C.UISEQUENCE
