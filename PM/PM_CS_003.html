Stored Procedure SP_GET_TANKLEVEL 정의 보기
CREATE PROCEDURE [dbo].[SP_GET_TANKLEVEL] 

--DECLARE

    @SITEID VARCHAR(40) = 'PL01', @ROOTEQUIPMENTID VARCHAR(40) = '', @EQUIPMENTCLASSID VARCHAR(40) ='', @EQUIPMENTID VARCHAR(40) = '', @TANKTYPE VARCHAR(40) = ''

    ,@ISPRODUCT VARCHAR(40) = 'N'

AS  

BEGIN 



    DECLARE @TANKLEVEL TABLE

    (

        TAGID VARCHAR(100)

        , EQUIPMENTID VARCHAR(20)

        , EQUIPMENTCLASSID VARCHAR(20)

        , ROOTEQUIPMENTID VARCHAR(10)

        , SITEID VARCHAR(10)

        , TAGNAME VARCHAR(100)

        , DATAMAPPINGTABLE VARCHAR(40)

        , DATAMAPPINGCOLUMN VARCHAR(40)

        , DATAMAPPINGPREVCOLUMN VARCHAR(40)

        , UNITID VARCHAR(40)

        , TAGTYPE VARCHAR(40)

    )

    ;



    DECLARE @VALUETABLE TABLE

    (

        TABLENAME VARCHAR(40)

        , COL_POS VARCHAR(10)

        , PROC_COLNAME VARCHAR(15)

        , PROC_VALUE VARCHAR(20)

    )

    ;



    DECLARE @STRSQT VARCHAR(MAX)

          , @DATAMAPPINGTABLE VARCHAR(100)

          , @SELECT_COLUMN VARCHAR(MAX)

    ;





    --

    -- TANK LEVEL 관련 TAG 조회

    --

    INSERT INTO @TANKLEVEL

    SELECT 

        OT.TAGID

        , OT.EQUIPMENTID

        , EC.EQUIPMENTCLASSID

        , OT.ROOTEQUIPMENTID

        , OT.SITEID

        , OT.TAGNAME

        , PDP.DATAMAPPINGTABLE 

        , PDP.DATAMAPPINGCOLUMN 

        , 'PREV' + PDP.DATAMAPPINGCOLUMN AS DATAMAPPINGPREVCOLUMN

        , PDP.UNITID

        , OT.TAGTYPE

    FROM CST_OPCTAGINFO OT

        INNER JOIN CST_PROCESSDATAPARAMETER PDP

            ON  OT.TAGID = PDP.PROCESSDATAPARAMETERID 

            AND OT.SITEID = PDP.SITEID 

            AND PDP.PROCESSDATADEFINITIONID = 'GetTankLevel'

            AND PDP.ISUSABLE = 'Usable'

        INNER JOIN CIM_EQUIPMENT E

            ON  OT.EQUIPMENTID = E.EQUIPMENTID 

            AND OT.SITEID = E.SITEID 

        INNER JOIN CIM_EQUIPMENTCLASS EC

            ON  E.EQUIPMENTCLASSID  = EC.EQUIPMENTCLASSID 

            AND E.SITEID = EC.SITEID 

    ;



    -- 조회 결과를 TABLE 별 분류하여 CURSOR 에 연결

    DECLARE CUR_LEVEL CURSOR FOR

        SELECT 

            T.DATAMAPPINGTABLE

            , STRING_AGG ( DATAMAPPINGCOLUMN + ', PREV' + DATAMAPPINGCOLUMN, ', '  ) WITHIN GROUP (ORDER BY T.DATAMAPPINGTABLE) AS SELECT_COLUMN

        FROM @TANKLEVEL T

        GROUP BY T.DATAMAPPINGTABLE



    -- CURSOR OPEN

    OPEN CUR_LEVEL

    

    -- TABLE 별로 WHILE & 결과를 TEMP TABLE 에 저장

    FETCH NEXT FROM CUR_LEVEL INTO @DATAMAPPINGTABLE, @SELECT_COLUMN

    WHILE(@@FETCH_STATUS=0)

    BEGIN

    

        SET @STRSQT = ' SELECT UNP.TABLENAME, LEFT(UNP.PROC_COLNAME, 3) AS COL_POS, UNP.PROC_COLNAME, UNP.PROC_VALUE ' +

              ' FROM ( SELECT ''' + @DATAMAPPINGTABLE + ''' AS TABLENAME, ' + @SELECT_COLUMN + ' FROM ' + @DATAMAPPINGTABLE + ') M' +

              ' UNPIVOT ( PROC_VALUE FOR PROC_COLNAME IN ( ' + @SELECT_COLUMN + ') ) UNP'

              ;

    

        INSERT @VALUETABLE EXEC (@STRSQT);



        FETCH NEXT FROM CUR_LEVEL INTO @DATAMAPPINGTABLE, @SELECT_COLUMN

    END



    -- CURSOR 삭제

    CLOSE CUR_LEVEL

    DEALLOCATE CUR_LEVEL

    

    

    IF @ISPRODUCT = 'Y'

    BEGIN

        -- 기본 정보와 PROCESS DATA 연결 조회

        SELECT 

            TL.ROOTEQUIPMENTID

            , EC.EQUIPMENTCLASSID

            --, E.TANKTYPE AS EQUIPMENTCLASSNAME

            , C.CODENAME AS TANKTYPE            

            , TL.EQUIPMENTID

            , E.EQUIPMENTNAME

            , E.TANKCAPA AS CAPACITY

            , '' AS PRODUCTDEFINITIONID

            , '' AS PRODUCTDEFINITIONNAME

            , ROUND(CONVERT(FLOAT, VT_PRE.PROC_VALUE), 2) AS PREVLEVEL  -- 이전 Level

            , NULL AS PREVMOL       -- 이전 농도

            , ROUND(VT_PRE.PROC_VALUE * E.TANKCAPA / 100.0, 2) AS PREVVOLUME    -- 이전 볼륨

            , NULL AS PREVCONTENT   -- 이전 함량

            , ROUND(CONVERT(FLOAT, VT_CUR.PROC_VALUE), 2) AS LEVEL

            , NULL AS MOL

            , ROUND(VT_CUR.PROC_VALUE * E.TANKCAPA / 100.0, 2) AS VOLUME

  , NULL AS CONTENT       

        FROM @TANKLEVEL TL

            LEFT OUTER JOIN @VALUETABLE VT_CUR

                ON  TL.DATAMAPPINGTABLE = VT_CUR.TABLENAME

                AND TL.DATAMAPPINGCOLUMN = VT_CUR.PROC_COLNAME

            LEFT OUTER JOIN @VALUETABLE VT_PRE

                ON  TL.DATAMAPPINGTABLE = VT_PRE.TABLENAME

                AND TL.DATAMAPPINGPREVCOLUMN = VT_PRE.PROC_COLNAME

            INNER JOIN CIM_EQUIPMENT E

                ON  TL.EQUIPMENTID = E.EQUIPMENTID 

                AND TL.SITEID = E.SITEID 

                AND E.TANKTYPE IS NOT NULL

            LEFT OUTER JOIN CIM_EQUIPMENTCLASS EC

                ON  E.EQUIPMENTCLASSID = EC.EQUIPMENTCLASSID 

                AND E.SITEID = EC.SITEID 

            LEFT OUTER JOIN CIM_CODE C

                ON  E.TANKTYPE = C.CODEID

                AND E.SITEID = C.SITEID

                AND C.CODECLASSID = 'TANKTYPE'              

        WHERE TL.SITEID = @SITEID

        AND (@ROOTEQUIPMENTID = '' OR E.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)

        AND (@TANKTYPE = '' OR E.TANKTYPE = @TANKTYPE)

        AND (@EQUIPMENTID = '' OR E.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%') 

        AND  E.TANKTYPE IN ( 'WIP', 'PROD', 'MATERIAL' )

    END

    ELSE 

    BEGIN 

        -- 기본 정보와 PROCESS DATA 연결 조회

        SELECT 

            TL.ROOTEQUIPMENTID

            , EC.EQUIPMENTCLASSID

            , C.CODENAME AS TANKTYPE

            , TL.EQUIPMENTID

            , E.EQUIPMENTNAME + ' (' + TL.TAGNAME  + ')' AS EQUIPMENTNAME

            , E.EQUIPMENTNAME 

            , E.MEDICINECODE AS ITEMID

--          , MD.MATERIALDEFINITIONNAME AS ITEMNAME

            , CASE WHEN ISNULL(MD.MATERIALDEFINITIONNAME, '') <> '' AND ISNULL(TL.UNITID, '') <> '' 

                        THEN ISNULL(MD.MATERIALDEFINITIONNAME, '') + ' (' + TL.UNITID + ')' 

                   ELSE MD.MATERIALDEFINITIONNAME

              END AS ITEMNAME           

            , E.TANKCAPA AS CAPACITY

--++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

--            , ROUND(CONVERT(FLOAT, VT_CUR.PROC_VALUE), 2) AS LEVEL

            , CASE WHEN ISNULL(TL.TAGTYPE, 'Level') = 'Weight' AND E.MEDICINECODE = '4A0501'

                        THEN ROUND( CONVERT(FLOAT, VT_CUR.PROC_VALUE) / 1.1 / ( E.TANKCAPA * 1000.0 ) * 100.0, 2)

                   ELSE ROUND(CONVERT(FLOAT, VT_CUR.PROC_VALUE), 2)

              END AS LEVEL

--++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++        

            --, ROUND(VT_CUR.PROC_VALUE * E.TANKCAPA / 100.0, 2) AS VOLUME

            , CASE WHEN ISNULL(TL.TAGTYPE, 'Level') = 'Weight' THEN ROUND(CONVERT(FLOAT, VT_CUR.PROC_VALUE), 2) ELSE ROUND(VT_CUR.PROC_VALUE * E.TANKCAPA / 100.0, 2) END AS VOLUME            

            , E.WARNINGQTY

            , E.SAFETYQTY AS SAFETYSTOCK

            , CASE WHEN ISNULL(TL.TAGTYPE, 'Level') = 'Weight' THEN 

                      CASE WHEN ISNULL(E.WARNINGQTY, 0) = 0 THEN 'NOR'

                           WHEN ROUND(CONVERT(FLOAT, VT_CUR.PROC_VALUE), 2) <= E.SAFETYQTY  THEN 'RED' -- 주의, 빨간색

                           WHEN ROUND(CONVERT(FLOAT, VT_CUR.PROC_VALUE), 2) <= E.WARNINGQTY THEN 'YEL' -- 관심, 노란색

                           ELSE 'NOR'

                      END

                   ELSE 

                      CASE WHEN ISNULL(E.WARNINGQTY, 0) = 0 THEN 'NOR'

                           WHEN ROUND(VT_CUR.PROC_VALUE * E.TANKCAPA / 100.0, 2) <= E.SAFETYQTY  THEN 'RED' -- 주의, 빨간색

                           WHEN ROUND(VT_CUR.PROC_VALUE * E.TANKCAPA / 100.0, 2) <= E.WARNINGQTY THEN 'YEL' -- 관심, 노란색

                           ELSE 'NOR'

                      END

              END AS WARNINGTYPE            

        FROM @TANKLEVEL TL

            LEFT OUTER JOIN @VALUETABLE VT_CUR

                ON  TL.DATAMAPPINGTABLE = VT_CUR.TABLENAME

                AND TL.DATAMAPPINGCOLUMN = VT_CUR.PROC_COLNAME

            LEFT OUTER JOIN @VALUETABLE VT_PRE

                ON  TL.DATAMAPPINGTABLE = VT_PRE.TABLENAME

                AND TL.DATAMAPPINGPREVCOLUMN = VT_PRE.PROC_COLNAME

            INNER JOIN CIM_EQUIPMENT E

                ON  TL.EQUIPMENTID = E.EQUIPMENTID 

                AND TL.SITEID = E.SITEID 

                AND E.TANKTYPE IS NOT NULL

            INNER JOIN CIM_EQUIPMENTCLASS EC

                ON  E.EQUIPMENTCLASSID = EC.EQUIPMENTCLASSID 

                AND E.SITEID = EC.SITEID 

            LEFT OUTER JOIN CIM_CODE C

                ON  E.TANKTYPE = C.CODEID

                AND E.SITEID = C.SITEID

                AND C.CODECLASSID = 'TANKTYPE'

            LEFT OUTER JOIN CIM_CODE C2

                ON  E.MEDICINECODE = C2.CODEID

                AND E.SITEID = C2.SITEID

                AND C2.CODECLASSID = 'MEDICINECODE' 

            LEFT OUTER JOIN CIM_MATERIALDEFINITION MD

                ON  E.MEDICINECODE = MD.MATERIALDEFINITIONID

                AND E.SITEID = MD.SITEID

        WHERE TL.SITEID = @SITEID

        AND (@ROOTEQUIPMENTID = '' OR E.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)

        AND (@TANKTYPE = '' OR E.TANKTYPE = @TANKTYPE)

        AND (@EQUIPMENTID = '' OR E.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%') 

        AND  E.TANKTYPE IN ( 'MATERIAL', 'PROD' )   

    END



END
