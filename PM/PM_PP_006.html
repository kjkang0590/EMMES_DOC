--Pallet 목록



	SELECT 

		LOT.PRODUCTDEFINITIONID 

		, LOT.PALLETID 

		, COUNT(*) AS CELLQTY

		, LOT.MODIFIER  

	FROM CIM_LOTHIST LOT

	WHERE 1=1

--		AND LOT.GRADE IS NOT NULL

		AND LOT.MATERIALTYPE = 'CELL'

		AND LOT.CUSTOMACTIVITY = 'LotDataMeasureReport'

		AND LOT.TRACKOUTTIME >= TO_DATe(:FROMDATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') AND LOT.TRACKOUTTIME < TO_DATE(:TODATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') + 1

		AND (:PRODUCTDEFINITIONID IS NULL OR LOT.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID)

		AND (:PALLETID IS NULL OR LOT.PALLETID LIKE '%'|| :PALLETID ||'%')

		AND (:OUTBOXID IS NULL OR LOT.OUTBOX LIKE '%'|| :OUTBOXID ||'%' )

		AND (:INBOXID IS NULL OR LOT.INBOX LIKE '%'|| :INBOXID ||'%')

		AND (:CELLID IS NULL OR LOT.LOTID LIKE '%'|| :CELLID ||'%')

		AND (:USERID IS NULL OR LOT.MODIFIER LIKE '%'|| :USERID ||'%' )

	GROUP BY LOT.PRODUCTDEFINITIONID, LOT.PALLETID, LOT.MODIFIER
