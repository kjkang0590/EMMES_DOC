--PM_PB_012_LOT 추적 조회 쿼리
--SelectMaterialInputByLotList
--Param : SITEID, FROMDATE, TODATE, LOTID, WORKORDERID, PRODUCTORDERID
------------------------------------------	
SELECT L.PRODUCTORDERID
	 , L.WORKORDERID
	 , L.LOTID
	 , PD.PRODUCTDEFINITIONID
	 , PD.PRODUCTDEFINITIONNAME
	 , L.PROCESSSEGMENTID
	 , PM.PROCESSSEGMENTNAME
	 , L.ORIGINALQTY
	 , L.RESULTQTY
	 , CASE WHEN L.STATE = 'Consumed' THEN 'Finished'
	        ELSE L.STATE
	   END AS LOTSTATE
	 , L.TRACKINTIME
	 , L.TRACKOUTTIME
	 , RL.MATERIALLOTID
	 , RL.QTY AS CONSUMEQTY
	 , L.STATE
  FROM CIM_LOT L WITH(NOLOCK)
 INNER JOIN CIM_PRODUCTORDER PO WITH(NOLOCK)
	ON L.PRODUCTORDERID = PO.PRODUCTORDERID
   AND L.SITEID = PO.SITEID
   AND PO.STATE IN ('Active', 'Finished')
 INNER JOIN CIM_WORKORDER WO WITH(NOLOCK)
	ON L.PRODUCTORDERID = WO.PRODUCTORDERID
   AND L.WORKORDERID = WO.WORKORDERID
   AND L.SITEID = WO.SITEID
  LEFT OUTER JOIN CIM_LOTMATERIALLOTREL RL WITH(NOLOCK)
	ON  L.LOTID = RL.LOTID
   AND L.SITEID = RL.SITEID
   AND L.PROCESSSEGMENTID = RL.PROCESSSEGMENTID
  LEFT OUTER JOIN CIM_PROCESSSEGMENT PM WITH(NOLOCK)
    ON L.PROCESSSEGMENTID = PM.PROCESSSEGMENTID
   AND L.SITEID = PM.SITEID
  LEFT OUTER JOIN CIM_PRODUCTDEFINITION PD WITH(NOLOCK)
	ON L.PRODUCTDEFINITIONID = PD.PRODUCTDEFINITIONID
   AND L.SITEID = PD.SITEID
 WHERE L.SITEID = @SITEID
   AND CONVERT(CHAR(10), L.CREATETIME, 23) BETWEEN @FROMDATE AND @TODATE
   AND ( @LOTID = '' OR L.LOTID LIKE '%' + @LOTID + '%' )
   AND ( @WORKORDERID = '' OR L.WORKORDERID = @WORKORDERID )
   AND ( @PRODUCTORDERID = '' OR L.PRODUCTORDERID = @PRODUCTORDERID )
   AND RL.MATERIALLOTID LIKE '%'+@MATLOTID+'%'
 ORDER BY L.PRODUCTORDERID, L.WORKORDERID, ISNULL(PM.REASONCODE, PM.WORKCENTER), L.PROCESSSEGMENTID ASC

--SelectCboMaterialInputByLotPo
--Param : SITEID, FROMDATE, TODATE
------------------------------------------	
SELECT L.PRODUCTORDERID AS CODE,
	   L.PRODUCTORDERID AS NAME
  FROM CIM_LOT L
 INNER JOIN CIM_PRODUCTORDER CP
    ON CP.SITEID = L.SITEID
   AND CP.PRODUCTORDERID = L.PRODUCTORDERID
   AND CP.STATE IN ('Active', 'Finished')
 WHERE L.SITEID = @SITEID
   AND CONVERT(CHAR(10), L.CREATETIME, 23) BETWEEN @FROMDATE AND @TODATE
   AND L.PRODUCTORDERID IS NOT NULL
 GROUP BY L.PRODUCTORDERID
 ORDER BY L.PRODUCTORDERID
