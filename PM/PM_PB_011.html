-- PM_PB_011 BP 사일로 투입 현황
-- SelectBpSiloMaterialInputList
-- Param : SITEID, USERID, LINEID, FROMDATE, TODATE, MATERIALLOTID, BPSILOSTATE
------------------------------------------
SELECT M.WORKDATE
	 , M.MATERIALLOTID
	 , M.MATERIALDEFINITIONID
	 , M.MATERIALDEFINITIONNAME
	 , M.KITTINGTIME
	 , M.ORIGINALQTY
	 , M.ORIGINALQTY - M.QTY AS USEDQTY
	 , M.QTY
	 , M.LOCATION
     , M.LOCATIONNAME
	 , M.BPSILOSTATE
	 , M.IFFLAG
	 , M.STATE
FROM
(
	SELECT
		CONVERT(VARCHAR(10), DATEADD(HOUR, 7, ML.KITTINGTIME), 120) AS WORKDATE
		, ML.MATERIALDEFINITIONID
		, CM.MATERIALDEFINITIONNAME
		, ML.MATERIALLOTID
		, ML.KITTINGTIME
		, ML.ORIGINALQTY
		, ML.QTY
		, ML.LOCATION
		, CF.FACILITYNAME AS LOCATIONNAME
		, ML.STATE
		, BS.IFFLAG
		, BS.CREATETIME
		, ML.ISKITTING
		, CASE WHEN ML.STATE IN ( 'Created', 'Active' ) THEN 'Active'
		       WHEN ML.STATE = 'Consumed' AND ML.QTY <= 0 THEN 'Terminate'
		       ELSE '-'
		  END AS BPSILOSTATE
		, DENSE_RANK() OVER(PARTITION BY BS.LOTID ORDER BY BS.CREATETIME DESC) AS RNK_NO
	FROM CST_MATERIALLOT ML WITH(NOLOCK)
	LEFT OUTER JOIN IFS_PRD_BPSAILOLOADING BS WITH(NOLOCK)
	ON  ML.MATERIALLOTID = BS.LOTID
	AND ML.SITEID  = BS.SITEID
	AND BS.IFFLAG IN ('N', 'Y', 'E')
	LEFT OUTER JOIN CIM_MATERIALDEFINITION CM WITH(NOLOCK)
	ON  CM.SITEID = ML.SITEID
	AND CM.MATERIALDEFINITIONID = ML.MATERIALDEFINITIONID
	LEFT OUTER JOIN CIM_FACILITY CF WITH(NOLOCK)
	ON  CF.SITEID = ML.SITEID
	AND CF.FACILITYID = ML.LOCATION
	WHERE ML.ISKITTING = 'Y'
	AND   ML.LOCATION IN ('SL30', 'SP20')
	AND   ML.STATE IN ('Create', 'Active', 'Consumed')
	AND   CONVERT(VARCHAR(10), DATEADD(HOUR, 7, ML.KITTINGTIME), 120) BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
) M
WHERE M.RNK_NO = 1
AND   (@MATERIALLOTID = '' OR M.MATERIALLOTID LIKE '%' + @MATERIALLOTID + '%')
AND   (@BPSILOSTATE = '' OR M.BPSILOSTATE = @BPSILOSTATE)
ORDER BY M.KITTINGTIME ASC
