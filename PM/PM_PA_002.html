--PM_PA_002_전처리 제품 포장 - 포장 목록

--SelectGrindingPackingList

--Param : SITEID, FROMDATE, TODATE, PRODUCTCLASSID, PRODUCTDEFINITIONID

------------------------------------------

SELECT S1.PACKINGTIME

     , S1.WORKORDERID

	 , S1.LOTID

	 , S1.PRODUCTDEFINITIONID

	 , S1.PRODUCTDEFINITIONNAME

	 , S1.PROCESSSEGMENTID

	 , S1.PACKINGQTY

	 , S1.UNITNAME AS UNITID

	 , S1.INSPRESULT

	 , S1.JUDGETIME

	 , CC.CODENAME AS ISLABELPRINT

	 , S1.PRINTCOUNT

  FROM (SELECT CL.LOTSHIPPEDTIME AS PACKINGTIME

             , CL.WORKORDERID

	         , CL.LOTID

             , CL.PRODUCTDEFINITIONID

             , CP.PRODUCTDEFINITIONNAME

             , CL.PROCESSSEGMENTID

             , CL.QTY AS PACKINGQTY

             , CU.UNITNAME

             , CI.FINALJUDGE AS INSPRESULT

             , CI.FINALDATETIME AS JUDGETIME

             , CASE WHEN PL.LOTID IS NULL OR PL.LOTID = '' THEN 'N'

                    ELSE 'Y'

                END AS ISLOTID

             , PL.PRINTCOUNT

             , CL.SITEID

             , CL.LASTEVENTTIME

             , DENSE_RANK() OVER (PARTITION BY CI.LOTID ORDER BY CI.REQDATETIME DESC) AS ORDERNO

          FROM CIM_LOT CL WITH(NOLOCK)

         INNER JOIN CIM_PRODUCTDEFINITION CP WITH(NOLOCK) ON CP.SITEID = CL.SITEID AND CP.PRODUCTDEFINITIONID = CL.PRODUCTDEFINITIONID

          LEFT JOIN CIM_INSPREQ CI WITH(NOLOCK) ON CI.SITEID = CL.SITEID AND CI.LOTID = CL.LOTID

          LEFT JOIN CST_PRODUCTLABEL PL WITH(NOLOCK) ON PL.SITEID = CL.SITEID AND PL.LOTID = CL.LOTID

          LEFT JOIN CIM_UNIT CU WITH(NOLOCK) ON CU.SITEID = CL.SITEID AND CU.UNITID = CL.UNITID

         WHERE CL.SITEID = @SITEID

           AND CL.STATE = 'Shipped'

           AND CL.PROCESSSEGMENTID = 'P1020'

           --AND CL.CREATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)

            -- 2026.05.06 박성진 SC2026040035 - 전처리 포장 조회 오류 ex) [전처리 포장에서 조회를 할때 2026-04-01 ~ 2026-04-25인데 2026-03-31 데이터가 조회됨]

	      	-- => 화면에 보여지는 데이터필드 [PACKINGTIME] 이므로 이해당 조건에 맞게 조건 변경함

           AND CL.LOTSHIPPEDTIME >= @FROMDATE

           AND CL.LOTSHIPPEDTIME <  DATEADD(DAY, 1, @TODATE)

           AND (@PRODUCTCLASSID = '' OR CP.PRODUCTCLASSID = @PRODUCTCLASSID)

           AND (@PRODUCTDEFINITIONID = '' OR CP.PRODUCTDEFINITIONID = @PRODUCTDEFINITIONID)

       ) S1

  LEFT JOIN CIM_CODE CC ON CC.SITEID = S1.SITEID AND CC.CODEID = S1.ISLOTID AND CC.CODECLASSID = 'ISLABELPRINT'

 WHERE S1.ORDERNO = 1

 ORDER BY S1.LASTEVENTTIME DESC, S1.LOTID

--PM_PA_002_전처리 제품 포장 - 포장 세부 LOT 목록
--SelectPackingInfoList
--Param : SITEID, LOTID
------------------------------------------
SELECT CL.LOTID
     , CL.WORKORDERID
     , CL.PRODUCTDEFINITIONID
     , CP.PRODUCTDEFINITIONNAME
     , CLM.QTY
     , CU.UNITID
  FROM CIM_LOTMATERIALLOTREL CLM WITH(NOLOCK)
 INNER JOIN CIM_LOT CL WITH(NOLOCK) ON CL.SITEID = CLM.SITEID AND CL.LOTID = CLM.MATERIALLOTID
  LEFT OUTER JOIN CIM_PRODUCTDEFINITION CP  WITH(NOLOCK) ON CP.SITEID = CL.SITEID AND CP.PRODUCTDEFINITIONID = CL.PRODUCTDEFINITIONID
  LEFT OUTER JOIN CIM_UNIT CU  WITH(NOLOCK) ON CU.SITEID = CLM.SITEID AND CU.UNITID = CLM.UNITID
 WHERE CLM.SITEID = @SITEID
   AND CLM.LOTID = @LOTID
 ORDER BY CLM.MATERIALLOTID

-- Manual Process :: ERP IF 취소 - Consume 조회
-- SelectErpConsumeIfInfo
-- Param : SITEID, LOTID
-- 2026.04.13 공정진행 소성 프로그램 오류
-- authid : hk, sj
------------------------------------------
WITH DistinctTIDState AS (
    -- 1단계: TID와 CUDFLAG 기준으로 중복을 제거하여 1차 집계
    SELECT TID, CUDFLAG
    FROM IFS_PRD_MATERIALCONSUME
    WHERE SITEID = @SITEID
      AND PRODUCTLOTID = @LOTID
    GROUP BY TID, CUDFLAG
),
StateCTIDs AS (
    -- 2단계: 전체 TID를 대상으로 최종 상태가 'C'(합이 1)인 TID들만 필터링
    SELECT TID
    FROM DistinctTIDState
    GROUP BY TID
    HAVING SUM(CASE WHEN CUDFLAG = 'C' THEN 1 
                    WHEN CUDFLAG = 'D' THEN -1 
                    ELSE 0 END) = 1
),
TargetMaxTID AS (
    -- 3단계: 'C' 상태로 살아남은 TID들 중에서 가장 최신(MAX) TID 1개만 추출
    SELECT MAX(TID) AS MaxTID
    FROM StateCTIDs
)
-- 4단계: 위에서 구한 단 1개의 최신 'C' 상태 TID를 조건으로 상세 데이터 조회 (맨 아래로 내림)
SELECT EVENTSEQ, 
       PRODUCTLOTID, 
       LOTID, 
       CUDFLAG, 
       IFFLAG, 
       TID   
FROM IFS_PRD_MATERIALCONSUME
WHERE SITEID = @SITEID
  AND PRODUCTLOTID = @LOTID
  AND TID = (SELECT MaxTID FROM TargetMaxTID) -- 구별된 단일 TID값으로 필터링
ORDER BY EVENTSEQ DESC
