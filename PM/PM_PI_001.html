테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

-- SelectProcessSegmentListForRework



WITH DT AS (

    SELECT DISTINCT

           PN.SITEID

         , PN.ISUSABLE

         , PN.PROCESSDEFINITIONID

         , PN.PROCESSSEGMENTID

         , PN.PROCESSNODESEQUENCE

         , NVL(PS.ISREWORKYN, 'N') AS ISREWORKYN

      FROM CIM_PRODUCTPROCESSREL PP

     INNER JOIN CIM_PROCESSNODE PN ON PN.SITEID = PP.SITEID AND PN.ISUSABLE = PP.ISUSABLE

       AND PN.PROCESSDEFINITIONID = PN.PROCESSDEFINITIONID

     INNER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = PN.SITEID AND PS.ISUSABLE = PN.ISUSABLE

       AND PS.PROCESSSEGMENTID = PN.PROCESSSEGMENTID

       AND PS.ISREWORKYN = 'Y'

     WHERE 1=1

       AND PP.SITEID = :SITEID

       AND PP.ISUSABLE = 'Usable'

       AND PP.ISDEFAULT = 'Y'

)

SELECT DISTINCT

       PN.PROCESSSEGMENTID

     , PS.PROCESSSEGMENTNAME

  FROM DT

 INNER JOIN CIM_PROCESSNODE PN ON PN.SITEID = DT.SITEID AND PN.ISUSABLE = DT.ISUSABLE

   AND PN.PROCESSDEFINITIONID = DT.PROCESSDEFINITIONID

   AND PN.PROCESSNODESEQUENCE > DT.PROCESSNODESEQUENCE

 INNER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = PN.SITEID AND PS.ISUSABLE = PN.ISUSABLE

   AND PS.PROCESSSEGMENTID = PN.PROCESSSEGMENTID

 INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL RL ON RL.SITEID = PS.SITEID AND RL.ISUSABLE = PS.ISUSABLE

   AND RL.PROCESSSEGMENTID = PS.PROCESSSEGMENTID

 INNER JOIN CIM_EQUIPMENT EQ ON EQ.SITEID = RL.SITEID AND EQ.ISUSABLE = RL.ISUSABLE

   AND EQ.EQUIPMENTID = RL.EQUIPMENTID

   AND 1 = CASE

                WHEN :PLANTID IS NULL THEN 1

                WHEN EQ.PLANTID IN (SELECT * FROM TABLE(FN_MES_SPLIT(:PLANTID, ','))) THEN 1

                ELSE 0

           END

   AND 1 = CASE

                WHEN :LINEID IS NULL THEN 1

                WHEN EQ.LINEID IN (SELECT * FROM TABLE(FN_MES_SPLIT(:LINEID, ','))) THEN 1

                ELSE 0

           END

ORDER BY 1

SELECT LT.LOTID

     , PN.PROCESSNODESEQUENCE

     , PN.PROCESSNODEID

     , PN.PROCESSNODENAME

     , PN.PROCESSSEGMENTID

     , PS.PROCESSSEGMENTNAME

     , PS.ISREWORKYN

  FROM CIM_LOT LT

 INNER JOIN CIM_PROCESSNODE CN ON CN.SITEID = LT.SITEID AND CN.ISUSABLE = LT.ISUSABLE

   AND CN.PROCESSNODEID = LT.PROCESSNODEID

 INNER JOIN CIM_PROCESSNODE PN ON PN.SITEID = CN.SITEID AND PN.ISUSABLE = CN.ISUSABLE

   AND PN.PROCESSDEFINITIONID = CN.PROCESSDEFINITIONID

   AND PN.PROCESSNODESEQUENCE < CN.PROCESSNODESEQUENCE

 INNER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = PN.SITEID --AND PS.ISUSABLE = PN.ISUSABLE

   AND PS.PROCESSSEGMENTID = PN.PROCESSSEGMENTID

   AND PS.ISREWORKYN = 'Y'

 WHERE 1=1

   AND LT.SITEID = :SITEID

   AND LT.LOTID = :LOTID

   AND LT.ISUSABLE = 'Usable'

 ORDER BY 1, 2
