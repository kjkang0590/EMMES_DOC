-- PM_PR_001 LOT 재공 조회 - 재공 현황
-- SelectWipList
-- Param : SITEID, LINEID, PROCESSTYPE, PROCESSSEGMENTID, MATERIALDEFINITIONID
------------------------------------------
SELECT CL.PROCESSTYPE
     , CC.CODENAME AS PROCESSTYPENAME
	 , CL.PROCESSSEGMENTID
	 , CP.PROCESSSEGMENTNAME
	 , CM.MATERIALDEFINITIONID
	 , CM.MATERIALDEFINITIONNAME
	 , COUNT(CL.LOTID) AS WIPLOTCNT
	 , SUM(CL.QTY) AS WIPQTY
	 , CU.UNITNAME
  FROM CIM_LOT CL WITH(NOLOCK)
 INNER JOIN CIM_PROCESSSEGMENT CP WITH(NOLOCK) ON CP.SITEID = CL.SITEID AND CP.PROCESSSEGMENTID = CL.PROCESSSEGMENTID
 INNER JOIN CIM_MATERIALDEFINITION CM WITH(NOLOCK) ON CM.SITEID = CL.SITEID AND CM.MATERIALDEFINITIONID = CL.MATERIALDEFINITIONID
 LEFT OUTER JOIN CIM_CODE CC WITH(NOLOCK) ON CC.SITEID = CL.SITEID AND CC.CODECLASSID = 'PROCESSTYPE' AND CC.CODEID = CL.PROCESSTYPE
 LEFT OUTER JOIN CIM_UNIT CU WITH(NOLOCK) ON CU.SITEID = CL.SITEID AND CU.UNITID = CL.UNITID
 WHERE CL.SITEID = @SITEID
   AND CL.STATE = 'Active'
   AND CL.LINEID = @LINEID
   AND (@PROCESSTYPE = '' OR CL.PROCESSTYPE = @PROCESSTYPE)
   AND (@PROCESSSEGMENTID = '' OR CL.PROCESSSEGMENTID = @PROCESSSEGMENTID)
   AND (@MATERIALDEFINITIONID = '' OR CL.MATERIALDEFINITIONID = @MATERIALDEFINITIONID)
 GROUP BY CL.PROCESSTYPE
     , CC.CODENAME
	 , CL.PROCESSSEGMENTID
	 , CP.PROCESSSEGMENTNAME
	 , CM.MATERIALDEFINITIONID
	 , CM.MATERIALDEFINITIONNAME
	 , CU.UNITNAME
 ORDER BY CL.PROCESSTYPE
     , CL.PROCESSSEGMENTID
     , CM.MATERIALDEFINITIONID

마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_PROCESSSEGMENT, CIM_MATERIALDEFINITION, CIM_UNIT

-- PM_PR_001 LOT 재공 조회 - LOT 목록
-- SelectWipLotList
-- Param : SITEID, LINEID, PROCESSTYPE, PROCESSSEGMENTID, MATERIALDEFINITIONID
------------------------------------------
SELECT CL.PROCESSSEGMENTID
	 , CP.PROCESSSEGMENTNAME
	 , CL.LOTID
	 , CL.WORKORDERID
	 , CL.PRODUCTORDERID
	 , CM.MATERIALDEFINITIONID
	 , CM.MATERIALDEFINITIONNAME
	 , CL.STATE
	 , CL.TRACKINTIME --투입시간
	 , DATEADD(HOUR, 2, CL.TRACKINTIME) AS EXPECTENDTIME --예상종료시간 : TRACKINTIME + READTIME (계산 필요)
	 , CL.MATERIALQTY --자재투입수 (갯수) 원자재 - ex. PACK 1개
	 , CL.MATERIALUNITID 
	 , CL.SUBMATERIALQTY --자재투입량 (무게)   - ex. PACK은 cell 300개 무게는 240KG (후처리때는 값없음)
	 , CL.QTY     --  생산량
	 , CU.UNITNAME  --  (전처리때는 값없음)
	 , CL.QTY AS CONVERT_LIQTY	-- 계산식 필요 (전처리때는 값없음)
	 , CL.QTY AS CONVERT_COQTY	-- 계산식 필요 (전처리때는 값없음)
	 , CL.QTY AS CONVERT_NIQTY	-- 계산식 필요 (전처리때는 값없음)
  FROM CIM_LOT CL WITH(NOLOCK)
 INNER JOIN CIM_PROCESSSEGMENT CP WITH(NOLOCK) ON CP.SITEID = CL.SITEID AND CP.PROCESSSEGMENTID = CL.PROCESSSEGMENTID
 INNER JOIN CIM_MATERIALDEFINITION CM WITH(NOLOCK) ON CM.SITEID = CL.SITEID AND CM.MATERIALDEFINITIONID = CL.MATERIALDEFINITIONID
  LEFT OUTER JOIN CIM_UNIT CU WITH(NOLOCK) ON CU.SITEID = CL.SITEID AND CU.UNITID = CL.UNITID
 WHERE CL.SITEID = @SITEID
   AND CL.STATE = 'Active'
   AND CL.LINEID = @LINEID
   AND (@PROCESSTYPE = '' OR CL.PROCESSTYPE = @PROCESSTYPE)
   AND (@PROCESSSEGMENTID = '' OR CP.PROCESSSEGMENTID = @PROCESSSEGMENTID)
   AND (@MATERIALDEFINITIONID = '' OR CM.MATERIALDEFINITIONID = @MATERIALDEFINITIONID)
 ORDER BY CL.PROCESSSEGMENTID
     , CL.LOTID
