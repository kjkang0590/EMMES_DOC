--PM_MA_003_자재 이력 조회
--SelectMaterialLotHist
--Param : SITEID, FROMDATE, TODATE, LOCATION, MATERIALCLASSID, MATERIALTYPE, MATERIALDEFINITIONID, MATERIALLOTID
------------------------------------------
SELECT ML.LASTEVENTTIME
     , ML.LOCATION
     , CF.FACILITYNAME AS LOCATIONNAME
     , ML.MATERIALLOTID
     , ML.MATERIALLOTNAME
     , ML.MATERIALDEFINITIONID
     , MD.MATERIALDEFINITIONNAME
     , ML.MATERIALCLASSID
     , MC.MATERIALCLASSNAME
     , ML.MATERIALTYPE
     , ML.STATE
     , ML.QTY
     , CU.UNITNAME
  FROM CST_MATERIALLOTHIST ML
  LEFT OUTER JOIN CIM_FACILITY CF ON CF.SITEID = ML.SITEID AND CF.FACILITYID = ML.LOCATION
  LEFT OUTER JOIN CIM_MATERIALCLASS MC ON MC.SITEID = ML.SITEID AND MC.MATERIALCLASSID = ML.MATERIALCLASSID
  LEFT OUTER JOIN CIM_MATERIALDEFINITION MD ON MD.SITEID = ML.SITEID AND MD.MATERIALDEFINITIONID = ML.MATERIALDEFINITIONID
  LEFT OUTER JOIN CIM_UNIT CU ON CU.SITEID = ML.SITEID AND CU.UNITID = ML.UNITID
 WHERE ML.SITEID = @SITEID
   AND ML.LASTEVENTTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@LOCATION = '' OR ML.LOCATION = @LOCATION)
   AND (@MATERIALCLASSID = '' OR ML.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALTYPE = '' OR ML.MATERIALTYPE = @MATERIALTYPE)
   AND (@MATERIALDEFINITIONID = '' OR ML.MATERIALDEFINITIONID = @MATERIALDEFINITIONID)
   AND (@MATERIALLOTID = '' OR ML.MATERIALLOTID LIKE '%' + @MATERIALLOTID + '%')
 ORDER BY ML.LASTEVENTTIME DESC

--SelectCboMaterialDefinitionByGroupType
--Param : SITEID, MATERIALCLASSID, MATERIALTYPE
------------------------------------------
SELECT MD.MATERIALDEFINITIONID AS CODE
     , MD.MATERIALDEFINITIONNAME AS NAME
     , MD.MATERIALCLASSID
     , MD.MATERIALTYPE
  FROM CIM_MATERIALDEFINITION MD WITH(NOLOCK)
 INNER JOIN CIM_MATERIALCLASS CM WITH(NOLOCK) ON CM.SITEID = MD.SITEID AND CM.MATERIALCLASSID = MD.MATERIALCLASSID
 WHERE MD.SITEID = @SITEID
   AND (@MATERIALCLASSID = '' OR MD.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALTYPE = '' OR MD.MATERIALTYPE = @MATERIALTYPE)
 ORDER BY MD.MATERIALDEFINITIONID
