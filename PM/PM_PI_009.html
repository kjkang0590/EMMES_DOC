-- SelectBatchListByBatchType

SELECT B.SITEID, B.BATCHID, B.COMMENTS, B.WORKORDERID, PD.PRODUCTDEFINITIONID, PD.PRODUCTDEFINITIONNAME
	, B.CREATETIME
	, COUNT(DISTINCT LCR.CARRIERID) CARRIERCOUNT
	, COUNT(DISTINCT LBR.LOTID) LOTCOUNT
FROM CIM_BATCH B
	LEFT JOIN CIM_LOTBATCHREL LBR
		ON 1=1
		AND LBR.ISUSABLE = B.ISUSABLE
		AND LBR.SITEID = B.SITEID
		AND LBR.BATCHID = B.BATCHID
	LEFT JOIN CIM_LOTCARRIERREL LCR
		ON 1=1
		AND LCR.ISUSABLE = LBR.ISUSABLE
		AND LCR.SITEID = LBR.SITEID
		AND LCR.LOTID = LBR.LOTID
	LEFT JOIN CIM_WORKORDER WO
		ON 1=1
		AND WO.ISUSABLE = B.ISUSABLE
		AND WO.SITEID = B.SITEID
		AND WO.WORKORDERID = B.WORKORDERID
	LEFT JOIN CIM_PRODUCTDEFINITION PD
		ON 1=1
		AND PD.ISUSABLE = WO.ISUSABLE
		AND PD.SITEID = WO.SITEID
		AND PD.PRODUCTDEFINITIONID = WO.PRODUCTDEFINITIONID
WHERE 1=1
	AND B.ISUSABLE = 'Usable'
	AND B.SITEID = :SITEID
	AND B.BATCHTYPE = :BATCHTYPE
GROUP BY B.SITEID, B.BATCHID, B.COMMENTS, B.WORKORDERID, B.CREATETIME, PD.PRODUCTDEFINITIONID, PD.PRODUCTDEFINITIONNAME
ORDER BY B.CREATETIME DESC

CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
