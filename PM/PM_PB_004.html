--PM_PB_014_자주검사 - 계층적 공정검사 체크 시트(LPA Check Sheet) 조회 쿼리
--SelectLPAResultList
--Param : SITEID, LINEID, WORKDATE, LPAGROUP, LPATYPE, INSPMANAGER
------------------------------------------
SELECT @WORKDATE AS WORKDATE
     , I.LPAGROUP
     , A.CODENAME AS LPAGROUPNAME
     , I.LPAITEMID
     , B.CODENAME AS LPATYPENAME
     , C.CODENAME AS PROCESSSEGMENTTYPENAME
     , D.CODENAME AS LPADIVISIONNAME
     , I.LPAITEMNAME
     , E.CODENAME AS INSPMANAGERNAME
     , V.INSPUSER
     , F.USERNAME AS INSPUSERNAME
     , CASE WHEN V.ITEMVALUE IS NULL OR V.ITEMVALUE = '' THEN 'NA' ELSE V.ITEMVALUE END AS ITEMVALUE
  FROM CST_LPAITEM I WITH(NOLOCK) 
  LEFT JOIN CST_LPARESULTITEM V WITH(NOLOCK) ON I.SITEID = V.SITEID AND I.LPAGROUP = V.LPAGROUP AND I.LPAITEMID = V.LPAITEMID AND V.WORKDATE = @WORKDATE
  LEFT JOIN CIM_CODE A WITH(NOLOCK) ON A.SITEID = I.SITEID AND A.CODEID = I.LPAGROUP AND A.CODECLASSID = 'LPAGROUP' 
  LEFT JOIN CIM_CODE B WITH(NOLOCK) ON B.SITEID = I.SITEID AND B.CODEID = I.LPATYPE AND B.CODECLASSID = 'LPATYPE'
  LEFT JOIN CIM_CODE C WITH(NOLOCK) ON C.SITEID = I.SITEID AND C.CODEID = I.PROCESSSEGMENTTYPE AND C.CODECLASSID = 'PROCESSSEGMENTTYPE'
  LEFT JOIN CIM_CODE D WITH(NOLOCK) ON D.SITEID = I.SITEID AND D.CODEID = I.LPADIVISION AND D.CODECLASSID = 'LPADIVISION'
  LEFT JOIN CIM_CODE E WITH(NOLOCK) ON E.SITEID = I.SITEID AND E.CODEID = I.INSPMANAGER AND E.CODECLASSID = 'INSPMANAGER'
  LEFT JOIN CIM_USER F WITH(NOLOCK) ON F.SITEID = V.SITEID AND F.USERID = V.INSPUSER
 WHERE I.SITEID = @SITEID
   AND I.ISUSABLE = 'Usable'
   AND (@LPAGROUP = '' OR I.LPAGROUP = @LPAGROUP)
   AND (@LPATYPE = '' OR I.LPATYPE = @LPATYPE)
   AND (@INSPMANAGER= '' OR I.INSPMANAGER = @INSPMANAGER)
 ORDER BY V.WORKDATE, V.LPAGROUP, I.UISEQUENCE

--PM_PB_014_자주검사조회 - 하단 특이사항, 부적합 세부 내용 조회 쿼리
--SelectLPACommentsList
--Param : SITEID, LINEID, FROMDATE, TODATE, LPAGROUP
------------------------------------------
SELECT L.WORKDATE
     , A.CODENAME AS LPAGROUP
     , L.COMMENTS
     , L.DESCRIPTION
  FROM CST_LPARESULT L WITH(NOLOCK)
  LEFT JOIN CIM_CODE A WITH(NOLOCK) ON A.SITEID = L.SITEID AND A.CODEID = L.LPAGROUP AND A.CODECLASSID = 'LPAGROUP'
 WHERE L.SITEID = @SITEID 
   AND L.LINEID = @LINEID
   AND L.WORKDATE BETWEEN @FROMDATE AND @TODATE
   AND (@LPAGROUP = '' OR L.LPAGROUP = @LPAGROUP)
 ORDER BY L.WORKDATE, L.LPAGROUP
