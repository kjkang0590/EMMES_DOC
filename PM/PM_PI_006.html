마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_INSPDEFINITIONITEMSPEC, CIM_STATE, CIM_RECIPEPARAMETER

--LOT 추적 (검사결과)

SELECT 
	  RSLT.INSPLOTID AS LOTID
	, SPEC.INSPTYPE AS INSPTYPE
	, SPEC.INSPITEMID AS ITEMID
	, SPEC.USL AS UPPERLIMIT
	, SPEC.LSL AS LOWERLIMIT
	, SPEC.TARGET AS TARGET
	, RSLT.INSPITEMVALUE AS VALUE
	, IR.STATE AS INSPRESULT
	, ST.STATENAME  AS INSPRESULTNAME
FROM CIM_INSPRESULT RSLT
LEFT OUTER JOIN CIM_INSPDEFINITIONITEMSPEC SPEC
	ON RSLT.INSPDEFINITIONITEMSPECSYSID = SPEC.INSPDEFINITIONITEMSPECSYSID 
LEFT OUTER JOIN CIM_INSPREQ IR ON IR.SITEID = RSLT.SITEID AND IR.INSPREQNO = RSLT.INSPREQNO 
LEFT OUTER JOIN CIM_STATE ST ON IR.SITEID = ST.SITEID AND IR.STATE = ST.STATEID AND ST.STATEMODELID = 'InspReqState'
WHERE 1=1
	AND RSLT.SITEID = :SITEID
	AND RSLT.INSPLOTID = :LOTID
	AND (:INSPRESULT IS NULL OR IR.STATE = :INSPRESULT)
	AND (:INSPTYPE IS NULL OR :INSPTYPE IN('IQC', 'OQC', 'PQC'))
UNION ALL
SELECT SR.LOTID 
     , 'SQC' AS INSPTYPE
     , SR.RECIPEITEMID AS ITEMID
     , SR.UPPERLIMIT
     , SR.LOWERLIMIT
     , RP.TARGET
     , SR.ITEMVALUE AS VALUE
     , SR.STATE AS INSPRESULT
     , ST.STATENAME AS INSPRESULTNAME
  FROM CUS_SQCRESULT SR 
 INNER JOIN CIM_RECIPEPARAMETER RP ON RP.SITEID = SR.SITEID AND RP.RECIPEDEFINITIONID = SR.RECIPEDEFINITIONID AND RP.RECIPEITEMID = SR.RECIPEITEMID
       AND RP.ISUSABLE = 'Usable'
 LEFT OUTER JOIN CIM_STATE ST ON SR.SITEID = ST.SITEID AND SR.STATE = ST.STATEID AND ST.STATEMODELID = 'SQCState'
 WHERE SR.SITEID = :SITEID 
   AND SR.LOTID = :LOTID
   AND (:INSPRESULT IS NULL OR SR.STATE = :INSPRESULT)
   AND (:INSPTYPE IS NULL OR :INSPTYPE IN('SQC'))
