SELECT CO.TAGID
     , CO.TAGNAME
     , CO.EQUIPMENTID
     , CE.EQUIPMENTNAME
     , 20 AS QTY -- 수량 테이블에서 입력 수량 가져와야한다 (쿼리 수정 필요)

 WHERE CO.SITEID = @SITEID

SQL 보기
------------------------------------------
--PM_PB_006_망초 관리
--SelectNa2So4List
--Param : SITEID, FROMDATE, TODATE
------------------------------------------
SELECT CL.LOTID
     , CL.PRODUCTDEFINITIONID
     , CP.PRODUCTDEFINITIONNAME
     , CL.QTY AS PACKINGQTY
     , CU.UNITNAME
     , CL.LOTSHIPPEDTIME AS PACKINGTIME
     , CL.CREATOR
     , CL.DESCRIPTION
  FROM CIM_LOT CL WITH(NOLOCK)
 INNER JOIN CIM_PRODUCTDEFINITION CP WITH(NOLOCK) ON CP.SITEID = CL.SITEID AND CP.PRODUCTDEFINITIONID = CL.PRODUCTDEFINITIONID
  LEFT OUTER JOIN CIM_UNIT CU WITH(NOLOCK) ON CU.SITEID = CL.SITEID AND CU.UNITID = CL.UNITID 
 WHERE CL.SITEID = @SITEID
   AND CL.STATE = 'Shipped'
   --AND CL.PROCESSSEGMENTID = 'P1090' --Shipped 처리 되어도 Processsegment가 제대로 박혀있는지 확인 필요
   AND CL.PRODUCTDEFINITIONID = '1PSSA1' -- 망초만 조회
   AND CL.LOTSHIPPEDTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
 ORDER BY CL.LASTEVENTTIME DESC, CL.LOTID
