마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_MATERIALDEFINITION, CIM_PROCESSSEGMENT, CIM_EQUIPMENT, CIM_CODE, CIM_STATE

--PM_PB_002 공정 진행 (반응) - Lot 목록 조회
--SelectLeachingProcessList
--Param : SITEID, LINEID, FROMDATE, TODATE, WOSTATE, MATERIALCLASSID, MATERIALDEFINITIONID, PRODUCTORDERID
------------------------------------------
SELECT WO.PRODUCTORDERID
     , WO.WORKORDERID
     , WO.PRODUCTDEFINITIONID AS ROOTMATERIALDEFINITIONID
     , WO.STATE AS WOSTATE
     , CC.CODENAME AS WOSTATENAME
     , CL.LOTID
     , CL.STATE AS LOTSTATUS
     , CS.STATENAME AS LOTSTATENAME
     , MD.MATERIALCLASSID
     , MD.MATERIALDEFINITIONID
     , MD.MATERIALDEFINITIONNAME
     , CL.PROCESSSEGMENTID
     , PC.PROCESSSEGMENTNAME
     , CL.EQUIPMENTID
     , CE.EQUIPMENTNAME
     , WO.REQUESTTIME
     , CL.TRACKINTIME
     , CL.TRACKOUTTIME
     , CL.MATERIALQTY
     , (CL.LITHIUMMETALQTY) AS EXPECTLIQTY
     , (CL.COBALTMETALQTY) AS EXPECTCOQTY
     , (CL.NICKELMETALQTY) AS EXPECTNIQTY
     , CASE WHEN CL.LOTID IS NULL AND ISNULL(TT.CNT, 0) = 0 THEN 'START'
            WHEN CL.LOTID IS NOT NULL AND ISNULL(TT.CNT, 0) = 0 THEN 'END'
            ELSE 'MOVE'
       END AS NEXTPROCESS
  FROM (SELECT DISTINCT W.WORKORDERID, W.PRODUCTORDERID, W.STATE, W.REQUESTTIME, W.SITEID, W.LINEID, W.PRODUCTDEFINITIONID, W.PROCESSTYPE, L.PRODUCTDEFINITIONID AS MATERIALDEFINITIONID
          FROM CIM_WORKORDER W
         INNER JOIN CIM_LOT L ON L.SITEID = W.SITEID AND L.WORKORDERID = W.WORKORDERID AND L.STATE IN ('Finished', 'Consumed') AND L.PROCESSSEGMENTID = 'P2010'
       ) AS WO
 INNER JOIN CIM_MATERIALDEFINITION MD ON MD.SITEID = WO.SITEID AND MD.MATERIALDEFINITIONID = WO.MATERIALDEFINITIONID
  LEFT JOIN CIM_LOT CL ON CL.SITEID = WO.SITEID AND CL.WORKORDERID = WO.WORKORDERID AND CL.PROCESSSEGMENTID = 'P2020'
  LEFT JOIN CIM_PROCESSSEGMENT PC ON PC.SITEID = CL.SITEID AND PC.PROCESSSEGMENTID = CL.PROCESSSEGMENTID
  LEFT JOIN CIM_EQUIPMENT CE ON CE.SITEID = CL.SITEID AND CE.EQUIPMENTID = CL.EQUIPMENTID 
  LEFT JOIN CIM_CODE CC ON CC.SITEID = WO.SITEID AND CC.CODEID = WO.STATE AND CC.CODECLASSID = 'ORDERSTATE'
  LEFT JOIN CIM_STATE CS ON CS.SITEID = CL.SITEID AND CS.STATEID = CL.STATE AND CS.STATEMODELID = 'LotState' and CS.ISUSABLE = 'Usable'
  LEFT JOIN (SELECT T.SITEID, T.PROCESSSEGMENTID, T.FROMEQUIPMENTID, COUNT(1) AS CNT
               FROM CST_TANKPATHNODE T WITH(NOLOCK)
              WHERE T.TOEQUIPMENTID <> 'END'
              GROUP BY T.SITEID, T.PROCESSSEGMENTID, T.FROMEQUIPMENTID
            ) AS TT ON TT.SITEID = CL.SITEID AND TT.PROCESSSEGMENTID = CL.PROCESSSEGMENTID AND TT.FROMEQUIPMENTID = CL.EQUIPMENTID
 WHERE WO.SITEID = @SITEID
   AND (@LINEID = '' OR WO.LINEID = @LINEID)
   AND (@WOSTATE = '' OR WO.STATE = @WOSTATE) 
   AND WO.PROCESSTYPE = 'P2000'
   AND WO.REQUESTTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@MATERIALCLASSID = '' OR MD.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALDEFINITIONID = '' OR MD.MATERIALDEFINITIONID = @MATERIALDEFINITIONID)
   AND (@PRODUCTORDERID = '' OR WO.PRODUCTORDERID LIKE '%' + @PRODUCTORDERID + '%')
 ORDER BY WO.PRODUCTORDERID, WO.WORKORDERID, CL.LOTID
;

    -- 1단계: TID와 CUDFLAG 기준으로 중복을 제거하여 1차 집계
    SELECT TID, CUDFLAG
    FROM IFS_PRD_MATERIALCONSUME
    WHERE SITEID = @SITEID
      AND PRODUCTLOTID = @LOTID
    GROUP BY TID, CUDFLAG
),
StateCTIDs AS (
    -- 2단계: 전체 TID를 대상으로 최종 상태가 'C'(합이 1)인 TID들만 필터링
    SELECT TID
    FROM DistinctTIDState
    GROUP BY TID
    HAVING SUM(CASE WHEN CUDFLAG = 'C' THEN 1 
                    WHEN CUDFLAG = 'D' THEN -1 
                    ELSE 0 END) = 1
),
TargetMaxTID AS (
    -- 3단계: 'C' 상태로 살아남은 TID들 중에서 가장 최신(MAX) TID 1개만 추출
    SELECT MAX(TID) AS MaxTID
    FROM StateCTIDs
)
-- 4단계: 위에서 구한 단 1개의 최신 'C' 상태 TID를 조건으로 상세 데이터 조회 (맨 아래로 내림)
SELECT EVENTSEQ, 
       PRODUCTLOTID, 
       LOTID, 
       CUDFLAG, 
       IFFLAG, 
       TID   
FROM IFS_PRD_MATERIALCONSUME
WHERE SITEID = @SITEID
  AND PRODUCTLOTID = @LOTID
  AND TID = (SELECT MaxTID FROM TargetMaxTID) -- 구별된 단일 TID값으로 필터링
ORDER BY EVENTSEQ DESC
