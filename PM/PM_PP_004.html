	                           INNER JOIN CIM_MATERIALLOT ML ON ML.SITEID = MPR.SITEID AND ML.MATERIALLOTID = MPR.MATERIALLOTID AND ML.ISUSABLE = 'Usable' 

                               INNER JOIN CIM_MATERIALDEFINITION MD ON MD.SITEID = ML.SITEID AND MD.MATERIALDEFINITIONID = ML.MATERIALDEFINITIONID AND MD.ISUSABLE = 'Usable' AND MD.PRODUCTTYPE='HALB'

                            ), 0) AS LOTQTY

	, SUM(LOT.ORIGINALQTY) AS INPUTQTY

	, SUM(LOT.QTY) + NVL((SELECT SUM(ML.QTY)

	                        FROM CIM_MATERIALLOTPORTREL MPR 

	                       INNER JOIN CIM_MATERIALLOT ML ON ML.SITEID = MPR.SITEID AND ML.MATERIALLOTID = MPR.MATERIALLOTID AND ML.ISUSABLE = 'Usable'

   AND EQP.ISUSABLE = 'Usable'

    ON SGMT.SITEID = PER.SITEID

   AND SGMT.PROCESSSEGMENTID = PER.PROCESSSEGMENTID 

   AND SGMT.ISUSABLE = 'Usable'

(

	SELECT 

		LINE.SITEID AS SITEID,

		LINE.FACILITYID AS LINEID, 

		LINE.FACILITYNAME AS LINENAME,

		LINE.PARENTFACILITYID AS PLANTID

	FROM CIM_FACILITY LINE

		WHERE 1=1

		AND SITEID = :SITEID

		AND UPPER(LINE.FACILITYTYPE) = 'LINE'

) FAC	

	ON EQP.LINEID = FAC.LINEID

LEFT OUTER JOIN 

(

	SELECT 

		LINE.SITEID AS SITEID,

		LINE.FACILITYID AS LINEID, 

		LINE.FACILITYNAME AS LINENAME,

		LINE.PARENTFACILITYID AS PLANTID

	FROM CIM_FACILITY LINE

		WHERE 1=1

		AND SITEID = :SITEID

		AND UPPER(LINE.FACILITYTYPE) = 'PLANTLINE'

) PLANTLINE

	ON EQP.SITEID = PLANTLINE.SITEID

	AND EQP.PLANTID = PLANTLINE.PLANTID 	

WHERE 1=1

	AND LOT.ISUSABLE = 'Usable'

	AND LOT.SITEID = :SITEID

	AND LOT.STATE IN('Active','Defect')

	AND LOT.PROCESSINGSTATE IN('WaitForRule', 'ProcessingRule')

	AND LOT.CREATETIME >= TO_DATE(:FROMDATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') AND LOT.CREATETIME < TO_DATE(:TODATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') + 1

	AND (:PLANTID IS NULL OR EQP.PLANTID = :PLANTID)

	AND (:PLANTLINEID IS NULL OR PLANTLINE.LINEID = :PLANTLINEID)

	AND (:LINEID IS NULL OR EQP.LINEID = :LINEID)

	AND (:EQUIPMENTID IS NULL OR EQP.EQUIPMENTID = :EQUIPMENTID)

	AND (:PROCESSSEGMENTID IS NULL OR LOT.PROCESSSEGMENTID = :PROCESSSEGMENTID)

	AND (:PRODUCTDEFINITIONID IS NULL OR LOT.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID)

	AND (:WORKORDERID IS NULL OR LOT.WORKORDERID = :WORKORDERID)

	AND (:BATCHID IS NULL OR REL.BATCHID = :BATCHID	)

GROUP BY 	PLANTLINE.SITEID, PLANTLINE.LINEID, PLANTLINE.LINENAME, FAC.LINEID, FAC.LINENAME, PER.PROCESSSEGMENTID , SGMT.PROCESSSEGMENTNAME , LOT.PRODUCTDEFINITIONID ,PER.EQUIPMENTID, EQP.EQUIPMENTNAME  

HAVING SUM(LOT.QTY) > 0

ORDER BY FAC.LINEID, PER.PROCESSSEGMENTID

SELECT 

	LOT.LOTID AS LOTID

	, LOT.PRODUCTDEFINITIONID AS PRODUCTDEFINITIONID

	, LOT.PRODUCTORDERID AS PRODUCTORDERID

	, LOT.WORKORDERID AS WORKORDERID

	, LOT.ORIGINALQTY AS INPUTQTY

	, LOT.QTY AS GOODQTY

	, LOT.DEFECTQTY AS DEFECTQTY

	, LOT.LOSSQTY AS LOSSQTY

	, LOT.ISHOLD AS ISHOLD

	, LOT.ISREWORK AS ISREWORK

	, LOT.GRADE AS GRADE

	, LOT.LOCATION AS LOCATION

	, LOT.TRACKINTIME AS STARTDATE

	, LOT.TRACKOUTTIME AS ENDDATE

	, TO_CHAR(LOT.TRACKINUSER) AS WORKUSERID

 FROM CIM_LOT LOT

LEFT JOIN CIM_LOTBATCHREL REL

   ON LOT.SITEID = REL.SITEID 

  AND LOT.LOTID = REL.LOTID 

  AND REL.ISUSABLE = 'Usable'

LEFT JOIN CIM_PROCESSDEFINITION PD 

   ON PD.PROCESSDEFINITIONID = LOT.PROCESSDEFINITIONID 

  AND PD.SITEID = LOT.SITEID

  AND PD.ISUSABLE = 'Usable'

LEFT JOIN CIM_PROCESSNODE PN 

   ON PN.PROCESSDEFINITIONID = PD.PROCESSDEFINITIONID 

  AND PN.SITEID = PD.SITEID 

  AND PN.ISUSABLE = 'Usable'

  AND PN.PROCESSNODEID = LOT.PROCESSNODEID 

LEFT JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER

	ON PER.SITEID = PN.SITEID 

   AND PER.PROCESSSEGMENTID = PN.PROCESSSEGMENTID 

   AND PER.ISUSABLE = 'Usable'

LEFT JOIN CIM_EQUIPMENT EQP 

	ON EQP.SITEID = PER.SITEID 

   AND EQP.EQUIPMENTID = PER.EQUIPMENTID 

   AND EQP.ISUSABLE = 'Usable'

LEFT JOIN CIM_PROCESSSEGMENT SGMT 

    ON SGMT.SITEID = PER.SITEID

   AND SGMT.PROCESSSEGMENTID = PER.PROCESSSEGMENTID 

   AND SGMT.ISUSABLE = 'Usable'

WHERE 1=1

	AND LOT.ISUSABLE = 'Usable'

	AND LOT.SITEID = :SITEID

	AND LOT.STATE= 'Active'

	AND LOT.PROCESSINGSTATE IN('WaitForRule', 'ProcessingRule')

	--AND LOT.CREATETIME >= TO_DATE(:FROMDATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') AND LOT.CREATETIME < TO_DATE(:TODATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') + 1

	AND (:PRODUCTDEFINITIONID IS NULL OR LOT.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID)

	AND (:PROCESSSEGMENTID IS NULL OR LOT.PROCESSSEGMENTID = :PROCESSSEGMENTID )

	AND (:LINEID IS NULL OR EQP.LINEID = :LINEID)

	AND (:EQUIPMENTID IS NULL OR EQP.EQUIPMENTID = :EQUIPMENTID)

UNION ALL

-- SelectLineList

SELECT 

	FACILITYID AS CODE, -- LINEID,

	FACILITYNAME AS NAME, -- LINENAME,

	SHORTNAME, SHOP, LINEGROUP

FROM CIM_FACILITY LINE

	WHERE 1=1

	AND UPPER(LINE.FACILITYTYPE) = 'LINE'

	AND LINE.SITEID = :SITEID

	AND LINE.PARENTFACILITYID IN 

	(

		SELECT FACILITYID FROM CIM_FACILITY PLANTLINE

		WHERE 1=1

			AND (:PLANTID IS NULL OR PLANTLINE.PARENTFACILITYID = :PLANTID)

			AND UPPER(PLANTLINE.FACILITYTYPE) = 'PLANTLINE'

			AND PLANTLINE.SITEID = :SITEID

	)

	AND LINE.ISUSABLE = 'Usable'

	AND (:SHOP IS NULL OR LINE.SHOP IN (SELECT COLUMN_VALUE FROM FN_MES_SPLIT(:SHOP, ',')))

ORDER BY UISEQUENCE
테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

-- SelectProcessSegmentList

SELECT DISTINCT EQ.SITEID, EQ.PLANTID, EQ.LINEID
	, PC.PROCESSSEGMENTID AS CODE
	, PC.PROCESSSEGMENTNAME AS NAME
	, PC.PROCESSTYPE
	, PC.BIZRULEID
	, CASE WHEN (SELECT COUNT(*) FROM CIM_EQUIPMENT SUB WHERE SUB.PARENTID = EQ.EQUIPMENTID) > 0 THEN 'Y' ELSE 'N' END ISSUBEQP
FROM CIM_EQUIPMENT EQ
	INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER
	ON 1=1
	AND PER.EQUIPMENTID = EQ.EQUIPMENTID
	AND PER.SITEID = EQ.SITEID
	AND PER.ISUSABLE = 'Usable'
	INNER JOIN CIM_PROCESSSEGMENT PC
	ON 1=1
	AND PC.PROCESSSEGMENTID = PER.PROCESSSEGMENTID
	AND PC.SITEID = PER.SITEID
	AND PC.ISUSABLE = 'Usable'
WHERE 1=1
	AND (:PLANTID IS NULL OR EQ.PLANTID = :PLANTID)
	AND (:LINEID IS NULL OR  EQ.LINEID = :LINEID)
ORDER BY EQ.PLANTID, EQ.LINEID, PC.PROCESSSEGMENTID
