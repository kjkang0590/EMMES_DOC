-- PM_PR_003 생산 실적 조회
-- SelectLotReportsList
-- Param : SITEID, LINEID, FROMDATE, TODATE, PROCESSTYPE, PRODUCTCLASSID, PRODUCTDEFINITIONID, PRODUCTORDERID, WORKORDERID
------------------------------------------
SELECT LMR.MATERIALLOTID AS LOTID
     , CL.PRODUCTORDERID
     , CL.WORKORDERID
     , LMR.QTY
     , LMR.UNITID
     , LMR.LOTID AS PACKLOTID
     , PC.PRODUCTCLASSNAME
     , CL.PRODUCTDEFINITIONID
     , CP.PRODUCTDEFINITIONNAME
     , CL.QTY AS PACKQTY
     , CL.UNITID AS PACKUNITID
     , CL.CREATETIME
     , IR.JUDGE AS FINALJUDGE
     , IR.INSPDATETIME
  FROM CIM_LOT CL WITH(NOLOCK)
 INNER JOIN CIM_PRODUCTDEFINITION CP WITH(NOLOCK) ON CP.SITEID = CL.SITEID AND CP.PRODUCTDEFINITIONID = CL.PRODUCTDEFINITIONID
 INNER JOIN CIM_PRODUCTCLASS PC WITH(NOLOCK) ON PC.SITEID = CP.SITEID AND PC.PRODUCTCLASSID = CP.PRODUCTCLASSID
 INNER JOIN CIM_LOTMATERIALLOTREL LMR WITH(NOLOCK) ON LMR.SITEID = CL.SITEID AND LMR.LOTID = CL.LOTID
  LEFT OUTER JOIN CIM_INSPREQLOT IR WITH(NOLOCK) ON IR.SITEID = CL.SITEID AND IR.INSPLOTID = CL.LOTID
 WHERE CL.SITEID = @SITEID
   AND CL.LINEID = @LINEID
   AND CL.CREATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@PROCESSTYPE = '' OR CL.PROCESSTYPE = @PROCESSTYPE)
   --AND (@PRODUCTCLASSID = '' OR CP.PRODUCTCLASSID = @PRODUCTCLASSID)
   AND CP.PRODUCTCLASSID = @PRODUCTCLASSID
   AND (@PRODUCTDEFINITIONID = '' OR CP.PRODUCTDEFINITIONID = @PRODUCTDEFINITIONID)
   AND (@PRODUCTORDERID = '' OR CL.PRODUCTORDERID = @PRODUCTORDERID)
   AND (@WORKORDERID = '' OR CL.WORKORDERID = @WORKORDERID)
