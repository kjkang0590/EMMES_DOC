Stored Procedure SP_GET_PROCESSDATA 정의 보기
CREATE PROCEDURE [dbo].[SP_GET_PROCESSDATA] 

@ROOTEQUIPMENTID VARCHAR(2), @EQUIPMENTCLASSID VARCHAR(20), @EQUIPMENTID VARCHAR(10), @SITEID VARCHAR(4), @TAGID VARCHAR(100)

AS

BEGIN



	DECLARE @PROCESSDATAINFO TABLE

	(

		ROWNUM INT

		, SITEID VARCHAR(10)

		, PROCESSDATAPARAMETERID VARCHAR(100)

		, ROOTEQUIPMENTID VARCHAR(10)

		, EQUIPMENTID VARCHAR(10)

		, EQUIPMENTNAME VARCHAR(100)

		, TAGID VARCHAR(200)

		, TAGNAME VARCHAR(200)

		, COLVALUES VARCHAR(40)

		, MODIFYTIME DATETIME

		, DATAMAPPINGTABLE VARCHAR(40)

		, DATAMAPPINGCOLUMN VARCHAR(40)

		, DCSTAGID VARCHAR(100)

	);





	WITH EQUIPMENTIFNO AS

	(

		SELECT DISTINCT

			R.PARENTID AS MAIN_EQUIPMENTID

			, R.CHILDID AS SUB_EQUIPMENTID

			, E.ROOTEQUIPMENTID

			, E.EQUIPMENTID

			, E.EQUIPMENTNAME

			, E.EQUIPMENTCLASSID

			, E.SITEID

			, OT.TAGID

			, OT.TAGNAME

			, OT.DCSTAGID

		FROM CIM_EQUIPMENT E WITH(NOLOCK)

			INNER JOIN CST_PUBLICREL R WITH(NOLOCK)

				ON  E.SITEID = R.SITEID 

				AND E.EQUIPMENTID = R.PARENTID 

				AND R.RELTYPE = 'EQUIPMENT'

			INNER JOIN CST_OPCTAGINFO OT WITH(NOLOCK)

				ON  OT.SITEID = R.SITEID

--				AND OT.EQUIPMENTID = R.CHILDID

--				AND ( OT.EQUIPMENTID = R.CHILDID OR OT.EQUIPMENTID = R.PARENTID)

--				AND (R.CHILDID IN(SELECT value FROM STRING_SPLIT(OT.EQUIPMENTID, ',')) OR R.PARENTID IN(SELECT value FROM STRING_SPLIT(OT.EQUIPMENTID, ','))) -- 2024/04/29 TEST 위해 주석으로 추가

				AND ( OT.EQUIPMENTID = R.PARENTID OR R.CHILDID = OT.DCSTAGID) -- 6/17 변경

				AND OT.ISUSABLE = 'Usable'				

		WHERE E.SITEID = @SITEID

		AND   E.ROOTEQUIPMENTID = @ROOTEQUIPMENTID

		AND   (@EQUIPMENTCLASSID = '' OR E.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)

		AND   (@EQUIPMENTID = '' OR E.EQUIPMENTID = @EQUIPMENTID)

		AND   E.ISUSABLE = 'Usable'

		--AND   OT.TAGID LIKE '%' + @TAGID + '%'

		AND   ISNULL(OT.DCSTAGID, '') LIKE '%' + @TAGID + '%'

	)

	INSERT INTO @PROCESSDATAINFO 

	SELECT 

    	  ROW_NUMBER() OVER(ORDER BY (SELECT 1)) AS ROWNUM

		, PDP.SITEID

		, PDP.PROCESSDATAPARAMETERID

		, EI.ROOTEQUIPMENTID

		, EI.EQUIPMENTID

		, EI.EQUIPMENTNAME

		, EI.TAGID

		, EI.TAGNAME

		, NULL AS COLVALUES

		, NULL AS MODIFYTIME

		, PDP.DATAMAPPINGTABLE 

		, PDP.DATAMAPPINGCOLUMN 

		, EI.DCSTAGID

	FROM CST_PROCESSDATAPARAMETER PDP WITH(NOLOCK)

		INNER JOIN EQUIPMENTIFNO EI

			ON  PDP.PROCESSDATAPARAMETERID = EI.TAGID

			AND PDP.SITEID = EI.SITEID

	WHERE PDP.SITEID = @SITEID

	AND   PDP.PROCESSDATADEFINITIONID  = 'EISMES_Tran_Process_Data'

	AND   PDP.ISUSABLE = 'Usable'

	;

	

	DECLARE @TABLENAMES VARCHAR(100)

	      , @COLUMNNAMES VARCHAR(100)

		  , @MAX_CNT INT 

		  , @NOW_NO INT 

		  , @RTN_VALUE VARCHAR(15)

		  , @RTN_TIME DATETIME

	;

	SELECT @MAX_CNT = COUNT(1), @NOW_NO = 1 FROM @PROCESSDATAINFO

	

	WHILE (@NOW_NO <= @MAX_CNT)

	BEGIN

		SELECT @TABLENAMES = DATAMAPPINGTABLE

		     , @COLUMNNAMES = DATAMAPPINGCOLUMN

		FROM @PROCESSDATAINFO

		WHERE ROWNUM = @NOW_NO

		

		EXEC SP_GET_PROCESSDATAVALUE @TABLENAMES, @COLUMNNAMES, @RTN_VALUE OUTPUT, @RTN_TIME OUTPUT;

		

		UPDATE @PROCESSDATAINFO SET COLVALUES = @RTN_VALUE, MODIFYTIME = @RTN_TIME WHERE ROWNUM = @NOW_NO;

		

		SET @NOW_NO = @NOW_NO + 1;

	END

	

	SELECT DISTINCT

		ROOTEQUIPMENTID

		, EQUIPMENTID

		, EQUIPMENTNAME

		, TAGID

		, TAGNAME

		, COLVALUES AS TAGVALUE

		, '' AS DESCRIPTION

		, MODIFYTIME AS CREATETIME

		, DCSTAGID

	FROM @PROCESSDATAINFO



END
Stored Procedure SP_GET_PROCESSDATAVALUE 정의 보기
CREATE PROCEDURE [dbo].[SP_GET_PROCESSDATAVALUE] 

@TABLENAMES VARCHAR(100), @COLUMNNAMES VARCHAR(MAX), @RTNVAL VARCHAR(15) OUTPUT, @RTNDATE DATETIME OUTPUT

AS

BEGIN

	DECLARE @RTNTABLE TABLE

	(

		RTNVALUES VARCHAR(15)

		, RTNDATE DATETIME 

	)

	

	DECLARE @SQL NVARCHAR(MAX)

	

    SET @SQL = 'SELECT ' + @COLUMNNAMES + ', MODIFYTIME FROM ' + @TABLENAMES

    INSERT @RTNTABLE EXEC (@SQL)

    

  	SELECT @RTNVAL = RTNVALUES, @RTNDATE = RTNDATE FROM @RTNTABLE

END

Stored Procedure SP_GET_PROCESSDATAHIST 정의 보기
CREATE PROCEDURE [dbo].[SP_GET_PROCESSDATAHIST] 

@FROMDATE VARCHAR(8), @TODATE VARCHAR(8), @TAGID VARCHAR(100), @SITEID VARCHAR(4), @EQUIPMENTID VARCHAR(40)

AS

BEGIN



	DECLARE @PROCESSDATAHISTINFO TABLE

	(

		ROOTEQUIPMENTID VARCHAR(10)

		, EQUIPMENTID VARCHAR(10)

		, EQUIPMENTNAME VARCHAR(100)

		, TAGID VARCHAR(100)

		, TAGNAME VARCHAR(100)

		, DATAMAPPINGTABLE VARCHAR(40)

		, DATAMAPPINGCOLUMN VARCHAR(40)

		, DCSTAGID VARCHAR(40)

	);



	DECLARE @PROCESSDATAVALUE TABLE

	(

		CREATETIME DATETIME

		, TAGVALUE VARCHAR(15)

	);



	INSERT INTO @PROCESSDATAHISTINFO

	SELECT 

		E.ROOTEQUIPMENTID

		, E.EQUIPMENTID

		, E.EQUIPMENTNAME

		, OT.TAGID

		, OT.TAGNAME

		, PDP.DATAMAPPINGTABLE + 'HIST' AS DATAMAPPINGTABLE

		, PDP.DATAMAPPINGCOLUMN	

		, OT.DCSTAGID 

	FROM CST_OPCTAGINFO OT

	INNER JOIN CST_PUBLICREL R

	   ON R.SITEID = OT.SITEID

--	  AND (R.CHILDID = OT.EQUIPMENTID OR R.PARENTID = OT.EQUIPMENTID)

	   AND ( OT.EQUIPMENTID = R.PARENTID OR R.CHILDID = OT.DCSTAGID)  -- 6/17 변경

	  AND OT.ISUSABLE = 'Usable'

	INNER JOIN CIM_EQUIPMENT E

	   ON E.SITEID = R.SITEID

	  AND E.EQUIPMENTID = R.PARENTID

	  AND R.RELTYPE = 'EQUIPMENT'

	/*  

	INNER JOIN CIM_EQUIPMENT E

	   ON  E.EQUIPMENTID = OT.EQUIPMENTID

	  AND E.SITEID = OT.SITEID

	*/

	INNER JOIN CST_PROCESSDATAPARAMETER PDP WITH(NOLOCK)

		ON  PDP.PROCESSDATAPARAMETERID = OT.TAGID

		AND PDP.SITEID = OT.SITEID

		AND PDP.PROCESSDATADEFINITIONID  = 'EISMES_Tran_Process_Data'		

	WHERE OT.SITEID = @SITEID --'PL01'

	AND   OT.TAGID =  @TAGID --'DCS.Unit15.Global.PK1504_AG18.Inp_SpeedFdbkData'-- @TAGID

	AND   E.EQUIPMENTID = @EQUIPMENTID	

	;



	DECLARE @QUERY_STRING VARCHAR(MAX)

		  , @DATAMAPPINGTABLE VARCHAR(40)

		  , @DATAMAPPINGCOLUMN VARCHAR(100)

	; 



	SELECT @DATAMAPPINGTABLE = DATAMAPPINGTABLE, @DATAMAPPINGCOLUMN = DATAMAPPINGCOLUMN FROM @PROCESSDATAHISTINFO;



	SET @QUERY_STRING = 'SELECT MODIFYTIME AS CREATETIME, ' +  @DATAMAPPINGCOLUMN + ' AS TAGVALUE FROM ' + @DATAMAPPINGTABLE

	                 + ' WHERE MODIFYTIME >= CONVERT(DATETIME, ''' + @FROMDATE + ' 07:00:00'')' + 

 	                 + ' AND MODIFYTIME <  DATEADD(DAY, 1, CONVERT(DATETIME, ''' + @TODATE + ' 07:00:00''))';

 	               

	INSERT INTO @PROCESSDATAVALUE EXEC (@QUERY_STRING)

	

	SELECT DISTINCT

		PI.ROOTEQUIPMENTID

		, PI.EQUIPMENTID

		, PI.EQUIPMENTNAME

		, PI.TAGID

		, PI.TAGNAME

		, PD.TAGVALUE

		, NULL AS DESCRIPTION

		, PD.CREATETIME

		, PI.DCSTAGID

	FROM @PROCESSDATAHISTINFO PI, @PROCESSDATAVALUE PD

	ORDER BY PD.CREATETIME DESC

	;

END
