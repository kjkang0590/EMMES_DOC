WITH BATCH AS
(
	SELECT DISTINCT
		WO.WORKORDERID , 
		WO.WORKORDERNAME ,
		RD.PRODUCTDEFINITIONID ,
		PD.PRODUCTDEFINITIONNAME ,
		BT.BATCHID ,
		BT.COMMENTS,
		CASE WHEN EQ.BATCHID = BT.BATCHID THEN EQ.BATCHID ELSE NULL END AS EQBATCHID,
		BT.STATE,
		BT.BATCHNAME ,
		PER.PROCESSSEGMENTID ,
		PS.PROCESSSEGMENTNAME ,  
		EQ.EQUIPMENTID ,
		EQ.EQUIPMENTNAME,
		EQ.SITEID,
		EQ.PLANTID,
		EQ.LINEID,
		BT.CREATOR,
		(SELECT FN_USERNAME(BT.SITEID, BT.CREATOR) FROM DUAL) AS CREATORNAME,
		BT.CREATETIME,
		BT.MODIFYTIME,
		BT.MODIFIER ,
		(SELECT FN_USERNAME(BT.SITEID, BT.MODIFIER) FROM DUAL) AS MODIFIERNAME,
		BT.PLANDATE
	FROM CIM_EQUIPMENT EQ 
		INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER 
			ON 1=1
			AND PER.EQUIPMENTID = EQ.EQUIPMENTID 
			AND PER.SITEID = EQ.SITEID 
			AND PER.ISUSABLE ='Usable'
		INNER JOIN CIM_PROCESSSEGMENT PS 
			ON 1=1
			AND PS.PROCESSSEGMENTID = PER.PROCESSSEGMENTID 
			AND PS.SITEID = PER.SITEID 
			AND PS.ISUSABLE = 'Usable'
		INNER JOIN CIM_RECIPEDEFINITION RD 
			ON 1=1
			AND RD.EQUIPMENTID = EQ.EQUIPMENTID 
			AND RD.PROCESSSEGMENTID = PER.PROCESSSEGMENTID 
			AND RD.SITEID  = EQ.SITEID 
			AND RD.ISUSABLE  = 'Usable'	
			AND RD.STATE = 'Active' 
		INNER JOIN CIM_PRODUCTDEFINITION PD 
			ON 1=1
			AND PD.PRODUCTDEFINITIONID = RD.PRODUCTDEFINITIONID 
			AND PD.SITEID = RD.SITEID
			AND PD.ISUSABLE = 'Usable'
		INNER JOIN CIM_WORKORDER WO
			ON 1=1
			AND WO.PRODUCTDEFINITIONID = PD.PRODUCTDEFINITIONID 
			AND WO.SITEID = PD.SITEID 
			AND WO.ISUSABLE = 'Usable'
		INNER JOIN CIM_BATCH BT 
			ON 1=1
			AND BT.WORKORDERID = WO.WORKORDERID 
			AND BT.SITEID = WO.SITEID 
			AND BT.ISUSABLE = 'Usable'
		    --AND BT.STATE IN ('Created', 'Active')
	WHERE 1=1
		AND PER.PROCESSSEGMENTID = :PROCESSSEGMENTID
		AND EQ.PLANTID = :PLANTID 
		AND EQ.LINEID = :LINEID
		AND EQ.SITEID = :SITEID
		AND (:EQUIPMENTID IS NULL OR EQ.EQUIPMENTID = :EQUIPMENTID)
		AND (:STATE IS NULL OR BT.STATE = :STATE)
		AND (BT.STATE IN ('Created', 'Active') OR BT.PLANDATE >= TO_DATE(:FROMDATE, 'YYYY-MM-DD'))
		AND (BT.STATE IN ('Created', 'Active') OR BT.PLANDATE < TO_DATE(:TODATE, 'YYYY-MM-DD') + 1)
		AND EQ.ISUSABLE = 'Usable'
),

CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
