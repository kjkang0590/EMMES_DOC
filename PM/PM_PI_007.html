------- 화성타입지정 Bach 목록 조회 쿼리

SELECT DISTINCT
     CB.WORKORDERID ,
     CB.BATCHID ,
     CB.COMMENTS,
     FORMATION.FORMATIONDEFINITIONID ,
     NVL(CPN.PROCESSNODESEQUENCE,0) AS PROCESSNODESEQUENCE ,
     CPN.PROCESSNODEID,
     CPS.PROCESSSEGMENTNAME ,
     FORMATION.PROCESSDEFINITIONID,
     CB.GRADECONTROLTYPE ,
     CB.GRADECONTROLCOUNT
  FROM CIM_BATCH CB
      LEFT JOIN CIM_CODE CC
               ON CC.CODEID = CB.GRADECONTROLTYPE
              AND CC.CODECLASSID = 'GRADECLASSIFYTYPE'
              AND CC.SITEID = CB.SITEID
              AND CC.ISUSABLE = 'Usable'
      LEFT JOIN CIM_PROCESSNODE CPN
             ON 1=1
            AND CPN.PROCESSDEFINITIONID = CB.BATCHID
            AND CPN.SITEID = CB.SITEID
            AND CPN.ISUSABLE = 'Usable'
            AND CPN.PROCESSNODESEQUENCE = 
            (
				SELECT 
					MAX(PD.PROCESSNODESEQUENCE) 
				  FROM CIM_LOTBATCHREL LBR
				 INNER JOIN CIM_PROCESSNODE PD
	   				ON 1=1
	   			   AND PD.PROCESSDEFINITIONID = LBR.BATCHID
	   			   AND PD.SITEID = LBR.SITEID
	   			   AND PD.ISUSABLE = LBR.ISUSABLE
				 INNER JOIN CIM_LOTHIST LH
	   				ON 1=1
	   			   AND LH.LOTID = LBR.LOTID
	   			   AND LH.SITEID = LBR.SITEID
	   			   AND LH.ISUSABLE = LBR.ISUSABLE
	   			   AND NVL(LH.HISTORYFLAG, 'Y') = 'Y'
	   			   AND LH.PROCESSDEFINITIONID = PD.PROCESSDEFINITIONID
	   		       AND LH.PROCESSNODEID = PD.PROCESSNODEID
		         WHERE 1=1
			       AND LBR.BATCHID = CB.BATCHID
				   AND LBR.SITEID = CB.SITEID
				   AND LBR.ISUSABLE = CB.ISUSABLE
			)
      LEFT JOIN CIM_PROCESSSEGMENT CPS
             ON 1=1
            AND CPS.PROCESSSEGMENTID  = CPN.PROCESSSEGMENTID
            AND CPS.SITEID = CPN.SITEID
            AND CPS.ISUSABLE = 'Usable'
      LEFT JOIN (
              SELECT
                  MAX(FORMATIONDEFINITIONID) AS FORMATIONDEFINITIONID ,
                  PROCESSDEFINITIONID
              FROM CIM_PROCESSNODE
              WHERE ISENDNODE = 'Y'
             GROUP BY PROCESSDEFINITIONID
          ) FORMATION
         ON 1=1
        AND FORMATION.PROCESSDEFINITIONID = CB.BATCHID
WHERE 1=1
   AND CB.STATE  IN ('Created' , 'Active' , 'Inspect' )
   AND CB.BATCHTYPE = 'Formation'
   AND CB.ISUSABLE = 'Usable'
   AND CB.SITEID = :SITEID
   AND (:WORKORDERID IS NULL OR UPPER(CB.WORKORDERID) LIKE :WORKORDERID || '%')     
   AND (:BATCHID IS NULL OR UPPER(CB.BATCHID) LIKE :BATCHID || '%')    
ORDER BY CB.BATCHID

-- SelectLineList

SELECT 

	FACILITYID AS CODE, -- LINEID,

	FACILITYNAME AS NAME, -- LINENAME,

	SHORTNAME, SHOP, LINEGROUP

FROM CIM_FACILITY LINE

	WHERE 1=1

	AND UPPER(LINE.FACILITYTYPE) = 'LINE'

	AND LINE.SITEID = :SITEID

	AND LINE.PARENTFACILITYID IN 

	(

		SELECT FACILITYID FROM CIM_FACILITY PLANTLINE

		WHERE 1=1

			AND (:PLANTID IS NULL OR PLANTLINE.PARENTFACILITYID = :PLANTID)

			AND UPPER(PLANTLINE.FACILITYTYPE) = 'PLANTLINE'

			AND PLANTLINE.SITEID = :SITEID

	)

	AND LINE.ISUSABLE = 'Usable'

	AND (:SHOP IS NULL OR LINE.SHOP IN (SELECT COLUMN_VALUE FROM FN_MES_SPLIT(:SHOP, ',')))

ORDER BY UISEQUENCE
테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

-- SelectProcessSegmentList

SELECT DISTINCT EQ.SITEID, EQ.PLANTID, EQ.LINEID
	, PC.PROCESSSEGMENTID AS CODE
	, PC.PROCESSSEGMENTNAME AS NAME
	, PC.PROCESSTYPE
	, PC.BIZRULEID
	, CASE WHEN (SELECT COUNT(*) FROM CIM_EQUIPMENT SUB WHERE SUB.PARENTID = EQ.EQUIPMENTID) > 0 THEN 'Y' ELSE 'N' END ISSUBEQP
FROM CIM_EQUIPMENT EQ
	INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER
	ON 1=1
	AND PER.EQUIPMENTID = EQ.EQUIPMENTID
	AND PER.SITEID = EQ.SITEID
	AND PER.ISUSABLE = 'Usable'
	INNER JOIN CIM_PROCESSSEGMENT PC
	ON 1=1
	AND PC.PROCESSSEGMENTID = PER.PROCESSSEGMENTID
	AND PC.SITEID = PER.SITEID
	AND PC.ISUSABLE = 'Usable'
WHERE 1=1
	AND (:PLANTID IS NULL OR EQ.PLANTID = :PLANTID)
	AND (:LINEID IS NULL OR  EQ.LINEID = :LINEID)
ORDER BY EQ.PLANTID, EQ.LINEID, PC.PROCESSSEGMENTID

테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

SELECT DISTINCT
     CB.WORKORDERID ,
     CB.BATCHID ,
     CB.COMMENTS,
     FORMATION.FORMATIONDEFINITIONID ,
     NVL(CPN.PROCESSNODESEQUENCE,0) AS PROCESSNODESEQUENCE ,
     CPN.PROCESSNODEID,
     CPS.PROCESSSEGMENTNAME ,
     FORMATION.PROCESSDEFINITIONID,
     CB.GRADECONTROLTYPE ,
     CB.GRADECONTROLCOUNT
  FROM CIM_BATCH CB
      LEFT JOIN CIM_CODE CC
               ON CC.CODEID = CB.GRADECONTROLTYPE
              AND CC.CODECLASSID = 'GRADECLASSIFYTYPE'
              AND CC.SITEID = CB.SITEID
              AND CC.ISUSABLE = 'Usable'
      LEFT JOIN CIM_PROCESSNODE CPN
             ON 1=1
            AND CPN.PROCESSDEFINITIONID = CB.BATCHID
            AND CPN.SITEID = CB.SITEID
            AND CPN.ISUSABLE = 'Usable'
            AND CPN.PROCESSNODESEQUENCE = 
            (
				SELECT 
					MAX(PD.PROCESSNODESEQUENCE) 
				  FROM CIM_LOTBATCHREL LBR
				 INNER JOIN CIM_PROCESSNODE PD
	   				ON 1=1
	   			   AND PD.PROCESSDEFINITIONID = LBR.BATCHID
	   			   AND PD.SITEID = LBR.SITEID
	   			   AND PD.ISUSABLE = LBR.ISUSABLE
				 INNER JOIN CIM_LOTHIST LH
	   				ON 1=1
	   			   AND LH.LOTID = LBR.LOTID
	   			   AND LH.SITEID = LBR.SITEID
	   			   AND LH.ISUSABLE = LBR.ISUSABLE
	   			   AND NVL(LH.HISTORYFLAG, 'Y') = 'Y'
	   			   AND LH.PROCESSDEFINITIONID = PD.PROCESSDEFINITIONID
	   		       AND LH.PROCESSNODEID = PD.PROCESSNODEID
		         WHERE 1=1
			       AND LBR.BATCHID = CB.BATCHID
				   AND LBR.SITEID = CB.SITEID
				   AND LBR.ISUSABLE = CB.ISUSABLE
			)
      LEFT JOIN CIM_PROCESSSEGMENT CPS
             ON 1=1
            AND CPS.PROCESSSEGMENTID  = CPN.PROCESSSEGMENTID
            AND CPS.SITEID = CPN.SITEID
            AND CPS.ISUSABLE = 'Usable'
      LEFT JOIN (
              SELECT
                  MAX(FORMATIONDEFINITIONID) AS FORMATIONDEFINITIONID ,
                  PROCESSDEFINITIONID
              FROM CIM_PROCESSNODE
              WHERE ISENDNODE = 'Y'
             GROUP BY PROCESSDEFINITIONID
          ) FORMATION
         ON 1=1
        AND FORMATION.PROCESSDEFINITIONID = CB.BATCHID
WHERE 1=1
   AND CB.STATE  IN ('Created' , 'Active' , 'Inspect' )
   AND CB.BATCHTYPE = 'Formation'
   AND CB.ISUSABLE = 'Usable'
   AND CB.SITEID = :SITEID
   AND (:WORKORDERID IS NULL OR UPPER(CB.WORKORDERID) LIKE :WORKORDERID || '%')     
   AND (:BATCHID IS NULL OR UPPER(CB.BATCHID) LIKE :BATCHID || '%')    
ORDER BY CB.BATCHID
