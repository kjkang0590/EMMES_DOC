CREATE PROCEDURE dbo.SP_PM_PB_004_SELECTPRODUCTDAILYREPORT

	@STARTDATE VARCHAR(8), @ENDDATE VARCHAR(8)

AS	

BEGIN

	SET @ENDDATE = (SELECT CASE WHEN CONVERT(DATE, REPLACE(GETDATE(), '-', '')) < CONVERT(DATE, @ENDDATE) THEN REPLACE(CONVERT(DATE, GETDATE()), '-', '')

       	    					ELSE @ENDDATE END); --AS ENDDATE

	

	WITH BASE_DATE AS

	(

	    ------------------------------------

	    -- 일자

	    ------------------------------------

	    SELECT CONVERT(VARCHAR(8), DATEADD(DAY, NUMBER-1, @STARTDATE), 112) AS WORKDATE

	    FROM MASTER..SPT_VALUES

	    WHERE TYPE = 'P'

	    AND   NUMBER BETWEEN 1 AND DATEDIFF(D, @STARTDATE, @ENDDATE)+1

	)

	, BP_TB_IN_TBL AS

	(

	    ------------------------------------

	    -- BP : BP 투입량(라벨기준)

	    ------------------------------------

	   /*

	    SELECT CONVERT(VARCHAR(8), DATEADD(HOUR, -7, ML.KITTINGTIME), 112) AS WORKDATE

	         , SUM(ML.ORIGINALQTY - ML.QTY ) AS BP_TB_QTY -- BP 톤백 QTY

	    FROM CST_MATERIALLOT ML WITH(NOLOCK)

	    WHERE ML.KITTINGTIME  BETWEEN CONVERT(DATETIME, @STARTDATE + ' 07:00:00') AND DATEADD(DAY, 1, CONVERT(DATETIME, @ENDDATE + ' 07:00:00'))

	    AND   ML.ISKITTING = 'Y'

	    AND   ML.STATE <> 'Transfer'

	    AND   ML.ORIGINALQTY <> ML.QTY

	    GROUP BY CONVERT(VARCHAR(8), DATEADD(HOUR, -7, ML.KITTINGTIME), 112)

	   */

        SELECT CONVERT(VARCHAR(8), DATEADD(HOUR, -7, ML.KITTINGTIME), 112) AS WORKDATE

             --, SUM(ML.ORIGINALQTY - ML.QTY ) AS BP_TB_QTY -- BP 톤백 QTY

            , SUM( CASE WHEN ML.STATE = 'Active' AND ML.ORIGINALQTY = ML.QTY THEN ML.QTY

                   WHEN ML.STATE = 'Active' AND ML.ORIGINALQTY > ML.QTY THEN ML.ORIGINALQTY - ML.QTY

                   WHEN ML.STATE = 'Consumed' THEN ML.ORIGINALQTY

                   ELSE 0

              END ) AS BP_TB_QTY        

        FROM CST_MATERIALLOT ML WITH(NOLOCK)

        WHERE ML.KITTINGTIME  BETWEEN CONVERT(DATETIME, @STARTDATE + ' 07:00:00') AND DATEADD(DAY, 1, CONVERT(DATETIME, @ENDDATE + ' 07:00:00'))

        AND   ML.ISKITTING = 'Y'

        AND   ML.STATE <> 'Transfer'

        GROUP BY CONVERT(VARCHAR(8), DATEADD(HOUR, -7, ML.KITTINGTIME), 112)

	)

	, KILN_IN_OUT_TBL AS

	(

	    ------------------------------------

	    -- BP : BP 투입 후 실무게 / kiln 총 투입량 / KILN 후 무게

	    ------------------------------------

	    SELECT M.WORKDATE

	         , SUM(BP_IN_SILO) AS BP_IN_SILO -- 투입후 실무게

	         , SUM(BP_IN_KILN) AS BP_IN_KILN -- KILN 투입량

	         , SUM(BP_OUT_KILN) AS BP_OUT_KILN -- KILN(소성) 후 무게

	    FROM

	    (

	        SELECT PD.WORKDATE

	             --, CASE WHEN PD.TAGID = 'DCS.Unit14.Global.PLC_PK1102[10]' THEN PD.EOH ELSE 0 END AS BP_IN_SILO  -- 투입후 실무게

	             , CASE WHEN PD.TAGID = 'DCS.Unit14.Global.PLC_PK1102[10]' THEN PD.INPUTVALUE ELSE 0 END AS BP_IN_SILO  -- 투입후 실무게

	             , CASE WHEN PD.TAGID = 'DCS.Unit14.Global.PLC_PK1102[10]' THEN PD.OUTPUTVALUE ELSE 0 END AS BP_IN_KILN -- KILN 투입량

	             , CASE WHEN PD.TAGID IN ( 'DCS.Unit14.Global.PLC_PK1102[11]', 'DCS.Unit14.Global.PLC_PK1102[12]') THEN PD.INPUTVALUE ELSE 0 END AS BP_OUT_KILN -- 소성 후 총 무게

	        FROM CST_RPT_TANKPRODUCE_DAY PD WITH(NOLOCK)

	        WHERE EQUIPMENTID = 'PK-1102'

	        AND   WORKDATE BETWEEN @STARTDATE AND @ENDDATE

	        AND   TAGID IN ( 'DCS.Unit14.Global.PLC_PK1102[10]', 'DCS.Unit14.Global.PLC_PK1102[11]','DCS.Unit14.Global.PLC_PK1102[12]')

	    ) M

	    GROUP BY M.WORKDATE

	)

	, PRIMIX_IN_TBL AS

	(

	    ------------------------------------

	    -- V-1401 투입량 -- 반응 공정 투입 정보로 변경 (진행전)

	    ------------------------------------

	   /*

	    SELECT PM.WORKDATE

	         , SUM(PM.INPUTVALUE) AS PRIMX_IN_QTY

	    FROM CST_RPT_TANKPRODUCE_DAY PM WITH(NOLOCK)

	    WHERE EQUIPMENTID IN ('V-1401-1', 'V-1401-2')

	    AND   WORKDATE BETWEEN @STARTDATE AND @ENDDATE

	    GROUP BY PM.WORKDATE

	   */

    	SELECT 

            CONVERT(VARCHAR(8), DATEADD(HOUR, -7, L.TRACKINTIME), 112) AS WORKDATE

            , SUM( L.MATERIALQTY ) AS PRIMX_IN_QTY

        FROM CIM_LOT L

        WHERE L.TRACKINTIME BETWEEN  CONVERT(DATETIME, @STARTDATE + ' 07:00:00') AND DATEADD(DAY, 1, CONVERT(DATETIME, @ENDDATE + ' 07:00:00'))

        AND   L.PROCESSSEGMENTID = 'P2020'

        GROUP BY CONVERT(VARCHAR(8), DATEADD(HOUR, -7, L.TRACKINTIME), 112)

	)

	, PROD_QTY_TBL AS

	(

	    ------------------------------------

	    -- 생산량

	    ------------------------------------

	    SELECT PV.WORKDATE

	         , ISNULL(PV.LI, 0) AS LI

	         , ISNULL(PV.CO, 0) AS CO

	         , ISNULL(PV.NI, 0) AS NI

	    FROM

	    (

	        SELECT CONVERT(VARCHAR(8), DATEADD(HOUR, -7, L.LOTFINISHEDTIME), 112) AS WORKDATE

	             , CASE WHEN L.PROCESSSEGMENTID = 'P2060' THEN 'LI'

	                    WHEN L.PROCESSSEGMENTID = 'P2070' THEN 'CO'

	                    WHEN L.PROCESSSEGMENTID = 'P2080' THEN 'NI'

	               END AS PRODUCTTYPE

	             , SUM( L.QTY ) AS QTY

	        FROM CIM_LOT L WITH(NOLOCK)

	        INNER JOIN CIM_PRODUCTDEFINITION PD WITH(NOLOCK)

	        ON  L.PRODUCTDEFINITIONID = PD.PRODUCTDEFINITIONID

	        AND L.SITEID = PD.SITEID

	        AND PD.PRODUCTCLASSID IN ( 'METAL', 'LIQUID', 'POWDER')

	        WHERE L.STATE = 'Shipped'   

	        AND   L.LOTFINISHEDTIME BETWEEN CONVERT(DATETIME, @STARTDATE + ' 07:00:00') AND DATEADD(DAY, 1, CONVERT(DATETIME, @ENDDATE + ' 07:00:00'))

	        GROUP BY CONVERT(VARCHAR(8), DATEADD(HOUR, -7, L.LOTFINISHEDTIME), 112), L.PROCESSSEGMENTID

	    ) P

	    PIVOT

	    (

	        SUM(QTY) FOR PRODUCTTYPE IN ( LI, CO, NI )

	    ) PV

	)

	SELECT BD.WORKDATE

	     , ISNULL(BT.BP_TB_QTY, 0) AS BP_TB_QTY

	     , ISNULL(KT.BP_IN_SILO, 0) AS BP_IN_SILO

	     , ISNULL(KT.BP_IN_KILN, 0) AS BP_IN_KILN

	     , ISNULL(KT.BP_OUT_KILN, 0) AS BP_OUT_KILN

	     , ISNULL(PT.PRIMX_IN_QTY, 0) AS PRIMX_IN_QTY

	     , ISNULL(PRDT.LI, 0) AS LI

	     , ISNULL(PRDT.CO, 0) AS CO

	     , ISNULL(PRDT.NI, 0) AS NI

	FROM BASE_DATE BD

	LEFT OUTER JOIN BP_TB_IN_TBL BT

	  ON BD.WORKDATE = BT.WORKDATE

	LEFT OUTER JOIN KILN_IN_OUT_TBL KT

	  ON BD.WORKDATE = KT.WORKDATE

	LEFT OUTER JOIN PRIMIX_IN_TBL PT

	  ON BD.WORKDATE = PT.WORKDATE

	LEFT OUTER JOIN PROD_QTY_TBL PRDT

	  ON BD.WORKDATE = PRDT.WORKDATE

	;

END

CREATE PROCEDURE dbo.SP_PM_PB_004_SELECTPRODUCTDAILYREPORT2

	@STARTDATE VARCHAR(8), @ENDDATE VARCHAR(8)

AS	

BEGIN

	SET @ENDDATE = (SELECT CASE WHEN REPLACE(CONVERT(VARCHAR, DATEADD(HOUR, -7, GETDATE()), 112), '-', '') < CONVERT(DATE, @ENDDATE) 

								THEN REPLACE(CONVERT(VARCHAR, DATEADD(HOUR, -7, GETDATE()), 112), '-', '')

								ELSE @ENDDATE END)

	

    --[ 생산일보 - 하단/좌측 그리드 ] 	    					

    SELECT 

        CASE WHEN M.SILONAME = 'A_RAW_BP_SILO' THEN 'Raw B.P Silo'

             WHEN M.SILONAME = 'B_A_BP_SILO' THEN 'Calcined B.P Silo A'

             WHEN M.SILONAME = 'C_B_BP_SILO' THEN 'Calcined B.P Silo B'

             ELSE ''

        END AS SILONAME

        , ISNULL(QTY, 0) AS QTY

    FROM 

    (

        SELECT 'A_RAW_BP_SILO' AS SILONAME UNION

        SELECT 'B_A_BP_SILO' AS SILONAME UNION

        SELECT 'C_B_BP_SILO' AS SILONAME 

    ) M

    LEFT OUTER JOIN 

    (

        SELECT

            SILONAME

            , ROW_NUMBER() OVER(ORDER BY SILONAME ASC) AS ROW_NO

            , QTY

        FROM

        (

            SELECT

                SUM(CASE WHEN PD.TAGID = 'DCS.Unit14.Global.PLC_PK1102[10]' THEN PD.EOH ELSE 0 END) AS A_RAW_BP_SILO  -- 투입후 실무게

                , SUM(CASE WHEN PD.TAGID = 'DCS.Unit14.Global.PLC_PK1102[11]' THEN PD.EOH ELSE 0 END) AS B_A_BP_SILO -- KILN 투입량

                , SUM(CASE WHEN PD.TAGID = 'DCS.Unit14.Global.PLC_PK1102[12]' THEN PD.EOH ELSE 0 END) AS C_B_BP_SILO -- 소성 후 총 무게

            FROM CST_RPT_TANKPRODUCE_DAY PD WITH(NOLOCK)

            WHERE EQUIPMENTID = 'PK-1102'

            AND   WORKDATE = ( CASE WHEN @ENDDATE > CONVERT(VARCHAR(8), DATEADD(HOUR, -7, GETDATE()) , 112)

                                        THEN CONVERT(VARCHAR(8), DATEADD(HOUR, -7, GETDATE()) , 112)

                                    ELSE @ENDDATE END )

            AND   TAGID IN ( 'DCS.Unit14.Global.PLC_PK1102[10]', 'DCS.Unit14.Global.PLC_PK1102[11]','DCS.Unit14.Global.PLC_PK1102[12]')

            GROUP BY PD.WORKDATE

        ) M

        UNPIVOT

        (

            QTY FOR SILONAME IN ( [A_RAW_BP_SILO], [B_A_BP_SILO], [C_B_BP_SILO] )

        ) A

    ) P

    ON M.SILONAME = P.SILONAME

	@STARTDATE VARCHAR(8), @ENDDATE VARCHAR(8)

AS	

BEGIN

	SET @ENDDATE = (SELECT CASE WHEN REPLACE(CONVERT(VARCHAR, DATEADD(HOUR, -7, GETDATE()), 112), '-', '') < CONVERT(DATE, @ENDDATE) 

								THEN REPLACE(CONVERT(VARCHAR, DATEADD(HOUR, -7, GETDATE()), 112), '-', '')

								ELSE @ENDDATE END)

	

	--+++++++++++++++++++++++++++++++++++++++++++++++++++++++

	-- 생산일보 하단 그리드 우측 부자재 QUERY

	--+++++++++++++++++++++++++++++++++++++++++++++++++++++++

	------------------------------------

	-- Chemical 

	-- H2SO4       4A0101  황산                 V-1706A SULFURIC ACID TANK (V-1706A LEVEL)                  [ DCS.Unit17_9x.Global.LI1006A.Val ]

	--                                         V-1706B SULFURIC ACID TANK (V-1706B LEVEL)                  [ DCS.Unit17_9x.Global.LI1006B.Val ]

	-- additive                                V-1426  ADDITIVE TANK (LT-141326 LT-141326  V-1426 LEVEL)   [ DCS.Unit14.Global.CCW_02_Data.HoldRegisters_4xxx[46] ]

	-- Na2CO3      4A0501  탄산나트륨, 소다회 (Kg)  PK-1406 POWDER HANDLING SYSTEM FOR SODIUM CARBONATE (PK-1406-H001 SODIUM CARBONATE SILO)    [ DCS.Unit14.Global.CCW_02_Data.HoldRegisters_4xxx[46] ]

	-- Na3PO4      5A0601  제삼인산나트륨 (Kg)     PK-1407 POWDER HANDLING SYSTEM FOR TRISODIUM PHOSPHATE (PK-1407-V02 FEEDHOPPER 1403)        [ DCS.Unit14.Global.PLC_PK140678[16] ]

	-- NaOH        4A0201  가성소다              V-1701A LIQUID CAUSTIC SODA STORAGE TANK (V-1701A LEVEL)    [ DCS.Unit17_9x.Global.LI1001A.Val ]

	--                                         V-1701B LIQUID CAUSTIC SODA STORAGE TANK (V-1701B LEVEL)    [ DCS.Unit17_9x.Global.LI1001B.Val ]

	-- H2O2        4A0401  과산화수소             V-1704  HYDROGEN PEROXIDE TANK (V-1704 LEVEL)               [ DCS.Unit17_9x.Global.LI1004.Val ]

	-- HCI(20%)    4A0301  염산                 V-1703  20wt% (6.2N) HYDROCHLORIC ACID TANK  (V-1703 LEVEL) [ DCS.Unit17_9x.Global.LI1003.Val ]

	------------------------------------

	SELECT 

	    M.CHEMICAL_NO

	    , SUM(M.INPUTQTY) AS INPUTQTY

	    , SUM(M.OUTPUTQTY) AS OUTPUTQTY

	    , SUM(M.EOH) AS EOH

	FROM 

	(

	    SELECT 

	        CASE WHEN TD.EQUIPMENTID IN ('V-1706A', 'V-1706B') THEN 'H2SO4'

	             WHEN TD.EQUIPMENTID = 'V-1426' THEN 'Additive'

	             WHEN TD.EQUIPMENTID = 'PK-1406' THEN 'Na2CO3'

	             WHEN TD.EQUIPMENTID = 'PK-1407' THEN 'Na3PO4'

	             WHEN TD.EQUIPMENTID IN ('V-1701A', 'V-1701B') THEN 'NaOH'

	             WHEN TD.EQUIPMENTID = 'V-1704' THEN 'H2O2'

	             WHEN TD.EQUIPMENTID = 'V-1703' THEN 'HCI(20%)'

	             ELSE TD.EQUIPMENTID

	         END AS CHEMICAL_NO

	        , SUM( CASE WHEN TD.EQUIPMENTID LIKE 'PK%' THEN TD.INPUTVALUE 

	                    WHEN TD.EQUIPMENTID LIKE 'V%' THEN TD.INPUT_M3

	               ELSE 0

	          END ) AS INPUTQTY  -- 입고        

	        , SUM( CASE WHEN TD.EQUIPMENTID LIKE 'PK%' THEN TD.OUTPUTVALUE 

	               WHEN TD.EQUIPMENTID LIKE 'V%' THEN TD.OUTPUT_M3

	               ELSE 0

	          END ) AS OUTPUTQTY  -- 사용        

	        , SUM( CASE WHEN TD.WORKDATE = @ENDDATE AND TD.EQUIPMENTID LIKE 'PK%' THEN TD.EOH 

	               WHEN TD.WORKDATE = @ENDDATE AND TD.EQUIPMENTID LIKE 'V%' THEN TD.EOH_M3

	               ELSE 0

	          END ) AS EOH  -- 재공량    

	    FROM CST_RPT_TANKPRODUCE_DAY TD

	        INNER JOIN CIM_EQUIPMENT E 

	            ON  TD.EQUIPMENTID = E.EQUIPMENTID 

	            AND TD.SITEID = E.SITEID 

	            AND E.TANKTYPE IN ( 'MATERIAL', 'PROD' )

	    WHERE TD.EQUIPMENTID IN (

	        'V-1706A', 'V-1706B', 'V-1426', 

	        'V-1701A', 'V-1701B', 'V-1704', 'V-1703',

	        'PK-1406', 'PK-1407' )

	    AND  TD.WORKDATE BETWEEN @STARTDATE AND @ENDDATE

	    GROUP BY TD.EQUIPMENTID

	) M

	GROUP BY CHEMICAL_NO

	;

END
