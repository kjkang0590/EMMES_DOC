마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_PRODUCTDEFINITION, CIM_FACILITY, CIM_STATE

--PM_CS_001_제품 입고 대기 현황 - 제품 목록
--SelectLocationbyProductList
--Param : SITEID, LINEID, FROMDATE, TODATE, LOCATION, PRODUCTDEFINITIONID, LOTID
------------------------------------------
SELECT M.FACILITYID
	 , M.FACILITYNAME
	 , M.LOTID
	 , M.PRODUCTDEFINITIONID
	 , M.PRODUCTDEFINITIONNAME
	 , M.QTY
	 , M.LOTSHIPPEDTIME 
	 , M.INSPREQNO
	 , M.STATE
	 , M.INSPECTIONSTATE
	 , M.FINALJUDGE
	 , M.INSPRECEIPTTIME
	 , M.FINALDATETIME
  FROM (
        SELECT CF.FACILITYID
             , CF.FACILITYNAME
             , CL.LOTID
             , CL.PRODUCTDEFINITIONID
             , CP.PRODUCTDEFINITIONNAME
             , CL.RESULTQTY AS QTY
             , CL.LOTSHIPPEDTIME 
             , IR.INSPREQNO
             , IR.STATE
             , CS.STATENAME AS INSPECTIONSTATE
             , IR.FINALJUDGE
             , IR.INSPRECEIPTTIME
             , IR.FINALDATETIME 
          FROM CIM_LOT CL
         INNER JOIN CIM_PRODUCTDEFINITION CP ON CP.SITEID = CL.SITEID AND CP.PRODUCTDEFINITIONID = CL.PRODUCTDEFINITIONID AND CP.PRODUCTTYPE = 'FERT'
          LEFT OUTER JOIN CIM_INSPREQ IR ON IR.SITEID = CP.SITEID AND IR.LOTID = CL.LOTID AND IR.INSPTYPE = 'OQC'
          LEFT OUTER JOIN CIM_FACILITY CF ON CF.SITEID = CL.SITEID AND CF.FACILITYID = CL.LOCATION
          LEFT OUTER JOIN CIM_STATE CS ON CS.SITEID = IR.SITEID AND CS.STATEID = IR.STATE AND CS.STATEMODELID = 'InspReqState'
         WHERE CL.SITEID = @SITEID
           AND CL.LINEID = @LINEID
           AND CL.CREATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
           AND (@LOCATION = '' OR CL.LOCATION = @LOCATION)
           AND (@PRODUCTDEFINITIONID = '' OR CL.PRODUCTDEFINITIONID = @PRODUCTDEFINITIONID)
           AND (@PACKLOTID = '' OR CL.LOTID LIKE '%' + @PACKLOTID + '%')
       ) M
   WHERE (@STATE = '' OR M.STATE = @STATE)
