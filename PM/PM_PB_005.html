--PM_PB_005_부산물 등록
--SelectByProductList
--Param : SITEID, LINEID, FROMDATE, TODATE, PRODUCTCLASSID, PRODUCTDEFINITIONID
------------------------------------------
SELECT CL.LOTID
     , CP.PRODUCTCLASSID
     , CC.PRODUCTCLASSNAME
	 , CL.PRODUCTDEFINITIONID
	 , CP.PRODUCTDEFINITIONNAME
          , CL.EQUIPMENTID
	 , CP.SHORTNAME
	 , CL.QTY
	 , CL.UNITID
	 , CL.CREATETIME
	 , CCU.USERNAME AS CREATOR
	 , CL.COMMENTS
          , CCL.PRINTCOUNT
  FROM CIM_LOT CL WITH(NOLOCK)
 INNER JOIN CIM_PRODUCTDEFINITION CP WITH(NOLOCK)
    ON CP.SITEID = CL.SITEID
   AND CP.PRODUCTDEFINITIONID = CL.PRODUCTDEFINITIONID
   AND CP.SHORTNAME = 'Waste'
   AND CP.PRODUCTTYPE = 'FERT'
  LEFT OUTER JOIN CIM_PRODUCTCLASS CC WITH(NOLOCK) ON CC.SITEID = CP.SITEID AND CC.PRODUCTCLASSID = CP.PRODUCTCLASSID
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK) ON CCU.SITEID = CL.SITEID AND CCU.USERID = CL.CREATOR
  LEFT OUTER JOIN CST_PRODUCTLABEL CCL WITH(NOLOCK) ON CCL.SITEID = CL.SITEID AND CCL.LOTID = CL.LOTID
 WHERE CL.SITEID = @SITEID
   AND CL.LINEID = @LINEID
   AND CL.CREATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@PRODUCTCLASSID = '' OR CP.PRODUCTCLASSID = @PRODUCTCLASSID)
   AND (@PRODUCTDEFINITIONID = '' OR CL.PRODUCTDEFINITIONID = @PRODUCTDEFINITIONID)
