--제품입고이력 



SELECT 

	TO_DATE(MAT.PALLETSTOREDDATE , 'YYYY-MM-DD HH24:MI:SS') AS STOREDATE  

	, CASE

		WHEN MAT.CUSTOMACTIVITY = 'CreatePalletLotIn' THEN '입고'

		WHEN MAT.CUSTOMACTIVITY = 'DeletePalletLotIn' THEN '입고취소'

	END AS GUBUN	

	, MAT.PREVLOCATION AS FROMLOCATION

	, (	SELECT FAC.FACILITYNAME 

	  	FROM CIM_FACILITY FAC 

		WHERE MAT.SITEID = FAC.SITEID 

			AND FAC.FACILITYID = MAT.PREVLOCATION ) AS FROMLOCATIONNAME --창고이름		

	, MAT.LOCATION AS TOLOCATION

	, (	SELECT FAC.FACILITYNAME 

	  	FROM CIM_FACILITY FAC 

		WHERE MAT.SITEID = FAC.SITEID 

			AND FAC.FACILITYID = MAT.LOCATION ) AS TOLOCATIONNAME --창고이름	

	, MAT.PRODUCTORDERID

	, DEF.PRODUCTDEFINITIONID

	, DEF.PRODUCTDEFINITIONNAME

	, MAT.PALLETID 

	, MAT.QTY 

	, MAT.UNITID 

FROM 

(

	SELECT

		*

	FROM 

	(

		SELECT

			ROW_NUMBER() OVER (PARTITION BY MAT.MATERIALLOTID ORDER BY MAT.LASTEVENTSEQ DESC) AS RNK, MAT.*, PAL.PALLETSTOREDDATE 

		FROM CIM_MATERIALLOTHIST MAT

		LEFT OUTER JOIN CUS_PALLETSTOREDHIST PAL

			ON MAT.PALLETID = PAL.PALLETID 

		WHERE 1=1

			AND MAT.ISUSABLE = 'Usable'

			AND MAT.SITEID = :SITEID 

			AND PAL.PALLETSTOREDDATE >= TO_DATE(:FROMDATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') AND PAL.PALLETSTOREDDATE < TO_DATE(:TODATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') + 1

			AND MAT.CUSTOMACTIVITY IN ('CreatePalletLotIn', 'DeletePalletLotIn')

			AND (:LOCATION IS NULL OR MAT.LOCATION = :LOCATION )

	) 

	WHERE RNK = 1

) MAT

LEFT OUTER JOIN CIM_PRODUCTORDER OD

	ON MAT.SITEID = OD.SITEID 

	AND MAT.PRODUCTORDERID = OD.PRODUCTORDERID 

LEFT OUTER JOIN CIM_PRODUCTDEFINITION DEF

	ON MAT.SITEID = DEF.SITEID 

	AND OD.PRODUCTDEFINITIONID = DEF.PRODUCTDEFINITIONID 

WHERE 1=1

	AND (DEF.PRODUCTDEFINITIONID LIKE '%'|| :PRODUCTDEFINITIONID ||'%' OR 

	DEF.PRODUCTDEFINITIONNAME LIKE '%'|| :PRODUCTDEFINITIONNAME ||'%')

ORDER BY MAT.PALLETSTOREDDATE

--FERT Location List

SELECT

	cf.FACILITYID AS CODE

	, cf.FACILITYNAME AS NAME

FROM CIM_FACILITY cf 

WHERE 1=1

	AND cf.SITEID = :SITEID 

	AND cf.ISUSABLE = 'Usable'

	AND cf.FACILITYCLASSID = 'WAREHOUSE'

	AND cf.FACILITYTYPE IN ('FERT')

ORDER BY cf.UISEQUENCE
