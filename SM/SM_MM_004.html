마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_MATERIALCLASS, CIM_MATERIALDEFINITION, CIM_UNIT
SQL 보기
--자재 입고 이력 목록
--
SELECT
	TO_CHAR(MAT.RECEIVEDTIME, 'YYYY-MM-DD HH24:MI:SS') AS STOREDATE --입고일
	, CASE
		WHEN MAT.CUSTOMACTIVITY = 'CreateMaterialin' THEN '입고'
		WHEN MAT.CUSTOMACTIVITY = 'DeleteMaterialin' THEN '입고취소'
	END AS GUBUN
	, MAT.PREVLOCATION AS FROMLOCATION --참고ID
	, FLOC.FACILITYNAME AS FROMLOCATIONNAME --창고이름
	, MAT.LOCATION AS TOLOCATION --참고ID
	, FLOC.FACILITYNAME AS TOLOCATIONNAME --창고이름			
	, CLS.MATERIALCLASSID --자재유형구분
	, CLS.MATERIALCLASSNAME --자재유형이름
	, DEF.MATERIALDEFINITIONID --자재 ID
	, DEF.MATERIALDEFINITIONNAME --자재이름
	, MAT.MATERIALLOTID --자재 LOT ID
	, MAT.ORIGINALQTY AS QTY -- 입고수량
	, MAT.UNITID -- 단위 ID
	, UNT.UNITNAME -- 단위이름
FROM CIM_MATERIALLOTHIST MAT
INNER JOIN CIM_FACILITY FLOC
	ON FLOC.FACILITYID  = MAT.PREVLOCATION 
	AND FLOC.SITEID = MAT.SITEID 
	AND FLOC.ISUSABLE = 'Usable'
INNER JOIN CIM_FACILITY TLOC
	ON TLOC.FACILITYID  = MAT.LOCATION 
	AND TLOC.SITEID = MAT.SITEID 
	AND TLOC.ISUSABLE = 'Usable'
INNER JOIN CIM_MATERIALCLASS CLS 
	ON CLS.MATERIALCLASSID = MAT.MATERIALCLASSID 
	AND CLS.SITEID = MAT.SITEID 
	AND CLS.ISUSABLE = 'Usable'
INNER JOIN CIM_MATERIALDEFINITION DEF 
	ON DEF.MATERIALDEFINITIONID = MAT.MATERIALDEFINITIONID 
	AND DEF.MATERIALCLASSID = MAT.MATERIALCLASSID 
	AND DEF.SITEID = MAT.SITEID 
	AND DEF.MATERIALDEFINITIONID LIKE '%' || :MATERIALDEFINITIONID || '%'
	AND DEF.MATERIALDEFINITIONNAME LIKE '%' || :MATERIALDEFINITIONNAME || '%'
	AND DEF.ISUSABLE = 'Usable'
LEFT JOIN CIM_UNIT UNT 
	ON UNT.UNITID = MAT.UNITID 
	AND UNT.SITEID = MAT.SITEID 
	AND UNT.ISUSABLE = 'Usable'
WHERE ( TO_CHAR(MAT.RECEIVEDTIME, 'YYYY-MM-DD') >= :FROMDATE AND TO_CHAR( MAT.RECEIVEDTIME, 'YYYY-MM-DD') <= :TODATE )
	AND (MAT.CUSTOMACTIVITY = 'CreateMaterialin' OR MAT.CUSTOMACTIVITY = 'DeleteMaterialin')
	AND MAT.ISUSABLE = 'Usable'
	AND MAT.STATE = 'Active'
	AND MAT.SITEID = :SITEID
	AND (:MATERIALCLASSID IS NULL OR MAT.MATERIALCLASSID = :MATERIALCLASSID )
	AND (:MATERIALTYPE IS NULL OR MAT.MATERIALTYPE = :MATERIALTYPE )
	AND (:LOCATION IS NULL OR MAT.LOCATION = :LOCATION )
ORDER BY MAT.MATERIALLOTID, MAT.RECEIVEDTIME
