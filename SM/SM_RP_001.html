#1 Query SelectRohMaterial

연결된 기능 (1번 표): #1 조회 - 그리드 SM_RP_0011 (자재재고현황1_자재재고현황)

  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)

    CIM_MATERIALLOT메인읽기   CIM_MATERIALDEFINITION읽기   CIM_MATERIALCLASS읽기   CIM_FACILITY읽기   CIM_CODE읽기
조인 구조도 테이블 3 · 조인 1
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_MATERIALDEFINITION, CIM_CODE
SQL 보기
--- 자재 재고 현황 조회

SELECT 

    CMT.LOCATION ,

    CF.FACILITYNAME AS LOCATIONNAME ,

    CMT.MATERIALCLASSID ,

	CMS.MATERIALCLASSNAME , 

	CMDN.PRODUCTTYPE ,

	 CC2.CODENAME AS PRODUCTTYPENAME ,

	CMT.MATERIALDEFINITIONID ,

	CMDN.MATERIALDEFINITIONNAME ,

	SUM(CMT.QTY) AS QTY ,

	CMT.UNITID 

  FROM CIM_MATERIALLOT CMT

 INNER JOIN CIM_MATERIALDEFINITION CMDN

         ON 1=1 

        AND CMDN.MATERIALDEFINITIONID = CMT.MATERIALDEFINITIONID

        AND CMDN.SITEID  = CMT.SITEID 

        AND CMDN.ISUSABLE  = 'Usable'         

 INNER JOIN CIM_MATERIALCLASS CMS

         ON 1=1 

        AND CMS.MATERIALCLASSID  = CMDN.MATERIALCLASSID

        AND CMS.SITEID  = CMDN.SITEID 

        AND CMS.ISUSABLE  = 'Usable' 

  INNER JOIN CIM_FACILITY CF 

         ON 1=1

        AND CF.FACILITYID = CMT.LOCATION

        AND CF.FACILITYCLASSID = 'WAREHOUSE'

        AND CF.FACILITYTYPE = 'ROH'

        AND CF.SITEID = CMT.SITEID 

        AND CF.ISUSABLE = 'Usable'

 INNER JOIN CIM_CODE CC2

         ON 1=1 

        AND CC2.CODEID  = CMDN.PRODUCTTYPE 

        AND CC2.CODECLASSID  = 'PRODUCTTYPE'

        AND CC2.SITEID  = CMDN.SITEID 

        AND CC2.ISUSABLE  = 'Usable'        

 WHERE 1=1

   AND CMT.SITEID = :SITEID 

   AND (:LOCATION IS NULL  OR CMT.LOCATION = :LOCATION)

   AND (:MATERIALCLASSID IS NULL  OR CMT.MATERIALCLASSID = :MATERIALCLASSID)

   AND (:MATERIALDEFINITIONID IS NULL  OR CMT.MATERIALDEFINITIONID = :MATERIALDEFINITIONID)

   AND (:MATERIALDEFINITIONNAME IS NULL  OR CMDN.MATERIALDEFINITIONNAME = :MATERIALDEFINITIONNAME)

 GROUP BY CMT.LOCATION ,  CF.FACILITYNAME , CMT.MATERIALCLASSID , CMS.MATERIALCLASSNAME ,  CMDN.PRODUCTTYPE ,  CC2.CODENAME , CMT.MATERIALDEFINITIONID , CMDN.MATERIALDEFINITIONNAME 

          , CMT.UNITID ,CF.UISEQUENCE ,CC2.UISEQUENCE

 ORDER BY CF.UISEQUENCE , CC2.UISEQUENCE DESC , CMT.MATERIALDEFINITIONID
