SQL 보기
--- D/O 계획 대비 실적 현황 조회

SELECT 

	TO_CHAR(CDO.DELIVERYPLANDATE , 'YYYY-MM-DD') AS DELIVERYPLANDATE,

	CDO.DELIVERYORDERID ,

	CDO.DELIVERYITEMNO,

	CDO.DELIVERYORDERTYPE,

	 CC.CODENAME AS DELIVERYORDERTYPENAME,

	CDO.CUSTOMERID ,

	CCR.CUSTOMERNAME,

	CDO.PRODUCTDEFINITIONID,

	CPD.PRODUCTDEFINITIONNAME ,

	CDO.DELIVERYDATE ,

	CDO.QTY ,

	NVL(CS.SHIPPEDQTY,'0') AS SHIPPEDQTY ,

	CDO.QTY - NVL(CS.SHIPPEDQTY,'0') AS LEFTQTY,

	CDO.UNITID

  FROM CIM_DELIVERYORDER CDO

 INNER JOIN CIM_CODE CC

         ON 1=1

        AND CC.CODEID = CDO.DELIVERYORDERTYPE

        AND CC.CODECLASSID = 'DELIVERYORDERTYPE'

        AND CC.SITEID = CDO.SITEID

        --AND CC.ISUSABLE = 'Usable'

 INNER JOIN CIM_CUSTOMER CCR

         ON 1=1

        AND CCR.CUSTOMERID = CDO.CUSTOMERID

        AND CCR.SITEID = CDO.SITEID

        --AND CCR.ISUSABLE = 'Usable'

 INNER JOIN CIM_PRODUCTDEFINITION CPD

         ON 1=1

        AND CPD.PRODUCTDEFINITIONID = CDO.PRODUCTDEFINITIONID

        AND CPD.SITEID = CDO.SITEID

        --AND CPD.ISUSABLE = 'Usable'

  LEFT JOIN (

  		SELECT 

  			DELIVERYORDERID ,

  			DELIVERYITEMNO,

  			SITEID,

  			SUM(QTY) AS SHIPPEDQTY

  		  FROM CUS_SHIPPED

  		 GROUP BY DELIVERYORDERID , DELIVERYITEMNO,SITEID

        )CS

         ON 1=1

        AND CS.DELIVERYORDERID = CDO.DELIVERYORDERID

        AND CS.DELIVERYITEMNO = CDO.DELIVERYITEMNO

        AND CS.SITEID = CDO.SITEID

        --AND CS.ISUSABLE = 'Usable'                        

 WHERE CDO.STATE <> 'Created'

   AND CDO.SITEID = :SITEID

   AND CDO.DELIVERYPLANDATE BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 1

   AND (:CUSTOMERID IS NULL  OR CDO.CUSTOMERID = :CUSTOMERID)

   AND (:DELIVERYORDERID IS NULL OR UPPER(CDO.DELIVERYORDERID) LIKE UPPER(:DELIVERYORDERID) || '%')

   AND (:PRODUCTDEFINITIONID IS NULL OR UPPER(CDO.PRODUCTDEFINITIONID) LIKE UPPER(:PRODUCTDEFINITIONID) || '%')

   AND (:PRODUCTDEFINITIONNAME IS NULL OR UPPER(CPD.PRODUCTDEFINITIONNAME) LIKE UPPER(:PRODUCTDEFINITIONNAME) || '%')
