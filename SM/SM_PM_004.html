------------------------ 출하 목록 쿼리

SELECT

	TO_DATE(TO_CHAR(CS.SHIPPEDDATE,'YYYY-MM-DD'),'YYYY-MM-DD') AS  SHIPPEDDATE,

	CS.SHIPPEDNO,

	CS.DELIVERYORDERID,

	CS.DELIVERYITEMNO,

	CS.CUSTOMERID,

	CC.CUSTOMERNAME,

	CD.DELIVERYORDERTYPE,

	CC2.CODENAME AS DELIVERYORDERTYPENAME,

	CMD.MATERIALCLASSID,

	CMC.MATERIALCLASSNAME,

	CS.MATERIALDEFINITIONID,

	CMD.MATERIALDEFINITIONNAME,

	CS.QTY,

	CS.UNITID,

	CS.SHIPOUT,

	CS.WAYBILLNO,

	CS.FROMLOCATION,

	CS.LOCATION,

	CS.STATE,

	CS.PRODUCTTYPE

  FROM CUS_SHIPPED CS

 INNER JOIN CIM_DELIVERYORDER CD

         ON 1=1

        AND CD.DELIVERYORDERID = CS.DELIVERYORDERID 

 INNER JOIN CIM_CUSTOMER CC

         ON 1=1

        AND CC.CUSTOMERID = CS.CUSTOMERID 

        AND CC.SITEID = CC.SITEID 

        AND CC.ISUSABLE = 'Usable'

 INNER JOIN CIM_MATERIALDEFINITION CMD

 		 ON 1=1

 		AND CMD.MATERIALDEFINITIONID = CS.MATERIALDEFINITIONID

 		AND CMD.SITEID = CS.SITEID

 		--AND CMD.ISUSABLE = 'Usable'

  LEFT JOIN CIM_MATERIALCLASS CMC

         ON 1=1

        AND CMC.MATERIALCLASSID = CMD.MATERIALCLASSID

        AND CMC.SITEID = CMD.SITEID

        AND CMC.ISUSABLE = 'Usable'

  LEFT JOIN CIM_CODE CC2

         ON 1=1

        AND CC2.CODEID = CD.DELIVERYORDERTYPE

        AND CC2.CODECLASSID = 'DELIVERYORDERTYPE'

        AND CC2.SITEID = :SITEID

        AND CC2.ISUSABLE = 'Usable'

 WHERE CS.SHIPPEDDATE BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND (TO_DATE(:TODATE, 'YYYY-MM-DD') + 1)

   AND CS.SITEID = :SITEID

   AND CS.ISUSABLE = 'Usable'

   AND (:DELIVERYORDERID IS NULL OR UPPER(CS.DELIVERYORDERID) LIKE UPPER(:DELIVERYORDERID) || '%')

   AND (:SHIPPEDNO IS NULL OR UPPER(CS.SHIPPEDNO) LIKE UPPER(:SHIPPEDNO) || '%')

   AND (:MATERIALDEFINITIONID IS NULL OR UPPER(CS.MATERIALDEFINITIONID) LIKE UPPER(:MATERIALDEFINITIONID) || '%')

   AND (:MATERIALDEFINITIONNAME IS NULL OR UPPER(CMD.MATERIALDEFINITIONNAME) LIKE UPPER(:MATERIALDEFINITIONNAME) || '%')  

   AND (:CUSTOMERID IS NULL OR CS.CUSTOMERID = :CUSTOMERID)
