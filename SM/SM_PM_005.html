--제품 출하 이력 



SELECT 

	TO_CHAR(DEL.DELIVERYDATE , 'YYYY-MM-DD') AS DELIVERYDATE  

	, DEL.DELIVERYORDERID

	, DEL.DELIVERYITEMNO

	, DEL.CUSTOMERID

	, DEL.DELIVERYORDERTYPE 

	, DEL.PRODUCTDEFINITIONID 

	, DEF.PRODUCTDEFINITIONNAME

	, DEL.QTY 

	, DEL.UNITID 

	, DEL.CUSTOMERNAME AS DESTINATIONNAME

	, '' AS DISPATCHNO

FROM 

(

	SELECT

		*

	FROM 

	(

		SELECT

			ROW_NUMBER() OVER (PARTITION BY DEL.DELIVERYORDERID ORDER BY DEL.LASTEVENTSEQ) AS RNK, DEL.*, CSTM.CUSTOMERNAME 

		FROM CIM_DELIVERYORDERHIST DEL

		LEFT OUTER JOIN CIM_CUSTOMER CSTM

			ON DEL.SITEID = CSTM.SITEID 

			AND DEL.CUSTOMERID = CSTM.CUSTOMERID 

		WHERE 1=1

			AND DEL.ISUSABLE = 'Usable'

			AND DEL.STATE = 'Active'

			AND DEL.SITEID = :SITEID 

			AND DEL.DELIVERYDATE >= TO_DATE(:FROMDATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') AND DEL.DELIVERYDATE < TO_DATE(:TODATE||' 00:00:00', 'YYYY-MM-DD HH24:MI:SS') + 1

			AND DEL.CUSTOMACTIVITY = 'ConfirmDeliveryOrder'

	) 

	WHERE RNK = 1

) DEL

LEFT OUTER JOIN CIM_PRODUCTDEFINITION DEF

	ON DEL.SITEID = DEF.SITEID 

	AND DEL.PRODUCTDEFINITIONID = DEF.PRODUCTDEFINITIONID 

WHERE 1=1

	AND (DEF.PRODUCTDEFINITIONID LIKE '%'|| :PRODUCTDEFINITIONID ||'%')

	AND (DEF.PRODUCTDEFINITIONNAME LIKE '%'|| :PRODUCTDEFINITIONNAME ||'%')

조인 구조도 테이블 3 · 조인 2

SQL 보기
--출하 상세 List



SELECT

	PAL.DELIVERYORDERID

	, DEF.PRODUCTTYPE

	, DEF.PRODUCTDEFINITIONID

	, DEF.PRODUCTDEFINITIONNAME

	, MAT.MATERIALLOTID AS LOTID

	, MAT.QTY

	, MAT.UNITID

FROM CUS_PALLETLOT PAL

LEFT OUTER JOIN CIM_MATERIALLOT MAT

	ON PAL.PALLETLOTID = MAT.PALLETID 

LEFT OUTER JOIN CIM_MATERIALDEFINITION DEF

	ON MAT.MATERIALDEFINITIONID = DEF.MATERIALDEFINITIONID 

LEFT OUTER JOIN CIM_PRODUCTORDER OD

	ON MAT.SITEID = OD.SITEID 

	AND MAT.PRODUCTORDERID = OD.PRODUCTORDERID 

LEFT OUTER JOIN CIM_PRODUCTDEFINITION DEF

	ON MAT.SITEID = DEF.SITEID 

	AND OD.PRODUCTDEFINITIONID = DEF.PRODUCTDEFINITIONID 	

WHERE 1=1

	AND PAL.SITEID = :SITEID

	AND PAL.DELIVERYORDERID = :DELIVERYORDERID
