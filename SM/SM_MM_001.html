Query (조회 서비스) : SelectPurchaseOrderItemList

  • ID 구성: SelectPurchaseOrderItemList / 클래스 THiRAMES.Service.UI / 버전 00001
  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)
  • 쿼리 파라메터: FROMDATE TODATE SITEID VENDORID PURCHASEORDER MATERIALCLASSID MATERIALDEFINITIONID MATERIALDEFINITIONNAME
  • 테이블(조회/쓰기 전체, 함수·SP 포함):
    CIM_MATERIALLOT메인읽기   CIM_VENDOR읽기   CIM_MATERIALDEFINITION읽기   CIM_MATERIALCLASS읽기   CIM_FACILITY읽기   CIM_INSPDEFINITION읽기   CIM_CODE읽기
조인 구조도 테이블 5 · 조인 3
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_VENDOR, CIM_MATERIALDEFINITION, CIM_CODE
SQL 보기
SELECT 
		MATERIALINDATE,
		PURCHASEORDERITEMNO,
		VENDORID,
		VENDORNAME,
		PURCHASEORDER,
		MATERIALCLASSID,
		MATERIALCLASSNAME,
		MATERIALDEFINITIONID,
		MATERIALDEFINITIONNAME,
		VENDORLOT,
		RECEIPTUNITTYPE,
		RECEIPTUNITTYPENAME,
		STATENAME,
		SUM(QTY) AS QTY,
		COUNT(OUTBOX) AS OUTBOXQTY,
		COUNT(INBOX) AS INBOXQTY,
		COUNT(CUSTOMERLOT) AS MATERIALLOTQTY,
		RECEIPTUNITQTY,
		UNITID,
		STORECHECK,
		INSPMETHOD,
		INSPMETHODNAME,
		REPEATCOUNT,
		INSPDEFINITIONNAME,
		REV,
		INSPDEFINITIONSYSID,
		INSPSPEC,
		INSPTARGET,
		REQTOTALQTY
FROM (
SELECT 
	CMT.CREATETIME AS MATERIALINDATE, 
	CMT.PURCHASEORDERITEMNO ,
	CMT.VENDORID ,
	CR.VENDORNAME ,
	CMT.PURCHASEORDER , 
	CMDN.MATERIALCLASSID ,
     CMS.MATERIALCLASSNAME ,
	CMT.MATERIALDEFINITIONID ,
	CMDN.MATERIALDEFINITIONNAME ,
	CMT.VENDORLOT ,
	CMDN.RECEIPTUNITTYPE ,
	  CC.CODENAME AS RECEIPTUNITTYPENAME ,
	CMT.OUTBOX  ,
	CMT.INBOX  ,
	CMT.CUSTOMERLOT , 
	CMT.QTY  AS RECEIPTUNITQTY,
	CMT.QTY ,
	CMT.UNITID ,
	CMT.STORECHECK  ,
	  CF.FACILITYNAME AS STATENAME, 
	 CASE WHEN CIN.INSPMETHOD IS NULL THEN CIN2.INSPMETHOD ELSE CIN.INSPMETHOD END AS INSPMETHOD,
	 CASE WHEN CIN.INSPMETHOD IS NULL THEN CC3.CODENAME ELSE CC2.CODENAME END AS INSPMETHODNAME ,
 	 CASE WHEN CIN.INSPMETHOD IS NULL THEN CIN2.REPEATCOUNT ELSE CIN.REPEATCOUNT END AS REPEATCOUNT ,
	 CASE WHEN CIN.INSPMETHOD IS NULL THEN CIN2.INSPDEFINITIONNAME ELSE CIN.INSPDEFINITIONNAME END AS INSPDEFINITIONNAME ,
	 CASE WHEN CIN.INSPMETHOD IS NULL THEN CIN2.REV ELSE CIN.REV END AS REV ,
	 CASE WHEN CIN.INSPDEFINITIONSYSID  IS NULL THEN CIN2.INSPDEFINITIONSYSID ELSE CIN.INSPDEFINITIONSYSID END AS INSPDEFINITIONSYSID ,
	 CASE WHEN CIN.INSPSPEC  IS NULL THEN CIN2.INSPSPEC ELSE CIN.INSPSPEC END AS INSPSPEC ,
	 'MATERIALLOTID' AS INSPTARGET , 
	 '0' AS REQTOTALQTY
FROM CIM_MATERIALLOT CMT
INNER JOIN CIM_VENDOR CR
		ON 1=1
		AND CR.VENDORID = CMT.VENDORID
		AND CR.SITEID  = CMT.SITEID 
		AND CR.ISUSABLE = 'Usable'
 INNER JOIN CIM_MATERIALDEFINITION CMDN
         ON 1=1 
        AND CMDN.MATERIALDEFINITIONID = CMT.MATERIALDEFINITIONID
        AND CMDN.SITEID  = CMT.SITEID 
        AND CMDN.ISUSABLE  = 'Usable' 
 INNER JOIN CIM_MATERIALCLASS CMS
         ON 1=1 
        AND CMS.MATERIALCLASSID  = CMDN.MATERIALCLASSID
        AND CMS.SITEID  = CMDN.SITEID 
        AND CMS.ISUSABLE  = 'Usable'
  INNER JOIN CIM_FACILITY CF
         ON 1=1 
        AND CF.FACILITYID  = CMT.LOCATION
        AND CF.SITEID  = CMDN.SITEID 
        AND CF.ISUSABLE  = 'Usable' 
-- INNER JOIN CIM_STATE CE
--         ON 1=1 
--        AND CE.STATEID  = CMT.STORECHECK 
--		AND CE.STATEMODELID = 'PurchaseOrderItemNoState'
--        AND CE.SITEID  = CMT.SITEID 
--        AND CE.ISUSABLE  = 'Usable'
LEFT JOIN CIM_INSPDEFINITION CIN
         ON 1=1 
        AND CIN.COPERATOR  = CMT.VENDORID 
        AND CIN.ITEMID  = CMT.MATERIALDEFINITIONID 
        AND CIN.INSPTYPE = 'IQC'
        AND CIN.STATE = 'Active'
        AND CIN.SITEID  = CMT.SITEID 
        AND CIN.ISUSABLE  = 'Usable' 
  LEFT JOIN CIM_INSPDEFINITION CIN2
         ON 1=1 
        AND CIN2.COPERATOR  = '-'
        AND CIN2.ITEMID  = CMT.MATERIALDEFINITIONID 
        AND CIN2.INSPTYPE = 'IQC'
        AND CIN2.STATE = 'Active'
        AND CIN2.SITEID  = CMT.SITEID 
        AND CIN2.ISUSABLE  = 'Usable'        
  LEFT JOIN CIM_CODE CC
         ON 1=1 
        AND CC.CODEID  = CMDN.RECEIPTUNITTYPE 
        AND CC.CODECLASSID  = 'RECEIPTUNITTYPE'
        AND CC.SITEID  = CMT.SITEID 
        AND CC.ISUSABLE  = 'Usable'
  LEFT JOIN CIM_CODE CC2
         ON 1=1 
        AND CC2.CODEID  = CIN.INSPMETHOD 
        AND CC2.CODECLASSID  = 'INSPMETHOD'
        AND CC2.SITEID  = CMT.SITEID 
        AND CC2.ISUSABLE  = 'Usable' 
  LEFT JOIN CIM_CODE CC3
         ON 1=1 
        AND CC3.CODEID  = CIN2.INSPMETHOD 
        AND CC3.CODECLASSID  = 'INSPMETHOD'
        AND CC3.SITEID  = CMT.SITEID 
        AND CC3.ISUSABLE  = 'Usable'
     
WHERE 1=1
   AND CMT.CREATETIME BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 1
   AND CMT.SITEID = :SITEID 
   AND CMT.ISUSABLE = 'Usable'
   AND (:VENDORID IS NULL  OR CMT.VENDORID = :VENDORID)
   AND (:PURCHASEORDER IS NULL OR UPPER(CMT.PURCHASEORDER) LIKE UPPER(:PURCHASEORDER) || '%')
   AND (:MATERIALCLASSID IS NULL  OR CMS.MATERIALCLASSID = :MATERIALCLASSID)
   AND CF.FACILITYID IN('1100','1200','1300')
--   AND (:STATE IS NULL  OR CMT.STATE = :STATE)
   AND (:MATERIALDEFINITIONID IS NULL  OR UPPER(CMT.MATERIALDEFINITIONID) LIKE UPPER(:MATERIALDEFINITIONID) || '%')
   AND (:MATERIALDEFINITIONNAME IS NULL  OR UPPER(CMDN.MATERIALDEFINITIONNAME) LIKE UPPER(:MATERIALDEFINITIONNAME) || '%')
   )
GROUP BY MATERIALINDATE,
		PURCHASEORDERITEMNO,
		VENDORID,
		VENDORNAME,
		PURCHASEORDER,
		MATERIALCLASSID,
		MATERIALCLASSNAME,
		MATERIALDEFINITIONID,
		MATERIALDEFINITIONNAME,
		VENDORLOT,
		RECEIPTUNITTYPE,
		RECEIPTUNITTYPENAME,
		RECEIPTUNITQTY,
		UNITID,
		STORECHECK,
		INSPMETHOD,
		INSPMETHODNAME,
		REPEATCOUNT,
		INSPDEFINITIONNAME,
		REV,
		INSPDEFINITIONSYSID,
		INSPSPEC,
		INSPTARGET,
		STATENAME,
		REQTOTALQTY
ORDER BY PURCHASEORDERITEMNO
