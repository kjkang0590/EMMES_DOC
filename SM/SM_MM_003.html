#1 Query SelectMaterialinLotList

연결된 기능 (1번 표): #1 조회 - 그리드 SM_MM_0031 (자재입고취소1_자재입고목록)

Query (조회 서비스) : SelectMaterialinLotList

  • ID 구성: SelectMaterialinLotList / 클래스 THiRAMES.Service.UI / 버전 00001
  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)
  • 쿼리 파라메터: FROMDATE TODATE SITEID LOCATION VENDORID MATERIALCLASSID MATERIALDEFINITIONID MATERIALDEFINITIONNAME
  • 테이블(조회/쓰기 전체, 함수·SP 포함):
    CIM_MATERIALLOT메인읽기   CIM_VENDOR읽기   CIM_MATERIALDEFINITION읽기   CIM_MATERIALCLASS읽기   CIM_CODE읽기
조인 구조도 테이블 2 · 조인 0
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_VENDOR, CIM_MATERIALDEFINITION, CIM_CODE
SQL 보기
---- 입고테이블(커스텀) LOT목록 조회 쿼리
SELECT 
	CMT.STOCKDATE AS MATERIALINDATE ,
	CMT.VENDORID ,
	CVR.VENDORNAME ,
	CMT.MATERIALCLASSID ,
	CMS.MATERIALCLASSNAME ,
	CMT.MATERIALDEFINITIONID ,
	CMDN.MATERIALDEFINITIONNAME ,
	CMT.VENDORLOT ,
   CMDN.RECEIPTUNITTYPE ,
	 CC.CODENAME AS  RECEIPTUNITTYPENAME ,
	CMT.OUTBOX ,
	CMT.INBOX ,
	CMT.MATERIALLOTID ,
	CMT.QTY ,
	CMT.UNITID ,
	CMDN.PRODUCTTYPE ,
	CMT.LOCATION ,
	CMT.STATE
  FROM CIM_MATERIALLOT CMT
 INNER JOIN CIM_VENDOR CVR
         ON 1=1
        AND CVR.VENDORID = CMT.VENDORID 
    	AND CVR.SITEID  = CMT.SITEID 
        AND CVR.ISUSABLE  = 'Usable'
 INNER JOIN CIM_MATERIALDEFINITION CMDN
         ON 1=1 
        AND CMDN.MATERIALDEFINITIONID = CMT.MATERIALDEFINITIONID
        AND CMDN.SITEID  = CMT.SITEID 
        AND CMDN.ISUSABLE  = 'Usable'         
 INNER JOIN CIM_MATERIALCLASS CMS
         ON 1=1 
        AND CMS.MATERIALCLASSID  = CMDN.MATERIALCLASSID
        AND CMS.SITEID  = CMDN.SITEID 
        AND CMS.ISUSABLE  = 'Usable' 
 INNER JOIN CIM_CODE CC
         ON 1=1 
        AND CC.CODEID  = CMDN.RECEIPTUNITTYPE 
        AND CC.CODECLASSID  = 'RECEIPTUNITTYPE'
        AND CC.SITEID  = CMDN.SITEID 
        AND CC.ISUSABLE  = 'Usable'         
 WHERE (TO_CHAR(CMT.STOCKDATE,'YYYY-MM-DD') >= :FROMDATE AND TO_CHAR(CMT.STOCKDATE,'YYYY-MM-DD') <= :TODATE )
   AND CMT.STATE = 'Active'
   AND CMT.QTY <> CMT.ORIGINALQTY 
   AND CMT.SITEID = :SITEID 
   AND CMT.LOCATION = :LOCATION
   AND (:VENDORID IS NULL  OR CMT.VENDORID = :VENDORID)
   AND (:MATERIALCLASSID IS NULL  OR CMT.MATERIALCLASSID = :MATERIALCLASSID)
   AND (:MATERIALDEFINITIONID IS NULL  OR CMT.MATERIALDEFINITIONID = :MATERIALDEFINITIONID)
   AND (:MATERIALDEFINITIONNAME IS NULL  OR CMDN.MATERIALDEFINITIONNAME = :MATERIALDEFINITIONNAME)
 ORDER BY CMT.STOCKDATE , CMT.MATERIALLOTID

SELECT 

	CMN.MATERIALINDATE ,

	CMN.MATERIALINID ,

	CMN.VENDORID ,

	CVR.VENDORNAME ,

	CMT.MATERIALCLASSID ,

	CMS.MATERIALCLASSNAME ,

	CMT.MATERIALDEFINITIONID ,

	CMDN.MATERIALDEFINITIONNAME ,

	CMT.VENDORLOT ,

   CMDN.RECEIPTUNITTYPE ,

	 CC.CODENAME AS  RECEIPTUNITTYPENAME ,

	CMT.INBOX ,

	CMT.QTY ,

         ON 1=1

        AND CMN.MATERIALLOTID = CMT.MATERIALLOTID 

        AND CMN.SITEID = CMT.SITEID 

        AND CMN.ISUSABLE = 'Usable'

        AND CMN.LOCATION = CMT.LOCATION

        AND CMN.MATERIALSAVETYPE = 'MATERIALIN'

 INNER JOIN CIM_VENDOR CVR

         ON 1=1

        AND CVR.VENDORID = CMN.VENDORID 

    	AND CVR.SITEID  = CMN.SITEID
