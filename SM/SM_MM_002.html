-------- 자재 입고 시 조회 쿼리
SELECT 
	CPOM.MATERIALINDATE ,
	CPOM.PURCHASEORDERITEMNO ,
	CPOM.VENDORID ,
	CR.VENDORNAME ,
	CPOM.PURCHASEORDER , 
	CPOM.MATERIALCLASSID ,
     CMS.MATERIALCLASSNAME ,
	CPOM.MATERIALDEFINITIONID ,
	CMDN.MATERIALDEFINITIONNAME ,
	CPOM.VENDORLOT ,
	CMDN.RECEIPTUNITTYPE  ,
	  CC.CODENAME AS RECEIPTUNITTYPENAME ,
	(CPOM.OUTBOXQTY - NVL(OUTBOX.OUTBOX,'0')) AS OUTBOXQTY ,
	CASE WHEN CPOM.INBOXQTY = 0 THEN CPOM.INBOXQTY
	ELSE (CPOM.INBOXQTY - NVL(INBOX.INBOX,'0')) END AS INBOXQTY,
	CASE WHEN CPOM.MATERIALLOTQTY = 0 THEN CPOM.MATERIALLOTQTY
	ELSE (CPOM.MATERIALLOTQTY - NVL(MATERIALLOT.MATERIALLOT,'0')) END AS MATERIALLOTQTY,
	CPOM.RECEIPTUNITQTY ,
	(CPOM.QTY -TOTAL.TOTAL) AS QTY ,
	CPOM.UNITID ,
	CMDN.PRODUCTTYPE 
  FROM CUS_PURCHASEORDERITEM CPOM
 INNER JOIN CIM_VENDOR CR
         ON 1=1 
        AND CR.VENDORID = CPOM.VENDORID
        AND CR.SITEID  = CPOM.SITEID 
        AND CR.ISUSABLE  = 'Usable'
 INNER JOIN CIM_MATERIALDEFINITION CMDN
         ON 1=1 
        AND CMDN.MATERIALDEFINITIONID = CPOM.MATERIALDEFINITIONID
        AND CMDN.SITEID  = CPOM.SITEID 
        AND CMDN.ISUSABLE  = 'Usable'    
 INNER JOIN CIM_MATERIALCLASS CMS
         ON 1=1 
        AND CMS.MATERIALCLASSID  = CMDN.MATERIALCLASSID
        AND CMS.SITEID  = CMDN.SITEID 
        AND CMS.ISUSABLE  = 'Usable'     
 INNER JOIN CIM_STATE CE
         ON 1=1 
        AND CE.STATEID  = CPOM.STATE
        AND CE.STATEMODELID = 'PurchaseOrderItemNoState'
        AND CE.SITEID  = CPOM.SITEID 
        AND CE.ISUSABLE  = 'Usable'  
  LEFT JOIN CIM_INSPDEFINITION CIN
         ON 1=1 
        AND CIN.COPERATOR  = CPOM.VENDORID 
        AND CIN.ITEMID  = CPOM.MATERIALDEFINITIONID 
        AND CIN.INSPTYPE = 'IQC'
        AND CIN.STATE = 'Active'
        AND CE.SITEID  = CPOM.SITEID 
        AND CE.ISUSABLE  = 'Usable' 
  LEFT JOIN CIM_CODE CC
         ON 1=1 
        AND CC.CODEID  = CMDN.RECEIPTUNITTYPE 
        AND CC.CODECLASSID  = 'RECEIPTUNITTYPE'
        AND CC.SITEID  = CPOM.SITEID 
        AND CC.ISUSABLE  = 'Usable'
  LEFT JOIN (
	   SELECT 
	 		CM.PURCHASEORDERITEMNO,
	 		CM.SITEID,
	 		COUNT(CM.OUTBOX) AS OUTBOX 
	   FROM CIM_MATERIALLOT CM
	  WHERE EXPIREDATE IS NOT NULL
	    AND STORECHECK IS NOT NULL
	  	AND CM.SITEID = :SITEID
	  	AND CM.ISUSABLE = 'Usable'
	  GROUP BY 	CM.PURCHASEORDERITEMNO, CM.SITEID
  ) OUTBOX
        ON 1=1
       AND OUTBOX.PURCHASEORDERITEMNO = CPOM.PURCHASEORDERITEMNO
       AND OUTBOX.SITEID = CPOM.SITEID
 LEFT JOIN (
	   SELECT 
	 		CM.PURCHASEORDERITEMNO,
	 		CM.SITEID,
	 		COUNT(CM.INBOX) AS INBOX 
	   FROM CIM_MATERIALLOT CM
	  WHERE EXPIREDATE IS NOT NULL
	    AND STORECHECK IS NOT NULL
	    AND STATE = 'Active'
	  	AND CM.SITEID = :SITEID
	  	AND CM.ISUSABLE = 'Usable'
	  GROUP BY 	CM.PURCHASEORDERITEMNO, CM.SITEID
  ) INBOX
      	ON 1=1
       AND INBOX.PURCHASEORDERITEMNO = CPOM.PURCHASEORDERITEMNO
       AND INBOX.SITEID = CPOM.SITEID
   LEFT JOIN (
	   SELECT 
	 		CM.PURCHASEORDERITEMNO,
	 		CM.SITEID,
	 		COUNT(CM.MATERIALLOTID) AS MATERIALLOT 
	   FROM CIM_MATERIALLOT CM
	  WHERE EXPIREDATE IS NOT NULL
	    AND STORECHECK IS NOT NULL
	    AND STATE = 'Active'
	  	AND CM.SITEID = :SITEID
	  	AND CM.ISUSABLE = 'Usable'
	  GROUP BY 	CM.PURCHASEORDERITEMNO, CM.SITEID
  ) MATERIALLOT
        ON 1=1
       AND MATERIALLOT.PURCHASEORDERITEMNO = CPOM.PURCHASEORDERITEMNO
       AND MATERIALLOT.SITEID = CPOM.SITEID
 LEFT JOIN (
	   SELECT 
	 		CM.PURCHASEORDERITEMNO,
	 		CM.SITEID,
	 		SUM(CM.QTY) AS TOTAL 
	   FROM CIM_MATERIALLOT CM
	  WHERE EXPIREDATE IS NOT NULL
	    AND STORECHECK IS NOT NULL
	    AND STATE = 'Active'
	  	AND CM.SITEID = :SITEID
	  	AND CM.ISUSABLE = 'Usable'
	  GROUP BY 	CM.PURCHASEORDERITEMNO, CM.SITEID
  ) TOTAL
        ON 1=1
       AND TOTAL.PURCHASEORDERITEMNO = CPOM.PURCHASEORDERITEMNO
       AND TOTAL.SITEID = CPOM.SITEID       
 WHERE 1=1
   AND CPOM.MATERIALINDATE BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 1
   AND (CPOM.STATE = 'Active' OR CPOM.STATE = 'Processing' OR(CPOM.STATE = 'Created' AND CIN.INSPMETHOD = 'PASS'))
   AND CPOM.SITEID = :SITEID 
   AND CPOM.ISUSABLE = 'Usable'
   AND (:VENDORID IS NULL  OR CPOM.VENDORID = :VENDORID)
   AND (:MATERIALCLASSID IS NULL  OR CPOM.MATERIALCLASSID = :MATERIALCLASSID)
   AND (:MATERIALDEFINITIONID IS NULL OR UPPER(CPOM.MATERIALDEFINITIONID) LIKE UPPER(:MATERIALDEFINITIONID) || '%')
   AND (:MATERIALDEFINITIONNAME IS NULL  OR UPPER(CMDN.MATERIALDEFINITIONNAME) LIKE UPPER(:MATERIALDEFINITIONNAME) || '%')
 ORDER BY PURCHASEORDERITEMNO
