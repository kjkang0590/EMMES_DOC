SELECT 

	CPL.DELIVERYORDERID ,

	CPL.PALLETLOTID AS PALLETID,

	CPL.CUSTOMERID ,

	CC.CUSTOMERNAME ,

	CPL.MATERIALCLASSID ,

	CMC.MATERIALCLASSNAME ,

	CPL.MATERIALDEFINITIONID ,

	CMD.MATERIALDEFINITIONNAME ,

	CPL.QTY ,

	CPL.UNITID ,

	CPL.PREVLOCATION ,

	CPL.PREVSTATE ,

	CPL.LOCATION ,

	CPL.STATE ,

	CMD.PRODUCTTYPE ,

	CPS.PALLETSTOREDNO 

  FROM CUS_PALLETLOT CPL

 INNER JOIN CIM_CUSTOMER CC

    ON 1=1

   AND CC.CUSTOMERID = CPL.CUSTOMERID 

  INNER JOIN CIM_MATERIALDEFINITION CMD

         ON 1=1

        AND CMD.MATERIALDEFINITIONID = CPL.MATERIALDEFINITIONID

        AND CMD.SITEID = :SITEID

        --AND CMD.ISUSABLE = :ISUSABLE 

 INNER JOIN CIM_MATERIALCLASS CMC

         ON 1=1

        AND CMC.MATERIALCLASSID = CPL.MATERIALCLASSID

        AND CMC.SITEID = :SITEID

       -- AND CMC.ISUSABLE = :ISUSABLE 

 INNER JOIN CUS_PALLETSTORED CPS

 		 ON 1=1

 		AND CPS.PALLETID =  CPL.PALLETLOTID 

 		AND CPS.PALLETSTOREDTYPE = 'IN'

 		AND CPS.SITEID = :SITEID

 		--AND CPS.ISUSABLE = :ISUSABLE 

 WHERE 1=1 

   AND CPL.STATE = 'Active'

   AND CPL.SITEID = :SITEID

   AND CPL.ISUSABLE = 'Usable'

   AND (:MATERIALDEFINITIONID IS NULL OR UPPER(CPL.MATERIALDEFINITIONID) LIKE UPPER(:MATERIALDEFINITIONID) || '%')

   AND (:MATERIALDEFINITIONNAME IS NULL OR UPPER(CMD.MATERIALDEFINITIONNAME) LIKE UPPER(:MATERIALDEFINITIONNAME) || '%')  

   AND (:CUSTOMERID IS NULL OR CPL.CUSTOMERID = :CUSTOMERID)

SELECT 

	CDO.DELIVERYORDERID ,

	CDO.DELIVERYITEMNO ,

	PALLET.PALLETID , 

	CC.CUSTOMERID ,

	CC.CUSTOMERNAME ,

	CMD.MATERIALCLASSID ,

	CMC.MATERIALCLASSNAME ,

	PALLET.MATERIALDEFINITIONID,

	CMD.MATERIALDEFINITIONNAME  ,

	PALLET.PALLETQTY AS QTY,

	CMD.UNITID ,

	CMD.LOCATION ,

	CMD.PRODUCTTYPE 

  FROM

  (

	SELECT 

		CML.PRODUCTORDERID ,

		CML.PALLETID,

		CML.MATERIALDEFINITIONID ,

		SUM(CML.QTY) AS PALLETQTY 

	 FROM CIM_MATERIALLOT CML

	WHERE PALLETID IS NOT NULL

	GROUP BY CML.PALLETID , CML.MATERIALDEFINITIONID , CML.PRODUCTORDERID

  ) PALLET

 INNER JOIN CIM_MATERIALDEFINITION CMD

         ON 1=1

        AND CMD.MATERIALDEFINITIONID = PALLET.MATERIALDEFINITIONID

        AND CMD.SITEID = :SITEID

        --AND CMD.ISUSABLE = :ISUSABLE 

 INNER JOIN CIM_MATERIALCLASS CMC

         ON 1=1

        AND CMC.MATERIALCLASSID = CMD.MATERIALCLASSID

        AND CMC.SITEID = :SITEID

       -- AND CMC.ISUSABLE = :ISUSABLE

 INNER JOIN CIM_DELIVERYORDER CDO

         ON 1=1

        AND CDO.DELIVERYORDERID = PALLET.PRODUCTORDERID

        AND CDO.SITEID = :SITEID

        --AND CDO.ISUSABLE = :ISUSABLE

 INNER JOIN CIM_CUSTOMER CC

         ON 1=1

        AND CC.CUSTOMERID = CDO.CUSTOMERID

        AND CC.SITEID = :SITEID

       -- AND CC.ISUSABLE = :ISUSABLE

  LEFT JOIN CUS_PALLETLOT CPL

  		 ON 1=1

  		AND CPL.PALLETLOTID = PALLET.PALLETID

  		AND CPL.SITEID = :SITEID

  		--AND CPL.ISUSABLE = :ISUSABLE

 WHERE 1=1

   AND CPL.PALLETLOTID IS NULL 

   AND (:MATERIALDEFINITIONID IS NULL OR UPPER(PALLET.MATERIALDEFINITIONID) LIKE UPPER(:MATERIALDEFINITIONID) || '%')

   AND (:MATERIALDEFINITIONNAME IS NULL OR UPPER(CMD.MATERIALDEFINITIONNAME) LIKE UPPER(:MATERIALDEFINITIONNAME) || '%')  

   AND (:CUSTOMERID IS NULL OR CDO.CUSTOMERID = :CUSTOMERID)
