SELECT REQ.INSPREQNO

     , REQ.FINALDATETIME

     , REQ.INSPMETHOD

     , CC.CODENAME AS INSPMETHODNAME

     , REQ.INSPTYPE

     , REQ.ITEMID

     , CASE WHEN @INSPTYPE = 'IQC' THEN MD.MATERIALDEFINITIONNAME 

            WHEN @INSPTYPE = 'OQC' THEN PD.PRODUCTDEFINITIONNAME

            ELSE '' END AS ITEMNAME

     , REQ.PROCESSSEGMENTID

     , PS.PROCESSSEGMENTNAME

     , CEC.EQUIPMENTCLASSID

     , REQ.EQUIPMENTID

     , CE.EQUIPMENTNAME

     , REQ.SAMPLELOTID

     , REQ.SAMPLEPOINTID

     , IR.INSPREQNO

     , CW.WORKORDERID

     , REQ.SAMPLEID

     , REQ.LOTID

     , REQ.FINALJUDGE

     , REQ.STATE

     , CS.STATENAME

     , REQ.COMMENTS

     , REQ.CREATETIME

     , RCU.USERNAME AS INSPRECEIPTUSERID

     , REQ.INSPRECEIPTTIME

     , ECU.USERNAME AS INSPENDUSERID

     , REQ.INSPDEFINITIONSYSID

     , ID.INSPDEFINITIONNAME

     , REQ.RECEPTDATETIME -- 2026.05.06 박성진 - SC2026040041 [검사결과조회] 컬럼 추가 UI [검사접수시간] : INSPRECEIPTTIME

     , REQ.INSPENDTIME  -- 2026.05.06 박성진 - SC2026040041 [검사결과조회] 컬럼 추가 UI [최종결과시간] : FINALRESULTTIME

     -- 2026.05.06 박성진 - SC2026040041 [진행시간] 컬럼 추가 UI [진행시간] : PROCESSTIME 

     , CASE WHEN REQ.RECEPTDATETIME IS NULL THEN '0'

		ELSE CONCAT(

	     DATEDIFF(SECOND, REQ.RECEPTDATETIME, ISNULL(REQ.INSPENDTIME, GETDATE())) / 86400

	     , 'D '

	     , RIGHT('00' + CONVERT(VARCHAR(2), (DATEDIFF(SECOND, REQ.RECEPTDATETIME, ISNULL(REQ.INSPENDTIME, GETDATE())) % 86400) / 3600), 2)

	     , ':'

	     , RIGHT('00' + CONVERT(VARCHAR(2), (DATEDIFF(SECOND, REQ.RECEPTDATETIME, ISNULL(REQ.INSPENDTIME, GETDATE())) % 3600) / 60), 2)

  LEFT JOIN CIM_MATERIALDEFINITION MD WITH(NOLOCK) ON MD.SITEID = REQ.SITEID AND MD.MATERIALDEFINITIONID = REQ.ITEMID

  LEFT JOIN CIM_PRODUCTDEFINITION PD WITH(NOLOCK) ON PD.SITEID = REQ.SITEID AND PD.PRODUCTDEFINITIONID = REQ.ITEMID

  LEFT JOIN CIM_INSPDEFINITION ID WITH(NOLOCK) ON ID.SITEID = REQ.SITEID AND ID.INSPDEFINITIONSYSID = REQ.INSPDEFINITIONSYSID

  LEFT JOIN CIM_LOT CL WITH(NOLOCK) ON CL.SITEID = REQ.SITEID AND CL.LOTID = REQ.LOTID

  LEFT JOIN CIM_WORKORDER CW WITH(NOLOCK) ON CW.SITEID = CL.SITEID AND CW.WORKORDERID = CL.WORKORDERID

 WHERE REQ.SITEID = @SITEID

   AND REQ.STATE = 'Confirm'

	  FROM CIM_INSPRESULT AS IR

	 	ON IIS.SITEID = IR.SITEID

	   AND IIS.INSPDEFINITIONITEMSPECSYSID = IR.INSPDEFINITIONITEMSPECSYSID

			   AND SAMPLEID = @SAMPLEID

     ORDER BY IIS.UISEQUENCE
