마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_PRODUCTDEFINITION, CIM_INSPITEMCLASS, CIM_EQUIPMENT, CIM_INSPDEFINITION, CIM_UNIT

--QM_AN_006_검사 결과 Trend - 검사 결과 목록 Grid, 검사 결과 Trend 차트 조회 쿼리
--SelectInspectionTrendList
--Param : SITEID, FROMDATE, TODATE, INSPTYPE, INSPITEMID, ITEMID, EQUIPMENTID, SAMPLEPOINTID, FINALJUDGE
------------------------------------------
SELECT A.INSPREQNO
     , A.ITEMID
	 , D.PRODUCTDEFINITIONNAME
	 , H.EQUIPMENTCLASSID
	 , G.EQUIPMENTID
	 , G.EQUIPMENTNAME
	 , A.SAMPLEPOINTID
	 , A.INSPTYPE
     , F.INSPITEMCLASSID
	 , F.INSPITEMCLASSNAME
	 , E.INSPITEMID
	 , E.INSPITEMNAME
     , A.INSPDEFINITIONSYSID
	 , I.INSPDEFINITIONNAME
     , L.WORKORDERID
     , B.INSPLOTID AS LOTID
     , B.USL
     , B.LSL
     , B.TARGET
	 , J.UNITNAME
	 , B.INSPITEMVALUE
	 , B.JUDGE
     , A.REQDATETIME AS REQUESTTIME
     , A.FINALDATETIME
     , A.SAMPLEID
  FROM CIM_INSPREQ A WITH(NOLOCK)
  LEFT JOIN CIM_INSPRESULT B WITH(NOLOCK) ON A.SITEID = B.SITEID AND A.INSPREQNO = B.INSPREQNO
  LEFT JOIN CIM_INSPDEFINITIONITEMSPEC C WITH(NOLOCK) ON B.SITEID = C.SITEID AND B.INSPDEFINITIONITEMSPECSYSID = C.INSPDEFINITIONITEMSPECSYSID
  LEFT JOIN CIM_PRODUCTDEFINITION D WITH(NOLOCK) ON A.SITEID = D.SITEID AND A.ITEMID = D.PRODUCTDEFINITIONID
  LEFT JOIN CIM_INSPITEM E WITH(NOLOCK) ON C.SITEID = E.SITEID AND C.INSPITEMID = E.INSPITEMID
  LEFT JOIN CIM_INSPITEMCLASS F WITH(NOLOCK) ON E.SITEID = F.SITEID AND E.INSPITEMCLASSID = F.INSPITEMCLASSID
  LEFT JOIN CIM_EQUIPMENT G WITH(NOLOCK) ON A.SITEID = G.SITEID AND A.EQUIPMENTID = G.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS H WITH(NOLOCK) ON G.SITEID = H.SITEID AND G.EQUIPMENTCLASSID = H.EQUIPMENTCLASSID
  LEFT JOIN CIM_INSPDEFINITION I WITH(NOLOCK) ON A.SITEID = I.SITEID AND A.INSPDEFINITIONSYSID = I.INSPDEFINITIONSYSID
  LEFT JOIN CIM_UNIT J WITH(NOLOCK) ON B.SITEID = J.SITEID AND B.UNITID = J.UNITID
  LEFT JOIN CIM_LOT K WITH(NOLOCK) ON B.SITEID = K.SITEID AND B.INSPLOTID = K.LOTID
  LEFT JOIN CIM_WORKORDER L WITH(NOLOCK) ON K.SITEID = L.SITEID AND K.WORKORDERID = L.WORKORDERID
 WHERE A.SITEID = @SITEID
   AND A.FINALDATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND A.INSPMETHOD = @INSPMETHOD
   AND (@INSPTYPE = '' OR A.INSPTYPE = @INSPTYPE)
   AND E.INSPITEMID IN(SELECT value FROM STRING_SPLIT(@INSPITEMID, ','))
   AND (@ITEMID = '' OR A.ITEMID = @ITEMID)
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)
   AND (@SAMPLEPOINTID = '' OR A.SAMPLEPOINTID LIKE '%' + @SAMPLEPOINTID + '%')
   AND (@FINALJUDGE = '' OR B.JUDGE = @FINALJUDGE)
   AND (@INSPITEMCLASSID = '' OR F.INSPITEMCLASSID = @INSPITEMCLASSID)
 ORDER BY A.REQDATETIME
