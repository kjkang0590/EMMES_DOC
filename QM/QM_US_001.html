마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_PROCESSSEGMENT, CIM_CODE, CIM_USER

------------------------------------------
--QM_US_001_부적합 관리 - 부적합 Lot 정보 조회 쿼리
--SelectIncongruityLotList
--Param : SITEID, FROMDATE, TODATE, LOTID, PROCESSSEGMENTID, MEASURESRESULT, CREATOR, PROCESSOR, ISHOLD
------------------------------------------
SELECT A.OCCURRENCEDATE
     , A.WRITINGDEPT
     , H.USERNAME AS CREATOR
     , A.INCONGRUITYTYPE
     , C.CODENAME AS INCONGRUITYTYPENAME
     , A.GRADE
     , D.CODENAME AS GRADENAME
     , A.LOTID
     , A.PROCESSSEGMENTID
     , B.PROCESSSEGMENTNAME
     , A.QTY
     , A.WORKDATE
     , A.WORKNO
     , A.INCONGRUITYNOTE
     , A.CAUSE
     , A.MEASURES
     , A.MEASURESRESULT
     , E.CODENAME AS MEASURESRESULTNAME
     , A.PROCESSRESULT
     , A.PROCESSDATE
     , A.PROCESSDEPT
     , F.DEPTNAME AS PROCESSDEPTNAME
     , A.PROCESSOR
     , F.USERNAME AS PROCESSORNAME
     , A.APPROVALRESULT
     , G.CODENAME AS APPROVALRESULTNAME
     , A.ISNOTICE
     , A.ISNOTICE AS ISNOTICENAME
     , L.ISHOLD
     , L.STATE AS LOTSTATE
  FROM CST_INCONGRUITY A WITH(NOLOCK)
  LEFT JOIN CIM_LOT L WITH(NOLOCK) ON A.SITEID = L.SITEID AND A.LOTID = L.LOTID
  LEFT JOIN CIM_PROCESSSEGMENT B WITH(NOLOCK) ON A.SITEID = B.SITEID AND A.PROCESSSEGMENTID = B.PROCESSSEGMENTID
  LEFT JOIN CIM_CODE C WITH(NOLOCK) ON A.SITEID = C.SITEID AND A.INCONGRUITYTYPE = C.CODEID AND C.CODECLASSID = 'INCONGRUITYTYPE'
  LEFT JOIN CIM_CODE D WITH(NOLOCK) ON A.SITEID = D.SITEID AND A.GRADE = D.CODEID AND D.CODECLASSID = 'GRADE'
  LEFT JOIN CIM_CODE E WITH(NOLOCK) ON A.SITEID = E.SITEID AND A.MEASURESRESULT = E.CODEID AND E.CODECLASSID = 'MEASURESRESULT'
  LEFT JOIN CIM_USER F WITH(NOLOCK) ON A.SITEID = F.SITEID AND A.PROCESSOR = F.USERID
  LEFT JOIN CIM_CODE G WITH(NOLOCK) ON A.SITEID = G.SITEID AND A.APPROVALRESULT = G.CODEID AND G.CODECLASSID = 'ISCONFIRMFORQALEADER'
  LEFT JOIN CIM_USER H WITH(NOLOCK) ON A.SITEID = H.SITEID AND A.CREATOR = H.USERID
 WHERE A.SITEID = @SITEID
   AND A.OCCURRENCEDATE BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@LOTID = '' OR A.LOTID = @LOTID)
   AND (@ISHOLD = '' OR L.ISHOLD = @ISHOLD)
   AND (@PROCESSSEGMENTID = '' OR A.PROCESSSEGMENTID = @PROCESSSEGMENTID)
   AND (@MEASURESRESULT = '' OR A.MEASURESRESULT = @MEASURESRESULT)
   AND (@CREATOR = '' OR H.USERNAME LIKE '%' + @CREATOR + '%')
   AND (@PROCESSOR = '' OR F.USERNAME LIKE '%' + @PROCESSOR + '%')
