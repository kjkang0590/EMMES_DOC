--QM_AN_001_Sample Label 출력 이력 조회 쿼리
--SelectSampleLabelList
--Param : SITEID, LINEID, INSPMETHOD, INSPTYPE
------------------------------------------
SELECT A.WORKDATE,
       A.INSPMETHOD,
       B.CODENAME AS REQUESTCATEGORY,
       A.INSPTYPE,
       C.CODENAME AS INSPECTIONTYPE,
       CONCAT(A.STARTLABELID, '~', A.ENDLABELID) AS SAMPLEBARCODE,
       A.PRINTCOUNT,
       D.PRINTERNAME AS PRINTERLOCATION,
       CONVERT(CHAR(19), A.CREATETIME, 20) AS PRINTTIME,
       A.CREATOR AS PRINTUSER
  FROM CST_SAMPLELABELPRINT A WITH(NOLOCK)
  LEFT JOIN CIM_CODE B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.INSPMETHOD = B.CODEID
   AND B.CODECLASSID = 'REQUESTCATEGORY'
  LEFT JOIN CIM_CODE C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.INSPTYPE = C.CODEID 
   AND C.CODECLASSID = 'INSPTYPE'
  LEFT JOIN RPS_PRINTERCONFIG D WITH(NOLOCK)
    ON A.SITEID = D.SITEID 
   AND A.PRINTERLOCATION = D.PRINTERID 
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND CONVERT(CHAR(10), A.CREATETIME, 23) BETWEEN CONVERT(DATE, @FROMDATE) AND CONVERT(DATE, @TODATE)
ORDER BY A.LASTEVENTSEQ DESC
