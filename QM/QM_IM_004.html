SELECT C.INSPREQNO, 
	   C.SITEID,
	   C.INSPDEFINITIONSYSID,
	   C.COPERATOR,
	   COP.COPERATORNAME,
	   C.ITEMID,
	   M.MATERIALDEFINITIONNAME AS ITEMNAME,
	   C.INSPTYPE,
	   CD1.CODENAME AS INSPTYPENAME,
	   C.INSPSPEC,
	   CD3.CODENAME AS INSPSPECNAME,
	   C.REV,
	   C.PROCESSNODEID,
	   C.PROCESSDEFINITIONID,
	   C.PROCESSSEGMENTID,
--	   PS.PROCESSSEGMENTNAME ,
	   C.STATE,
	   ST.STATENAME,
	   C.INSPMETHOD,
	   CD2.CODENAME AS INSPMETHODNAME,
	   C.REPEATCOUNT,
	   C.CURRENTCOUNT,
	   C.REQDATETIME,
	   C.RECEPTDATETIME,
	   C.FINALDATETIME,
	   C.FINALJUDGE,
	   C.ACTIVITY,
	   C.PREVACTIVITY,
	   C.CUSTOMACTIVITY,
	   C.PREVCUSTOMACTIVITY,
	   C.ISUSABLE,
	   C.DESCRIPTION,
	   C.REASONCODE,
	   C.COMMENTS,
	   C.CREATOR,
	   (SELECT FN_USERNAME(C.SITEID, C.CREATOR) FROM DUAL) AS CREATOR2,
	   C.CREATETIME,
	   C.MODIFIER,
	   (SELECT FN_USERNAME(C.SITEID, C.MODIFIER) FROM DUAL) AS MODIFIER2,
	   C.MODIFYTIME,
	   C.LASTEVENTTIME,
	   C.TID
  FROM CIM_INSPREQ C
	LEFT OUTER JOIN (
	  			SELECT VD.VENDORID AS CID, VD.VENDORNAME AS COPERATORNAME, VD.SITEID FROM CIM_VENDOR VD
	  			UNION ALL
	  			SELECT CT.CUSTOMERID AS CID, CT.CUSTOMERNAME AS COPERATORNAME, CT.SITEID FROM CIM_CUSTOMER CT
	  		   ) COP 
		ON C.SITEID = COP.SITEID
	  	AND C.COPERATOR = COP.CID
	INNER JOIN CIM_MATERIALDEFINITION M
		ON 1=1
		AND C.SITEID = M.SITEID 
		AND C.ITEMID = M.MATERIALDEFINITIONID
--	INNER JOIN CIM_PROCESSSEGMENT PS
--	   ON 1=1
--	  AND PS.SITEID = C.SITEID
--	  AND PS.PROCESSSEGMENTID = C.PROCESSSEGMENTID
--	  AND PS.ISUSABLE = 'Usable'
	INNER JOIN CIM_STATE ST 
		ON 1=1
		AND C.SITEID = ST.SITEID
		AND C.STATE = ST.STATEID
		AND ST.STATEMODELID = 'InspReqState'
	INNER JOIN CIM_CODE CD1
		ON 1=1
		AND C.SITEID = CD1.SITEID 
		AND C.INSPTYPE = CD1.CODEID
		AND CD1.CODECLASSID = 'INSPTYPE'
	INNER JOIN CIM_CODE CD2
		ON 1=1
		AND C.SITEID = CD2.SITEID 
		AND C.INSPMETHOD = CD2.CODEID
		AND CD2.CODECLASSID = 'INSPMETHOD'
	LEFT OUTER JOIN CIM_CODE CD3
		ON 1=1
		AND C.SITEID = CD3.SITEID 
		AND C.INSPSPEC= CD3.CODEID
		AND CD3.PARENTID = 'OQC'
 WHERE (:INSPREQNO IS NULL OR INSPREQNO LIKE :INSPREQNO || '%')
   AND (:COPERATOR IS NULL OR C.COPERATOR = :COPERATOR)
   AND (:INSPTYPE IS NULL OR C.INSPTYPE = :INSPTYPE)
   AND (:INSPSPEC IS NULL OR C.INSPSPEC = :INSPSPEC)
   AND (:ITEMID IS NULL OR C.ITEMID = :ITEMID)
   AND (:INSPMETHOD IS NULL OR C.INSPMETHOD = :INSPMETHOD)
   AND (:STATE IS NULL OR C.STATE IN (SELECT * FROM TABLE(FN_MES_SPLIT(:STATE, ','))))
   AND (:REQDATETIME IS NULL OR C.REQDATETIME BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 0.99999)
   AND (:RECEPTDATETIME IS NULL OR C.RECEPTDATETIME BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 0.99999)
   AND (:ISUSABLE IS NULL OR C.ISUSABLE = :ISUSABLE)
   AND C.SITEID = :SITEID
ORDER BY C.INSPREQNO, C.REQDATETIME
테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

SELECT C.*, (SELECT FN_USERNAME(C.SITEID, C.CREATOR) FROM DUAL) AS CREATOR2, (SELECT FN_USERNAME(C.SITEID, C.MODIFIER) FROM DUAL) AS MODIFIER2

  FROM CIM_INSPDEFINITIONITEMSPEC C

 WHERE 1=1

   AND (:INSPDEFINITIONSYSID IS NULL OR C.INSPDEFINITIONSYSID = :INSPDEFINITIONSYSID)

   AND (:ISUSABLE IS NULL OR C.ISUSABLE = :ISUSABLE)

   AND C.SITEID = :SITEID

ORDER BY C.INSPDEFINITIONSYSID, C.UISEQUENCE

CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
