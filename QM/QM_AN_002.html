--QM_AN_002_검사 의뢰 요청 - 검사의뢰 요청 목록 조회 쿼리
--SelectInspectionRequestList
--Param : SITEID, FROMDATE, TODATE, INSPMETHOD, INSPTYPE, SAMPLELOTID, EQUIPMENTID, CREATOR
------------------------------------------
SELECT A.INSPREQNO,
       A.INSPMETHOD,
       A.INSPTYPE,
       A.COPERATOR,
       A.ITEMID,
       CASE WHEN @INSPTYPE = 'IQC' THEN H.MATERIALDEFINITIONNAME 
            WHEN @INSPTYPE = 'OQC' THEN I.PRODUCTDEFINITIONNAME
            ELSE '' END AS ITEMNAME,
       A.INSPDEFINITIONSYSID,
       G.INSPDEFINITIONNAME,
       A.STATE,
       B.STATENAME,
       D.EQUIPMENTCLASSID,
       A.EQUIPMENTID,
       C.EQUIPMENTNAME,
       A.SAMPLELOTID,
       A.SAMPLEPOINTID,
       A.CREATOR AS CREATORID,
       ISNULL(E.USERNAME, A.CREATOR) AS CREATOR,
       A.REQDATETIME AS CREATETIME,
       A.COMMENTS,
       A.MODIFIER AS MODIFIERID,
       F.USERNAME AS MODIFIER,
       A.MODIFYTIME,
       K.WORKORDERID,
       A.LOTID,
       A.SAMPLEID
  FROM CIM_INSPREQ A WITH(NOLOCK)
  LEFT JOIN CIM_STATE B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.STATE = B.STATEID
   AND B.STATEMODELID = 'InspReqState'
  LEFT JOIN CIM_EQUIPMENT C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.EQUIPMENTID = C.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS D WITH(NOLOCK)
    ON C.SITEID = D.SITEID
   AND C.EQUIPMENTCLASSID = D.EQUIPMENTCLASSID
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID 
   AND A.CREATOR = E.USERID
  LEFT JOIN CIM_USER F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.MODIFIER = F.USERID
  LEFT JOIN CIM_INSPDEFINITION G WITH(NOLOCK)
    ON A.SITEID = G.SITEID
   AND A.INSPDEFINITIONSYSID = G.INSPDEFINITIONSYSID
  LEFT JOIN CIM_MATERIALDEFINITION H WITH(NOLOCK)
    ON A.SITEID = H.SITEID
   AND A.ITEMID = H.MATERIALDEFINITIONID
  LEFT JOIN CIM_PRODUCTDEFINITION I WITH(NOLOCK)
    ON A.SITEID = I.SITEID
   AND A.ITEMID = I.PRODUCTDEFINITIONID
  LEFT JOIN CIM_LOT J WITH(NOLOCK)
    ON A.SITEID = J.SITEID
   AND A.LOTID = J.LOTID
  LEFT JOIN CIM_WORKORDER K WITH(NOLOCK)
    ON J.SITEID = K.SITEID
   AND J.WORKORDERID = K.WORKORDERID
 WHERE A.SITEID = @SITEID
   AND A.STATE = 'Request'
   AND A.REQDATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@INSPMETHOD = '' OR A.INSPMETHOD = @INSPMETHOD)
   AND (@INSPTYPE = '' OR A.INSPTYPE = @INSPTYPE)
   AND (@SAMPLELOTID = '' OR A.SAMPLELOTID LIKE '%' + @SAMPLELOTID + '%')
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)
   AND (@SAMPLEPOINTID = '' OR A.SAMPLEPOINTID = @SAMPLEPOINTID)
   AND (@CREATOR = '' OR E.USERNAME LIKE '%' + @CREATOR + '%')
 ORDER BY A.MODIFYTIME DESC, A.CREATETIME DESC

--QM_AN_002_검사 의뢰 요청 EQP&LQC 검사 사양 조회 쿼리
--SelectInspdefinitionsysidByInspReq2
--Param : SITEID, COPERATOR, ITEMID, SAMPLEPOINTID
------------------------------------------
SELECT INSPDEFINITIONSYSID AS CODE,
       INSPDEFINITIONNAME AS NAME
  FROM CIM_INSPDEFINITION WITH(NOLOCK)
 WHERE SITEID = @SITEID
   AND INSPTYPE IN('EQP','LQC')
   AND (@COPERATOR = '' OR COPERATOR = @COPERATOR)
   AND (ITEMID = @ITEMID OR ITEMID = @SAMPLEPOINTID)
   AND ISUSABLE = 'Usable'
 ORDER BY CODE
