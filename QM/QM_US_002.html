--QM_US_002_검교정 관리 - 검사기 그룹 조회 쿼리
--SelectCalibrationGroupList
--Param : SITEID, LINEID, CALIBRATIONCLASSNAME, ISUSABLE
------------------------------------------
SELECT A.CALIBRATIONCLASSID,
       A.CALIBRATIONCLASSNAME,
       A.LINEID,
       A.ISUSABLE,
       A.CREATOR AS CREATORID,
       B.USERNAME AS CREATOR,
       A.CREATETIME,
       A.MODIFIER AS MODIFIERID,
       C.USERNAME AS MODIFIER,
       A.MODIFYTIME  
  FROM CST_CALIBRATIONCLASS A WITH(NOLOCK)
  LEFT JOIN CIM_USER B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.CREATOR = B.USERID
  LEFT JOIN CIM_USER C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.MODIFIER = C.USERID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND (@CALIBRATIONCLASSID= '' OR A.CALIBRATIONCLASSID= @CALIBRATIONCLASSID)
   AND (@ISUSABLE = '' OR A.ISUSABLE = @ISUSABLE)
 ORDER BY A.CREATETIME

--QM_US_002_검교정 관리 - 검사기 목록 조회 쿼리
--SelectCalibrationList
--Param : SITEID, LINEID, CALIBRATIONCLASSID
------------------------------------------
SELECT A.CALIBRATIONDEFINITIONID,
	   A.CALIBRATIONDEFINITIONNAME,
	   A.CALIBRATIONCLASSID,
	   G.CODENAME AS CALIBRATIONCLASSNAME,
	   A.CALIBRATIONTYPE,
	   A.LOCATION,
	   A.MEASURERANGE,
	   A.INTERVARLDAY,
	   A.CALIBRATIONCYCLE,
	   A.CALIBRATIONDATE,
	   C.CORRECTIONID,
            A.CALIBRATIONNEXTDATE,
	   A.CALIBRATIONMODEL,
	   A.EQUIPMENTID,
	   IIF(A.EQUIPMENTNAME IS NOT NULL, A.EQUIPMENTNAME, D.EQUIPMENTNAME) AS EQUIPMENTNAME,
	   A.DEPTID,
	   A.ISUSABLE,
	   A.CREATOR AS CREATORID,
	   E.USERNAME AS CREATOR,
	   A.CREATETIME,
	   A.MODIFIER AS MODIFERID,
	   F.USERNAME AS MODIFIER,
	   A.MODIFYTIME
  FROM CST_CALIBRATIONDEFINITION A WITH(NOLOCK)
 INNER JOIN CST_CALIBRATIONCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID 
   AND A.LINEID = B.LINEID
   AND A.CALIBRATIONCLASSID = B.CALIBRATIONCLASSID
  LEFT JOIN (
	  SELECT CR.*
	    FROM (
	    	SELECT ROW_NUMBER() OVER(PARTITION BY CALIBRATIONDEFINITIONID ORDER BY CREATETIME DESC) AS ROWNUM, CORRECTIONID, CALIBRATIONDEFINITIONID, SITEID
		  	  FROM CST_CALIBRATIONRESULT WITH(NOLOCK)
	    ) CR
	   WHERE CR.ROWNUM = 1
  ) C
    ON A.SITEID = C.SITEID
   AND A.CALIBRATIONDEFINITIONID = C.CALIBRATIONDEFINITIONID
  LEFT JOIN CIM_EQUIPMENT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.EQUIPMENTID = D.EQUIPMENTID
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.CREATOR = E.USERID
  LEFT JOIN CIM_USER F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.MODIFIER = F.USERID
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON A.SITEID = G.SITEID
   AND A.CALIBRATIONCLASSID = G.CODEID
   AND G.CODECLASSID = 'CALIBRATIONGROUP' 
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.CALIBRATIONCLASSID = @CALIBRATIONCLASSID
 ORDER BY A.CREATEDATE
