--QM_AN_003_검사 의뢰 접수 - 검사의뢰 요청 목록 조회 쿼리

--SelectInspectionReceiptList

--Param : SITEID, FROMDATE, TODATE, INSPMETHOD, INSPTYPE, STATE, SAMPLELOTID, EQUIPMENTID, CREATOR, INSPRECEIPTUSERID

------------------------------------------

SELECT A.INSPREQNO,

       A.INSPMETHOD,

       D.CODENAME AS INSPMETHODNAME,

       A.INSPTYPE,

       A.ITEMID,

       CASE WHEN @INSPTYPE = 'IQC' THEN M.MATERIALDEFINITIONNAME 

            WHEN @INSPTYPE = 'OQC' THEN N.PRODUCTDEFINITIONNAME

            ELSE '' END AS ITEMNAME,

       A.COPERATOR,

       A.STATE,

       E.STATENAME,

       C.EQUIPMENTCLASSID,

       A.EQUIPMENTID,

       B.EQUIPMENTNAME,

       A.SAMPLELOTID,

       A.SAMPLEPOINTID,

       K.SAMPLEPOINTNAME,

       A.CREATOR AS CREATORID,

       ISNULL(F.USERNAME, A.CREATOR) AS CREATOR,

       A.CREATETIME,

       A.COMMENTS,

       A.MODIFIER AS MODIFIERID,

       G.USERNAME AS MODIFIER,

       A.MODIFYTIME,

       A.INSPRECEIPTUSERID,

       H.USERNAME AS INSPRECEIPTUSERNAME,

       A.RECEPTDATETIME AS INSPRECEIPTTIME,

       A.INSPSTARTUSERID,

       I.USERNAME AS INSPSTARTUSERNAME,

       A.INSPSTARTTIME,

       A.REASONCODE AS REASONCODEID,

       J.CODENAME AS REASONCODE,

       A.DESCRIPTION,

       P.WORKORDERID,

       A.LOTID,

       A.INSPDEFINITIONSYSID,

       L.INSPDEFINITIONNAME,

       A.SAMPLEID

  FROM CIM_INSPREQ A WITH(NOLOCK)

  LEFT JOIN CIM_EQUIPMENT B WITH(NOLOCK)

    ON A.SITEID = B.SITEID

   AND A.EQUIPMENTID = B.EQUIPMENTID

  LEFT JOIN CIM_EQUIPMENTCLASS C WITH(NOLOCK)

    ON B.SITEID = C.SITEID

   AND B.EQUIPMENTCLASSID = C.EQUIPMENTCLASSID

  LEFT JOIN CIM_CODE D WITH(NOLOCK)

    ON A.SITEID = D.SITEID

   AND A.INSPMETHOD = D.CODEID

   AND D.CODECLASSID = 'REQUESTCATEGORY'

  LEFT JOIN CIM_STATE E WITH(NOLOCK)

    ON A.SITEID = E.SITEID

   AND A.STATE = E.STATEID

   AND E.STATEMODELID = 'InspReqState'

  LEFT JOIN CIM_USER F WITH(NOLOCK)

    ON A.SITEID = F.SITEID 

   AND A.CREATOR = F.USERID

  LEFT JOIN CIM_USER G WITH(NOLOCK)

    ON A.SITEID = G.SITEID

   AND A.MODIFIER = G.USERID

  LEFT JOIN CIM_USER H WITH(NOLOCK)

    ON A.SITEID = H.SITEID

   AND A.INSPRECEIPTUSERID = H.USERID

  LEFT JOIN CIM_USER I WITH(NOLOCK)

    ON A.SITEID = I.SITEID

   AND A.INSPSTARTUSERID = I.USERID

  LEFT JOIN CIM_CODE J WITH(NOLOCK)

    ON A.SITEID = J.SITEID

   AND A.REASONCODE = J.CODEID

   AND J.CODECLASSID = 'REJECTCODE'

  LEFT JOIN CST_SAMPLEPOINT K WITH(NOLOCK)

    ON A.SITEID = K.SITEID

   AND A.SAMPLEPOINTID = K.SAMPLEPOINTID

  LEFT JOIN CIM_INSPDEFINITION L WITH(NOLOCK)

    ON A.SITEID = L.SITEID

   AND A.INSPDEFINITIONSYSID = L.INSPDEFINITIONSYSID

  LEFT JOIN CIM_MATERIALDEFINITION M WITH(NOLOCK)

    ON A.SITEID = M.SITEID

   AND A.ITEMID = M.MATERIALDEFINITIONID

  LEFT JOIN CIM_PRODUCTDEFINITION N WITH(NOLOCK)

    ON A.SITEID = N.SITEID

   AND A.ITEMID = N.PRODUCTDEFINITIONID

  LEFT JOIN CIM_LOT O WITH(NOLOCK)

    ON A.SITEID = O.SITEID

   AND A.LOTID = O.LOTID

  LEFT JOIN CIM_WORKORDER P WITH(NOLOCK)

    ON O.SITEID = P.SITEID

   AND O.WORKORDERID = P.WORKORDERID

 WHERE A.SITEID = @SITEID

   AND A.STATE IN('Request', 'Receipt', 'Inspection', 'Return')

   AND A.CREATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)

   AND (@INSPMETHOD = '' OR A.INSPMETHOD = @INSPMETHOD)

   AND (@INSPTYPE = '' OR A.INSPTYPE = @INSPTYPE)

   AND (@STATE = '' OR A.STATE = @STATE)

   AND (@SAMPLELOTID = '' OR A.SAMPLELOTID LIKE '%' + @SAMPLELOTID + '%')

   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)

   AND (@CREATOR = '' OR F.USERNAME LIKE '%' + @CREATOR + '%')

   --AND (@INSPRECEIPTUSERID = '' OR A.INSPRECEIPTUSERID LIKE '%' + @INSPRECEIPTUSERID + '%')

   -- 2026.04.30 박성진 - 화면에서 요청파라미터의 값이 'USERNAME'이므로 해당 건이 CIM_USER의 USERNAME으로 찾을수있게 수정

   AND (

       LTRIM(RTRIM(ISNULL(@INSPRECEIPTUSERID, ''))) = ''

    OR A.INSPRECEIPTUSERID LIKE '%' + LTRIM(RTRIM(@INSPRECEIPTUSERID)) + '%'

    OR A.INSPRECEIPTUSERID IN (

           SELECT U.USERID

             FROM CIM_USER U WITH(NOLOCK)

            WHERE U.SITEID = A.SITEID

              AND U.USERNAME LIKE '%' + LTRIM(RTRIM(@INSPRECEIPTUSERID)) + '%'

       )

)



 ORDER BY A.LASTEVENTTIME DESC, A.CREATETIME DESC

------------------------------------------
SELECT CI.INSPREQNO 	AS INSPREQNO 
	 , CI.LOTID 		AS LOTID 
	 , CS.EQUIPMENTID	AS EQUIPMENTID 
	 , CONVERT(VARCHAR, CI.REQDATETIME, 120) AS REQDATETIME
	 , CI.INSPTYPE  	AS INSPTYPE
	 , CI.ERPINSPTYPE   AS ERPINSPTYPE 
FROM CIM_INSPREQ CI WITH (NOLOCK)
LEFT OUTER JOIN CST_SAMPLEPOINT CS ON CI.SITEID = CS.SITEID AND CI.SAMPLEPOINTID = CS.SAMPLEPOINTID 
WHERE CI.SITEID = @SITEID
AND CI.SAMPLELOTID = @SAMPLELOTID
