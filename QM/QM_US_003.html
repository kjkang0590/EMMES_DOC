--QM_US_003_검교정 이력 조회 쿼리
--SelectCalibrationResultList
--Param : SITEID, LINEID, FROMDATE, TODATE, CALIBRATIONCLASSID, CALIBRATIONDEFINITIONID, DEPTID, ISUSABLE
------------------------------------------
SELECT B.EQUIPMENTID,
	   IIF(B.EQUIPMENTNAME IS NOT NULL, B.EQUIPMENTNAME, F.EQUIPMENTNAME) AS EQUIPMENTNAME,
	   C.CALIBRATIONCLASSNAME,
	   G.CODENAME AS CALIBRATIONTYPE,
	   B.LOCATION,
	   A.CALIBRATIONDATE,
	   A.CORRECTIONID,
	   B.INTERVARLDAY,
	   H.CODENAME AS CALIBRATIONCYCLE,
	   CASE WHEN B.CALIBRATIONCYCLE = 'D' THEN DATEADD(DAY, B.INTERVARLDAY, A.CALIBRATIONDATE)
	        WHEN B.CALIBRATIONCYCLE = 'W' THEN DATEADD(WK, B.INTERVARLDAY, A.CALIBRATIONDATE)
	        WHEN B.CALIBRATIONCYCLE = 'M' THEN DATEADD(MONTH, B.INTERVARLDAY, A.CALIBRATIONDATE)
	        WHEN B.CALIBRATIONCYCLE = 'Y' THEN DATEADD(YEAR, B.INTERVARLDAY, A.CALIBRATIONDATE)
	        ELSE NULL END AS CALIBRATIONNEXTDATE,
	   B.CALIBRATIONMODEL,
	   B.MEASURERANGE,
	   D.CODENAME AS DEPTNAME,
	   E.CODENAME AS ISUSABLE
  FROM CST_CALIBRATIONRESULT A WITH(NOLOCK)
  LEFT JOIN CST_CALIBRATIONDEFINITION B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.LINEID = B.LINEID
   AND A.CALIBRATIONDEFINITIONID = B.CALIBRATIONDEFINITIONID
  LEFT JOIN CST_CALIBRATIONCLASS C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.LINEID = C.LINEID
   AND B.CALIBRATIONCLASSID = C.CALIBRATIONCLASSID
  LEFT JOIN CIM_CODE D WITH(NOLOCK)
    ON D.CODECLASSID = 'DEPT'
   AND B.SITEID = D.SITEID
   AND B.DEPTID = D.CODEID
  LEFT JOIN CIM_CODE E WITH(NOLOCK)
    ON E.CODECLASSID = 'ISUSABLE'
   AND B.SITEID = E.SITEID
   AND B.ISUSABLE = E.CODEID
  LEFT JOIN CIM_EQUIPMENT F WITH(NOLOCK)
    ON B.SITEID = F.SITEID
   AND B.EQUIPMENTID = F.EQUIPMENTID
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON B.SITEID = G.SITEID
   AND B.CALIBRATIONTYPE = G.CODEID
   AND G.CODECLASSID = 'CALIBRATIONTYPE'
  LEFT JOIN CIM_CODE H WITH(NOLOCK)
    ON B.SITEID = H.SITEID
   AND B.CALIBRATIONCYCLE = H.CODEID
   AND H.CODECLASSID = 'CALIBRATIONCYCLE'
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.LASTEVENTTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@CALIBRATIONCLASSID = '' OR C.CALIBRATIONCLASSID = @CALIBRATIONCLASSID)
   AND (@EQUIPMENTID = '' OR B.EQUIPMENTID = @EQUIPMENTID)
   AND (@DEPTID = '' OR B.DEPTID = @DEPTID)
   AND (@ISUSABLE = '' OR B.ISUSABLE = @ISUSABLE)
 ORDER BY A.LASTEVENTTIME

--SelectCboCalibrationGroupList
--Param : SITEID, LINEID
------------------------------------------
SELECT A.CALIBRATIONCLASSID AS CODE,
	   B.CODENAME AS NAME
  FROM CST_CALIBRATIONCLASS A WITH(NOLOCK)
  LEFT JOIN CIM_CODE B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.CALIBRATIONCLASSID = B.CODEID
   AND B.CODECLASSID = 'CALIBRATIONGROUP'
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.ISUSABLE = 'Usable'
 ORDER BY A.CALIBRATIONCLASSID
