--QM_AN_004_검사 결과 등록 - 검사목록 조회 쿼리
--SelectInspectionConfirmList
--Param : SITEID, FROMDATE, TODATE, INSPMETHOD, STATE, FINALJUDGE, SAMPLELOTID, LOTID, EQUIPMENTID, INSPENDUSERID
------------------------------------------
SELECT A.INSPREQNO
     , A.INSPMETHOD
     , J.CODENAME AS INSPMETHODNAME
     , A.INSPTYPE AS INSPTARGET
     , A.ITEMID
     , CASE WHEN @INSPTYPE = 'IQC' THEN H.MATERIALDEFINITIONNAME 
            WHEN @INSPTYPE = 'OQC' THEN I.PRODUCTDEFINITIONNAME
            ELSE '' END AS ITEMNAME
     , A.SAMPLELOTID
     , A.INSPDEFINITIONSYSID
     , C.INSPDEFINITIONNAME
     , A.STATE
     , K.STATENAME
     , G.WORKORDERID
     , A.LOTID
     , E.EQUIPMENTCLASSID
     , A.EQUIPMENTID
     , D.EQUIPMENTNAME
     , A.SAMPLEPOINTID
     , A.FINALDATETIME
     , A.INSPENDUSERID
     , L.USERNAME AS INSPENDUSERNAME
     , CASE WHEN A.REPEATCOUNT IS NULL THEN 1 ELSE A.REPEATCOUNT + 1 END AS REPARTCOUNT
     , A.FINALJUDGE
     , A.CREATETIME
     , A.INSPENDTIME
     , A.SAMPLEID
     , M.USERNAME AS CREATOR
     , A.COMMENTS
  FROM CIM_INSPREQ A WITH(NOLOCK)
  LEFT JOIN CIM_INSPDEFINITION C WITH(NOLOCK) ON A.SITEID = C.SITEID AND A.INSPDEFINITIONSYSID = C.INSPDEFINITIONSYSID
  LEFT JOIN CIM_EQUIPMENT D WITH(NOLOCK) ON A.SITEID = D.SITEID AND A.EQUIPMENTID = D.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS E WITH(NOLOCK) ON D.SITEID = E.SITEID AND D.EQUIPMENTCLASSID = E.EQUIPMENTCLASSID
  LEFT JOIN CIM_LOT F WITH(NOLOCK) ON A.SITEID = F.SITEID AND A.LOTID = F.LOTID
  LEFT JOIN CIM_WORKORDER G WITH(NOLOCK) ON F.SITEID = G.SITEID AND F.WORKORDERID = G.WORKORDERID
  LEFT JOIN CIM_MATERIALDEFINITION H WITH(NOLOCK) ON A.SITEID = H.SITEID AND A.ITEMID = H.MATERIALDEFINITIONID
  LEFT JOIN CIM_PRODUCTDEFINITION I WITH(NOLOCK) ON A.SITEID = I.SITEID AND A.ITEMID = I.PRODUCTDEFINITIONID
  LEFT JOIN CIM_CODE J WITH(NOLOCK) ON A.SITEID = J.SITEID AND A.INSPMETHOD = J.CODEID AND J.CODECLASSID = 'REQUESTCATEGORY'
  LEFT JOIN CIM_STATE K WITH(NOLOCK) ON A.SITEID = K.SITEID AND A.STATE = K.STATEID AND K.STATEMODELID = 'InspReqState'
  LEFT JOIN CIM_USER L WITH(NOLOCK) ON A.SITEID = L.SITEID AND A.INSPENDUSERID = L.USERID
  LEFT JOIN CIM_USER M WITH(NOLOCK) ON A.SITEID = M.SITEID AND A.CREATOR = M.USERID
 WHERE A.SITEID = @SITEID
   AND A.STATE IN('Inspection', 'Completed')
   AND A.INSPSTARTTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@INSPMETHOD = '' OR A.INSPMETHOD = @INSPMETHOD)
   AND (@STATE = '' OR A.STATE = @STATE)
   AND (@FINALJUDGE = '' OR A.FINALJUDGE = @FINALJUDGE)
   AND (UPPER(@SAMPLELOTID) = '' OR UPPER(A.SAMPLELOTID) LIKE '%' + UPPER(@SAMPLELOTID) + '%')
   AND (@LOTID = '' OR A.LOTID LIKE '%' + @LOTID + '%')
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)
   AND (@INSPENDUSERID = '' OR L.USERNAME LIKE '%' + @INSPENDUSERID + '%')
   AND (@INSPTYPE = '' OR A.INSPTYPE = @INSPTYPE)
  ORDER BY A.INSPREQNO

CREATE PROCEDURE [dbo].[SP_QM_AN_004_SELECTINSPDESCRESULTLIST]

@SITEID VARCHAR(40), @INSPREQNO VARCHAR(40), @INSPDEFINITIONSYSID VARCHAR(40)

AS 



	DECLARE @COLNAMES NVARCHAR(MAX) = ''

	DECLARE @SQL NVARCHAR(MAX) = ''

	

	SELECT @COLNAMES = @COLNAMES + '[' + II.INSPITEMNAME + '],'

	  FROM CIM_INSPRESULT AS IR

	 INNER JOIN CIM_INSPDEFINITIONITEMSPEC AS IIS

	 	ON IIS.SITEID = IR.SITEID

	   AND IIS.INSPDEFINITIONITEMSPECSYSID = IR.INSPDEFINITIONITEMSPECSYSID

	   AND IR.INSPREQNO = @INSPREQNO

	 INNER JOIN CIM_INSPITEM II

	    ON II.SITEID = IIS.SITEID AND II.INSPITEMID = IIS.INSPITEMID

	 WHERE IIS.SITEID = @SITEID

	   AND IIS.INSPDEFINITIONSYSID = @INSPDEFINITIONSYSID

	   --AND IIS.ISUSABLE = 'Usable'

     ORDER BY IIS.UISEQUENCE

	

	IF (@COLNAMES = '')

		BEGIN

			SET @COLNAMES = NULL

		END

	ELSE

		BEGIN

			SET @COLNAMES = LEFT(@COLNAMES, LEN(@COLNAMES)-1)

		END

	

    IF (@COLNAMES IS NOT NULL)

    	BEGIN

			SET @SQL = '

			SELECT CASE WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''INSPITEMCLASSNAME'' THEN ''검사유형''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''INSPITEMNAME'' THEN ''검사항목''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''UNITID'' THEN ''단위코드''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''UNITNAME'' THEN ''단위''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''INSPITEMVALUE'' THEN ''검사값''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''FINALJUDGE'' THEN ''OK/NG''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''INSPDEFINITIONSPECSYSID'' THEN ''검사항목스펙SYSID''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''ISUSABLE'' THEN ''사용여부''

						WHEN SUBSTRING(ITEM, 5, LEN(ITEM)) = ''TARGET'' THEN ''Target''

						ELSE SUBSTRING(ITEM, 5, LEN(ITEM)) END AS 검사항목, ' + @COLNAMES + ',

						ROW_NUMBER() OVER(ORDER BY ORDERSEQ) AS ROWNUM

			  FROM (SELECT *, SUBSTRING(ITEM, 1, 3) AS ORDERSEQ

			          FROM (SELECT II.INSPITEMNAME

								, CONVERT(VARCHAR, IIC.INSPITEMCLASSNAME) AS A01_INSPITEMCLASSNAME

								, CONVERT(VARCHAR, II.INSPITEMNAME) AS A02_INSPITEMNAME

								, CONVERT(VARCHAR, IR.UNITID) AS A03_UNITID

								, IIF(CONVERT(VARCHAR, IR.UNITID) IS NOT NULL, CONVERT(VARCHAR, CCR.UNITNAME), '''') AS A03_UNITNAME

								, IIF(CONVERT(VARCHAR, IR.USL) IS NOT NULL, CONVERT(VARCHAR, IR.USL), CONVERT(VARCHAR, IDS.USL)) AS A04_USL

			                    , IIF(CONVERT(VARCHAR, IR.LSL) IS NOT NULL, CONVERT(VARCHAR, IR.LSL), CONVERT(VARCHAR, IDS.LSL)) AS A05_LSL

								, IIF(CONVERT(VARCHAR, IR.TARGET) IS NOT NULL, CONVERT(VARCHAR, IR.TARGET), CONVERT(VARCHAR, IDS.TARGET)) AS A06_TARGET

								, CONVERT(VARCHAR, IIF(IR.INSPITEMVALUE IS NULL, '''', IR.INSPITEMVALUE)) AS A07_INSPITEMVALUE

								, CONVERT(VARCHAR, (CASE WHEN CONVERT(FLOAT, CONVERT(DECIMAL(38, 4), INSPITEMVALUE)) >= IIF(CONVERT(FLOAT, IR.LSL) IS NOT NULL, CONVERT(FLOAT, IR.LSL), CONVERT(FLOAT, IDS.LSL))

														  AND CONVERT(FLOAT, CONVERT(DECIMAL(38, 4), INSPITEMVALUE)) <= IIF(CONVERT(FLOAT, IR.USL) IS NOT NULL, CONVERT(FLOAT, IR.USL), CONVERT(FLOAT, IDS.USL)) THEN ' + '''OK''' + '

														 WHEN IIF(CONVERT(FLOAT, IR.LSL) IS NOT NULL, CONVERT(FLOAT, IR.LSL), CONVERT(FLOAT, IDS.LSL)) IS NULL

														  AND CONVERT(FLOAT, CONVERT(DECIMAL(38, 4), INSPITEMVALUE)) <= IIF(CONVERT(FLOAT, IR.USL) IS NOT NULL, CONVERT(FLOAT, IR.USL), CONVERT(FLOAT, IDS.USL)) THEN ' + '''OK''' + '

														 WHEN IIF(CONVERT(FLOAT, IR.USL) IS NOT NULL, CONVERT(FLOAT, IR.USL), CONVERT(VARCHAR, IDS.USL)) IS NULL

														  AND CONVERT(FLOAT, CONVERT(DECIMAL(38, 4), INSPITEMVALUE)) >= IIF(CONVERT(FLOAT, IR.LSL) IS NOT NULL, CONVERT(FLOAT, IR.LSL), CONVERT(FLOAT, IDS.LSL)) THEN ' + '''OK''' + '

														 WHEN IIF(CONVERT(FLOAT, IR.LSL) IS NOT NULL, CONVERT(FLOAT, IR.LSL), CONVERT(FLOAT, IDS.LSL)) IS NULL 

														  AND IIF(CONVERT(FLOAT, IR.USL) IS NOT NULL, CONVERT(FLOAT, IR.USL), CONVERT(FLOAT, IDS.USL)) IS NULL THEN ' + '''OK''' + '

														 ELSE ' + '''NG''' + '

													END)) AS A08_FINALJUDGE

								, IIF(CONVERT(VARCHAR, IR.INSPDEFINITIONITEMSPECSYSID) IS NOT NULL, CONVERT(VARCHAR, IR.INSPDEFINITIONITEMSPECSYSID), CONVERT(VARCHAR, IDS.INSPDEFINITIONITEMSPECSYSID)) AS A09_INSPDEFINITIONSPECSYSID

								, IIF(CONVERT(VARCHAR, IR.ISUSABLE) IS NOT NULL, CONVERT(VARCHAR, IR.ISUSABLE), CONVERT(VARCHAR, IDS.ISUSABLE)) AS A10_ISUSABLE

			                  FROM CIM_INSPRESULT IR

			                  LEFT OUTER JOIN CIM_INSPDEFINITIONITEMSPEC IDS

			                    ON IDS.SITEID = IR.SITEID

			                   AND IDS.INSPDEFINITIONITEMSPECSYSID = IR.INSPDEFINITIONITEMSPECSYSID

							   AND IR.INSPREQNO = ''' + @INSPREQNO + '''

			                  LEFT OUTER JOIN CIM_INSPITEM II

			                    ON II.SITEID = IDS.SITEID

			                   AND II.INSPITEMID = IDS.INSPITEMID

							  LEFT OUTER JOIN CIM_INSPITEMCLASS IIC

							    ON IIC.SITEID = II.SITEID

							   AND IIC.INSPITEMCLASSID = II.INSPITEMCLASSID

							  LEFT OUTER JOIN CIM_UNIT CCR

								ON CCR.SITEID = IR.SITEID

							   AND CCR.UNITID = IR.UNITID

			                 WHERE IDS.SITEID = ''' + @SITEID + '''

							   AND IDS.INSPDEFINITIONSYSID = ''' + @INSPDEFINITIONSYSID + '''

			               ) AS INSPRESULT

			       UNPIVOT ([ITEMVALUE] FOR ITEM IN ([A01_INSPITEMCLASSNAME],[A02_INSPITEMNAME],[A03_UNITID],[A03_UNITNAME],[A04_USL],[A05_LSL],[A06_TARGET],[A07_INSPITEMVALUE],[A08_FINALJUDGE],[A09_INSPDEFINITIONSPECSYSID],[A10_ISUSABLE])) AS UNPIVOT_INSPRESULT

			       ) AS UNPIVOT_INSP

			 PIVOT (MAX(ITEMVALUE) FOR [INSPITEMNAME] IN (' + @COLNAMES + ')) AS PIVOT_INSPRESULT

			'

		

		EXEC (@SQL)

		END

	ELSE

		BEGIN

			SET @SQL = 'SELECT ''S'' AS errCode' 

  	  		EXEC (@SQL)

		END;
