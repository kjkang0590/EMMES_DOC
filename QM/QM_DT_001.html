#1 Query SelectDeltavBatchList

연결된 기능 (1번 표): #1 조회 - 그리드 QM_DT_0011 (델타V관리1_Batch목록)

Query (조회 서비스) : SelectDeltavBatchList

  • ID 구성: SelectDeltavBatchList / 클래스 THiRAMES.Service.UI / 버전 00001
  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)
  • 쿼리 파라메터: PLANTLINEID SITEID FROMDATE TODATE PRODUCTDEFINITIONID WORKORDERTYPE PLANTID CONTROLBATCHID BATCHID
  • 테이블(조회/쓰기 전체, 함수·SP 포함):
    CUS_FOMULARESULT메인읽기   CIM_LOTTRACKING읽기   CIM_PROCESSSEGMENT읽기   CIM_LOTBATCHREL읽기   CIM_BATCH읽기   CIM_WORKORDER읽기   CIM_EQUIPMENT읽기   CIM_FACILITY읽기
조인 구조도 테이블 6 · 조인 5
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_PROCESSSEGMENT, CIM_EQUIPMENT
SQL 보기
-- SelectDeltavBatchList

SELECT EQ.PLANTID 
     , FC.PARENTFACILITYID PLANTLINEID
     , PS.PROCESSSEGMENTNAME 
     , FR.EQUIPMENTID
     , EQ.EQUIPMENTNAME 
     , FR.CONTROLBATCHID
     , FR.BATCHID
     , LT.PRODUCTDEFINITIONID
     , BT.WORKORDERID
     , SUM(CASE WHEN FR.JUDGE = 'OK' THEN 1 ELSE 0 END) AS GOODQTY
     , SUM(CASE WHEN FR.JUDGE = 'NG' THEN 1 ELSE 0 END) AS DEFECTQTY
  FROM CUS_FOMULARESULT FR 
 INNER JOIN CIM_LOTTRACKING LT ON LT.SITEID = FR.SITEID AND LT.LOTID = FR.RESULTLOTID AND FR.PROCESSNODEID = LT.PROCESSNODEID 
 INNER JOIN CIM_PROCESSSEGMENT PS ON LT.SITEID = PS.SITEID AND LT.PROCESSSEGMENTID = PS.PROCESSSEGMENTID
 INNER JOIN CIM_LOTBATCHREL LBR ON LBR.SITEID = LT.SITEID AND LBR.LOTID = LT.LOTID 
 INNER JOIN CIM_BATCH BT ON BT.SITEID = LBR.SITEID AND BT.BATCHID = LBR.BATCHID AND BT.BATCHTYPE = PS.PROCESSTYPE 
 INNER JOIN CIM_WORKORDER WO ON WO.SITEID = BT.SITEID AND WO.WORKORDERID = BT.WORKORDERID 
 INNER JOIN CIM_EQUIPMENT EQ ON EQ.SITEID = FR.SITEID AND EQ.EQUIPMENTID = FR.EQUIPMENTID  
 INNER JOIN CIM_FACILITY FC ON FC.SITEID = LT.SITEID AND FC.PARENTFACILITYID = :PLANTLINEID AND LINEGROUP = '5700'
 WHERE FR.SITEID = :SITEID 
   AND FR.MODIFYTIME >= TO_DATE(:FROMDATE, 'YYYY-MM-DD')
   AND FR.MODIFYTIME <  TO_DATE(:TODATE, 'YYYY-MM-DD') + 1
   AND (:PRODUCTDEFINITIONID IS NULL OR LT.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID)
   AND (:WORKORDERTYPE IS NULL OR WO.WORKORDERTYPE = :WORKORDERTYPE)
   AND EQ.PLANTID = :PLANTID
   AND (:CONTROLBATCHID IS NULL OR FR.CONTROLBATCHID = :CONTROLBATCHID)
   AND (:BATCHID IS NULL OR FR.BATCHID = :BATCHID)
   AND FR.FOMULADEFINITIONID = 'FORMATIONCELLDELTAV'
 GROUP BY EQ.PLANTID, FC.PARENTFACILITYID, PS.PROCESSSEGMENTNAME, FR.EQUIPMENTID, EQ.EQUIPMENTNAME, FR.CONTROLBATCHID, FR.BATCHID, LT.PRODUCTDEFINITIONID, BT.WORKORDERID
