-- SelectInspReqHistory 검사이력 조회문 

------------------------------------

SELECT R.INSPREQNO,

	   R.SITEID,

	   R.COPERATOR,

	   R.ITEMID,

	   D.INSPDEFINITIONNAME,

	   R.REV,

	   R.INSPTYPE,

	   R.STATE,

	   R.FINALJUDGE,

	   R.REQDATETIME,

	   (SELECT FN_USERNAME(R.SITEID, R.CREATOR) FROM DUAL) AS REQUSER,

	   R.RECEPTDATETIME,

	   (SELECT FN_USERNAME(RHC.SITEID, RHC.CREATOR) FROM DUAL) AS RECEPTUSER,

	   RHI.LASTEVENTTIME AS INSPDATATIME,

	   (SELECT FN_USERNAME(RHI.SITEID, RHI.CREATOR) FROM DUAL) AS INSPUSER,

	   R.FINALDATETIME,

	   (SELECT FN_USERNAME(RHF.SITEID, RHF.CREATOR) FROM DUAL) AS FINALUSER

  FROM CIM_INSPREQ R

	INNER JOIN CIM_INSPDEFINITION D

		ON 1=1

	   AND R.SITEID = D.SITEID

	   AND R.COPERATOR = D.COPERATOR

	   AND R.ITEMID = D.ITEMID 

	   AND R.INSPTYPE = D.INSPTYPE 

	   AND R.REV = D.REV

	LEFT OUTER JOIN CIM_INSPREQHIST RHC

		ON 1=1

	   AND R.SITEID = RHC.SITEID

	   AND R.INSPREQNO = RHC.INSPREQNO

	   AND RHC.CUSTOMACTIVITY = 'ReceptInspection'

	   AND RHC.LASTEVENTSEQ = (SELECT MAX(T.LASTEVENTSEQ)

	   							   FROM CIM_INSPREQHIST T

	   							  WHERE 1=1

	   							    AND T.CUSTOMACTIVITY = 'ReceptInspection'

	   							    AND T.SITEID = RHC.SITEID

								    AND T.INSPREQNO = RHC.INSPREQNO)

	LEFT OUTER JOIN CIM_INSPREQHIST RHI

		ON 1=1

	   AND R.SITEID = RHI.SITEID

	   AND R.INSPREQNO = RHI.INSPREQNO

	   AND RHI.CUSTOMACTIVITY = 'InspectionResult'

	   AND RHI.LASTEVENTSEQ = (SELECT MAX(T1.LASTEVENTSEQ)

	   							   FROM CIM_INSPREQHIST T1

	   							  WHERE 1=1

	   							    AND T1.CUSTOMACTIVITY = 'InspectionResult'

	   							    AND T1.SITEID = RHI.SITEID

								    AND T1.INSPREQNO = RHI.INSPREQNO)

	LEFT OUTER JOIN CIM_INSPREQHIST RHF

		ON 1=1

	   AND R.SITEID = RHF.SITEID

	   AND R.INSPREQNO = RHF.INSPREQNO

	   AND RHF.CUSTOMACTIVITY = 'CloseInspReq'

	   AND RHF.LASTEVENTSEQ = (SELECT MAX(T1.LASTEVENTSEQ)

	   							   FROM CIM_INSPREQHIST T1

	   							  WHERE 1=1

	   							    AND T1.CUSTOMACTIVITY = 'CloseInspReq'

	   							    AND T1.SITEID = RHF.SITEID

								    AND T1.INSPREQNO = RHF.INSPREQNO)

 WHERE 1=1

   AND (:INSPTYPE IS NULL OR R.INSPTYPE = :INSPTYPE)   

   AND (:REQDATETIME IS NULL OR R.REQDATETIME BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 0.99999)

   AND (:RECEPTDATETIME IS NULL OR R.RECEPTDATETIME BETWEEN TO_DATE(:FROMDATE, 'YYYY-MM-DD') AND TO_DATE(:TODATE, 'YYYY-MM-DD') + 0.99999)

   AND (:STATE IS NULL OR R.STATE IN (SELECT * FROM TABLE(FN_MES_SPLIT(:STATE, ','))))

   AND (:INSPREQNO IS NULL OR R.INSPREQNO LIKE :INSPREQNO || '%')

   AND (:COPERATOR IS NULL OR R.COPERATOR = :COPERATOR)

   AND (:ITEMID IS NULL OR R.ITEMID = :ITEMID)

   AND (:INSPSPEC IS NULL OR R.INSPSPEC = :INSPSPEC)

   AND (:FINALJUDGE IS NULL OR R.FINALJUDGE = :FINALJUDGE)

   AND (:ISUSABLE IS NULL OR R.ISUSABLE = :ISUSABLE)

   AND R.SITEID = :SITEID
테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
