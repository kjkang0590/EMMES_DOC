SELECT PS.PROCESSSEGMENTNAME, DPR.*, DC.DEFECTCLASSNAME 

FROM CUS_DEFECTPROCESSSEGMENTREL DPR

	INNER JOIN CIM_PROCESSSEGMENT PS

		ON DPR.SITEID = PS.SITEID

		AND DPR.PROCESSSEGMENTID = PS.PROCESSSEGMENTID

	INNER JOIN CIM_DEFECT DF

		ON DPR.SITEID = DF.SITEID

		AND DPR.DEFECTID = DF.DEFECTID

	INNER JOIN CIM_DEFECTCLASS DC

		ON DC.SITEID = DF.SITEID

		AND DC.DEFECTCLASSID = DF.DEFECTCLASSID

WHERE 1=1

	AND DPR.SITEID = :SITEID

	AND DPR.PROCESSSEGMENTID = :PROCESSSEGMENTID

	AND DPR.DEFECTID LIKE :DEFECTID + '%'

	AND DF.DEFECTNAME LIKE :DEFECTNAME + '%'

	AND (:ISUSABLE IS NULL OR DPR.ISUSABLE = :ISUSABLE)

	AND DC.DEFECTCLASSNAME LIKE :DEFECTCLASSNAME + '%'
