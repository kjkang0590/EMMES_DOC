-- SelectRecipeDefinitionList4Master : 레시피 정의 기준정보 조회문

------------------------------------

-- INPUT : SITEID, EQUIPMENTID, PROCESSSEGMENTID, PRODUCTORDERID, PRODUCTORDERTYPE, RECIPEDEFINITIONTYPE

SELECT RD.*,

	   dbo.FN_USERNAME(RD.SITEID, RD.CREATOR)  AS CREATOR2, 

	   dbo.FN_USERNAME(RD.SITEID, RD.MODIFIER) AS MODIFIER2

  FROM CIM_RECIPEDEFINITION RD

 WHERE (:ISUSABLE IS NULL OR RD.ISUSABLE = :ISUSABLE)

   AND (:EQUIPMENTID IS NULL OR RD.EQUIPMENTID = :EQUIPMENTID)

   AND (:PROCESSSEGMENTID IS NULL OR RD.PROCESSSEGMENTID = :PROCESSSEGMENTID)

   AND (:PRODUCTDEFINITIONID IS NULL OR RD.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID)

   AND (:PRODUCTORDERTYPE IS NULL OR RD.PRODUCTORDERTYPE = :PRODUCTORDERTYPE)

   AND (:RECIPEDEFINITIONTYPE IS NULL OR RD.RECIPEDEFINITIONTYPE = :RECIPEDEFINITIONTYPE)

   AND RD.SITEID = :SITEID

ORDER BY RD.RECIPEDEFINITIONNAME, RD.VERSION, RD.SUBVERSION
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

	   AND PS.SITEID = PD.SITEID
