--MD_PM_007_설비별 Tag 관리
--SelectEquipmentTagList
--Param : SITEID, LINEID, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID, TAGID, ISUSABLE
------------------------------------------
SELECT *
FROM (
    SELECT OT.TAGID
         , OT.TAGNAME
         , OT.DESCRIPTION
         , OT.ROOTEQUIPMENTID
         , CE.EQUIPMENTCLASSID
         , OT.TAGTYPE AS EQUIPMENTCLASSNAME
         , OT.EQUIPMENTID
         , EQ.EQUIPMENTNAME
         , OT.DCSTAGID
         , OT.ISUSABLE
         , CCU.USERNAME AS CREATOR
         , OT.CREATETIME
         , MCU.USERNAME AS MODIFIER
         , OT.MODIFYTIME
         , '' AS PARENTID
         , '' AS CHILDID
    FROM CST_OPCTAGINFO OT WITH(NOLOCK)
    LEFT OUTER JOIN CIM_EQUIPMENT EQ WITH(NOLOCK)
        ON EQ.SITEID = OT.SITEID
        AND EQ.EQUIPMENTID = OT.EQUIPMENTID
    LEFT OUTER JOIN CIM_EQUIPMENTCLASS CE WITH(NOLOCK)
        ON CE.SITEID = EQ.SITEID 
        AND CE.EQUIPMENTCLASSID = EQ.EQUIPMENTCLASSID 
    LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK)
        ON CCU.SITEID = OT.SITEID
        AND CCU.USERID = OT.CREATOR
    LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK)
        ON MCU.SITEID = OT.SITEID
        AND MCU.USERID = OT.MODIFIER
    WHERE OT.SITEID = @SITEID
        AND OT.LINEID = @LINEID
        AND (@ROOTEQUIPMENTID = '' OR OT.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)
        AND (@EQUIPMENTCLASSID = '' OR EQ.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
        AND (@EQUIPMENTID = '' OR EQ.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%')
        AND (@TAGID = '' OR OT.TAGID LIKE '%' + REPLACE(REPLACE(@TAGID, '[', '\['), ']', '\]') + '%' ESCAPE '\')
        AND (@ISUSABLE = '' OR OT.ISUSABLE = @ISUSABLE)
  UNION ALL
	 SELECT OT.TAGID
	     , OT.TAGNAME
	     , OT.DESCRIPTION
	     , OT.ROOTEQUIPMENTID
	     , OT.EQUIPMENTCLASSID
	     , OT.TAGTYPE AS EQUIPMENTCLASSNAME     
	     , OT.EQUIPMENTID
	     , EQ.EQUIPMENTNAME
	     , OT.DCSTAGID
	     , OT.ISUSABLE
	     , CCU.USERNAME AS CREATOR
	     , OT.CREATETIME
	     , MCU.USERNAME AS MODIFIER
	     , OT.MODIFYTIME
              , CP.PARENTID
              , CP.CHILDID
	  FROM CST_OPCTAGINFO OT WITH(NOLOCK)
	  INNER JOIN CST_PUBLICREL CP WITH(NOLOCK)
	  	ON CP.SITEID = OT.SITEID
	  	AND CP.CHILDID = OT.EQUIPMENTID
	  LEFT OUTER JOIN CIM_EQUIPMENT EQ WITH(NOLOCK)
	    ON EQ.SITEID = CP.SITEID
	   AND EQ.EQUIPMENTID = CP.CHILDID
	  LEFT OUTER JOIN CIM_EQUIPMENTCLASS CE WITH(NOLOCK)
	    ON CE.SITEID = EQ.SITEID 
	   AND CE.EQUIPMENTCLASSID = EQ.EQUIPMENTCLASSID 
	  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK)
	    ON CCU.SITEID = OT.SITEID
	   AND CCU.USERID = OT.CREATOR
	  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK)
	    ON MCU.SITEID = OT.SITEID
	   AND MCU.USERID = OT.MODIFIER
	 WHERE OT.SITEID = @SITEID
	   AND OT.LINEID = @LINEID
	   AND (@ROOTEQUIPMENTID = '' OR OT.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)
	   AND (@EQUIPMENTID = '' OR CP.PARENTID LIKE '%' + @EQUIPMENTID + '%')
	   AND (@TAGID = '' OR OT.TAGID LIKE '%' + REPLACE(REPLACE(@TAGID, '[', '\['), ']', '\]') + '%' ESCAPE '\')
	   AND (@ISUSABLE = '' OR OT.ISUSABLE = @ISUSABLE)	
) AS Result
   ORDER BY CASE WHEN EQUIPMENTID = @EQUIPMENTID THEN 0 ELSE 1 END

------------------------------------------	
SELECT
	EC.EQUIPMENTCLASSID AS CODE
	, EC.EQUIPMENTCLASSNAME AS NAME
FROM CIM_EQUIPMENTCLASS EC WITH(NOLOCK),
(
	SELECT
		E.EQUIPMENTCLASSID
		, E.SITEID
	FROM CIM_EQUIPMENT E WITH(NOLOCK), CST_PUBLICREL R WITH(NOLOCK)
	WHERE E.EQUIPMENTID = R.PARENTID
	AND   E.SITEID = R.SITEID
	AND   E.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
	GROUP BY E.EQUIPMENTCLASSID, E.SITEID
) E
WHERE EC.EQUIPMENTCLASSID = E.EQUIPMENTCLASSID
AND   EC.SITEID = E.SITEID
