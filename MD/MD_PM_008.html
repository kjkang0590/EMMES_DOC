SELECT A.ROOTEQUIPMENTID,
	   A.EQUIPMENTCLASSID,
	   A.EQUIPMENTCLASSNAME,
	   A.EQUIPMENTLEVEL,
	   A.ISUSABLE,
	   A.DESCRIPTION,
	   B.USERNAME AS CREATOR,
	   A.CREATETIME,
	   C.USERNAME AS MODIFIER,
	   A.MODIFYTIME,
            A.COMMENTS
  FROM CIM_EQUIPMENTCLASS A WITH(NOLOCK)
  LEFT JOIN CIM_USER B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.CREATOR = B.USERID
  LEFT JOIN CIM_USER C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.MODIFIER = C.USERID
 WHERE A.SITEID = @SITEID
   AND A.ISUSABLE = @ISUSABLE

SELECT A.EQUIPMENTID,
	   A.EQUIPMENTNAME,
	   A.EQUIPMENTCLASSID,
	   B.EQUIPMENTCLASSNAME,
	   A.EQUIPMENTLEVEL,
	   A.PARENTID,
	   A.PROCESSTYPE,

            A.COMMENTS

  LEFT JOIN CIM_EQUIPMENTCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.EQUIPMENTCLASSID = B.EQUIPMENTCLASSID

    ON A.SITEID = C.SITEID

   AND C.RELATIONTYPE = 'MD_PM_008'

  LEFT JOIN CIM_USER D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.CREATOR = D.USERID
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.MODIFIER = E.USERID
 WHERE A.SITEID = @SITEID
   AND A.ISUSABLE = @ISUSABLE
   AND A.EQUIPMENTCLASSID = @EQUIPMENTCLASSID
   AND A.EQUIPMENTLEVEL = 1
