------------- 등급코드관리 조회 쿼리

SELECT 

	CC.SITEID,

	NVL2(CGD.GRADETYPE , CGD.GRADETYPE , :GRADETYPE) AS GRADETYPE ,

	CC.CODEID AS GRADEID ,

	CC.CODENAME  AS GRADENAME ,

	CGD.LSL ,

	CGD.USL ,

	CGD.ISDEFECT ,

	CGD.FOMULADEFINITIONID ,

	CGD.UISEQUENCE , 

	CGD.ISUSABLE 

  FROM CIM_CODE CC

  LEFT JOIN CIM_GRADEDEFINITION CGD

  		 ON CGD.SITEID = CC.SITEID 

  		AND CGD.ISUSABLE = 'Usable'

  		AND CGD.GRADEID = CC.CODEID

  		AND CGD.GRADETYPE = :GRADETYPE

 WHERE CC.SITEID = :SITEID 

   AND CC.ISUSABLE = 'Usable'

   AND CC.CODECLASSID = 'GRADE'
