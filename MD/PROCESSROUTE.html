SELECT 
	PD.*,
	dbo.FN_USERNAME(SITEID, CREATOR) AS CREATOR2,
	dbo.FN_USERNAME(SITEID, MODIFIER) AS MODIFIER2
  FROM CIM_PROCESSDEFINITION PD
 WHERE 1=1  
   AND PD.SITEID = @SITEID 
   AND (@PROCESSTYPE = '' OR PD.PROCESSTYPE LIKE @PROCESSTYPE + '%')
   AND (@STATE = ''  OR PD.STATE = @STATE)
   AND (@ISUSABLE = '' OR PD.ISUSABLE = @ISUSABLE)

CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
