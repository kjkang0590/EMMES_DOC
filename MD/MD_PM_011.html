--MD_PM_011_자주검사 항목 관리
--SelectLPAItemList
--Param : SITEID, LINEID, LPAGROUP, LPATYPE, PROCESSSEGMENTTYPE, INSPMANAGER, ISUSABLE
------------------------------------------
SELECT LI.LPAGROUP
     , LI.LPATYPE
     , LI.PROCESSSEGMENTTYPE
     , LI.LPADIVISION
     , LI.INSPMANAGER
     , LI.LPAITEMID
     , LI.LPAITEMNAME
     , LI.UISEQUENCE
     , LI.ISUSABLE
     , CCU.USERNAME AS CREATOR
     , LI.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , LI.MODIFYTIME
  FROM CST_LPAITEM LI
  LEFT JOIN CIM_USER CCU ON CCU.SITEID = LI.SITEID AND CCU.USERID = LI.CREATOR
  LEFT JOIN CIM_USER MCU ON MCU.SITEID = LI.SITEID AND MCU.USERID = LI.MODIFIER
 WHERE LI.SITEID = @SITEID
   AND (@LPAGROUP = '' OR LI.LPAGROUP = @LPAGROUP)
   AND (@LPATYPE = '' OR LI.LPATYPE = @LPATYPE)
   AND (@PROCESSSEGMENTTYPE = '' OR LI.PROCESSSEGMENTTYPE = @PROCESSSEGMENTTYPE)
   AND (@INSPMANAGER = '' OR LI.INSPMANAGER = @INSPMANAGER)
   AND (@ISUSABLE = '' OR LI.ISUSABLE = @ISUSABLE)
 ORDER BY LI.LPAGROUP
     , LI.LPATYPE
     , LI.PROCESSSEGMENTTYPE
     , LI.LPADIVISION
     , LI.UISEQUENCE
     , LI.LPAITEMID
