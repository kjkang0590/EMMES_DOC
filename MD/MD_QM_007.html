--MD_QM_007_Sample Point 관리
--SelectSamplePointList
--Param : SITEID, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID, SAMPLEPOINTID, SAMPLETYPE, ISUSABLE
------------------------------------------
SELECT SP.SAMPLEPOINTID
     , SP.SAMPLEPOINTNAME
     , SP.DESCRIPTION
     , SP.SAMPLETYPE
     , SP.EQUIPMENTID
     , SP.EQUIPMENTCLASSID
     , SE.PROCESSSEGMENTNAME
     , CC.CODENAME AS PROCESSTYPE
     , SP.ISUSABLE
     , CCU.USERNAME AS CREATOR
     , SP.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , SP.MODIFYTIME
  FROM CST_SAMPLEPOINT SP WITH(NOLOCK)
  LEFT OUTER JOIN CIM_PROCESSSEGMENT SE WITH(NOLOCK)
    ON SE.SITEID = SP.SITEID
   AND SE.PROCESSSEGMENTID = SP.PROCESSSEGMENTID
  LEFT OUTER JOIN CIM_CODE CC WITH(NOLOCK)
    ON CC.SITEID = SP.SITEID
   AND CC.CODECLASSID = 'PROCESSTYPE'
   AND CC.CODEID = SP.PROCESSTYPE
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK)
    ON CCU.SITEID = SP.SITEID
   AND CCU.USERID = SP.CREATOR
  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK)
    ON MCU.SITEID = SP.SITEID
   AND MCU.USERID = SP.MODIFIER
 WHERE SP.SITEID = @SITEID
   AND (@ROOTEQUIPMENTID = '' OR SP.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)
   AND (@EQUIPMENTCLASSID = '' OR SP.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
   AND (@EQUIPMENTID = '' OR SP.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%')
   AND (@SAMPLEPOINTID = '' OR SP.SAMPLEPOINTID LIKE '%' + @SAMPLEPOINTID + '%')
   AND (@SAMPLETYPE = '' OR SP.SAMPLETYPE = @SAMPLETYPE)
   AND SP.ISUSABLE = @ISUSABLE
 ORDER BY SP.SAMPLEPOINTID
