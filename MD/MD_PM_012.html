--MD_PM_012 Process Data 설정 - 등록 Process 목록 조회
--SelectOpcTagInfoList
--Param : SITEID, PROCESSDATADEFINITIONID, ROOTEQUIPMENTID, EQUIPMENTID, DCSTAGID
------------------------------------------      
SELECT OT.DCSTAGID
	 , OT.TAGNAME
     , OT.EQUIPMENTDINFO
     , OT.ROOTEQUIPMENTID
     , OT.EQUIPMENTID
     , E.EQUIPMENTNAME
     , STRING_AGG(DP.PROCESSDATADEFINITIONID, ',' ) WITHIN GROUP(ORDER BY DP.PROCESSDATADEFINITIONID) AS PROCESSDATADEFINITIONID
     , DP.DATAMAPPINGTABLE
     , DP.DATAMAPPINGCOLUMN
     , OT.TAGID AS OPCTAG
FROM CST_OPCTAGINFO OT
LEFT OUTER JOIN CST_PROCESSDATAPARAMETER DP
ON  OT.TAGID = DP.PROCESSDATAPARAMETERID 
AND OT.SITEID = DP.SITEID
AND DP.PROCESSDATADEFINITIONID = @PROCESSDATADEFINITIONID
--AND DP.DATAMAPPINGTABLE = ( CASE WHEN @DATAMAPPINGTABLE = '' THEN DP.DATAMAPPINGTABLE ELSE @DATAMAPPINGTABLE END )
LEFT OUTER JOIN CIM_EQUIPMENT E
ON OT.EQUIPMENTID = E.EQUIPMENTID
AND OT.SITEID = E.SITEID
WHERE OT.SITEID = @SITEID
AND   DP.ISUSABLE = 'Usable'
AND   DP.PROCESSDATACLASSID = 'REGISTER'
AND   ISNULL(OT.ROOTEQUIPMENTID, '') = (CASE WHEN @ROOTEQUIPMENTID = '' THEN ISNULL(OT.ROOTEQUIPMENTID, '') ELSE @ROOTEQUIPMENTID END)
AND   ISNULL(OT.EQUIPMENTID, '') = (CASE WHEN @EQUIPMENTID = '' THEN ISNULL(OT.EQUIPMENTID, '') ELSE @EQUIPMENTID END)
AND   ISNULL(OT.DCSTAGID, '') LIKE (CASE WHEN @DCSTAGID = '' THEN '%' + ISNULL(OT.DCSTAGID, '') + '%' ELSE '%' + @DCSTAGID + '%' END)
GROUP BY OT.DCSTAGID, OT.EQUIPMENTDINFO, OT.ROOTEQUIPMENTID, OT.EQUIPMENTID, DP.DATAMAPPINGTABLE, DP.DATAMAPPINGCOLUMN, OT.TAGID, E.EQUIPMENTNAME, OT.TAGNAME
ORDER BY DP.DATAMAPPINGTABLE, CONVERT(INT, REPLACE(DP.DATAMAPPINGCOLUMN, 'VALUE', ''))
