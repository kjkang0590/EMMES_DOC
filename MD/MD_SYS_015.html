#1 Query SelectAlarmDefinitionList

연결된 기능 (1번 표): #1 조회 - 그리드 MD_SYS_0151 (알람수신대상관리1_알람목록)

Query (조회 서비스) : SelectAlarmDefinitionList

  • ID 구성: SelectAlarmDefinitionList / 클래스 THiRAMES.Service.UI / 버전 00001
  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)
  • 쿼리명: MD_SYS_015_알림 전달 대상 관리
  • 쿼리 파라메터: SITEID
  • 테이블(조회/쓰기 전체, 함수·SP 포함):
    CIM_ALARMDEFINITION메인읽기   CIM_ALARMCLASS읽기   CST_ALARMDEFUSERREL읽기   CST_MAILINGCLASS읽기
조인 구조도 테이블 2 · 조인 1
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_ALARMCLASS, CST_MAILINGCLASS
SQL 보기
------------------------------------------
--MD_SYS_015_알림 전달 대상 관리
--SelectAlarmDefinitionList
--Param : SITEID
------------------------------------------
SELECT AD.ALARMCLASSID
     , AC.ALARMCLASSNAME
     , AD.ALARMDEFINITIONID
     , AD.ALARMDEFINITIONNAME
     , AD.ALARMLEVEL
     , AD.DESCRIPTION
     , MC.USERCLASSID AS ALARMDELIVERYGROUPID
     , MC.USERCLASSNAME AS ALARMDELIVERYGROUPNAME
     , AD.ISUSABLE
  FROM CIM_ALARMDEFINITION AD WITH(NOLOCK)
 INNER JOIN CIM_ALARMCLASS AC WITH(NOLOCK)
    ON AC.SITEID = AD.SITEID
   AND AC.ALARMCLASSID = AD.ALARMCLASSID
  LEFT OUTER JOIN CST_ALARMDEFUSERREL ADU WITH(NOLOCK)
    ON ADU.SITEID = AD.SITEID
   AND ADU.ALARMDEFINITIONID = AD.ALARMDEFINITIONID
  LEFT OUTER JOIN CST_MAILINGCLASS MC WITH(NOLOCK)
    ON MC.SITEID = ADU.SITEID
   AND MC.USERCLASSID = ADU.USERCLASSID
 WHERE AD.SITEID = @SITEID
   AND (@ALARMCLASSID = '' OR AD.ALARMCLASSID = @ALARMCLASSID)
   AND (@ALARMDEFINITIONID = '' OR AD.ALARMDEFINITIONID = @ALARMDEFINITIONID)
 ORDER BY AC.ALARMCLASSID, AD.ALARMDEFINITIONID

SQL 보기
------------------------------------------
--MD_SYS_015_알림 전달 대상 관리
--SelectAlarmDefUserList
--Param : SITEID, ALARMDELIVERYGROUPID(USERCLASS)
------------------------------------------
SELECT MCR.USERCLASSID AS ALARMDELIVERYGROUPID
     , MC.USERCLASSNAME AS ALARMDELIVERYGROUPNAME
     , CU.USERNAME
     , MCR.USERID
     , CU.DEPTNAME
     , CU.POSITIONNAME
  FROM CST_MAILINGCLASSREL MCR WITH(NOLOCK)
  LEFT OUTER JOIN CST_MAILINGCLASS MC WITH(NOLOCK)
    ON MC.SITEID = MCR.SITEID
   AND MC.USERCLASSID = MCR.USERCLASSID
  LEFT OUTER JOIN CIM_USER CU WITH(NOLOCK)
    ON CU.SITEID = MCR.SITEID
   AND CU.USERID = MCR.USERID
 WHERE MCR.SITEID = @SITEID
   AND (@ALARMDELIVERYGROUPID = '' OR MCR.USERCLASSID = @ALARMDELIVERYGROUPID)
 ORDER BY MCR.USERCLASSID, CU.DEPTNAME, CU.POSITIONNAME, CU.USERNAME
