--MD_QM_002_Recipe 관리
--SelectRecipeList
--Param : SITEID, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID, ISUSABLE
------------------------------------------
SELECT RI.ROOTEQUIPMENTID
     , RI.EQUIPMENTID
     , RI.EQUIPMENTNAME
     , RI.RECIPEITEMID
     , RI.RECIPEITEMNAME
     , RI.TAGID
     , RI.UNITID
     , RI.LCL
     , RI.UCL
     , RI.LSL
     , RI.USL
     , RI.TARGET
     , RI.INEQUATION
     , RI.ISUSABLE
     , RI.UISEQUENCE
     , RI.RECIPEDESCRIPTION
     , F.ATTACHFILENAME AS RECIPEGUIDE
     , F.FILEPATH
  FROM CST_RECIPEINFO RI WITH(NOLOCK)
 INNER JOIN CIM_EQUIPMENT CE WITH(NOLOCK) ON CE.SITEID = RI.SITEID AND CE.EQUIPMENTID = RI.EQUIPMENTID
  LEFT JOIN CIM_ATTACHMENT F WITH(NOLOCK) ON F.SITEID = RI.SITEID AND F.RELATIONID = RI.EQUIPMENTID AND F.ISUSABLE = 'Usable'
 WHERE RI.SITEID = @SITEID
   AND (@ROOTEQUIPMENTID = '' OR CE.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)
   AND (@EQUIPMENTCLASSID = '' OR CE.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
   AND (@EQUIPMENTID = '' OR CE.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%')
   AND RI.ISUSABLE = @ISUSABLE
 ORDER BY RI.UISEQUENCE, RI.EQUIPMENTID, RI.RECIPEITEMID
