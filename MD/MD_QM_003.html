--MD_QM_003_검사 항목 관리
--SelectINSPItemGroupList
--Param : SITEID, INSPITEMCLASSID, ISUSABLE
------------------------------------------
SELECT IC.INSPITEMCLASSID
     , IC.INSPITEMCLASSNAME
     , IC.ISUSABLE
     , IC.DESCRIPTION
     , IC.UISEQUENCE
     , CCU.USERNAME AS CREATOR
     , IC.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , IC.MODIFYTIME
  FROM CIM_INSPITEMCLASS IC WITH(NOLOCK)
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK) ON CCU.SITEID = IC.SITEID AND CCU.USERID = IC.CREATOR
  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK) ON MCU.SITEID = IC.SITEID AND MCU.USERID = IC.MODIFIER
 WHERE IC.SITEID = @SITEID
   AND (@INSPITEMCLASSID = '' OR IC.INSPITEMCLASSID = @INSPITEMCLASSID)
   AND IC.ISUSABLE = @ISUSABLE
 ORDER BY IC.UISEQUENCE, IC.INSPITEMCLASSID

--MD_QM_003_검사 항목 관리
--SelectINSPItemList
--Param : SITEID, INSPITEMCLASSID
------------------------------------------
SELECT IC.INSPITEMCLASSID
     , IC.INSPITEMCLASSNAME
     , II.INSPMETHOD
     , II.INSPITEMID
     , II.INSPITEMNAME
     , II.SHORTNAME
     , II.ISUSABLE
     , II.DESCRIPTION
     , II.UISEQUENCE
     , CCU.USERNAME AS CREATOR
     , II.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , II.MODIFYTIME
  FROM CIM_INSPITEMCLASS IC WITH(NOLOCK)
 INNER JOIN CIM_INSPITEM II WITH(NOLOCK) ON II.SITEID = IC.SITEID AND II.INSPITEMCLASSID = IC.INSPITEMCLASSID
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK) ON CCU.SITEID = II.SITEID AND CCU.USERID = II.CREATOR
  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK) ON MCU.SITEID = II.SITEID AND MCU.USERID = II.MODIFIER
 WHERE IC.SITEID = @SITEID
   AND (@INSPITEMCLASSID = '' OR IC.INSPITEMCLASSID = @INSPITEMCLASSID)
   AND (@ISUSABLE = '' OR II.ISUSABLE = @ISUSABLE)
 ORDER BY II.UISEQUENCE, II.INSPITEMCLASSID, II.INSPITEMID
