/**************************************************************************

 * REPORT CPK 조회

 *  CPK 는 SUMMARY 없음

 *

 *    DECLARE

 *       @SITEID VARCHAR(40) = 'PL01'                   -- SITEID

 *       , @FROMDATE DATETIME = '2024-07-01 07:00:00'   -- 시작 일시

 *       , @TODATE DATETIME= '2024-07-08 07:00:00'      -- 종료 일시

 *       , @CPKITEM VARCHAR(MAX) = 'KILN01,KILN02,KILN03,KILN04,KILN05,KILN06,KILN07,KILN08'    -- CPKITEM 코드를 ',' 로 연결

 *       , @ISGRID VARCHAR(1) = 'N'                     -- Y : GRID DATA 조회 | N : CHART DATA 조회

 * 

 *     EXEC SP_RPT_CPK_SELECT @SITEID, @FROMDATE, @TODATE, @CPKITEM, @ISGRID

 **************************************************************************/

CREATE PROCEDURE [dbo].[SP_RPT_CPK_SELECT] 

    @SITEID VARCHAR(40), @FROMDATE DATETIME, @TODATE DATETIME, @CPKITEM VARCHAR(MAX), @ISGRID VARCHAR(1) = 'Y'

AS  

--DECLARE

--        @SITEID VARCHAR(40) = 'PL01'

--        , @FROMDATE DATETIME = '2024-07-01 07:00:00'

--        , @TODATE DATETIME= '2024-07-01 07:30:00'

--        , @CPKITEM VARCHAR(MAX) = 'KILN01,KILN02,KILN03,KILN04,KILN05,KILN06,KILN07,KILN08'

--        , @ISGRID VARCHAR(1) = 'Y'

BEGIN 



    

    -- PROCESS DATA 용 변수

    DECLARE @STRSQT NVARCHAR(MAX)

          , @DATAMAPPINGTABLE NVARCHAR(100)

          , @SELECT_CURRENT_COLUMN NVARCHAR(MAX)

          , @PARAMS NVARCHAR(MAX)

          , @STARTDATE DATETIME

          , @ENDDATE   DATETIME          

    ;    

        

    

    DECLARE @CPK_VALUE TABLE

    (

        TABLENAME  VARCHAR(40)

        , COLNAME    VARCHAR(40)

        , MODIFYTIME DATETIME

        , NOW_VALUE  FLOAT

    );        

    

    DECLARE @CPK_ITEM_LIST TABLE

    (

        CPKITEMID           VARCHAR(40)

        , CPKITEMNAME       VARCHAR(40)

        , SITEID            VARCHAR(40)

        , UNIT              VARCHAR(40)

        , UNITNAME          VARCHAR(100)

        , TARGET            FLOAT

        , USL               FLOAT

        , LSL               FLOAT

        , CPKTARGET         FLOAT

        , TAGID             VARCHAR(100)

        , DCSTAGID          VARCHAR(40)

        , UISEQUENCE        SMALLINT

        , DATAMAPPINGTABLE  VARCHAR(40)

        , DATAMAPPINGCOLUMN VARCHAR(40)

    )

    ;

        

    --##########################################################################

    -- 기준정보 조회 : * START * 

    --##########################################################################

    INSERT INTO @CPK_ITEM_LIST

    SELECT 

        CD.CPKITEMID

        , CD.CPKITEMNAME

        , CD.SITEID

        , CD.UNIT 

        , C.CODENAME AS UNITNAME

        , CD.TARGET

        , CD.USL

        , CD.LSL

        , CD.CPKTARGET

        , CD.TAGID

        , CD.DCSTAGID

        , CD.UISEQUENCE

        , PM.DATAMAPPINGTABLE

        , PM.DATAMAPPINGCOLUMN

    FROM RPT_PROCESSCPK_DEFINITION CD

        INNER JOIN dbo.FN_MES_SPLIT(@CPKITEM, ',') SP

            ON  CD.CPKITEMID = SP.VALUE

        INNER JOIN CST_PROCESSDATAPARAMETER PM

            ON  CD.TAGID = PM.PROCESSDATAPARAMETERID

            AND CD.SITEID = PM.SITEID

            AND PM.PROCESSDATADEFINITIONID = 'EISMES_Tran_Process_Data'

            AND PM.ISUSABLE = 'Usable'

        LEFT OUTER JOIN CIM_CODE C

            ON  CD.UNIT = C.CODEID

            AND CD.SITEID = C.SITEID

            AND C.CODECLASSID = 'RPT_UNIT'

    WHERE CD.SITEID = @SITEID

    ORDER BY CD.UISEQUENCE    

    ;

    --##########################################################################

    -- 기준정보 조회 : * END * 

    --##########################################################################









    --##########################################################################

    -- PROCESS DATA 에서 DATA 수집 : * START * 

    --##########################################################################

    BEGIN

        

        -- 조회 결과를 TABLE 별 분류하여 CURSOR 에 연결

        DECLARE CUR_LEVEL CURSOR FOR

            SELECT 

T.DATAMAPPINGTABLE

                , STRING_AGG ( DATAMAPPINGCOLUMN, ', ') WITHIN GROUP (ORDER BY T.DATAMAPPINGTABLE) AS SELECT_CURRENT_COLUMN

            FROM @CPK_ITEM_LIST T

            GROUP BY T.DATAMAPPINGTABLE        

            

            

        -- CURSOR OPEN

        OPEN CUR_LEVEL

        

    

        -- TABLE 별로 WHILE & 결과를 TEMP TABLE 에 저장

        FETCH NEXT FROM CUR_LEVEL INTO @DATAMAPPINGTABLE, @SELECT_CURRENT_COLUMN 

        WHILE(@@FETCH_STATUS=0)

        BEGIN

            

            SET @PARAMS = N'@P_STARTTIME DATETIME, @P_ENDTIME DATETIME'

            

            SET @STRSQT = N'

                    SELECT 

                        TABLENAME

                        , COLNAME

                        , MODIFYTIME

                        , CONVERT(FLOAT, NOW_VALUE ) AS NOW_VALUE

                    FROM 

                    ( 

                        SELECT ''' + @DATAMAPPINGTABLE + ''' AS TABLENAME, 

                            MODIFYTIME,

                            ' + @SELECT_CURRENT_COLUMN + '

                        FROM ' + @DATAMAPPINGTABLE + 'HIST WITH(NOLOCK)

                        WHERE @P_STARTTIME <= MODIFYTIME AND MODIFYTIME < @P_ENDTIME 

                    ) M 

                    UNPIVOT 

                    ( 

                        NOW_VALUE FOR COLNAME IN ( ' + @SELECT_CURRENT_COLUMN + ')

                    ) UNPA

                    ORDER BY TABLENAME, COLNAME, MODIFYTIME'

            ;

        

--            PRINT '------------------------------------------------------------------------------'

--            PRINT @STRSQT    



            INSERT @CPK_VALUE (TABLENAME, COLNAME, MODIFYTIME, NOW_VALUE)

            EXECUTE sp_executesql @STRSQT, 

                          @PARAMS, 

                          @P_STARTTIME = @FROMDATE,

                          @P_ENDTIME = @TODATE

            ;    

    

            FETCH NEXT FROM CUR_LEVEL INTO @DATAMAPPINGTABLE, @SELECT_CURRENT_COLUMN

        END

    

        -- CURSOR 삭제

        CLOSE CUR_LEVEL

        DEALLOCATE CUR_LEVEL

    END 

    ;

    --##########################################################################

    -- PROCESS DATA 에서 DATA 수집 : # END #

    --##########################################################################    





    --

    -- GRID DATA

    --

    IF @ISGRID = 'Y'

    BEGIN

        WITH CPK_SUM_DATA AS

        (

            SELECT 

                CV.TABLENAME

                , CV.COLNAME

                , MAX(CV.NOW_VALUE) AS MAX_VALUE

                , MIN(CV.NOW_VALUE) AS MIN_VALUE

                , MAX(CV.NOW_VALUE) - MIN(CV.NOW_VALUE) AS RANGE_VALUE

                , ROUND(AVG(CV.NOW_VALUE),2) AS AVG_VALUE

                , STDEV(CV.NOW_VALUE) AS STDEV_VALUE

            FROM @CPK_VALUE CV

            GROUP BY CV.TABLENAME, CV.COLNAME

        )

        SELECT 

            CPKITEMID

            , CPKITEMNAME

            , UNITNAME

            , DCSTAGID

            , UISEQUENCE

            , ITEMTYPE

            , FORMAT(ROUND(ITEMVALUE,2), '####0.00') AS ITEMVALUE

        FROM 

        (

            SELECT 

                CL.CPKITEMID

                , CL.CPKITEMNAME

                , CL.UNITNAME

                , CL.DCSTAGID

                , ROUND(CD.MAX_VALUE, 2) AS [Max]

                , ROUND(CD.MIN_VALUE, 2) AS [Min]

                , (ROUND(CD.MAX_VALUE, 2) - ROUND(CD.MIN_VALUE, 2)) AS [Range]

                , CD.AVG_VALUE AS [Avg]

                , CD.STDEV_VALUE AS [STDEV]

--                , (CL.USL - CD.AVG_VALUE) / (CD.STDEV_VALUE * 3) AS UPPER_CP -- 상한 CP

--                , (CD.AVG_VALUE - CL.LSL) / (CD.STDEV_VALUE * 3) AS LOW_CP -- 하한 CP

                , ABS( (CL.USL + CL.LSL) / 2 - CD.AVG_VALUE ) / ( (CL.USL - CL.LSL) / 2 )  AS [K_VALUE] -- K : ABS[ (USL + LSL) / 2 - AVG ] / [ (USL-LSL) /2 ]

                , CASE WHEN CD.STDEV_VALUE = 0 OR (CL.USL - CL.LSL) = 0 THEN 0

--                       WHEN ( 1.0 - (ABS( (CL.USL + CL.LSL) / 2 - CD.AVG_VALUE ) / ( (CL.USL - CL.LSL)) / 2 ) ) < 0 THEN (CL.USL - CL.LSL) / (6 * CD.STDEV_VALUE) 

--                       ELSE ( 1.0 - (ABS( (CL.USL + CL.LSL) / 2 - CD.AVG_VALUE ) / ( (CL.USL - CL.LSL)) / 2 ) ) * (CL.USL - CL.LSL) / (6 * CD.STDEV_VALUE)

                       WHEN ((CL.USL - CD.AVG_VALUE) / (CD.STDEV_VALUE * 3)) > ((CD.AVG_VALUE - CL.LSL) / (CD.STDEV_VALUE * 3)) THEN (CD.AVG_VALUE - CL.LSL) / (CD.STDEV_VALUE * 3)

                       ELSE (CL.USL - CD.AVG_VALUE) / (CD.STDEV_VALUE * 3)

                  END AS [CPK] -- CPK = 상한 CP 와 하한 CP 중 작은 값 (2024-08-13 박경진 전임 요청)

                , CL.CPKTARGET AS [CPK Target]

                , CL.UISEQUENCE

            FROM CPK_SUM_DATA CD

                INNER JOIN @CPK_ITEM_LIST CL

                    ON  CD.TABLENAME = CL.DATAMAPPINGTABLE

                    AND CD.COLNAME = CL.DATAMAPPINGCOLUMN

        ) M

        UNPIVOT 

        (

            ITEMVALUE FOR ITEMTYPE IN ( [Max], [Min],[Range],[Avg],[STDEV],CPK,[CPK Target] )

        ) UNP

        ORDER BY UNP.UISEQUENCE

    END 

    --

    -- CHART DATA

    --

    ELSE IF @ISGRID = 'N'

    BEGIN 

        

        SELECT

            CL.CPKITEMID

            , CL.CPKITEMNAME

            , CV.MODIFYTIME

            , CL.TARGET

            , CL.USL

            , CL.LSL

            , CV.NOW_VALUE AS VALUE

        FROM @CPK_VALUE CV

            INNER JOIN @CPK_ITEM_LIST CL

                ON  CV.TABLENAME = CL.DATAMAPPINGTABLE

                AND CV.COLNAME   = CL.DATAMAPPINGCOLUMN

        ORDER BY CL.CPKITEMID, CV.MODIFYTIME                

        

    END



END
테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
