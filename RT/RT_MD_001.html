--MD_PM_013 FMB 기준정보 설정 조회 쿼리
--SelectFMBDefinitionList
--Param : SITEID, PROCESSDATADEFINITIONID, EQUIPMENTID, DCSTAGID
------------------------------------------
SELECT C.COMMENTS,
           A.TAGID,
	   A.DCSTAGID,
	   A.TAGNAME,
	   A.EQUIPMENTID,
	   C.DESCRIPTION,
	   C.LSL,
	   C.USL,
	   C.NOTOPERATINGVALUE,
	   C.ISAPPLY,
            IIF(C.ISAPPLY IS NULL, 'A', 'U') AS ROWSTATE
  FROM CST_OPCTAGINFO A WITH(NOLOCK)
 INNER JOIN CST_PROCESSDATAPARAMETER B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.TAGID = B.PROCESSDATAPARAMETERID
  LEFT JOIN RPT_FMBDEFINITION C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.TAGID = C.TAGID
 WHERE A.SITEID = @SITEID
   AND C.ISUSABLE = 'Usable'
   AND B.PROCESSDATADEFINITIONID = @PROCESSDATADEFINITIONID
   AND A.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%'
   AND A.DCSTAGID LIKE '%' + @DCSTAGID + '%'
 ORDER BY C.SEQNO
