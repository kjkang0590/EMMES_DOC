-- REPORT 용 조회 기간 일자/주차/월 생성

--  SELECT * FROM dbo.FN_RPT_DAY_WEEK_MONTH('D', '20240101', '20241231')

--  PARAM :

--     @TYPES : 일,주,월 [ D | W | M ]

--     @STARTDATE  : 시작일,주,월  [ 20241231 | 202453 | 202412 ]

--     @ENDDATE  : 시작일,주,월  [ 20241231 | 202453 | 202412 ]

-------------------------------------------------------------

CREATE FUNCTION [dbo].[FN_RPT_DAY_WEEK_MONTH]

(

    @TYPES VARCHAR(40)

    , @STARTDATE VARCHAR(40)

    , @ENDDATE VARCHAR(40)

)

RETURNS @RETARRAY TABLE

(

    WORKDATE    VARCHAR(8)

    , YEARS     VARCHAR(4)

    , YEARMONTH VARCHAR(6)

    , YEARWEEK  VARCHAR(6)

)

AS  

BEGIN  

    

    INSERT @RETARRAY( WORKDATE,YEARS,YEARMONTH,YEARWEEK)

    SELECT 

        WORKDATE,YEARS,YEARMONTH,YEARWEEK

    FROM 

    (

        SELECT 

            CONVERT(VARCHAR(8), WORKDATE, 112) AS WORKDATE

            , CONVERT(VARCHAR(4), WORKDATE, 112) AS YEARS

            , CONVERT(VARCHAR(6), WORKDATE, 112) AS YEARMONTH

            , CONVERT(VARCHAR(4), WORKDATE, 112) + FORMAT(DATEPART(WEEK, WORKDATE), '00') AS YEARWEEK

        FROM 

        (

            SELECT 

                DATEADD(DAY, NUMBER, CONVERT(DATETIME, LEFT(@STARTDATE, 4) + '-01-01')) AS WORKDATE

            FROM master..spt_values 

            WHERE TYPE = 'P'

            AND   NUMBER < DATEDIFF(DAY, CONVERT(DATETIME, LEFT(@STARTDATE, 4) + '-01-01'), CONVERT(DATETIME, LEFT(@ENDDATE, 4) + '-12-31'))

        ) M

    ) P

    WHERE  

        ( 

            (

                ( @TYPES = 'D' ) AND P.WORKDATE BETWEEN @STARTDATE AND @ENDDATE

            )

            OR 

            (

                ( @TYPES = 'W' ) AND P.YEARWEEK BETWEEN @STARTDATE AND @ENDDATE

            )

            OR 

            (

                ( @TYPES = 'M' ) AND P.YEARMONTH BETWEEN @STARTDATE AND @ENDDATE

            )

        )

    ;



    RETURN

END
