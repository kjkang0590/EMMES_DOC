-- 종합 REPORT : 생산량 조회

--   

--  SELECT * FROM dbo.FN_RPT_TOTAL_01A_PRODUCTION('D', '20240101', '20241231', '')

--  PARAM :

--     @TYPES : 일,주,월 [ D | W | M ]

--     @STARTDATE  : 시작일,주,월  [ 20241231 | 202453 | 202412 ]

--     @ENDDATE    : 시작일,주,월  [ 20241231 | 202453 | 202412 ]

--     @SELECTTYPE : 전체, Ton, M3 [ '' | Ton | M3]

-------------------------------------------------------------

CREATE FUNCTION [dbo].[FN_RPT_TOTAL_01A_PRODUCTION]

(

    @TYPES VARCHAR(40)

    , @STARTDATE VARCHAR(40)

    , @ENDDATE VARCHAR(40)

    , @SELECTTYPE VARCHAR(10) = ''

)

RETURNS @RETARRAY TABLE

(

    TOPICS      VARCHAR(40)

    , SUBTOPICS VARCHAR(40)

    , UNITID    VARCHAR(40)

    , PLANQTY   VARCHAR(40)

    , RESULTQTY VARCHAR(40)

    , GAPQTY    VARCHAR(40)

    , UISEQUENCE INT

    

)

AS  

BEGIN  

    

        DECLARE @RTN_DATA TABLE 

        (

            TOPICS      VARCHAR(40)

            , SUBTOPICS VARCHAR(40)

            , UNITID    VARCHAR(40)

            , PLANQTY   VARCHAR(40)

            , RESULTQTY VARCHAR(40)

            , GAPQTY    VARCHAR(40)

            , UISEQUENCE INT        

        )

    

        SET @TYPES = REPLACE(@TYPES, 'P', 'D'); -- CASE WHEN @TYPES = 'P' THEN 'D' ELSE @TYPES END 

    

        ------------------------------------------------------

        -- 전처리 정보

        ------------------------------------------------------

        INSERT INTO @RTN_DATA

        SELECT

            '전처리(BM)' AS TOPICS

            , '투입(CELL)' AS SUBTOPICS

            , 'Ton' AS UNITID

            , ROUND( SUM( ISNULL(PREV.PLAN_BM_NORMAL_INPUT, 0) ), 2)   AS PLANQTY

            , ROUND( SUM( ISNULL(PREV.RESULT_BM_NORMAL_INPUT, 0) ), 2) AS RESULTQTY

            , ROUND( SUM( ISNULL(PREV.RESULT_BM_NORMAL_INPUT, 0) ) - SUM( ISNULL(PREV.PLAN_BM_NORMAL_INPUT, 0) ), 2) AS GAPQTY

            , 1010 AS UISEQUENCE

        FROM dbo.FN_RPT_DAY_WEEK_MONTH( @TYPES, @STARTDATE, @ENDDATE) M

            LEFT OUTER JOIN RPT_PRODUCTION_PREV_PLAN_RESULT PREV

                ON  M.WORKDATE = PREV.WORKDATE

        WHERE  CASE WHEN @TYPES = 'D' THEN PREV.WORKDATE WHEN @TYPES = 'W' THEN PREV.YEARWEEK ELSE PREV.YEARMONTH END

                BETWEEN @STARTDATE AND @ENDDATE

        UNION ALL 

        SELECT

            '전처리(BM)' AS TOPICS

            , '배출(Ton Bag)' AS SUBTOPICS

            , 'Ton' AS UNITID

            , ROUND( SUM( ISNULL(PREV.PLAN_BP_NORMAL_OUTPUT, 0) ), 2)   AS PLANQTY

            , ROUND( SUM( ISNULL(PREV.RESULT_BP_NORMAL_OUTPUT, 0) ), 2) AS RESULTQTY

            , ROUND( SUM( ISNULL(PREV.RESULT_BP_NORMAL_OUTPUT, 0) ) - SUM( ISNULL(PREV.PLAN_BP_NORMAL_OUTPUT, 0) ), 2) AS GAPQTY

            , 1020 AS UISEQUENCE

        FROM dbo.FN_RPT_DAY_WEEK_MONTH( @TYPES, @STARTDATE, @ENDDATE) M

            LEFT OUTER JOIN RPT_PRODUCTION_PREV_PLAN_RESULT PREV

                ON M.WORKDATE = PREV.WORKDATE

        WHERE  CASE WHEN @TYPES = 'D' THEN PREV.WORKDATE WHEN @TYPES = 'W' THEN PREV.YEARWEEK ELSE PREV.YEARMONTH END

                BETWEEN @STARTDATE AND @ENDDATE    

        ;

    

        ------------------------------------------------------

        -- 후처리 정보

        ------------------------------------------------------    

        WITH NEXT_PROCESS_DATA AS

        (

            SELECT

                M.WORKDATE

                , M.YEARWEEK 

                , M.YEARMONTH

                ----------------------------------------------------------

                -- 계획

                , ISNULL(PL.PLAN_BM_SILO_INPUT  , 0) AS PLAN_BM_SILO_INPUT

                , ISNULL(PL.PLAN_FIRING_INPUT   , 0) AS PLAN_FIRING_INPUT

                , ISNULL(PL.PLAN_FIRING_OUTPUT  , 0) AS PLAN_FIRING_OUTPUT

                --, ISNULL(PL.PLAN_PREMIX_INPUT   , 0) AS PLAN_PREMIX_INPUT -- ##

                , ISNULL(PL.PLAN_PREMIX_OUTPUT   , 0) AS PLAN_PREMIX_OUTPUT -- ##

                --, ISNULL(PL.PLAN_REACTOR_INPUT  , 0) AS PLAN_REACTOR_INPUT -- ##

                , ISNULL(PL.PLAN_REACTOR_OUTPUT  , 0) AS PLAN_REACTOR_OUTPUT -- ##

                , ISNULL(PL.PLAN_DRY_OUTPUT     , 0) AS PLAN_DRY_OUTPUT

                , ISNULL(PL.PLAN_PK1501_INPUT   , 0) AS PLAN_PK1501_INPUT

                -- Ni/Co/Mg Sulfate 

                , ISNULL(PL.PLAN_V1517NCM_SULFATE_INPUT , 0) AS PLAN_V1517NCM_SULFATE_INPUT  -- ##

                , ISNULL(PL.PLAN_PK1502_INPUT           , 0) AS PLAN_PK1502_INPUT

                -- Ni/Mg Sulfate(V1523) 

                , ISNULL(PL.PLAN_V1523NM_SULFATE_INPUT  , 0) AS PLAN_V1523NM_SULFATE_INPUT  -- ##

                , ISNULL(PL.PLAN_PK1503_INPUT           , 0) AS PLAN_PK1503_INPUT

                -- Ni Sulfate(V1526) 

                , ISNULL(PL.PLAN_V1526N_SULFATE_INPUT   , 0) AS PLAN_V1526N_SULFATE_INPUT  -- ##

                , ISNULL(PL.PLAN_PK1504_INPUT           , 0) AS PLAN_PK1504_INPUT

                -- Ni Sulfate(V1531)

                , ISNULL(PL.PLAN_V1531N_SULFATE_INPUT   , 0) AS PLAN_V1531N_SULFATE_INPUT  -- ##

                -- CR FEED TANK

                , ISNULL(PL.PLAN_CR_FEEDTANK_INPUT      , 0) AS PLAN_CR_FEEDTANK_INPUT  -- ##

                , ISNULL(PL.PLAN_LI2CO3_OUTPUT  , 0) AS PLAN_LI2CO3_OUTPUT

                , ISNULL(PL.PLAN_NI3SO4_OUTPUT  , 0) AS PLAN_NI3SO4_OUTPUT

                , ISNULL(PL.PLAN_CO_EXT_OUTPUT  , 0) AS PLAN_CO_EXT_OUTPUT

                , ISNULL(PL.PLAN_LI3PO4_OUTPUT  , 0) AS PLAN_LI3PO4_OUTPUT

                ----------------------------------------------------------

                -- 결과

                , ISNULL(RL.RESULT_BM_SILO_INPUT, 0) / 1000.0 AS RESULT_BM_SILO_INPUT       -- KG -> TON

                , ISNULL(RL.RESULT_FIRING_INPUT , 0) / 1000.0 AS RESULT_FIRING_INPUT

                , ISNULL(RL.RESULT_FIRING_OUTPUT, 0) / 1000.0 AS RESULT_FIRING_OUTPUT

                --, ISNULL(RL.RESULT_PREMIX_INPUT , 0) AS RESULT_PREMIX_INPUT  -- ##

                , ISNULL(RL.RESULT_PREMIX_OUTPUT , 0) / 1000.0 AS RESULT_PREMIX_OUTPUT  -- ##

                --, ISNULL(RL.RESULT_REACTOR_INPUT, 0) AS RESULT_REACTOR_INPUT -- ##

                , ISNULL(RL.RESULT_REACTOR_OUTPUT, 0) / 1000.0 AS RESULT_REACTOR_OUTPUT -- ##

                , ISNULL(RL.RESULT_DRY_OUTPUT   , 0)  / 1000.0 AS RESULT_DRY_OUTPUT

                , ISNULL(RL.RESULT_PK1501_INPUT , 0) AS RESULT_PK1501_INPUT

                -- Ni/Co/Mg Sulfate 

                , ISNULL(RL.RESULT_V1517NCM_SULFATE_INPUT, 0) AS RESULT_V1517NCM_SULFATE_INPUT   -- ##

                , ISNULL(RL.RESULT_PK1502_INPUT          , 0) AS RESULT_PK1502_INPUT

                -- Ni/Mg Sulfate(V1523) 

                , ISNULL(RL.RESULT_V1523NM_SULFATE_INPUT , 0) AS RESULT_V1523NM_SULFATE_INPUT   -- ##

                , ISNULL(RL.RESULT_PK1503_INPUT          , 0) AS RESULT_PK1503_INPUT

                -- Ni Sulfate(V1526) 

                , ISNULL(RL.RESULT_V1526N_SULFATE_INPUT  , 0) AS RESULT_V1526N_SULFATE_INPUT   -- ##

                , ISNULL(RL.RESULT_PK1504_INPUT          , 0) AS RESULT_PK1504_INPUT

                -- Ni Sulfate(V1531)

                , ISNULL(RL.RESULT_V1531N_SULFATE_INPUT  , 0) AS RESULT_V1531N_SULFATE_INPUT   -- ##

                -- CR FEED TANK

                , ISNULL(RL.RESULT_CR_FEEDTANK_INPUT     , 0) AS RESULT_CR_FEEDTANK_INPUT   -- ##

                , ISNULL(RL.RESULT_LI2CO3_OUTPUT, 0) / 1000.0 AS RESULT_LI2CO3_OUTPUT

                , ISNULL(RL.RESULT_NI3SO4_OUTPUT, 0) / 1000.0 AS RESULT_NI3SO4_OUTPUT

                , ISNULL(RL.RESULT_CO_EXT_OUTPUT, 0) AS RESULT_CO_EXT_OUTPUT

                , ISNULL(RL.RESULT_LI3PO4_OUTPUT, 0) / 1000.0 AS RESULT_LI3PO4_OUTPUT

                -------------------------------------------------------------

                -- GAP

                , ( ISNULL(RL.RESULT_BM_SILO_INPUT, 0) / 1000.0) - ISNULL(PL.PLAN_BM_SILO_INPUT  , 0) AS GAP_BM_SILO_INPUT 

                , ( ISNULL(RL.RESULT_FIRING_INPUT , 0) / 1000.0) - ISNULL(PL.PLAN_FIRING_INPUT   , 0) AS GAP_FIRING_INPUT  

                , ( ISNULL(RL.RESULT_FIRING_OUTPUT, 0) / 1000.0)- ISNULL(PL.PLAN_FIRING_OUTPUT  , 0) AS GAP_FIRING_OUTPUT 

                , ( ISNULL(RL.RESULT_PREMIX_OUTPUT, 0) / 1000.0) - ISNULL(PL.PLAN_PREMIX_OUTPUT  , 0) AS GAP_PREMIX_OUTPUT   -- ##

                , ( ISNULL(RL.RESULT_REACTOR_OUTPUT,0) / 1000.0) - ISNULL(PL.PLAN_REACTOR_OUTPUT , 0) AS GAP_REACTOR_OUTPUT  -- ##

                , ( ISNULL(RL.RESULT_DRY_OUTPUT   , 0) / 1000.0) - ISNULL(PL.PLAN_DRY_OUTPUT     , 0) AS GAP_DRY_OUTPUT    

                , ISNULL(RL.RESULT_PK1501_INPUT , 0) - ISNULL(PL.PLAN_PK1501_INPUT   , 0) AS GAP_PK1501_INPUT  

            -- Ni/Co/Mg Sulfate

                , ISNULL(RL.RESULT_V1517NCM_SULFATE_INPUT , 0) - ISNULL(PL.PLAN_V1517NCM_SULFATE_INPUT, 0) AS GAP_V1517NCM_SULFATE_INPUT    -- ##

                , ISNULL(RL.RESULT_PK1502_INPUT           , 0) - ISNULL(PL.PLAN_PK1502_INPUT          , 0) AS GAP_PK1502_INPUT  

                -- Ni/Mg Sulfate(V1523)

                , ISNULL(RL.RESULT_V1523NM_SULFATE_INPUT  , 0) - ISNULL(PL.PLAN_V1523NM_SULFATE_INPUT , 0) AS GAP_V1523NM_SULFATE_INPUT    -- ##

                , ISNULL(RL.RESULT_PK1503_INPUT           , 0) - ISNULL(PL.PLAN_PK1503_INPUT          , 0) AS GAP_PK1503_INPUT  

                -- Ni Sulfate(V1526)

                , ISNULL(RL.RESULT_V1526N_SULFATE_INPUT   , 0) - ISNULL(PL.PLAN_V1526N_SULFATE_INPUT  , 0) AS GAP_V1526N_SULFATE_INPUT    -- ##

                , ISNULL(RL.RESULT_PK1504_INPUT           , 0) - ISNULL(PL.PLAN_PK1504_INPUT          , 0) AS GAP_PK1504_INPUT  

                -- Ni Sulfate(V1531)

                , ISNULL(RL.RESULT_V1531N_SULFATE_INPUT   , 0) - ISNULL(PL.PLAN_V1531N_SULFATE_INPUT  , 0) AS GAP_V1531N_SULFATE_INPUT    -- ##

                -- CR FEED TANK

                , ISNULL(RL.RESULT_CR_FEEDTANK_INPUT, 0) - ISNULL(PL.PLAN_CR_FEEDTANK_INPUT, 0) AS GAP_CR_FEEDTANK_INPUT    -- ##

                , ( ISNULL(RL.RESULT_LI2CO3_OUTPUT, 0) / 1000.0) - ISNULL(PL.PLAN_LI2CO3_OUTPUT  , 0) AS GAP_LI2CO3_OUTPUT 

                , ( ISNULL(RL.RESULT_NI3SO4_OUTPUT, 0) / 1000.0) - ISNULL(PL.PLAN_NI3SO4_OUTPUT  , 0) AS GAP_NI3SO4_OUTPUT 

                , ISNULL(RL.RESULT_CO_EXT_OUTPUT, 0) - ISNULL(PL.PLAN_CO_EXT_OUTPUT  , 0) AS GAP_CO_EXT_OUTPUT 

                , ( ISNULL(RL.RESULT_LI3PO4_OUTPUT, 0) / 1000.0) - ISNULL(PL.PLAN_LI3PO4_OUTPUT  , 0) AS GAP_LI3PO4_OUTPUT 

            FROM dbo.FN_RPT_DAY_WEEK_MONTH( @TYPES, @STARTDATE, @ENDDATE) M

                LEFT OUTER JOIN RPT_PRODUCTION_PLAN PL

                    ON  M.WORKDATE = PL.WORKDATE

                LEFT OUTER JOIN RPT_PRODUCTION_RESULT RL

                    ON  M.WORKDATE = RL.WORKDATE 

        ) 

        INSERT INTO @RTN_DATA

        SELECT 

         '전처리(BP SILO)' AS TOPICS

            , '투입' AS SUBTOPICS

            , 'Ton' AS UNITID

            , ROUND( SUM( ISNULL(PD.PLAN_BM_SILO_INPUT, 0) ), 2)   AS PLANQTY

            , ROUND( SUM( ISNULL(PD.RESULT_BM_SILO_INPUT, 0) ), 2) AS RESULTQTY

            , ROUND( SUM( ISNULL(PD.GAP_BM_SILO_INPUT, 0) ), 2) AS GAPQTY

            , 2010 AS UISEQUENCE

        FROM NEXT_PROCESS_DATA PD

        UNION ALL

        SELECT 

            '소성' AS TOPICS

            , '투입' AS SUBTOPICS

            , 'Ton' AS UNITID

            , ROUND( SUM( ISNULL(PD.PLAN_FIRING_INPUT, 0) ), 2)   AS PLANQTY

            , ROUND( SUM( ISNULL(PD.RESULT_FIRING_INPUT, 0) ), 2) AS RESULTQTY

            , ROUND( SUM( ISNULL(PD.RESULT_FIRING_INPUT, 0) ) - SUM( ISNULL(PD.PLAN_FIRING_INPUT, 0) ), 2) AS GAPQTY

            , 2020 AS UISEQUENCE

        FROM NEXT_PROCESS_DATA PD        

        UNION ALL

        SELECT 

            '소성' AS TOPICS

            , '배출' AS SUBTOPICS

            , 'Ton' AS UNITID

            , ROUND( SUM( ISNULL(PD.PLAN_FIRING_OUTPUT, 0) ), 2)   AS PLANQTY

            , ROUND( SUM( ISNULL(PD.RESULT_FIRING_OUTPUT, 0) ), 2) AS RESULTQTY

            , ROUND( SUM( ISNULL(PD.RESULT_FIRING_OUTPUT, 0) ) - SUM( ISNULL(PD.PLAN_FIRING_OUTPUT, 0) ), 2) AS GAPQTY

            , 2030 AS UISEQUENCE

        FROM NEXT_PROCESS_DATA PD          

        UNION ALL

        SELECT 

            '믹싱(V-1401)' AS TOPICS

            , '배출' AS SUBTOPICS   --배출

(

    @TYPES VARCHAR(40)

    , @STARTDATE VARCHAR(40)

    , @ENDDATE VARCHAR(40)

)

RETURNS @RETARRAY TABLE

(

    WORKDATE    VARCHAR(8)

    , YEARS     VARCHAR(4)

    , YEARMONTH VARCHAR(6)

    , YEARWEEK  VARCHAR(6)

)

AS  

BEGIN  

    

    INSERT @RETARRAY( WORKDATE,YEARS,YEARMONTH,YEARWEEK)

    SELECT 

        WORKDATE,YEARS,YEARMONTH,YEARWEEK

    FROM 

    (

        SELECT 

            CONVERT(VARCHAR(8), WORKDATE, 112) AS WORKDATE

            , CONVERT(VARCHAR(4), WORKDATE, 112) AS YEARS

            , CONVERT(VARCHAR(6), WORKDATE, 112) AS YEARMONTH

            , CONVERT(VARCHAR(4), WORKDATE, 112) + FORMAT(DATEPART(WEEK, WORKDATE), '00') AS YEARWEEK

        FROM 

        (

            SELECT 

                DATEADD(DAY, NUMBER, CONVERT(DATETIME, LEFT(@STARTDATE, 4) + '-01-01')) AS WORKDATE

            FROM master..spt_values 

            WHERE TYPE = 'P'

            AND   NUMBER < DATEDIFF(DAY, CONVERT(DATETIME, LEFT(@STARTDATE, 4) + '-01-01'), CONVERT(DATETIME, LEFT(@ENDDATE, 4) + '-12-31'))

        ) M

    ) P

    WHERE  

        ( 

            (

                ( @TYPES = 'D' ) AND P.WORKDATE BETWEEN @STARTDATE AND @ENDDATE

            )

            OR 

            (

                ( @TYPES = 'W' ) AND P.YEARWEEK BETWEEN @STARTDATE AND @ENDDATE

            )

            OR 

            (

                ( @TYPES = 'M' ) AND P.YEARMONTH BETWEEN @STARTDATE AND @ENDDATE

            )

        )
