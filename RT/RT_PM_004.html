/**********************************************************************************

 * 

 * 부자재 사용량 조회 QUERY

 *   DECLARE @SITEID VARCHAR(40)       -- SITEID : PL01 > 필수

 *         , @TYPES VARCHAR(40)        -- 일/주/월 구분 : [ D | W | M ]  > 필수

 *         , @STARTDATE VARCHAR(40)    -- 조회 FROM 일자  D - 20240701 / W : 202427 / M : 202407  > 필수

 *         , @ENDDATE VARCHAR(40)      -- 조회 TO 일자 :  D - 20240731 / W : 202431 / M : 202407  > 필수

 *         , @PROCESSTYPE VARCHAR(40)  -- LI / NCM(침전) / NCM(추출) 공정 구분 : [ LI | SEDIMENT | EXTRACTION ] > 필수

 *      EXEC SP_RPT_SUBMATERIAL_SELECT @SITEID, @TYPES, @STARTDATE, @ENDDATE, @PROCESSTYPE

 *     EX)    EXEC SP_RPT_SUBMATERIAL_SELECT 'PL01', 'D', '20240701', '20240731', 'LI'

 *            EXEC SP_RPT_SUBMATERIAL_SELECT 'PL01', 'W', '202427', '202431', 'SEDIMENT'

 *            EXEC SP_RPT_SUBMATERIAL_SELECT 'PL01', 'M', '202401', '202412', 'EXTRACTION'

 * 

 **********************************************************************************/

CREATE PROCEDURE [dbo].[SP_RPT_SUBMATERIAL_SELECT] 

    @SITEID VARCHAR(40), @TYPES VARCHAR(40), @STARTDATE VARCHAR(40), @ENDDATE VARCHAR(40), @PROCESSTYPE VARCHAR(40)

AS  

BEGIN 



    ---

    --- LI 공정

    ---

    IF @PROCESSTYPE = 'LI'

    BEGIN 

         WITH BASE_DATA AS 

         (

            -- SUMMARY DATA 조회

            SELECT 

                SM.WORKDATE

                , SM.YEARWEEK

                , SM.YEARMONTH

                , SM.V14011_ADDITIVE 

                , SM.V14011_H2SO4    

                , SM.V14011_REWATER  

                , SM.V14011_PRIWATER 

                , SM.V14012_ADDITIVE 

                , SM.V14012_H2SO4    

                , SM.V14012_REWATER  

                , SM.V14012_PRIWATER 

                , SM.V14081_LICAUS   

                , SM.V14082_LICAUS   

                , SM.V14083_LICAUS

            FROM RPT_SUBMATERIAL_SUM SM

            WHERE CASE WHEN @TYPES = 'D' THEN SM.WORKDATE WHEN @TYPES = 'W' THEN SM.YEARWEEK ELSE SM.YEARMONTH END

                    BETWEEN @STARTDATE AND @ENDDATE

         )

         SELECT 

            C.CODENAME

            , M.WORKDATE

            , M.V14011_ADDITIVE 

            , M.V14011_H2SO4    

            , M.V14011_REWATER  

            , M.V14011_PRIWATER 

            , M.V14012_ADDITIVE 

            , M.V14012_H2SO4    

            , M.V14012_REWATER  

            , M.V14012_PRIWATER 

            , M.V14081_LICAUS   

            , M.V14082_LICAUS   

            , M.V14083_LICAUS

         FROM 

         (

             SELECT

                'LI' AS PROCESSTYPE

                , CASE WHEN @TYPES = 'D' THEN LEFT(WD.RPTDATE, 4) + '-' + SUBSTRING(WD.RPTDATE, 5, 2) + '-' + RIGHT(WD.RPTDATE, 2)  -- 일 : 2024-01-01

                       WHEN @TYPES = 'W' THEN LEFT(WD.RPTDATE, 4) + '-'  + RIGHT(WD.RPTDATE, 2) + @TYPES                            -- 주 : 2024-01W

                       ELSE LEFT(WD.RPTDATE, 4) + '-'  + RIGHT(WD.RPTDATE, 2) + @TYPES                                              -- 월 : 2024-01M

                  END AS WORKDATE   -- 일/주/월 에 따라 WORKDATE 표현 방법  

                , ISNULL(M.V14011_ADDITIVE, 0) AS V14011_ADDITIVE

                , ISNULL(M.V14011_H2SO4   , 0) AS V14011_H2SO4

                , ISNULL(M.V14011_REWATER , 0) AS V14011_REWATER

                , ISNULL(M.V14011_PRIWATER, 0) AS V14011_PRIWATER

                , ISNULL(M.V14012_ADDITIVE, 0) AS V14012_ADDITIVE

                , ISNULL(M.V14012_H2SO4   , 0) AS V14012_H2SO4

                , ISNULL(M.V14012_REWATER , 0) AS V14012_REWATER

                , ISNULL(M.V14012_PRIWATER, 0) AS V14012_PRIWATER

                , ISNULL(M.V14081_LICAUS  , 0) AS V14081_LICAUS

                , ISNULL(M.V14082_LICAUS  , 0) AS V14082_LICAUS

                , ISNULL(M.V14083_LICAUS  , 0) AS V14083_LICAUS

             FROM

                (

                    -- 조회 기간에 대한 일자, 주차, 월 값을 가져옴

                    SELECT 

                        CASE WHEN @TYPES = 'D' THEN WD.WORKDATE WHEN @TYPES = 'W' THEN WD.YEARWEEK ELSE WD.YEARMONTH END AS RPTDATE

                    FROM dbo.FN_RPT_DAY_WEEK_MONTH( @TYPES, @STARTDATE, @ENDDATE) WD   

                    WHERE CASE WHEN @TYPES = 'D' THEN WD.WORKDATE WHEN @TYPES = 'W' THEN WD.YEARWEEK ELSE WD.YEARMONTH END

                            BETWEEN @STARTDATE AND @ENDDATE

                    GROUP BY CASE WHEN @TYPES = 'D' THEN WD.WORKDATE WHEN @TYPES = 'W' THEN WD.YEARWEEK ELSE WD.YEARMONTH END    

                ) WD

                LEFT OUTER JOIN 

                (

                    -- LI DATA 조회

                    SELECT 

                        CASE WHEN @TYPES = 'D' THEN SM.WORKDATE

                               WHEN @TYPES = 'W' THEN SM.YEARWEEK

                               ELSE SM.YEARMONTH 

                          END AS RPTDATE    

                        , SUM( ISNULL(SM.V14011_ADDITIVE, 0) ) AS V14011_ADDITIVE

                        , SUM( ISNULL(SM.V14011_H2SO4   , 0) ) AS V14011_H2SO4

                        , SUM( ISNULL(SM.V14011_REWATER , 0) ) AS V14011_REWATER

                        , SUM( ISNULL(SM.V14011_PRIWATER, 0) ) AS V14011_PRIWATER

                        , SUM( ISNULL(SM.V14012_ADDITIVE, 0) ) AS V14012_ADDITIVE

                        , SUM( ISNULL(SM.V14012_H2SO4   , 0) ) AS V14012_H2SO4

                        , SUM( ISNULL(SM.V14012_REWATER , 0) ) AS V14012_REWATER

                        , SUM( ISNULL(SM.V14012_PRIWATER, 0) ) AS V14012_PRIWATER

                        , SUM( ISNULL(SM.V14081_LICAUS  , 0) ) AS V14081_LICAUS

                        , SUM( ISNULL(SM.V14082_LICAUS  , 0) ) AS V14082_LICAUS

                        , SUM( ISNULL(SM.V14083_LICAUS  , 0) ) AS V14083_LICAUS

                    FROM BASE_DATA SM

                    GROUP BY CASE WHEN @TYPES = 'D' THEN SM.WORKDATE WHEN @TYPES = 'W' THEN SM.YEARWEEK ELSE SM.YEARMONTH END  -- 일/주/월 에 따라 컬럼변경

                ) M

                ON  M.RPTDATE = WD.RPTDATE

            UNION ALL 

            --

            -- 조회 기간 내에 값이 있는 COUNT

            --

            SELECT 

                'LI' AS PROCESSTYPE

                , 'N EA' AS WORKDATE

                , SUM( CASE WHEN BD.V14011_ADDITIVE > 0 THEN 1 ELSE 0 END ) AS V14011_ADDITIVE

                , SUM( CASE WHEN BD.V14011_H2SO4    > 0 THEN 1 ELSE 0 END ) AS V14011_H2SO4

                , SUM( CASE WHEN BD.V14011_REWATER  > 0 THEN 1 ELSE 0 END ) AS V14011_REWATER

                , SUM( CASE WHEN BD.V14011_PRIWATER > 0 THEN 1 ELSE 0 END ) AS V14011_PRIWATER

                , SUM( CASE WHEN BD.V14012_ADDITIVE > 0 THEN 1 ELSE 0 END ) AS V14012_ADDITIVE

                , SUM( CASE WHEN BD.V14012_H2SO4    > 0 THEN 1 ELSE 0 END ) AS V14012_H2SO4

                , SUM( CASE WHEN BD.V14012_REWATER  > 0 THEN 1 ELSE 0 END ) AS V14012_REWATER

                , SUM( CASE WHEN BD.V14012_PRIWATER > 0 THEN 1 ELSE 0 END ) AS V14012_PRIWATER

                , SUM( CASE WHEN BD.V14081_LICAUS   > 0 THEN 1 ELSE 0 END ) AS V14081_LICAUS

                , SUM( CASE WHEN BD.V14082_LICAUS   > 0 THEN 1 ELSE 0 END ) AS V14082_LICAUS

                , SUM( CASE WHEN BD.V14083_LICAUS   > 0 THEN 1 ELSE 0 END ) AS V14083_LICAUS

            FROM 

            (

                 SELECT

                        CASE WHEN @TYPES = 'D' THEN BD.WORKDATE WHEN @TYPES = 'W' THEN BD.YEARWEEK ELSE BD.YEARMONTH END AS WORKDATE

                        , SUM(ISNULL(BD.V14011_ADDITIVE, 0)) AS V14011_ADDITIVE

                        , SUM(ISNULL(BD.V14011_H2SO4, 0))    AS V14011_H2SO4

                        , SUM(ISNULL(BD.V14011_REWATER, 0))  AS V14011_REWATER

                        , SUM(ISNULL(BD.V14011_PRIWATER, 0)) AS V14011_PRIWATER

                        , SUM(ISNULL(BD.V14012_ADDITIVE, 0)) AS V14012_ADDITIVE

                        , SUM(ISNULL(BD.V14012_H2SO4, 0))    AS V14012_H2SO4

                        , SUM(ISNULL(BD.V14012_REWATER, 0))  AS V14012_REWATER

                        , SUM(ISNULL(BD.V14012_PRIWATER, 0)) AS V14012_PRIWATER

                        , SUM(ISNULL(BD.V14081_LICAUS, 0))   AS V14081_LICAUS

                        , SUM(ISNULL(BD.V14082_LICAUS, 0))   AS V14082_LICAUS

                        , SUM(ISNULL(BD.V14083_LICAUS, 0))   AS V14083_LICAUS

                 FROM BASE_DATA BD            

                 GROUP BY CASE WHEN @TYPES = 'D' THEN BD.WORKDATE WHEN @TYPES = 'W' THEN BD.YEARWEEK ELSE BD.YEARMONTH END             

            ) BD

            UNION ALL

             -- 

             -- 조회 기간내 합산(SUM)

             --

             SELECT

                    'LI' AS PROCESSTYPE

                    , 'SUM' AS WORKDATE

                    , SUM(ISNULL(BD.V14011_ADDITIVE, 0)) AS V14011_ADDITIVE

                    , SUM(ISNULL(BD.V14011_H2SO4, 0))    AS V14011_H2SO4

                    , SUM(ISNULL(BD.V14011_REWATER, 0))  AS V14011_REWATER

                    , SUM(ISNULL(BD.V14011_PRIWATER, 0)) AS V14011_PRIWATER

                    , SUM(ISNULL(BD.V14012_ADDITIVE, 0)) AS V14012_ADDITIVE

                    , SUM(ISNULL(BD.V14012_H2SO4, 0))    AS V14012_H2SO4

                    , SUM(ISNULL(BD.V14012_REWATER, 0))  AS V14012_REWATER

                    , SUM(ISNULL(BD.V14012_PRIWATER, 0)) AS V14012_PRIWATER

                    , SUM(ISNULL(BD.V14081_LICAUS, 0))   AS V14081_LICAUS

                    , SUM(ISNULL(BD.V14082_LICAUS, 0))   AS V14082_LICAUS

                    , SUM(ISNULL(BD.V14083_LICAUS, 0))   AS V14083_LICAUS

             FROM BASE_DATA BD

            UNION ALL

            --

            -- 조회 기간내 평균 ( 합산값/COUNT ) .. 일/주/월별 평균값 계산함.

            --

            SELECT 

                'LI' AS PROCESSTYPE

                , 'AVG' AS WORKDATE

                , CASE WHEN SUM(BD.V14011_ADDITIVE) > 0 THEN SUM(BD.V14011_ADDITIVE) / SUM( CASE WHEN BD.V14011_ADDITIVE > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14011_ADDITIVE

                , CASE WHEN SUM(BD.V14011_H2SO4)    > 0 THEN SUM(BD.V14011_H2SO4)    / SUM( CASE WHEN BD.V14011_H2SO4    > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14011_H2SO4

                , CASE WHEN SUM(BD.V14011_REWATER)  > 0 THEN SUM(BD.V14011_REWATER)  / SUM( CASE WHEN BD.V14011_REWATER  > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14011_REWATER

                , CASE WHEN SUM(BD.V14011_PRIWATER) > 0 THEN SUM(BD.V14011_PRIWATER) / SUM( CASE WHEN BD.V14011_PRIWATER > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14011_PRIWATER

                , CASE WHEN SUM(BD.V14012_ADDITIVE) > 0 THEN SUM(BD.V14012_ADDITIVE) / SUM( CASE WHEN BD.V14012_ADDITIVE > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14012_ADDITIVE

                , CASE WHEN SUM(BD.V14012_H2SO4)    > 0 THEN SUM(BD.V14012_H2SO4)    / SUM( CASE WHEN BD.V14012_H2SO4    > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14012_H2SO4

                , CASE WHEN SUM(BD.V14012_REWATER)  > 0 THEN SUM(BD.V14012_REWATER)  / SUM( CASE WHEN BD.V14012_REWATER  > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14012_REWATER

                , CASE WHEN SUM(BD.V14012_PRIWATER) > 0 THEN SUM(BD.V14012_PRIWATER) / SUM( CASE WHEN BD.V14012_PRIWATER > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14012_PRIWATER

                , CASE WHEN SUM(BD.V14081_LICAUS)   > 0 THEN SUM(BD.V14081_LICAUS)   / SUM( CASE WHEN BD.V14081_LICAUS   > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14081_LICAUS

                , CASE WHEN SUM(BD.V14082_LICAUS)   > 0 THEN SUM(BD.V14082_LICAUS)   / SUM( CASE WHEN BD.V14082_LICAUS   > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14082_LICAUS

                , CASE WHEN SUM(BD.V14083_LICAUS)   > 0 THEN SUM(BD.V14083_LICAUS)   / SUM( CASE WHEN BD.V14083_LICAUS   > 0 THEN 1 ELSE 0 END ) ELSE 0 END AS V14083_LICAUS

            FROM 

            (

                 SELECT

                        CASE WHEN @TYPES = 'D' THEN BD.WORKDATE WHEN @TYPES = 'W' THEN BD.YEARWEEK ELSE BD.YEARMONTH END AS WORKDATE

                        , SUM(ISNULL(BD.V14011_ADDITIVE, 0)) AS V14011_ADDITIVE

                        , SUM(ISNULL(BD.V14011_H2SO4, 0))    AS V14011_H2SO4

                        , SUM(ISNULL(BD.V14011_REWATER, 0))  AS V14011_REWATER

                        , SUM(ISNULL(BD.V14011_PRIWATER, 0)) AS V14011_PRIWATER

                        , SUM(ISNULL(BD.V14012_ADDITIVE, 0)) AS V14012_ADDITIVE

                        , SUM(ISNULL(BD.V14012_H2SO4, 0))    AS V14012_H2SO4

                        , SUM(ISNULL(BD.V14012_REWATER, 0))  AS V14012_REWATER

                        , SUM(ISNULL(BD.V14012_PRIWATER, 0)) AS V14012_PRIWATER

                        , SUM(ISNULL(BD.V14081_LICAUS, 0))   AS V14081_LICAUS

                        , SUM(ISNULL(BD.V14082_LICAUS, 0))   AS V14082_LICAUS

                        , SUM(ISNULL(BD.V14083_LICAUS, 0))   AS V14083_LICAUS

                 FROM BASE_DATA BD            

                 GROUP BY CASE WHEN @TYPES = 'D' THEN BD.WORKDATE WHEN @TYPES = 'W' THEN BD.YEARWEEK ELSE BD.YEARMONTH END             

            ) BD        

        ) M

        LEFT OUTER JOIN CIM_CODE C

            ON   C.CODECLASSID  = 'RPT_SUBMATERIAL_GROUP'

            AND  C.SITEID = @SITEID

            AND  C.CODEID = M.PROCESSTYPE

    END 



    --

    -- NCM 침출

    --

    IF @PROCESSTYPE = 'SEDIMENT'

    BEGIN 

        WITH BASE_DATA AS 

         (

            SELECT 

                SM.WORKDATE

                , SM.YEARWEEK

                , SM.YEARMONTH

                , V15011_H2SO4

                , V15011_H2O2

                , V15012_H2SO4

                , V15012_H2O2

                , V15013_H2SO4

                , V15013_H2O2

                , V15014_H2SO4

                , V15014_H2O2

                , V15015_H2SO4

                , V15015_H2O2

                , V15016_H2SO4

                , V15016_H2O2

                , V15041_H2SO4

                , V15041_H2O2

                , V15042_H2SO4

                , V15042_H2O2

                , V15043_H2SO4

                , V15043_H2O2

                , V15044_H2SO4

                , V15044_H2O2

                , V15045_H2SO4

                , V15045_H2O2

                , V15046_H2SO4

                , V15046_H2O2

                , V15121_H2SO4

                , V15122_H2SO4

            FROM RPT_SUBMATERIAL_SUM SM

            WHERE CASE WHEN @TYPES = 'D' THEN SM.WORKDATE WHEN @TYPES = 'W' THEN SM.YEARWEEK ELSE SM.YEARMONTH END

                    BETWEEN @STARTDATE AND @ENDDATE                    

         )

         SELECT

            C.CODENAME

            , M.WORKDATE

            , M.V15011_H2SO4

            , M.V15011_H2O2

            , M.V15012_H2SO4

            , M.V15012_H2O2

            , M.V15013_H2SO4

            , M.V15013_H2O2

            , M.V15014_H2SO4

            , M.V15014_H2O2

            , M.V15015_H2SO4

            , M.V15015_H2O2

            , M.V15016_H2SO4

            , M.V15016_H2O2

            , M.V15041_H2SO4

            , M.V15041_H2O2

            , M.V15042_H2SO4

            , M.V15042_H2O2

            , M.V15043_H2SO4

            , M.V15043_H2O2

            , M.V15044_H2SO4

            , M.V15044_H2O2

            , M.V15045_H2SO4

            , M.V15045_H2O2

            , M.V15046_H2SO4

            , M.V15046_H2O2

            , M.V15121_H2SO4

            , M.V15122_H2SO4

         FROM 

         (

             SELECT

                'SEDIMENT' AS PROCESSTYPE

                , CASE WHEN @TYPES = 'D' THEN LEFT(WD.RPTDATE, 4) + '-' + SUBSTRING(WD.RPTDATE, 5, 2) + '-' + RIGHT(WD.RPTDATE, 2)

                       WHEN @TYPES = 'W' THEN LEFT(WD.RPTDATE, 4) + '-'  + RIGHT(WD.RPTDATE, 2) + @TYPES

                       ELSE LEFT(WD.RPTDATE, 4) + '-'  + RIGHT(WD.RPTDATE, 2) + @TYPES 

                  END AS WORKDATE

                , ISNULL(SM.V15011_H2SO4, 0) AS V15011_H2SO4

                , ISNULL(SM.V15011_H2O2,  0) AS V15011_H2O2

                , ISNULL(SM.V15012_H2SO4, 0) AS V15012_H2SO4

                , ISNULL(SM.V15012_H2O2,  0) AS V15012_H2O2

                , ISNULL(SM.V15013_H2SO4, 0) AS V15013_H2SO4

                , ISNULL(SM.V15013_H2O2,  0) AS V15013_H2O2

                , ISNULL(SM.V15014_H2SO4, 0) AS V15014_H2SO4

                , ISNULL(SM.V15014_H2O2,  0) AS V15014_H2O2

                , ISNULL(SM.V15015_H2SO4, 0) AS V15015_H2SO4

                , ISNULL(SM.V15015_H2O2,  0) AS V15015_H2O2

                , ISNULL(SM.V15016_H2SO4, 0) AS V15016_H2SO4

                , ISNULL(SM.V15016_H2O2,  0) AS V15016_H2O2

                , ISNULL(SM.V15041_H2SO4, 0) AS V15041_H2SO4

                , ISNULL(SM.V15041_H2O2,  0) AS V15041_H2O2

                , ISNULL(SM.V15042_H2SO4, 0) AS V15042_H2SO4

--     @TYPES : 일,주,월 [ D | W | M ]

--     @STARTDATE  : 시작일,주,월  [ 20241231 | 202453 | 202412 ]

--     @ENDDATE  : 시작일,주,월  [ 20241231 | 202453 | 202412 ]

CREATE FUNCTION [dbo].[FN_RPT_DAY_WEEK_MONTH]

(

    @TYPES VARCHAR(40)

    , @STARTDATE VARCHAR(40)

    , @ENDDATE VARCHAR(40)

)

RETURNS @RETARRAY TABLE

(

    WORKDATE    VARCHAR(8)

    , YEARS     VARCHAR(4)

    , YEARMONTH VARCHAR(6)

    , YEARWEEK  VARCHAR(6)

)

AS  

BEGIN  

    

    INSERT @RETARRAY( WORKDATE,YEARS,YEARMONTH,YEARWEEK)

    SELECT 

        WORKDATE,YEARS,YEARMONTH,YEARWEEK

    FROM 

    (

        SELECT 

            CONVERT(VARCHAR(8), WORKDATE, 112) AS WORKDATE

            , CONVERT(VARCHAR(4), WORKDATE, 112) AS YEARS

            , CONVERT(VARCHAR(6), WORKDATE, 112) AS YEARMONTH

            , CONVERT(VARCHAR(4), WORKDATE, 112) + FORMAT(DATEPART(WEEK, WORKDATE), '00') AS YEARWEEK

        FROM 

        (

            SELECT 

                DATEADD(DAY, NUMBER, CONVERT(DATETIME, LEFT(@STARTDATE, 4) + '-01-01')) AS WORKDATE

            FROM master..spt_values 

            WHERE TYPE = 'P'

            AND   NUMBER < DATEDIFF(DAY, CONVERT(DATETIME, LEFT(@STARTDATE, 4) + '-01-01'), CONVERT(DATETIME, LEFT(@ENDDATE, 4) + '-12-31'))

        ) M

    ) P

    WHERE  

        ( 

            (

                ( @TYPES = 'D' ) AND P.WORKDATE BETWEEN @STARTDATE AND @ENDDATE

            )

            OR 

            (

                ( @TYPES = 'W' ) AND P.YEARWEEK BETWEEN @STARTDATE AND @ENDDATE

            )

            OR 

            (

                ( @TYPES = 'M' ) AND P.YEARMONTH BETWEEN @STARTDATE AND @ENDDATE

            )

        )

    ;



    RETURN

END
