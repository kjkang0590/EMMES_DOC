--RT_MD_003_설비 가동율 기준정보 조회 쿼리
--SelectOperationRateDefinitionList
--Param : SITEID, UNIT
------------------------------------------
SELECT A.UNIT,
       IIF(B.CODENAME IS NULL, 'BM공정', B.CODENAME) AS UNITNAME,
       A.ITEMID,
       A.ITEMNAME,
       A.ITEMCOUNT,
       A.DESCRIPTION,
       A.ISUSETIMERATE,
       A.ISUSEPERFORMANCERATE,
       A.TIMERATE,
       A.PERFORMANCERATE,
       A.COMPRERATE
  FROM RPT_OPERATIONRATE_DEFINITION A WITH(NOLOCK)
  LEFT JOIN CIM_CODE B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.UNIT = B.CODEID
   AND B.CODECLASSID = 'RPT_UNIT'
 WHERE A.SITEID = @SITEID
   AND (@UNIT = '' OR A.UNIT = @UNIT)
   AND  (B.CODENAME IS NULL OR A.ISUSABLE = 'Usable')
 ORDER BY A.UISEQUENCE
