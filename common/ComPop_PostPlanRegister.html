--RT_MD_002_계획대비 생산량 기준정보 - 후처리 계획 등록 POPUP 조회 쿼리

SELECT US1.RPT_DATE,
	   MAX(US1.WORKDATE) AS WORKDATE,
	   MAX(US1.YEARMONTH) AS YEARMONTH,
	   MAX(US1.YEARWEEK) AS YEARWEEK,
	   MAX(US1.PLAN_BM_SILO_INPUT) AS PLAN_BM_SILO_INPUT,
	   MAX(US1.PLAN_FIRING_INPUT) AS PLAN_FIRING_INPUT,
	   MAX(US1.PLAN_FIRING_OUTPUT) AS PLAN_FIRING_OUTPUT,
	   MAX(US1.PLAN_PREMIX_INPUT) AS PLAN_PREMIX_INPUT,
	   MAX(US1.PLAN_REACTOR_INPUT) AS PLAN_REACTOR_INPUT,
	   MAX(US1.PLAN_DRY_INPUT) AS PLAN_DRY_INPUT,
	   MAX(US1.PLAN_DRY_OUTPUT) AS PLAN_DRY_OUTPUT,
	   MAX(US1.PLAN_LI2CO3_OUTPUT) AS PLAN_LI2CO3_OUTPUT,
	   MAX(US1.PLAN_LI3PO4_OUTPUT) AS PLAN_LI3PO4_OUTPUT,
	   MAX(US1.PLAN_PK1501_INPUT) AS PLAN_PK1501_INPUT,
	   MAX(US1.PLAN_PK1502_INPUT) AS PLAN_PK1502_INPUT,
	   MAX(US1.PLAN_PK1503_INPUT) AS PLAN_PK1503_INPUT,
	   MAX(US1.PLAN_PK1504_INPUT) AS PLAN_PK1504_INPUT,
	   MAX(US1.PLAN_CO_EXT_OUTPUT) AS PLAN_CO_EXT_OUTPUT,
	   MAX(US1.PLAN_NI_EXT_OUTPUT) AS PLAN_NI_EXT_OUTPUT,
	   MAX(US1.PLAN_NI3SO4_OUTPUT) AS PLAN_NI3SO4_OUTPUT
  FROM (
  	SELECT S1.RPT_DATE,
       '' AS WORKDATE,
	   '' AS YEARMONTH,
	   '' AS YEARWEEK,
	   '' AS PLAN_BM_SILO_INPUT,
	   '' AS PLAN_FIRING_INPUT,
	   '' AS PLAN_FIRING_OUTPUT,
	   '' AS PLAN_PREMIX_INPUT,
	   '' AS PLAN_REACTOR_INPUT,
	   '' AS PLAN_DRY_INPUT,
	   '' AS PLAN_DRY_OUTPUT,
	   '' AS PLAN_LI2CO3_OUTPUT,
	   '' AS PLAN_LI3PO4_OUTPUT,
	   '' AS PLAN_PK1501_INPUT,
	   '' AS PLAN_PK1502_INPUT,
	   '' AS PLAN_PK1503_INPUT,
	   '' AS PLAN_PK1504_INPUT,
	   '' AS PLAN_CO_EXT_OUTPUT,
	   '' AS PLAN_NI_EXT_OUTPUT,
	   '' AS PLAN_NI3SO4_OUTPUT
  FROM (
  	SELECT REPLACE(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), '-', '') AS RPT_DATE
  	  FROM MASTER..SPT_VALUES A WITH(NOLOCK)
     WHERE A.TYPE = 'P' 
       AND SUBSTRING(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), 1, 7) = SUBSTRING(@SEARCHDATE, 1, 7)
  ) S1
UNION
SELECT S1.RPT_DATE,
       PP.WORKDATE,
	   PP.YEARMONTH,
	   PP.YEARWEEK,
	   PP.PLAN_BM_SILO_INPUT,
	   PP.PLAN_FIRING_INPUT,
	   PP.PLAN_FIRING_OUTPUT,
	   PP.PLAN_PREMIX_INPUT,
	   PP.PLAN_REACTOR_INPUT,
	   PP.PLAN_DRY_INPUT,
	   PP.PLAN_DRY_OUTPUT,
	   PP.PLAN_LI2CO3_OUTPUT,
	   PP.PLAN_LI3PO4_OUTPUT,
	   PP.PLAN_PK1501_INPUT,
	   PP.PLAN_PK1502_INPUT,
	   PP.PLAN_PK1503_INPUT,
	   PP.PLAN_PK1504_INPUT,
	   PP.PLAN_CO_EXT_OUTPUT,
	   PP.PLAN_NI_EXT_OUTPUT,
	   PP.PLAN_NI3SO4_OUTPUT
  FROM (
  	SELECT REPLACE(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), '-', '') AS RPT_DATE
  	  FROM MASTER..SPT_VALUES A WITH(NOLOCK)
     WHERE A.TYPE = 'P' 
       AND SUBSTRING(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), 1, 7) = SUBSTRING(@SEARCHDATE, 1, 7)
  ) S1
  LEFT JOIN RPT_PRODUCTION_PLAN PP WITH(NOLOCK)
    ON S1.RPT_DATE = PP.WORKDATE
 WHERE PP.SITEID = @SITEID
  ) US1
GROUP BY US1.RPT_DATE
