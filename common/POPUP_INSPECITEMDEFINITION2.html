--QM_AN_002~QM_AN_005 검사 의뢰 요청,접수,등록,검사 결과 등록,조회
--SelectInspDefListForInspReq
--Param : SITEID, INSPDEFINITIONSYSID
------------------------------------------
SELECT ID.INSPDEFINITIONSYSID
     , ID.INSPDEFINITIONNAME
     , ID.INSPTYPE
     , REQ.INSPMETHOD AS INSPMETHODCODE
     , CCR.CODENAME AS INSPMETHOD
     , ID.ISUSABLE
     , CCI.CODENAME AS ISUSABLENAME
     , ID.DESCRIPTION
  FROM CIM_INSPDEFINITION ID WITH(NOLOCK)
  LEFT JOIN (
  	SELECT TOP 1 SITEID, INSPMETHOD, INSPDEFINITIONSYSID 
  	  FROM CIM_INSPREQ WITH(NOLOCK)
  	 WHERE SITEID = @SITEID
  	   AND INSPDEFINITIONSYSID = @INSPDEFINITIONSYSID
  ) REQ
    ON REQ.SITEID = ID.SITEID
   AND REQ.INSPDEFINITIONSYSID = ID.INSPDEFINITIONSYSID
  LEFT JOIN CIM_CODE CCI WITH(NOLOCK)
    ON CCI.SITEID = ID.SITEID
   AND CCI.CODEID = ID.ISUSABLE
   AND CCI.CODECLASSID = 'ISUSABLE'
  LEFT JOIN CIM_CODE CCR WITH(NOLOCK)
    ON CCR.SITEID = REQ.SITEID
   AND CCR.CODEID = REQ.INSPMETHOD
   AND CCR.CODECLASSID = 'REQUESTCATEGORY'
 WHERE ID.SITEID = @SITEID
   AND ID.INSPDEFINITIONSYSID = @INSPDEFINITIONSYSID
 ORDER BY ID.INSPDEFINITIONSYSID

--SelectCboUnitByDivision
--Param : SITEID, UNITDIVISION

SELECT UNITID AS CODE
    , UNITNAME AS NAME
    , UISEQUENCE
  FROM CIM_UNIT WITH(NOLOCK)
 WHERE SITEID = @SITEID
   AND ISUSABLE = 'Usable'
   AND (@UNITDIVISION = '' OR COMMENTS LIKE '%' + @UNITDIVISION + '%')
 ORDER BY UISEQUENCE, UNITID
