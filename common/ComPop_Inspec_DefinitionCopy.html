--MD_QM_004_검사 표준 IQC - 검사 사양 목록 조회
--SelectIQCINSPDefList
--Param : SITEID, COPERATOR, MATERIALCLASSID, MATERIALDEFINITIONID, ISUSABLE, INSPDEFINITIONNAME
------------------------------------------
SELECT CID.INSPDEFINITIONSYSID
     , CID.INSPDEFINITIONNAME
     , CID.DESCRIPTION
     , CID.COPERATOR
     , CID.ITEMID
     , CM.MATERIALDEFINITIONNAME
     , CID.INSPTYPE
     , CID.ISUSABLE
     , CCU.USERNAME AS CREATOR
     , CID.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , CID.MODIFYTIME
  FROM CIM_INSPDEFINITION CID WITH(NOLOCK)
 INNER JOIN CIM_MATERIALDEFINITION CM WITH(NOLOCK) ON CM.SITEID = CID.SITEID AND CM.MATERIALDEFINITIONID = CID.ITEMID
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK) ON CCU.SITEID = CID.SITEID AND CCU.USERID = CID.CREATOR
  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK) ON MCU.SITEID = CID.SITEID AND MCU.USERID = CID.MODIFIER
 WHERE CID.SITEID = @SITEID
   AND CID.ISUSABLE = @ISUSABLE
   AND CID.INSPTYPE = 'IQC'
   AND (@COPERATOR = '' OR CID.COPERATOR = @COPERATOR)
   AND (@MATERIALCLASSID = '' OR CM.MATERIALCLASSID = @MATERIALCLASSID)
   AND (@MATERIALDEFINITIONID = '' OR CID.ITEMID LIKE '%' + @MATERIALDEFINITIONID + '%')
   AND (@INSPDEFINITIONNAME = '' OR CID.INSPDEFINITIONNAME LIKE '%' + @INSPDEFINITIONNAME + '%')
 ORDER BY CID.INSPDEFINITIONNAME, CID.INSPDEFINITIONSYSID

------------------------------------------
--MD_QM_004_검사 표준 IQC - 검사 항목 스펙 조회
--SelectINSPDefItemSpecList
--Param : SITEID, INSPDEFINITIONSYSID
------------------------------------------
SELECT IIS.INSPDEFINITIONSYSID
     , IIS.INSPDEFINITIONITEMSPECSYSID
     , CIC.INSPITEMCLASSID
     , CIC.INSPITEMCLASSNAME
     , IIS.INSPITEMID
     , CI.INSPITEMNAME
     , IIS.DESCRIPTION
     , IIS.UISEQUENCE
     , IIS.USL
     , IIS.LSL
     , IIS.UCL
     , IIS.LCL
     , IIS.TARGET
     , IIS.UNITID
     , IIS.ISUSABLE
     , CCU.USERNAME AS CREATOR
     , IIS.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , IIS.MODIFYTIME
  FROM CIM_INSPDEFINITIONITEMSPEC IIS WITH(NOLOCK)
  INNER JOIN CIM_INSPITEM CI WITH(NOLOCK) ON IIS.SITEID = CI.SITEID AND IIS.INSPITEMID = CI.INSPITEMID
  INNER JOIN CIM_INSPITEMCLASS CIC WITH(NOLOCK) ON CIC.SITEID = CI.SITEID AND CIC.INSPITEMCLASSID = CI.INSPITEMCLASSID
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK) ON CCU.SITEID = IIS.SITEID AND CCU.USERID = IIS.CREATOR
  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK) ON MCU.SITEID = IIS.SITEID AND MCU.USERID = IIS.MODIFIER
 WHERE IIS.SITEID = @SITEID
   AND IIS.INSPDEFINITIONSYSID = @INSPDEFINITIONSYSID
   AND (@ISUSABLE = '' OR IIS.ISUSABLE = @ISUSABLE)
 ORDER BY IIS.INSPDEFINITIONSYSID, IIS.UISEQUENCE, IIS.INSPDEFINITIONITEMSPECSYSID
