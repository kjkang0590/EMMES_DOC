--PM_PB_001 공정 진행 (반응) - POPUP 환산수량 조회 (종료 POPUP)
--SelectExchangeMetalStartQty
--Param : SITEID, MATERIALDEFINITIONID, LOTID
------------------------------------------
SELECT CASE WHEN B.PROCESSSEGMENTID = 'P2030' THEN 'Li'
            WHEN B.PROCESSSEGMENTID = 'P2040' THEN 'Co'
            WHEN B.PROCESSSEGMENTID = 'P2050' THEN 'Ni'
       END AS PRODUCT
     , CASE WHEN B.PROCESSSEGMENTID = 'P2030' THEN L.LITHIUMMETALQTY
            WHEN B.PROCESSSEGMENTID = 'P2040' THEN L.COBALTMETALQTY
            WHEN B.PROCESSSEGMENTID = 'P2050' THEN L.NICKELMETALQTY
       END AS QTY
  FROM CIM_BOM B WITH(NOLOCK)
 INNER JOIN CIM_PRODUCTDEFINITION P WITH(NOLOCK) ON P.SITEID = B.SITEID AND P.PRODUCTDEFINITIONID = B.MATERIALDEFINITIONID AND P.PRODUCTTYPE = 'WIP'
 INNER JOIN CIM_LOT L  WITH(NOLOCK) ON L.SITEID = @SITEID AND L.LOTID = @LOTID 
 WHERE B.SITEID = @SITEID
   AND B.PRODUCTDEFINITIONID = @MATERIALDEFINITIONID
   AND B.MATERIALDEFINITIONID <> '2LSS' --망초 제외
   AND B.PROCESSSEGMENTID IN (SELECT NEXTPROCESSNODEID FROM CIM_PROCESSPATH WHERE PROCESSNODEID = 'P2020')

--[SERVICE] Metal 환산수량으로 BP 환산수량 계산
--GetCalculateBPExchangeQty
--Param : TOTALINPUTQTY, LI_METAL_QTY, CO_METAL_QTY, NI_METAL_QTY
------------------------------------------
SELECT CONVERT(float, @TOTALINPUTQTY) AS TOTALINPUTQTY
     , CONVERT(float, A.LI_BP_QTY) AS LI_BP_QTY
     , CONVERT(float, A.CO_BP_QTY) AS CO_BP_QTY
     , CONVERT(float, @TOTALINPUTQTY) - CONVERT(float, A.LI_BP_QTY) - CONVERT(float, A.CO_BP_QTY) AS NI_BP_QTY
--     , CONVERT(float, A.LI_BP_QTY) + CONVERT(float, A.CO_BP_QTY) + CONVERT(float, @TOTALINPUTQTY) - CONVERT(float, A.LI_BP_QTY) - CONVERT(float, A.CO_BP_QTY) AS TOTALSUMQTY
  FROM (
        SELECT ROUND(CONVERT(float, @TOTALINPUTQTY) * CONVERT(float,  @LI_METAL_QTY) / (CONVERT(float, @LI_METAL_QTY) + CONVERT(float, @CO_METAL_QTY) + CONVERT(float, @NI_METAL_QTY )), 2) AS LI_BP_QTY
             , ROUND(CONVERT(float, @TOTALINPUTQTY) * CONVERT(float, @CO_METAL_QTY) / (CONVERT(float, @LI_METAL_QTY) + CONVERT(float, @CO_METAL_QTY) + CONVERT(float, @NI_METAL_QTY)), 2) AS CO_BP_QTY
             , ROUND(CONVERT(float, @TOTALINPUTQTY) * CONVERT(float, @NI_METAL_QTY) / (CONVERT(float, @LI_METAL_QTY) + CONVERT(float, @CO_METAL_QTY) + CONVERT(float, @NI_METAL_QTY )), 2) AS NI_BP_QTY
       ) AS A
