--MD_PM_007_설비별 Tag 관리 - 상위 설비 조회
--SelectEmmeParentEquipmentList
--Param : SITEID, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID
------------------------------------------
SELECT A.EQUIPMENTID,
	   A.EQUIPMENTNAME,
	   B.EQUIPMENTCLASSNAME AS EQUIPMENTCLASSID,
	   A.EQUIPMENTLEVEL,
	   A.PARENTID,
	   G.CODENAME AS PROCESSTYPE,
	   C.PROCESSSEGMENTNAME AS PROCESSSEGMENTID,
	   H.CODENAME AS ROOTEQUIPMENTID,
	   A.TANKCAPA,
	   A.TANKSIZE,
	   A.TANKTYPE,
	   I.CODENAME AS MEDICINECODE,
	   A.WARNINGQTY,
	   A.SAFETYQTY,
	   J.CODENAME AS ISUSABLE,
	   D.ATTACHFILENAME AS EQUIPMENTIMAGE,
	   E.USERNAME AS CREATOR,
	   A.CREATETIME,
	   F.USERNAME AS MODIFIER,
	   A.MODIFYTIME
  FROM CIM_EQUIPMENT A WITH(NOLOCK)
  LEFT JOIN CIM_EQUIPMENTCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.EQUIPMENTCLASSID = B.EQUIPMENTCLASSID
  LEFT JOIN CIM_PROCESSSEGMENT C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.PROCESSSEGMENTID = C.PROCESSSEGMENTID
  LEFT JOIN CIM_ATTACHMENT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.EQUIPMENTID = D.RELATIONID
   AND D.RELATIONTYPE = 'MD_PM_008'
   AND D.ISUSABLE = 'Usable'
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.CREATOR = E.USERID
  LEFT JOIN CIM_USER F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.MODIFIER = F.USERID
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON A.SITEID = G.SITEID
   AND A.PROCESSTYPE = G.CODEID
   AND G.CODECLASSID = 'PROCESSTYPE'
  LEFT JOIN CIM_CODE H WITH(NOLOCK)
    ON A.SITEID = H.SITEID
   AND A.ROOTEQUIPMENTID = H.CODEID
   AND H.CODECLASSID = 'ROOTEQUIPMENT'
  LEFT JOIN CIM_CODE I WITH(NOLOCK)
    ON A.SITEID = I.SITEID
   AND A.MEDICINECODE = I.CODEID
   AND I.CODECLASSID = 'MEDICINECODE'
  LEFT JOIN CIM_CODE J WITH(NOLOCK)
    ON A.SITEID = J.SITEID
   AND A.ISUSABLE = J.CODEID
   AND I.CODECLASSID = 'ISUSABLE'
 WHERE A.SITEID = @SITEID
   AND A.ROOTEQUIPMENTID = @ROOTEQUIPMENTID 
   AND A.EQUIPMENTCLASSID = @EQUIPMENTCLASSID
   AND A.EQUIPMENTID = @EQUIPMENTID
   AND A.ISUSABLE = 'Usable'

SELECT A.EQUIPMENTID AS PARENTID,
	   B.CHILDID AS EQUIPMENTID,
	   C.EQUIPMENTCLASSNAME,
	   F.CODENAME AS ROOTEQUIPMENTID,
	   E.CODENAME AS PROCESSTYPE,
	   D.PROCESSSEGMENTNAME AS PROCESSSEGMENTID

 INNER JOIN CST_PUBLICREL B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.EQUIPMENTID = B.PARENTID
  LEFT JOIN CIM_EQUIPMENT A2 WITH(NOLOCK)
    ON A.SITEID = A2.SITEID
   AND B.CHILDID = A2.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS C WITH(NOLOCK)
    ON A2.SITEID = C.SITEID
   AND A2.EQUIPMENTCLASSID = C.EQUIPMENTCLASSID
  LEFT JOIN CIM_PROCESSSEGMENT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.PROCESSSEGMENTID = D.PROCESSSEGMENTID
  LEFT JOIN CIM_CODE E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.PROCESSTYPE = E.CODEID
   AND E.CODECLASSID = 'PROCESSTYPE'
  LEFT JOIN CIM_CODE F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.ROOTEQUIPMENTID = F.CODEID
   AND F.CODECLASSID = 'ROOTEQUIPMENT'
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON A.SITEID = G.SITEID
   AND A.ISUSABLE = G.CODEID
   AND G.CODECLASSID = 'ISUSABLE'
 WHERE A.SITEID = @SITEID
   AND A.ISUSABLE = 'Usable'
   AND B.PARENTID = @PARENTID
