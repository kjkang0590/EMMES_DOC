SQL 보기
------------------------------------------
--MD_QM_002_Recipe 관리
--SelectRecipeHistList
--Param : SITEID, EQUIPMENTID, RECIPEITEMID
------------------------------------------
SELECT LASTEVENTTIME 
     , ROOTEQUIPMENTID
     , EQUIPMENTID
     , EQUIPMENTNAME
     , RECIPEITEMNAME
     , TAGID
     , UNITID
     , LCL
     , UCL
     , LSL
     , USL
     , TARGET
     , INEQUATION
     , ISUSABLE
     , RECIPEDESCRIPTION
  FROM CST_RECIPEINFOHIST WITH(NOLOCK)
 WHERE SITEID = @SITEID
   AND (@EQUIPMENTID = '' OR EQUIPMENTID = @EQUIPMENTID)
   AND (@RECIPEITEMID = '' OR RECIPEITEMID = @RECIPEITEMID)
 ORDER BY LASTEVENTSEQ DESC, EQUIPMENTID ASC, RECIPEITEMID ASC

SQL 보기
------------------------------------------
--공통 단위 조회
--SelectCboUnitByDivision
--Param : SITEID, UNITDIVISION
------------------------------------------
SELECT UNITID AS CODE
    , UNITNAME AS NAME
    , UISEQUENCE
  FROM CIM_UNIT WITH(NOLOCK)
 WHERE SITEID = @SITEID
   AND ISUSABLE = 'Usable'
   AND (@UNITDIVISION = '' OR COMMENTS LIKE '%' + @UNITDIVISION + '%')
 ORDER BY UISEQUENCE, UNITID
