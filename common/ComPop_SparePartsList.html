--EM_PB_005_BM 관리 - Spare Parts 목록 Grid 조회 쿼리
--SelectSpartComPopSearch
--Param : SITEID
------------------------------------------
SELECT MAX(D.LOCATION)          AS LOCATION,
       MAX(E.FACILITYNAME)      AS LOCATIONNAME,
       A.SPARTCLASSID,
       MAX(B.SPARTCLASSNAME)    AS SPARTCLASSNAME,
       A.SPARTDEFINITIONID      AS SPARTDEFINITION,
       MAX(A.SPARTDEFINITIONNAME) AS SPARTDEFINITIONNAME,
       ISNULL(SUM(D.QTY), 0)    AS QTY,          -- ★ 변경: A.QTY → D.QTY(실제 재고)
       MAX(A.UNITID)            AS UNITID,
       MAX(A.SAFETYSTOCK)       AS SAFETYSTOCK
  FROM CIM_SPARTDEFINITION A WITH(NOLOCK)
  LEFT JOIN CIM_SPARTCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.SPARTCLASSID = B.SPARTCLASSID
  LEFT JOIN CST_MATERIALLOT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.SPARTDEFINITIONID = D.MATERIALDEFINITIONID
   AND D.STATE = 'Active'                        -- ★ 추가: 종료/폐기 Lot 제외
  LEFT JOIN CIM_FACILITY E WITH(NOLOCK)
    ON D.SITEID = E.SITEID
   AND D.LOCATION = E.FACILITYID
 WHERE A.SITEID = @SITEID
 GROUP BY A.SPARTCLASSID, A.SPARTDEFINITIONID
 ORDER BY A.SPARTCLASSID;

SELECT CODEID AS CODE,
       CODENAME AS NAME,
       UISEQUENCE
FROM CIM_CODE 
WHERE CODECLASSID = :CODECLASSID
      AND ISUSABLE= 'Usable' 
      AND SITEID = :SITEID
ORDER BY UISEQUENCE
