-- SelectInspDefinitionListByInspItem 검사표준 조회문(검사항목으로) = QMS_Inspdefinition_101

-- INPUT = SITEID, ISUSABLE, INSPITEMCLASSID, INSPITEMCLASSNAME

SELECT C.*, 

	   (SELECT FN_USERNAME(C.SITEID, C.CREATOR) FROM DUAL) AS CREATOR2, 

	   (SELECT FN_USERNAME(C.SITEID, C.MODIFIER) FROM DUAL) AS MODIFIER2

  FROM CIM_INSPDEFINITION C 

  	INNER JOIN CIM_INSPDEFINITIONITEMSPEC S ON C.SITEID = S.SITEID

  										   AND C.COPERATOR = S.COPERATOR

  										   AND C.ITEMID = S.ITEMID

  										   AND C.INSPTYPE = S.INSPTYPE

  										   AND C.REV = S.REV

	INNER JOIN CIM_INSPITEM I ON C.SITEID = I.SITEID 

							 AND S.INSPITEMID = I.INSPITEMID

 WHERE 1=1

   AND (:INSPITEMID IS NULL OR I.INSPITEMID = :INSPITEMID)

   AND C.SITEID = :SITEID

ORDER BY C.INSPDEFINITIONSYSID
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
