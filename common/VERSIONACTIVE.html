-- InsertProcessDataParameter4RecipeActive : 레시피 활성시 PROCESSDATAPARAMETER 저장

------------------------------------

-- INPUT : SITEID, RECIPEDEFINITIONID

--

INSERT INTO CIM_PROCESSDATAPARAMETER 

(

    PROCESSDATAPARAMETERID

  , PROCESSDATADEFINITIONID

  , SITEID

  , PROCESSDATAPARAMETERNAME 

  , DATAMAPPINGCOLUMN 

  , ISUSABLE 

  , CREATOR 

  , CREATETIME 

  , MODIFIER 

  , MODIFYTIME 

  , LASTEVENTTIME 

  , TID

)

WITH IV AS (

  SELECT 

      RD.RECIPEDEFINITIONID AS PROCESSDATADEFINITIONID

    , RD.SITEID

    , TO_NUMBER(NVL(SUBSTR(PP.DATAMAPPINGCOLUMN, 10, 40), '0')) AS MAXITEM

  FROM CIM_RECIPEDEFINITION RD

  LEFT OUTER JOIN CIM_PROCESSDATAPARAMETER PP ON PP.PROCESSDATADEFINITIONID= RD.RECIPEDEFINITIONID 

  WHERE 1=1

    AND RD.SITEID = :SITEID 

    AND RD.RECIPEDEFINITIONID = :RECIPEDEFINITIONID -- '11MIX001012NN00102'

)

, DT AS (

  SELECT

      RP.RECIPEITEMID         AS PROCESSDATAPARAMETERID

    , RP.RECIPEDEFINITIONID   AS PROCESSDATADEFINITIONID

    , RP.SITEID

    , RI.RECIPEITEMNAME       AS PROCESSDATAPARAMETERNAME

    , 'Usable' AS ISUSABLE

    , RP.CREATOR

    , RP.CREATETIME 

    , RP.MODIFIER 

    , RP.MODIFYTIME 

    , RP.LASTEVENTTIME 

    , RP.TID

    , IV.MAXITEM

    , RANK() OVER (PARTITION BY RP.RECIPEDEFINITIONID, RP.RECIPEITEMID ORDER BY RP.RECIPEPARAMETERID) AS RRANK

  FROM IV

  INNER JOIN CIM_RECIPEDEFINITION RD ON RD.SITEID = IV.SITEID

    AND RD.RECIPEDEFINITIONID = IV.PROCESSDATADEFINITIONID

  INNER JOIN CIM_RECIPEPARAMETER RP ON RP.SITEID = RD.SITEID 

    AND RP.RECIPEDEFINITIONID = RD.RECIPEDEFINITIONID 

  LEFT OUTER JOIN CIM_RECIPEITEM RI ON RI.SITEID = RD.SITEID

    AND RI.RECIPEITEMID = RP.RECIPEITEMID 

  WHERE 1=1

    AND NOT EXISTS (SELECT 1 FROM CIM_PROCESSDATAPARAMETER PP

                    WHERE 1=1

                      AND PP.SITEID = RP.SITEID

                      AND PP.PROCESSDATADEFINITIONID = RP.RECIPEDEFINITIONID

                      AND PP.PROCESSDATAPARAMETERID = RP.RECIPEITEMID

                  ) 

)

, FD AS (

  SELECT

      DT.*

    , 'ITEMVALUE'||

      TO_CHAR(DT.MAXITEM +

      RANK() OVER (PARTITION BY DT.PROCESSDATADEFINITIONID ORDER BY DT.PROCESSDATAPARAMETERID)

      ) AS DATAMAPPINGCOLUMN

    , RANK() OVER (PARTITION BY DT.PROCESSDATADEFINITIONID ORDER BY DT.PROCESSDATAPARAMETERID) AS DRANK

  FROM DT

  WHERE 1=1

    AND DT.RRANK = 1

)

  SELECT 

      FD.PROCESSDATAPARAMETERID

    , FD.PROCESSDATADEFINITIONID

    , FD.SITEID

    , FD.PROCESSDATAPARAMETERNAME

    , FD.DATAMAPPINGCOLUMN

    , FD.ISUSABLE

    , FD.CREATOR

    , FD.CREATETIME

    , FD.MODIFIER

    , FD.MODIFYTIME

    , FD.LASTEVENTTIME

    , FD.TID

  FROM FD

  ORDER BY 2,1
