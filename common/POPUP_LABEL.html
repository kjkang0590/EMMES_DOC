마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_MATERIALDEFINITION, CIM_PRODUCTDEFINITION
SQL 보기
------------------------------------------
--PM_PA_002_전처리 제품 포장 - 파쇄 W/O 목록
--SelectGrindingWOList
--Param : SITEID
------------------------------------------
SELECT WO.WORKORDERID
     , CONVERT(DATE, WO.REQUESTTIME) AS PLANDATE
     , WO.QTY AS PLANQTY
     , CONVERT(FLOAT, WO.QTY) * CONVERT(FLOAT, B.SUBQTY) AS PLANWEIGHT
     , WO.PRODUCTDEFINITIONID AS MATERIALDEFINITIONID
     , M.MATERIALDEFINITIONNAME
     , WO.TOPRODUCTDEFINITIONID AS PRODUCTDEFINITIONID
     , P.PRODUCTDEFINITIONNAME
  FROM CIM_WORKORDER WO WITH(NOLOCK)
 INNER JOIN CIM_BOM B WITH(NOLOCK)
    ON B.SITEID = WO.SITEID
   AND B.PRODUCTDEFINITIONID = WO.PRODUCTDEFINITIONID
   AND B.MATERIALDEFINITIONID = WO.TOPRODUCTDEFINITIONID
   AND B.WORKCENTER = 'WC12'
  LEFT JOIN CIM_MATERIALDEFINITION M ON M.SITEID = WO.SITEID AND M.MATERIALDEFINITIONID = WO.PRODUCTDEFINITIONID
  LEFT JOIN CIM_PRODUCTDEFINITION P ON P.SITEID = WO.SITEID AND P.PRODUCTDEFINITIONID = WO.TOPRODUCTDEFINITIONID
 WHERE WO.SITEID = @SITEID
   AND WO.STATE = 'Active'
