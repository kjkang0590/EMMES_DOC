--QM_US_001_부적합 등록 - Hold 대상 Lot 조회 공통 팝업
--SelectHoldingLotList
--Param : SITEID, FROMDATE, TODATE, PRODUCTDEFINITIONID
------------------------------------------
SELECT R.LOTSHIPPEDTIME
     , R.PRODUCTORDERID
     , R.WORKORDERID
     , R.LOTID
     , R.PRODUCTDEFINITIONID
     , R.PRODUCTDEFINITIONNAME
     , R.PROCESSSEGMENTID
     , R.PROCESSSEGMENTNAME
     , R.STATE
     , R.QTY
     , R.UNITID
     , R.FINALJUDGE
  FROM (SELECT A.LOTSHIPPEDTIME
             , A.PRODUCTORDERID
             , A.WORKORDERID
             , A.LOTID
             , A.PRODUCTDEFINITIONID
             , C.PRODUCTDEFINITIONNAME
             , A.PROCESSSEGMENTID
             , B.PROCESSSEGMENTNAME
             , A.STATE
             , A.QTY
             , A.UNITID
             , I.FINALJUDGE
             , DENSE_RANK() OVER (PARTITION BY I.LOTID ORDER BY I.REQDATETIME DESC) AS ORDERNO
         FROM CIM_LOT A WITH(NOLOCK)
        INNER JOIN CIM_INSPREQ I WITH(NOLOCK) ON I.SITEID = A.SITEID AND I.LOTID = A.LOTID AND I.INSPTYPE = 'OQC' AND I.FINALJUDGE = 'NG'
         LEFT JOIN CIM_PROCESSSEGMENT B WITH(NOLOCK) ON A.SITEID = B.SITEID AND A.PROCESSSEGMENTID = B.PROCESSSEGMENTID
         LEFT JOIN CIM_PRODUCTDEFINITION C WITH(NOLOCK) ON A.SITEID = C.SITEID AND A.PRODUCTDEFINITIONID = C.PRODUCTDEFINITIONID
        WHERE A.SITEID = @SITEID
          AND A.PRODUCTTYPE = 'FERT'
          AND NOT EXISTS (SELECT LOTID FROM CST_INCONGRUITY CI WHERE CI.SITEID = A.SITEID AND CI.LOTID = A.LOTID)
          AND A.LOTSHIPPEDTIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
          AND (@PRODUCTDEFINITIONID = '' OR A.PRODUCTDEFINITIONID = @PRODUCTDEFINITIONID)
       ) R
   WHERE R.ORDERNO = 1
   ORDER BY R.LOTSHIPPEDTIME DESC
