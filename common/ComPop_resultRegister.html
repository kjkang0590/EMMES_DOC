--EM_PB_005_BM 관리 - Spare Parts 소진 목록 Grid 조회 쿼리
--SelectSpartConsumeSearch
--Param : SITEID
------------------------------------------
SELECT A.BMID,
	   A.LOCATION,
	   B.FACILITYNAME AS LOCATIONNAME,
	   D.SPARTCLASSID,
	   E.SPARTCLASSNAME,
	   A.SPARTDEFINITION,
	   D.SPARTDEFINITIONNAME,
	   A.QTY,
	   A.UNITID,
	   D.SAFETYSTOCK
  FROM CST_SPARTCONSUME A WITH(NOLOCK)
  LEFT JOIN CIM_FACILITY B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.LOCATION = B.FACILITYID
  LEFT JOIN CIM_SPARTDEFINITION D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.SPARTDEFINITION = D.SPARTDEFINITIONID
  LEFT JOIN CIM_SPARTCLASS E WITH(NOLOCK)
    ON D.SITEID = E.SITEID
   AND D.SPARTCLASSID = E.SPARTCLASSID
 WHERE A.SITEID = @SITEID
   AND A.BMID = @BMID
 ORDER BY A.SPARTDEFINITION

#5 Query SelectSpartComPopSearch

연결된 기능 (1번 표): #7 #findBtn1 click

Query (조회 서비스) : SelectSpartComPopSearch

  • ID 구성: SelectSpartComPopSearch / 클래스 THiRAMES.Service.UI / 버전 00001
  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)
  • 쿼리명: EM_PB_005_BM 관리 - Spare Parts 목록 Grid 조회 쿼리
  • 쿼리 파라메터: SITEID
  • 테이블(조회/쓰기 전체, 함수·SP 포함):
    CIM_SPARTDEFINITION메인읽기   CIM_SPARTCLASS읽기   CST_MATERIALLOT읽기   CIM_FACILITY읽기
조인 구조도 테이블 2 · 조인 1
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_SPARTCLASS, CIM_FACILITY
SQL 보기
------------------------------------------
--EM_PB_005_BM 관리 - Spare Parts 목록 Grid 조회 쿼리
--SelectSpartComPopSearch
--Param : SITEID
------------------------------------------
SELECT MAX(D.LOCATION)          AS LOCATION,
       MAX(E.FACILITYNAME)      AS LOCATIONNAME,
       A.SPARTCLASSID,
       MAX(B.SPARTCLASSNAME)    AS SPARTCLASSNAME,
       A.SPARTDEFINITIONID      AS SPARTDEFINITION,
       MAX(A.SPARTDEFINITIONNAME) AS SPARTDEFINITIONNAME,
       ISNULL(SUM(D.QTY), 0)    AS QTY,          -- ★ 변경: A.QTY → D.QTY(실제 재고)
       MAX(A.UNITID)            AS UNITID,
       MAX(A.SAFETYSTOCK)       AS SAFETYSTOCK
  FROM CIM_SPARTDEFINITION A WITH(NOLOCK)
  LEFT JOIN CIM_SPARTCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.SPARTCLASSID = B.SPARTCLASSID
  LEFT JOIN CST_MATERIALLOT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.SPARTDEFINITIONID = D.MATERIALDEFINITIONID
   AND D.STATE = 'Active'                        -- ★ 추가: 종료/폐기 Lot 제외
  LEFT JOIN CIM_FACILITY E WITH(NOLOCK)
    ON D.SITEID = E.SITEID
   AND D.LOCATION = E.FACILITYID
 WHERE A.SITEID = @SITEID
 GROUP BY A.SPARTCLASSID, A.SPARTDEFINITIONID
 ORDER BY A.SPARTCLASSID;
