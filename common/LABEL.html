-------------------- 라벨발행 팝업 조회 쿼리

SELECT

	CPOI.MATERIALINDATE ,

	CPOI.PURCHASEORDERITEMNO ,

	LABEL.BOX AS RECEIPTUNITTYPE,

	CC.CODENAME AS RECEIPTUNITTYPENAME,

	CPOI.VENDORID,

	CVR.VENDORNAME,

	CPOI.PURCHASEORDER ,

	CPOI.VENDORLOT ,

	CMD.MATERIALCLASSID ,

	CMC.MATERIALCLASSNAME,

	CPOI.MATERIALDEFINITIONID ,

	CMD.MATERIALDEFINITIONNAME ,

	LABEL.LABEL

  FROM CUS_PURCHASEORDERITEM CPOI

 INNER JOIN CIM_VENDOR CVR

 	     ON 1=1

 	    AND CVR.VENDORID = CPOI.VENDORID 

 	    AND CVR.SITEID = CPOI.SITEID 

 	    AND CVR.ISUSABLE = :ISUSABLE

 INNER JOIN CIM_MATERIALDEFINITION CMD

 	     ON 1=1

 	    AND CMD.MATERIALDEFINITIONID = CPOI.MATERIALDEFINITIONID 

 	    AND CMD.SITEID = CPOI.SITEID 

 	    AND CMD.ISUSABLE = :ISUSABLE

 INNER JOIN CIM_MATERIALCLASS CMC

 	     ON 1=1

 	    AND CMC.MATERIALCLASSID = CMD.MATERIALCLASSID 

 	    AND CMC.SITEID = CPOI.SITEID 

 	    AND CMC.ISUSABLE = :ISUSABLE	    

 INNER JOIN(

  	    SELECT

 	        'OUTBOX' AS BOX,

 	    	PURCHASEORDERITEMNO ,

 	    	OUTBOX AS LABEL

 	      FROM CIM_MATERIALLOT

 	     WHERE PURCHASEORDERITEMNO = :PURCHASEORDERITEMNO

 	     GROUP BY PURCHASEORDERITEMNO , OUTBOX

 	     UNION ALL 

 	    SELECT

 	        'INBOX' AS BOX,

 	    	PURCHASEORDERITEMNO ,

 	    	INBOX AS LABEL

 	      FROM CIM_MATERIALLOT

 	     WHERE PURCHASEORDERITEMNO = :PURCHASEORDERITEMNO

 	     GROUP BY PURCHASEORDERITEMNO , INBOX

 	     UNION ALL 

 	    SELECT

 	        'LOT' AS BOX,

 	    	PURCHASEORDERITEMNO ,

 	    	MATERIALLOTID AS LABEL

 	      FROM CIM_MATERIALLOT

 	     WHERE PURCHASEORDERITEMNO = :PURCHASEORDERITEMNO

 	     GROUP BY PURCHASEORDERITEMNO , MATERIALLOTID

		)LABEL 

		 ON 1=1

	    AND LABEL.PURCHASEORDERITEMNO = CPOI.PURCHASEORDERITEMNO

  LEFT JOIN CIM_CODE CC

 	     ON 1=1

 	    AND CC.CODEID = LABEL.BOX 

 	    AND CC.CODECLASSID = 'RECEIPTUNITTYPE'

 	    AND CC.SITEID = CPOI.SITEID 

 	    AND CC.ISUSABLE = :ISUSABLE 

  LEFT JOIN CIM_CODE CC2

 	     ON 1=1

 	    AND CC2.CODEID = CMD.RECEIPTUNITTYPE 

 	    AND CC2.CODECLASSID = 'RECEIPTUNITTYPE'

 	    AND CC2.SITEID = CPOI.SITEID 

 	    AND CC2.ISUSABLE = :ISUSABLE  	    

 WHERE 1=1

   AND LABEL.LABEL IS NOT NULL

   AND CPOI.SITEID = :SITEID

   AND CPOI.ISUSABLE = :ISUSABLE

   AND (:RECEIPTUNITTYPE IS NULL  OR LABEL.BOX = :RECEIPTUNITTYPE)

   AND CC.UISEQUENCE <= CC2.UISEQUENCE
