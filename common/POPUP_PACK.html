마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_MATERIALDEFINITION, CIM_PRODUCTDEFINITION

--PM_PA_002_전처리 제품 포장 - 포장 대상 W/O 목록
--SelectGrindingPackingInfoList
--Param : SITEID
------------------------------------------
SELECT WO.WORKORDERID
     , CONVERT(DATE, WO.REQUESTTIME) AS PLANDATE
     , WO.QTY AS PLANQTY
     , CONVERT(FLOAT, WO.QTY) * CONVERT(FLOAT, B.SUBQTY) AS PLANWEIGHT
     , WO.PRODUCTDEFINITIONID AS MATERIALDEFINITIONID
     , M.MATERIALDEFINITIONNAME
     , ISNULL(MAT.QTY, 0) AS MATERIALQTY
     , CONVERT(FLOAT, ISNULL(MAT.QTY, 0)) * CONVERT(FLOAT, B.SUBQTY) AS MATERIALWEIGHT
     , WO.TOPRODUCTDEFINITIONID AS PRODUCTDEFINITIONID
     , P.PRODUCTDEFINITIONNAME
  FROM CIM_WORKORDER WO WITH(NOLOCK)
 INNER JOIN CIM_BOM B WITH(NOLOCK)
    ON B.SITEID = WO.SITEID
   AND B.PRODUCTDEFINITIONID = WO.PRODUCTDEFINITIONID
   AND B.MATERIALDEFINITIONID = WO.TOPRODUCTDEFINITIONID
   AND B.WORKCENTER = 'WC12'
  LEFT JOIN CIM_MATERIALDEFINITION M WITH(NOLOCK) ON M.SITEID = WO.SITEID AND M.MATERIALDEFINITIONID = WO.PRODUCTDEFINITIONID
  LEFT JOIN CIM_PRODUCTDEFINITION P WITH(NOLOCK) ON P.SITEID = WO.SITEID AND P.PRODUCTDEFINITIONID = WO.TOPRODUCTDEFINITIONID
  LEFT JOIN (SELECT MAX(SITEID) AS SITEID, 'PRODUCT' AS MATERIALTYPE, PRODUCTDEFINITIONID AS MATERIALDEFINITIONID, SUM(QTY) AS QTY
               FROM CIM_LOT WITH(NOLOCK)
              WHERE SITEID = @SITEID
                AND PROCESSSEGMENTID = 'P1010'
                AND STATE IN ('Finished', 'Consumed')
                AND QTY > 0
              GROUP BY PRODUCTDEFINITIONID
              UNION
             SELECT MAX(SITEID) AS SITEID, 'MATERIAL' AS MATERIALTYPE, MATERIALDEFINITIONID, SUM(QTY) AS QTY
               FROM CST_MATERIALLOT WITH(NOLOCK)
              WHERE SITEID = @SITEID
                AND STATE IN ('Active', 'Consumed')
                AND QTY > 0
                AND LOCATION NOT LIKE 'SP%'
              GROUP BY MATERIALDEFINITIONID
            ) AS MAT ON MAT.SITEID = WO.SITEID AND MAT.MATERIALDEFINITIONID = WO.PRODUCTDEFINITIONID
 WHERE WO.SITEID = @SITEID
   AND WO.STATE = 'Active'
