--RT_PM_002_설비 가동율 - BM 가동율 등록 POPUP 조회 쿼리
--SelectOperationRatePrevPopList
--Param : SITEID, SEARCHDATE
------------------------------------------
SELECT US1.RPT_DATE,
       MAX(US1.WORKDATE) AS WORKDATE,
       MAX(US1.TIME_PREV_BM_NORMAL_RATE) AS TIME_PREV_BM_NORMAL_RATE,
       MAX(US1.PER_PREV_BM_NORMAL_RATE) AS PER_PREV_BM_NORMAL_RATE,
       MAX(US1.TIME_PREV_BM_NOCHARGE_RATE) AS TIME_PREV_BM_NOCHARGE_RATE,
       MAX(US1.PER_PREV_BM_NOCHARGE_RATE) AS PER_PREV_BM_NOCHARGE_RATE,
       MAX(US1.TIME_PREV_DRY_RATE) AS TIME_PREV_DRY_RATE,
       MAX(US1.PER_PREV_DRY_RATE) AS PER_PREV_DRY_RATE,
       MAX(US1.TIME_PREV_PACKING_RATE) AS TIME_PREV_PACKING_RATE,
       MAX(US1.PER_PREV_PACKING_RATE) AS PER_PREV_PACKING_RATE,
       MAX(US1.ROW_STATE) AS ROW_STATE
  FROM (
  	SELECT S1.RPT_DATE,
	    '' AS WORKDATE,
	    '' AS TIME_PREV_BM_NORMAL_RATE,
	    '' AS PER_PREV_BM_NORMAL_RATE,
	    '' AS TIME_PREV_BM_NOCHARGE_RATE,
	    '' AS PER_PREV_BM_NOCHARGE_RATE,
	    '' AS TIME_PREV_DRY_RATE,
	    '' AS PER_PREV_DRY_RATE,
	    '' AS TIME_PREV_PACKING_RATE,
	    '' AS PER_PREV_PACKING_RATE,
             'A' AS ROW_STATE
  	  FROM (
	  	SELECT REPLACE(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), '-', '') AS RPT_DATE
	  	  FROM MASTER..SPT_VALUES A WITH(NOLOCK)
	     WHERE A.TYPE = 'P' 
	       AND SUBSTRING(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), 1, 7) = SUBSTRING(@SEARCHDATE, 1, 7)
	  ) S1
 	 UNION
	SELECT S1.RPT_DATE,
	       OS.WORKDATE,
	       OS.TIME_PREV_BM_NORMAL_RATE,
		   OS.PER_PREV_BM_NORMAL_RATE,
		   OS.TIME_PREV_BM_NOCHARGE_RATE,
		   OS.PER_PREV_BM_NOCHARGE_RATE,
		   OS.TIME_PREV_DRY_RATE,
		   OS.PER_PREV_DRY_RATE,
		   OS.TIME_PREV_PACKING_RATE,
		   OS.PER_PREV_PACKING_RATE,
                     'U' AS ROW_STATE
	  FROM (
	  	SELECT REPLACE(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), '-', '') AS RPT_DATE
		  FROM MASTER..SPT_VALUES A WITH(NOLOCK)
		 WHERE A.TYPE = 'P'
		   AND SUBSTRING(CONVERT(CHAR(10), DATEADD(d, NUMBER, CONCAT(@SEARCHDATE, '-01')), 120), 1, 7) = SUBSTRING(@SEARCHDATE, 1, 7)
	  ) S1
	  LEFT JOIN RPT_OPERATIONRATE_SUM OS WITH(NOLOCK)
	    ON S1.RPT_DATE = OS.WORKDATE
	 WHERE OS.SITEID = @SITEID
  ) US1
 GROUP BY US1.RPT_DATE
