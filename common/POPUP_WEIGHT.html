--PM_PA_002_전처리 제품 포장 - 무게 등록 시 바코드 리딩한 LOT 조회
--SelectGrindBarcodeLot
--Param : SITEID, LOTID
------------------------------------------
SELECT L.LOTID
     , W.WORKORDERID
     , CONVERT(DATE, W.REQUESTTIME) AS PLANDATE
     , L.PRODUCTDEFINITIONID
     , P.PRODUCTDEFINITIONNAME
  FROM CIM_LOT L WITH(NOLOCK)
 INNER JOIN CIM_WORKORDER W WITH(NOLOCK) ON W.SITEID = L.SITEID AND W.WORKORDERID = L.WORKORDERID
  LEFT JOIN CIM_PRODUCTDEFINITION P WITH(NOLOCK) ON P.SITEID = L.SITEID AND P.PRODUCTDEFINITIONID = L.PRODUCTDEFINITIONID
 WHERE L.SITEID = @SITEID
   AND L.LOTID = @LOTID
   AND W.STATE = 'Active'
   AND L.PROCESSSEGMENTID = 'P1020'
