마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_MATERIALDEFINITION, CIM_STATE, CIM_CODE

-- SelectInspDefinition4RequestPopup = 검사의뢰 팝업 검사표준 조회문

------------------------------------

-- INPUT = SITEID, ITEMID, COPERATOR, INSPTYPE

--

WITH IDBASE AS

(

	SELECT IDBASE.*

  	      FROM CIM_INSPDEFINITION IDBASE

  	     WHERE 1=1

  	       AND IDBASE.SITEID = :SITEID

		   AND IDBASE.ITEMID = :ITEMID

		   AND IDBASE.STATE = 'Active'

),

ID1 AS

(

	SELECT ID1.*

  	      FROM IDBASE ID1

  	    WHERE 1=1

		   AND ID1.COPERATOR = :COPERATOR

),

IDTEMP AS

(

SELECT ID.*,

	   COP.COPERATORNAME,

	   M.MATERIALDEFINITIONNAME,

	   ST.STATENAME,

	   CD1.CODENAME AS INSPTYPENAME,

	   CD2.CODENAME AS INSPMETHODNAME,

	   CD3.CODENAME AS INSPSPECNAME

  FROM 

  	   (SELECT ID1.*

  	      FROM ID1

		UNION ALL 

		SELECT ID2.*

  	      FROM IDBASE ID2

  	     WHERE 1=1

		   AND ID2.COPERATOR = '-'

		   AND 0 = (SELECT COUNT(ID1.COPERATOR)

			  	      FROM ID1)

	   ) ID

	LEFT OUTER JOIN (

	  			SELECT VD.VENDORID AS CID, VD.VENDORNAME AS COPERATORNAME, VD.SITEID FROM CIM_VENDOR VD

	  			UNION ALL

	  			SELECT CT.CUSTOMERID AS CID, CT.CUSTOMERNAME AS COPERATORNAME, CT.SITEID FROM CIM_CUSTOMER CT

	  		   ) COP 

		ON ID.SITEID = COP.SITEID

	  	AND ID.COPERATOR = COP.CID

	LEFT OUTER JOIN CIM_MATERIALDEFINITION M 

		ON 1=1 

		AND ID.SITEID = M.SITEID 

		AND ID.ITEMID = M.MATERIALDEFINITIONID

	LEFT OUTER JOIN CIM_STATE ST 

		ON 1=1

		AND ID.SITEID = ST.SITEID

		AND ID.STATE = ST.STATEID

		AND ST.STATEMODELID = 'InspDefinitionState'

	LEFT OUTER JOIN CIM_CODE CD1

		ON 1=1

		AND ID.SITEID = CD1.SITEID 

		AND ID.INSPTYPE = CD1.CODEID

		AND CD1.CODECLASSID = 'INSPTYPE'

	LEFT OUTER JOIN CIM_CODE CD2

		ON 1=1

		AND ID.SITEID = CD2.SITEID 

		AND ID.INSPMETHOD = CD2.CODEID

		AND CD2.CODECLASSID = 'INSPMETHOD'

	LEFT OUTER JOIN CIM_CODE CD3

		ON 1=1

		AND ID.SITEID = CD3.SITEID 

		AND ID.INSPSPEC = CD3.CODEID

		AND CD3.PARENTID = 'OQC'

 WHERE 1=1

   AND ID.SITEID = :SITEID 

   AND ID.INSPTYPE = :INSPTYPE

   AND ID.ITEMID = :ITEMID

ORDER BY ID.INSPDEFINITIONSYSID

)

SELECT * FROM IDTEMP
