--PM_PB_002 공정 진행 (반응) - Tank 이동 이력 조회
--SelectTankMoveHistList
--Param : SITEID, LOTID
------------------------------------------
SELECT LH.LOTID
        , LH.STATE
        , CS.STATENAME 
        , LH.EQUIPMENTID
        , LH.CUSTOMACTIVITY
        , LH.TANKINTIME
        , LH.TANKOUTTIME   
  FROM CIM_LOTHIST LH WITH(NOLOCK)
  LEFT JOIN CIM_STATE CS WITH(NOLOCK)
    ON CS.SITEID = LH.SITEID 
   AND CS.STATEID = LH.STATE 
   AND CS.STATEMODELID = 'LotState' 
   AND CS.ISUSABLE = 'Usable'
 WHERE LH.SITEID = @SITEID
   AND LH.LOTID = @LOTID
   AND LH.ISUSABLE = 'Usable'
 ORDER BY LH.LASTEVENTSEQ DESC
