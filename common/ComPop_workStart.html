     , B.PRODUCTDEFINITIONID
     , B.MATERIALDEFINITIONID
     , B.SUBQTY
  FROM CIM_BOM B

   AND B.WORKCENTER = 'WC12'

SELECT RECEIVEDTIME, MATERIALLOTID, MATERIALDEFINITIONID, MATERIALCLASSID, LOCATION, STATE, QTY, UNITID, KITTINGTIME, SITEID
  FROM CST_MATERIALLOT
