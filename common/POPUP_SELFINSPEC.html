마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_SHIFT, CIM_USER, CIM_CODE
SQL 보기
------------------------------------------
--PM_PB_004_자주검사 - 계층적 공정검사 체크 시트(LPA Check Sheet) 조회 쿼리(Popup)
--SelectLPAResultListForPopup
--Param : SITEID, LINEID, LPASYSID
------------------------------------------
SELECT A.LPASYSID, 
       A.WORKDATE, 
       A.SHIFTID,
       A.ISSQC,
       D.SHIFTNAME,
       A.CREATOR,
       E.USERNAME AS CREATORNAME,
       A.COMMENTS,
       A.DESCRIPTION,
       B.ITEMVALUE,
       C.LPAGROUP,
       F.CODENAME AS LPAGROUPNAME,
       C.LPATYPE,
       G.CODENAME AS LPATYPENAME,
       C.PROCESSSEGMENTTYPE,
       C.LPAITEMID,
       C.LPAITEMNAME
  FROM CST_LPARESULT A WITH(NOLOCK)
  LEFT JOIN CST_LPARESULTITEM B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.LPASYSID = B.LPASYSID
  LEFT JOIN CST_LPAITEM C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.LPAITEMID = C.LPAITEMID 
  LEFT JOIN CIM_SHIFT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.SHIFTID = D.SHIFTID
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.CREATOR = E.USERID
  LEFT JOIN CIM_CODE F WITH(NOLOCK)
    ON C.SITEID = F.SITEID
   AND C.LPAGROUP = F.CODEID
   AND F.CODECLASSID = 'LPAGROUP'
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON C.SITEID = G.SITEID
   AND C.LPATYPE = G.CODEID
   AND G.CODECLASSID = 'LPATYPE'
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.LPASYSID = @LPASYSID
 ORDER BY A.WORKDATE, A.SHIFTID, C.UISEQUENCE
