-- 반응 투입 (TKIN) 시 METAL 무게 환산 (2024-04-17)

--  톤백 METAL 함유량을 찾지 못한 경우 : DEFAULT 함유량 (임의값)으로 계산하는 로직 추가

--  PARAM 

--    @SITEID : 'PL01'

--    @FIR_PRODUCTDEFINITIONID : 소성 LOT PRODUCTDEFINITIONID 

--    @CONSUMEQTY : 소진 대상 수량

--    @ISUIDATA : Y - UI 조회용, N - 소진 정보용.

-- EXEC SP_GET_LEACHINGMETALQTY_V3 'PL01', '2WKO0NCM6-22', 5020, 'N'

--------------------------------

CREATE PROCEDURE [dbo].[SP_GET_LEACHINGMETALQTY_V3] 

    @SITEID VARCHAR(40), @FIR_PRODUCTDEFINITIONID VARCHAR(40), @CONSUMEQTY INT, @ISUIDATA VARCHAR(1) = 'Y'

AS  

BEGIN 



    ---------------------------------------------

    -- TEMP TABLE 생성

    ---------------------------------------------

    DECLARE @LEACHINGMETALINFO TABLE

    (

        LOTID VARCHAR(40)

        , SITEID VARCHAR(40)

        , CONSUMEQTY FLOAT

        , USED_QTY FLOAT

        , LITHIUMMETALQTY FLOAT

        , COBALTMETALQTY FLOAT

        , NICKELMETALQTY FLOAT

        , WORKORDERID VARCHAR(40)

    )

    ;



    ---------------------------------------------

    -- METAL 환산 수량 계산 및 TEMP TABLE 저장 : START

    ---------------------------------------------

    WITH METAL_BASEDATA AS 

    (

        SELECT 

            M.LOTID

            , M.SITEID

            --, M.RESULTQTY  -- 검증용

            --, M.QTY        -- 검증용

            --, M.LITHIUMMETALQTY AS FIR_LITHIUMMETALQTY  -- 검증용

            --, M.COBALTMETALQTY AS FIR_COBALTMETALQTY    -- 검증용

            --, M.NICKELMETALQTY AS FIR_NICKELMETALQTY    -- 검증용

            --, M.START_QTY  -- 검증용

            --, M.END_QTY    -- 검증용

            , M.USED_QTY

            , M.USED_QTY / M.RESULTQTY * M.LITHIUMMETALQTY AS LITHIUMMETALQTY

            , M.USED_QTY / M.RESULTQTY * M.COBALTMETALQTY AS COBALTMETALQTY

            , M.USED_QTY / M.RESULTQTY * M.NICKELMETALQTY AS NICKELMETALQTY

            , M.WORKORDERID

        FROM 

        (

            SELECT 

                MB.LOTID

                , MB.SITEID

                , MB.RESULTQTY

                , MB.QTY

                , MB.LITHIUMMETALQTY

                , MB.COBALTMETALQTY

                , MB.NICKELMETALQTY

                , MB.START_QTY

                , MB.END_QTY

                , CASE WHEN MB.END_QTY - @CONSUMEQTY  <= 0 THEN ( MB.END_QTY - MB.START_QTY )

                       ELSE @CONSUMEQTY - MB.START_QTY 

                  END AS USED_QTY

                , MB.WORKORDERID

            FROM 

            (

                SELECT 

                    MA.LOTID

                    , MA.SITEID

                    , MA.RESULTQTY

                    , MA.QTY

                    , MA.LITHIUMMETALQTY

                    , MA.COBALTMETALQTY

                    , MA.NICKELMETALQTY

                    , ISNULL( LAG(MA.END_QTY) OVER(ORDER BY MA.CREATETIME ASC), 0) AS START_QTY

                    , MA.END_QTY

                    , MA.CREATETIME

                    , MA.WORKORDERID

                FROM 

                (

                    SELECT 

                        L.LOTID 

                        , L.SITEID 

                        , L.ORIGINALQTY 

                        , L.RESULTQTY

                        , L.QTY 

                        , L.PREVQTY

                        , L.STATE 

                        , L.PRODUCTDEFINITIONID 

                        , L.LITHIUMMETALQTY

                        , L.COBALTMETALQTY

                        , L.NICKELMETALQTY

                        , L.CREATETIME

                        , SUM(L.QTY) OVER(ORDER BY L.CREATETIME ASC) AS END_QTY

                        , L.WORKORDERID

                    FROM CIM_LOT L

                    WHERE L.SITEID = 'PL01'

                    AND   L.PROCESSSEGMENTID = 'P2010'

                    AND   L.STATE = 'Finished'

                    AND   L.PRODUCTDEFINITIONID = @FIR_PRODUCTDEFINITIONID

                ) MA

            ) MB

            WHERE MB.END_QTY <= @CONSUMEQTY

  OR    ( MB.START_QTY < @CONSUMEQTY AND @CONSUMEQTY <= MB.END_QTY )

        ) M

    ) 

    INSERT INTO @LEACHINGMETALINFO

    SELECT MT.LOTID 

         , MT.SITEID

         , @CONSUMEQTY AS CONSUMEQTY

         , MT.USED_QTY

         , MT.LITHIUMMETALQTY

         , MT.COBALTMETALQTY

         , MT.NICKELMETALQTY

         , MT.WORKORDERID

      FROM METAL_BASEDATA MT

    ;  

    ---------------------------------------------

    -- METAL 환산 수량 계산 및 TEMP TABLE 저장 : END

    ---------------------------------------------

  

    ---------------------------------------------

    -- 결과값 조회

    -- @ISUIDATA - Y : LI, CO, NI 의 공정별 SUM 값 SUM 값

     --            N : LI, CO, NI 의 SUM 값

    ---------------------------------------------

    IF @ISUIDATA = 'Y'

    BEGIN

        SELECT 

            M.PRODUCT

            , ROUND( SUM(QTY), 2) AS QTY

        FROM 

        (

            SELECT 

                  CASE WHEN B.PROCESSSEGMENTID = 'P2030' THEN 'Li'

                       WHEN B.PROCESSSEGMENTID = 'P2040' THEN 'Co'

                       WHEN B.PROCESSSEGMENTID = 'P2050' THEN 'Ni'

                  END AS PRODUCT

                , CASE WHEN B.PROCESSSEGMENTID = 'P2030' THEN MB.LITHIUMMETALQTY

                       WHEN B.PROCESSSEGMENTID = 'P2040' THEN MB.COBALTMETALQTY

                       WHEN B.PROCESSSEGMENTID = 'P2050' THEN MB.NICKELMETALQTY

                  END AS QTY

            FROM @LEACHINGMETALINFO MB

                INNER JOIN CIM_WORKORDER WO

                    ON  MB.WORKORDERID = WO.WORKORDERID 

                    AND MB.SITEID = WO.SITEID

                INNER JOIN CIM_BOM B

                    ON  WO.SITEID = B.SITEID 

                    AND WO.PRODUCTDEFINITIONID = B.PRODUCTDEFINITIONID

                    AND B.PROCESSSEGMENTID IN (SELECT NEXTPROCESSNODEID FROM CIM_PROCESSPATH WHERE PROCESSNODEID = 'P2020') 

                    AND B.MATERIALDEFINITIONID <> '2LSS' --망초 제외

                    AND B.MATERIALDEFINITIONID LIKE '2L%'

        ) M

        GROUP BY M.PRODUCT

    END

    ELSE

    BEGIN

--        SELECT M.LOTID

--             , M.CONSUMEQTY

--             , M.USED_QTY

--             , ROUND(CONVERT(FLOAT, M.LITHIUMMETALQTY), 2) AS LITHIUMMETALQTY

--             , ROUND(CONVERT(FLOAT, M.COBALTMETALQTY), 2) AS COBALTMETALQTY

--             , ROUND(CONVERT(FLOAT, M.NICKELMETALQTY), 2) AS NICKELMETALQTY

--        FROM 

--        (

--            SELECT ISNULL(MT.LOTID, 'TOTALSUM') AS LOTID

--                 , @CONSUMEQTY AS CONSUMEQTY

--                 , SUM(MT.USED_QTY) AS USED_QTY

--                 , SUM(MT.LITHIUMMETALQTY) AS LITHIUMMETALQTY

--                 , SUM(MT.COBALTMETALQTY) AS COBALTMETALQTY

--                 , SUM(MT.NICKELMETALQTY) AS NICKELMETALQTY

--              FROM @LEACHINGMETALINFO MT

--             GROUP BY MT.LOTID WITH ROLLUP

--         ) M

        SELECT M.LOTID

             , M.CONSUMEQTY

             , M.USED_QTY

             , ROUND(CONVERT(FLOAT, M.LITHIUMMETALQTY), 2) AS LITHIUMMETALQTY

             , ROUND(CONVERT(FLOAT, M.COBALTMETALQTY), 2) AS COBALTMETALQTY

             , ROUND(CONVERT(FLOAT, M.NICKELMETALQTY), 2) AS NICKELMETALQTY

             , M.WORKORDERID

        FROM 

        (

            SELECT ISNULL(MT.LOTID, 'TOTALSUM') AS LOTID

                 , @CONSUMEQTY AS CONSUMEQTY

                 , SUM(MT.USED_QTY) AS USED_QTY

                 , SUM(MT.LITHIUMMETALQTY) AS LITHIUMMETALQTY

                 , SUM(MT.COBALTMETALQTY) AS COBALTMETALQTY

                 , SUM(MT.NICKELMETALQTY) AS NICKELMETALQTY

                 , MIN(L.WORKORDERID) AS WORKORDERID

              FROM @LEACHINGMETALINFO MT

                LEFT OUTER JOIN CIM_LOT L

                    ON  MT.LOTID = L.LOTID

                    AND L.SITEID = @SITEID

             GROUP BY MT.LOTID WITH ROLLUP

         ) M        

    END



END

--[SERVICE] Metal 환산수량으로 BP 환산수량 계산
--GetCalculateBPExchangeQty
--Param : TOTALINPUTQTY, LI_METAL_QTY, CO_METAL_QTY, NI_METAL_QTY
------------------------------------------
SELECT CONVERT(float, @TOTALINPUTQTY) AS TOTALINPUTQTY
     , CONVERT(float, A.LI_BP_QTY) AS LI_BP_QTY
     , CONVERT(float, A.CO_BP_QTY) AS CO_BP_QTY
     , CONVERT(float, @TOTALINPUTQTY) - CONVERT(float, A.LI_BP_QTY) - CONVERT(float, A.CO_BP_QTY) AS NI_BP_QTY
--     , CONVERT(float, A.LI_BP_QTY) + CONVERT(float, A.CO_BP_QTY) + CONVERT(float, @TOTALINPUTQTY) - CONVERT(float, A.LI_BP_QTY) - CONVERT(float, A.CO_BP_QTY) AS TOTALSUMQTY

        , SITEID VARCHAR(40)

        , CONSUMEQTY FLOAT

        , USED_QTY FLOAT

        , LITHIUMMETALQTY FLOAT

        , COBALTMETALQTY FLOAT

        , NICKELMETALQTY FLOAT

        , WORKORDERID VARCHAR(40)

    )

    ;



    ---------------------------------------------

    -- METAL 환산 수량 계산 및 TEMP TABLE 저장 : START

    ---------------------------------------------

    WITH METAL_BASEDATA AS 

    (

        SELECT 

            M.LOTID

            , M.SITEID

            --, M.RESULTQTY  -- 검증용

            --, M.QTY        -- 검증용

            --, M.LITHIUMMETALQTY AS FIR_LITHIUMMETALQTY  -- 검증용

            --, M.COBALTMETALQTY AS FIR_COBALTMETALQTY    -- 검증용

            --, M.NICKELMETALQTY AS FIR_NICKELMETALQTY    -- 검증용

            --, M.START_QTY  -- 검증용

            --, M.END_QTY    -- 검증용

            , M.USED_QTY

            , M.USED_QTY / M.RESULTQTY * M.LITHIUMMETALQTY AS LITHIUMMETALQTY

            , M.USED_QTY / M.RESULTQTY * M.COBALTMETALQTY AS COBALTMETALQTY

            , M.USED_QTY / M.RESULTQTY * M.NICKELMETALQTY AS NICKELMETALQTY

            , M.WORKORDERID

        FROM 

        (

            SELECT 

                MB.LOTID

                , MB.SITEID

                , MB.RESULTQTY

                , MB.QTY

                , MB.LITHIUMMETALQTY

                , MB.COBALTMETALQTY

                , MB.NICKELMETALQTY

                , MB.START_QTY

                , MB.END_QTY

                , CASE WHEN MB.END_QTY - @CONSUMEQTY  <= 0 THEN ( MB.END_QTY - MB.START_QTY )

                       ELSE @CONSUMEQTY - MB.START_QTY 

                  END AS USED_QTY

                , MB.WORKORDERID

            FROM 

            (

                SELECT 

                    MA.LOTID

                    , MA.SITEID

                    , MA.RESULTQTY

                    , MA.QTY

                    , MA.LITHIUMMETALQTY

                    , MA.COBALTMETALQTY

                    , MA.NICKELMETALQTY

                    , ISNULL( LAG(MA.END_QTY) OVER(ORDER BY MA.CREATETIME ASC), 0) AS START_QTY

                    , MA.END_QTY

                    , MA.CREATETIME

                    , MA.WORKORDERID

                FROM 

                (

                    SELECT 

                        L.LOTID 

                        , L.SITEID 

                        , L.ORIGINALQTY 

                        , L.RESULTQTY

                        , L.QTY 

                        , L.PREVQTY

                        , L.STATE 

                        , L.PRODUCTDEFINITIONID 

                        , L.LITHIUMMETALQTY

                        , L.COBALTMETALQTY

                        , L.NICKELMETALQTY

                        , L.CREATETIME

                        , SUM(L.QTY) OVER(ORDER BY L.CREATETIME ASC) AS END_QTY

                        , L.WORKORDERID

                    FROM CIM_LOT L

                    WHERE L.SITEID = 'PL01'

                    AND   L.PROCESSSEGMENTID = 'P2010'

                    AND   L.STATE = 'Finished'

                    AND   L.PRODUCTDEFINITIONID = @FIR_PRODUCTDEFINITIONID

                ) MA

            ) MB

            WHERE MB.END_QTY <= @CONSUMEQTY

  OR    ( MB.START_QTY < @CONSUMEQTY AND @CONSUMEQTY <= MB.END_QTY )

        ) M

    ) 

    INSERT INTO @LEACHINGMETALINFO

    SELECT MT.LOTID 

         , MT.SITEID

         , @CONSUMEQTY AS CONSUMEQTY

         , MT.USED_QTY

         , MT.LITHIUMMETALQTY

         , MT.COBALTMETALQTY

         , MT.NICKELMETALQTY

         , MT.WORKORDERID

      FROM METAL_BASEDATA MT

    ;  

    ---------------------------------------------

    -- METAL 환산 수량 계산 및 TEMP TABLE 저장 : END

    ---------------------------------------------

  

    ---------------------------------------------

    -- 결과값 조회

    -- @ISUIDATA - Y : LI, CO, NI 의 공정별 SUM 값 SUM 값

     --            N : LI, CO, NI 의 SUM 값

    ---------------------------------------------

    IF @ISUIDATA = 'Y'

    BEGIN

        SELECT 

            M.PRODUCT

            , ROUND( SUM(QTY), 2) AS QTY

        FROM 

        (

            SELECT 

                  CASE WHEN B.PROCESSSEGMENTID = 'P2030' THEN 'Li'

                       WHEN B.PROCESSSEGMENTID = 'P2040' THEN 'Co'

                       WHEN B.PROCESSSEGMENTID = 'P2050' THEN 'Ni'

                  END AS PRODUCT

                , CASE WHEN B.PROCESSSEGMENTID = 'P2030' THEN MB.LITHIUMMETALQTY

                       WHEN B.PROCESSSEGMENTID = 'P2040' THEN MB.COBALTMETALQTY

                       WHEN B.PROCESSSEGMENTID = 'P2050' THEN MB.NICKELMETALQTY

                  END AS QTY

            FROM @LEACHINGMETALINFO MB

                INNER JOIN CIM_WORKORDER WO

                    ON  MB.WORKORDERID = WO.WORKORDERID 

                    AND MB.SITEID = WO.SITEID

                INNER JOIN CIM_BOM B

                    ON  WO.SITEID = B.SITEID 

                    AND WO.PRODUCTDEFINITIONID = B.PRODUCTDEFINITIONID

                    AND B.PROCESSSEGMENTID IN (SELECT NEXTPROCESSNODEID FROM CIM_PROCESSPATH WHERE PROCESSNODEID = 'P2020') 

                    AND B.MATERIALDEFINITIONID <> '2LSS' --망초 제외

                    AND B.MATERIALDEFINITIONID LIKE '2L%'

        ) M

        GROUP BY M.PRODUCT

    END

    ELSE

    BEGIN

--        SELECT M.LOTID

--             , M.CONSUMEQTY

--             , M.USED_QTY

--             , ROUND(CONVERT(FLOAT, M.LITHIUMMETALQTY), 2) AS LITHIUMMETALQTY

--             , ROUND(CONVERT(FLOAT, M.COBALTMETALQTY), 2) AS COBALTMETALQTY

--             , ROUND(CONVERT(FLOAT, M.NICKELMETALQTY), 2) AS NICKELMETALQTY

--        FROM 

--        (

--            SELECT ISNULL(MT.LOTID, 'TOTALSUM') AS LOTID

--                 , @CONSUMEQTY AS CONSUMEQTY

--                 , SUM(MT.USED_QTY) AS USED_QTY

--                 , SUM(MT.LITHIUMMETALQTY) AS LITHIUMMETALQTY

--                 , SUM(MT.COBALTMETALQTY) AS COBALTMETALQTY

--                 , SUM(MT.NICKELMETALQTY) AS NICKELMETALQTY

--              FROM @LEACHINGMETALINFO MT

--             GROUP BY MT.LOTID WITH ROLLUP

--         ) M

        SELECT M.LOTID

             , M.CONSUMEQTY

             , M.USED_QTY

             , ROUND(CONVERT(FLOAT, M.LITHIUMMETALQTY), 2) AS LITHIUMMETALQTY

             , ROUND(CONVERT(FLOAT, M.COBALTMETALQTY), 2) AS COBALTMETALQTY

             , ROUND(CONVERT(FLOAT, M.NICKELMETALQTY), 2) AS NICKELMETALQTY

             , M.WORKORDERID

        FROM 

        (

            SELECT ISNULL(MT.LOTID, 'TOTALSUM') AS LOTID

                 , @CONSUMEQTY AS CONSUMEQTY

                 , SUM(MT.USED_QTY) AS USED_QTY

                 , SUM(MT.LITHIUMMETALQTY) AS LITHIUMMETALQTY

                 , SUM(MT.COBALTMETALQTY) AS COBALTMETALQTY

                 , SUM(MT.NICKELMETALQTY) AS NICKELMETALQTY

                 , MIN(L.WORKORDERID) AS WORKORDERID

              FROM @LEACHINGMETALINFO MT

                LEFT OUTER JOIN CIM_LOT L

                    ON  MT.LOTID = L.LOTID

                    AND L.SITEID = @SITEID

             GROUP BY MT.LOTID WITH ROLLUP

         ) M        

    END



END
