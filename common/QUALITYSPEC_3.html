-- SelectInspDefinitionItemSpecByInspDefinitionSysId = 검사항목스펙 조회문(INSPDEFINITIONSYSID)

------------------------------------

-- INPUT = SITEID, ISUSABLE, INSPDEFINITIONSYSID

SELECT C.*,

	   COP.COPERATORNAME,

	   M.MATERIALDEFINITIONNAME,

	   CD1.CODENAME AS INSPTYPENAME,

	   CD2.CODENAME AS DATATYPENAME,

	   CD3.CODENAME AS CALCULATETYPENAME,

	   CD4.CODENAME AS INSPSPECNAME,

	   II.INSPITEMNAME,

	   (SELECT FN_USERNAME(C.SITEID, C.CREATOR) FROM DUAL) AS CREATOR2, 

	   (SELECT FN_USERNAME(C.SITEID, C.MODIFIER) FROM DUAL) AS MODIFIER2

  FROM CIM_INSPDEFINITIONITEMSPEC C

	LEFT OUTER JOIN (

	  			SELECT VD.VENDORID AS CID, VD.VENDORNAME AS COPERATORNAME, VD.SITEID FROM CIM_VENDOR VD

	  			UNION ALL

	  			SELECT CT.CUSTOMERID AS CID, CT.CUSTOMERNAME AS COPERATORNAME, CT.SITEID FROM CIM_CUSTOMER CT

	  		   ) COP 

		ON C.SITEID = COP.SITEID

	  	AND C.COPERATOR = COP.CID

	LEFT OUTER JOIN CIM_MATERIALDEFINITION M 

		ON 1=1

		AND C.SITEID = M.SITEID 

		AND C.ITEMID = M.MATERIALDEFINITIONID

	LEFT OUTER JOIN CIM_MATERIALCLASS MC 

		ON 1=1 

		AND C.SITEID = M.SITEID

		AND M.MATERIALCLASSID = MC.MATERIALCLASSID

	INNER JOIN CIM_CODE CD1

		ON 1=1

		AND C.SITEID = CD1.SITEID 

		AND C.INSPTYPE = CD1.CODEID

		AND CD1.CODECLASSID = 'INSPTYPE'

	LEFT OUTER JOIN CIM_CODE CD2

		ON 1=1

		AND C.SITEID = CD2.SITEID 

		AND C.DATATYPE = CD2.CODEID

		AND CD2.CODECLASSID = 'DATATYPE'

	LEFT OUTER JOIN CIM_CODE CD3

		ON 1=1

		AND C.SITEID = CD3.SITEID 

		AND C.CALCULATETYPE = CD3.CODEID

		AND CD3.CODECLASSID = 'CALCULATETYPE'

	LEFT OUTER JOIN CIM_CODE CD4

		ON 1=1

		AND C.SITEID = CD4.SITEID 

		AND C.INSPSPEC = CD4.CODEID

		AND CD4.CODECLASSID = 'INSPSPEC'

	INNER JOIN CIM_INSPITEM II

		ON 1=1

		AND C.SITEID = II.SITEID

		AND C.INSPITEMID = II.INSPITEMID

 WHERE 1=1

   AND C.INSPDEFINITIONSYSID = :INSPDEFINITIONSYSID

   AND (:ISUSABLE IS NULL OR C.ISUSABLE = :ISUSABLE)

   AND C.SITEID = :SITEID

ORDER BY C.INSPDEFINITIONSYSID, C.UISEQUENCE
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
