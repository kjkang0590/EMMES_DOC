-- SelectWorkOrderListByProcessSegmentId



	

SELECT DISTINCT WO.WORKORDERID, WO.WORKORDERNAME, WO.PRODUCTDEFINITIONID, PRD.PRODUCTDEFINITIONNAME, WO.QTY 

FROM CIM_PROCESSSEGMENT PS

	INNER JOIN CIM_PROCESSNODE PN

		ON 1=1

		AND PN.PROCESSSEGMENTID = PS.PROCESSSEGMENTID

		AND PN.SITEID = PS.SITEID

		AND PN.ISSTARTNODE = 'Y'

		AND PN.ISUSABLE = 'Usable'

	INNER JOIN CIM_PROCESSDEFINITION PD

		ON 1=1

		AND PD.PROCESSDEFINITIONID = PN.PROCESSDEFINITIONID

		AND PD.SITEID = PN.SITEID

		AND PD.ISUSABLE = 'Usable'

	INNER JOIN CIM_PRODUCTPROCESSREL PPR

		ON 1=1

		AND PPR.PROCESSDEFINITIONID = PD.PROCESSDEFINITIONID

		AND PPR.SITEID = PD.SITEID

		AND PPR.ISUSABLE = 'Usable'

	INNER JOIN CIM_WORKORDER WO

		ON 1=1

		AND WO.PRODUCTDEFINITIONID = PPR.PRODUCTDEFINITIONID

		AND WO.SITEID = PPR.SITEID

		AND WO.STATE = 'Active'

		AND WO.ISUSABLE = 'Usable'

	INNER JOIN CIM_PRODUCTDEFINITION PRD

		ON 1=1

		AND PRD.PRODUCTDEFINITIONID = WO.PRODUCTDEFINITIONID

		AND PRD.SITEID = WO.SITEID	

		AND PRD.ISUSABLE = 'Usable'

	WHERE 1=1

		AND PS.PROCESSSEGMENTID = :PROCESSSEGMENTID

		AND PS.SITEID = :SITEID

		AND PS.ISUSABLE = 'Usable'

ORDER BY WO.WORKORDERID
