-------------- 화성타입지정 Node 조회 쿼리

 SELECT

  	CPN.FORMATIONDEFINITIONID ,

 	CPN.FORMATIONVERSION ,

 	CPN.PROCESSDEFINITIONID ,

    CPN.PROCESSNODEID ,

    CPN.PROCESSNODENAME ,

 	CPN.PROCESSNODETYPE ,

 	CPN.PROCESSNODESEQUENCE ,

 	CPN.ISSTARTNODE ,

 	CPN.ISENDNODE ,

 	CPN.NEXTPROCESSNODEID ,

 	CPN.DESCRIPTION ,

 	CPN."CYCLE" ,

 	CFD.FORMATIONDEFINITIONNAME ,

 	CPN.PROCESSSEGMENTID ,

    CPS.PROCESSSEGMENTNAME ,

 	CPN.RECIPEDEFINITIONID ,

 	CPN.RECIPEDEFINITIONNAME 

   FROM CIM_PROCESSNODE CPN

   LEFT JOIN CIM_FORMATIONDEFINITION CFD 

     ON 1=1

    AND CFD.FORMATIONDEFINITIONID = CPN.FORMATIONDEFINITIONID 

    AND CFD.FORMATIONVERSION  = CPN.FORMATIONVERSION 

    AND CFD.SITEID = CPN.SITEID 

    AND CFD.ISUSABLE = 'Usable'

   LEFT JOIN CIM_PROCESSSEGMENT CPS 

     ON 1=1

    AND CPS.PROCESSSEGMENTID  = CPN.PROCESSSEGMENTID

    AND CPS.SITEID = CPN.SITEID 

    AND CPS.ISUSABLE = 'Usable'  

  WHERE 1=1

    AND CPN.SITEID = :SITEID

    AND CPN.ISUSABLE = 'Usable'

    AND CPN.PROCESSDEFINITIONID = :BATCHID

  ORDER BY CPN.PROCESSNODESEQUENCE
#3 Query SelectCurrentProcessNodeSequence

연결된 기능 (1번 표): #3 event

Query (조회 서비스) : SelectCurrentProcessNodeSequence

  • ID 구성: SelectCurrentProcessNodeSequence / 클래스 THiRAMES.Service.UI / 버전 00001
  • 정의 위치: DB 저장 쿼리 (CIM_STOREDQUERY)
  • 쿼리 파라메터: SITEID WORKORDERID BATCHID
  • 테이블(조회/쓰기 전체, 함수·SP 포함):
    CIM_BATCH메인읽기   CIM_LOTHIST읽기   CIM_LOTBATCHREL읽기   CIM_PROCESSNODE읽기
조인 구조도 테이블 3 · 조인 2
INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_PROCESSNODE
SQL 보기
------- 화성타입지정 Current Sequence 조회 쿼리

SELECT 

	 NVL(CPN.PROCESSNODESEQUENCE,0) AS PROCESSNODESEQUENCE

  FROM CIM_BATCH CB

  LEFT JOIN (

	 	SELECT

	 		CLB.BATCHID ,

	 		MAX(CL.PROCESSNODEID) AS PROCESSNODEID 

		  FROM CIM_LOTHIST CL 

	     INNER JOIN CIM_LOTBATCHREL CLB 

	             ON CLB.LOTID  = CL.LOTID

	            AND CLB.BATCHID = CL.PROCESSDEFINITIONID

	            AND CLB.SITEID = CL.SITEID

	            AND CLB.ISUSABLE = 'Usable'

	     INNER JOIN CIM_BATCH CB 

	             ON CB.BATCHID = CLB.BATCHID 

	            AND CB.BATCHTYPE = 'Formation'

	     GROUP BY CLB.BATCHID

  	  ) NODE 

     ON 1=1

    AND NODE.BATCHID = CB.BATCHID

  LEFT JOIN CIM_PROCESSNODE CPN 

         ON 1=1

        AND CPN.PROCESSNODEID = NODE.PROCESSNODEID

        AND CPN.SITEID = CB.SITEID 

        AND CPN.ISUSABLE = 'Usable'

 WHERE 1=1

   AND CB.BATCHTYPE = 'Formation'

   AND CB.ISUSABLE = 'Usable'

   AND CB.SITEID = :SITEID

   AND CB.WORKORDERID = :WORKORDERID     

   AND CB.BATCHID = :BATCHID    

 ORDER BY CB.BATCHID
