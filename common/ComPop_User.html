--ComPop_User
--SelectUserListForComPop
--Param : SITEID, DEPTID, POSITIONID, USERNAME
------------------------------------------
SELECT U.DEPTID
     , U.DEPTNAME
     , U.POSITIONID
     , U.POSITIONNAME
     , U.USERNAME
     , U.USERID
     , U.ISUSABLE
     , CCU.USERNAME AS CREATOR
     , U.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , U.MODIFYTIME
  FROM CIM_USER U WITH(NOLOCK)
  LEFT OUTER JOIN CIM_USER CCU WITH(NOLOCK)
    ON CCU.SITEID = U.SITEID
   AND CCU.USERID = U.CREATOR
  LEFT OUTER JOIN CIM_USER MCU WITH(NOLOCK)
    ON MCU.SITEID = U.SITEID
   AND MCU.USERID = U.MODIFIER
 WHERE U.SITEID = @SITEID
   AND (@DEPTID = '' OR U.DEPTID = @DEPTID)
   AND (@POSITIONID = '' OR U.POSITIONID = @POSITIONID)
   AND (@USERNAME = '' OR U.USERNAME LIKE '%' + @USERNAME + '%')
 ORDER BY U.DEPTID
     , U.POSITIONID
     , U.UISEQUENCE
     , U.USERNAME
