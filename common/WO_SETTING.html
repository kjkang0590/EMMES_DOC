---------------- CUS_PRODUCTWORKORDERREL 조회
SELECT
	CPR.WORKORDERPRODUCTITEMID ,
	CPD.PRODUCTDEFINITIONNAME AS WORKORDERPRODUCTITEMNAME ,
	CPR.SITEID ,
	CPR.UISEQUENCE ,
	CPR.SCHEDULEUNIT ,
	CPR.ISUSABLE ,
	CPR.DESCRIPTION ,
	CPR.REASONCODE ,
	CPR.COMMENTS 
  FROM CUS_PRODUCTWORKORDERREL CPR
 INNER JOIN CIM_PRODUCTDEFINITION CPD
    ON 1=1
   AND CPD.SITEID = CPR.SITEID
   AND CPD.ISUSABLE = 'Usable'
   AND CPD.PRODUCTDEFINITIONID = CPR.WORKORDERPRODUCTITEMID
 WHERE 1=1
   AND CPR.SITEID = :SITEID
   AND CPR.ISUSABLE = 'Usable'
   AND CPR.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID
