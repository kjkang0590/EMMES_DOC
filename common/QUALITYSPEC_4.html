-- INPUT = SITEID, INSPTYPE

SELECT A.CODE,

	   A.NAME

  FROM (

  	    SELECT UNISTR('IQC,PQC,OQC') AS INSPTYPE,

  	    	   '-' AS CODE,

  	    	   UNISTR('-') AS NAME

  	      FROM DUAL

  	    UNION ALL

  	    SELECT 'PQC,OQC' AS INSPTYPE,

  	    	   CUSTOMERID AS CODE,

	   		   CUSTOMERNAME AS NAME

		  FROM CIM_CUSTOMER

		 WHERE ISUSABLE='Usable' 

		   AND SITEID = :SITEID

		UNION ALL  

		SELECT 'IQC' AS INSPTYPE,

			   VENDORID AS CODE,

			   VENDORNAME AS NAME

		  FROM CIM_VENDOR

		 WHERE ISUSABLE ='Usable' 

		   AND SITEID = :SITEID

	   ) A

 WHERE 1=1

   AND (:INSPTYPE IS NULL OR A.INSPTYPE LIKE '%' || :INSPTYPE || '%')
