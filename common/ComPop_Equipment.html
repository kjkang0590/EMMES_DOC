마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_EQUIPMENTCLASS, CIM_PROCESSSEGMENT, CIM_USER
SQL 보기
------------------------------------------
--ComPop_Equipment
--SelectEquipmentListForComPop
--Param : SITEID, LINEID, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID
------------------------------------------
SELECT CE.ROOTEQUIPMENTID
     , CC.EQUIPMENTCLASSID
     , CC.EQUIPMENTCLASSNAME
     , CE.EQUIPMENTID
     , CE.EQUIPMENTNAME
     , CE.EQUIPMENTTYPE
     , CE.PROCESSTYPE
     , CE.PROCESSSEGMENTID
     , PS.PROCESSSEGMENTNAME
     , CE.ISUSABLE
     , CE.UISEQUENCE
     , CE.DESCRIPTION
     , CCU.USERNAME AS CREATOR
     , CE.CREATETIME
     , MCU.USERNAME AS MODIFIER
     , CE.MODIFYTIME
  FROM CIM_EQUIPMENT CE WITH(NOLOCK)
 INNER JOIN CIM_EQUIPMENTCLASS CC WITH(NOLOCK) ON CC.SITEID = CE.SITEID AND CC.EQUIPMENTCLASSID = CE.EQUIPMENTCLASSID
--  LEFT JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PE WITH(NOLOCK) ON PE.SITEID = CE.SITEID AND PE.EQUIPMENTID = CE.EQUIPMENTID
  LEFT JOIN CIM_PROCESSSEGMENT PS WITH(NOLOCK) ON PS.SITEID = CE.SITEID AND PS.PROCESSSEGMENTID = CE.PROCESSSEGMENTID
  LEFT JOIN CIM_USER CCU WITH(NOLOCK) ON CCU.SITEID = CE.SITEID AND CCU.USERID = CE.CREATOR
  LEFT JOIN CIM_USER MCU WITH(NOLOCK) ON MCU.SITEID = CE.SITEID AND MCU.USERID = CE.MODIFIER
 WHERE CE.SITEID = @SITEID
   AND CE.LINEID = @LINEID
   AND (@ROOTEQUIPMENTID = '' OR CE.ROOTEQUIPMENTID = @ROOTEQUIPMENTID)
   AND (@EQUIPMENTCLASSID = '' OR CC.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
   AND (@EQUIPMENTID = '' OR UPPER(CE.EQUIPMENTID) LIKE UPPER('%' + @EQUIPMENTID + '%'))
 ORDER BY CC.ROOTEQUIPMENTID
     , CC.EQUIPMENTCLASSID
     , CE.UISEQUENCE
     , CE.EQUIPMENTID

INNER JOIN LEFT JOIN 기준(FROM) 테이블 · 박스 안 ON 줄은 해당 테이블의 조인 조건 (정적 분석, 서브쿼리/UNION은 근사)
SQL 보기
------------------------------------------
--[공통]설비그룹 조회 - UNIT 별  
--SelectCboEquipmentClassList
--Param : SITEID, ROOTEQUIPMENTID
------------------------------------------	
SELECT
	EC.EQUIPMENTCLASSID AS CODE
	, EC.EQUIPMENTCLASSNAME AS NAME
FROM CIM_EQUIPMENTCLASS EC WITH(NOLOCK),
(
	SELECT
		E.EQUIPMENTCLASSID
		, E.SITEID
	FROM CIM_EQUIPMENT E WITH(NOLOCK), CST_PUBLICREL R WITH(NOLOCK)
	WHERE E.EQUIPMENTID = R.PARENTID
	AND   E.SITEID = R.SITEID
	AND   E.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
	GROUP BY E.EQUIPMENTCLASSID, E.SITEID
) E
WHERE EC.EQUIPMENTCLASSID = E.EQUIPMENTCLASSID
AND   EC.SITEID = E.SITEID
