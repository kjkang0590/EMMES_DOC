SELECT CODEID AS CODE,
       CODENAME AS NAME,
       UISEQUENCE
FROM CIM_CODE 
WHERE CODECLASSID = :CODECLASSID

      AND SITEID = :SITEID
ORDER BY UISEQUENCE
