-- SelectProductProcessRelPopUp 라우팅설정 팝업 조회문
------------------------------------
-- INPUT = SITEID, PRODUCTDEFINITIONID
SELECT
    PD.SITEID
  , PD.PRODUCTDEFINITIONID
  , PD.SITEID 
  , PF.PROCESSDEFINITIONID 
  , PF.PROCESSDEFINITIONNAME 
  , ISNULL(FL.ISDEFAULT, 'N') AS ISDEFAULT
  , CASE WHEN FL.PRODUCTDEFINITIONID IS NOT NULL THEN 'Usable' ELSE 'UnUsable' END AS ISUSABLE
  , FL.DESCRIPTION
  , FL.COMMENTS
FROM CIM_PRODUCTDEFINITION PD
JOIN CIM_PROCESSDEFINITION PF 
	ON PF.PROCESSTYPE = PD.PRODUCTDEFINITIONTYPE
  AND PF.SITEID = PD.SITEID
  AND PF.ISUSABLE ='Usable'
LEFT OUTER JOIN CIM_PRODUCTPROCESSREL FL 
	ON FL.PROCESSDEFINITIONID = PF.PROCESSDEFINITIONID
  AND FL.PRODUCTDEFINITIONID = PD.PRODUCTDEFINITIONID 
  AND FL.SITEID = PF.SITEID
  AND FL.ISUSABLE = 'Usable'
WHERE 1=1
  AND PD.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID
ORDER BY FL.PRODUCTDEFINITIONID, CASE WHEN FL.ISDEFAULT ='Y' THEN 0 ELSE 1 END, PF.PROCESSDEFINITIONID
