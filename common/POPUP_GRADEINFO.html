SQL 보기
SELECT 

	CG.GRADETYPE ,

	CG.GRADEID , 

	CG.GRADENAME 

  FROM CIM_GRADEDEFINITION CG

 WHERE CG.SITEID = :SITEID

   AND CG.ISUSABLE = 'Usable'

   AND CG.GRADETYPE = :PRODUCTDEFINITIONID

 ORDER BY UISEQUENCE
