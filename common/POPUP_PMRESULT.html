--EM_PB_004_PM 결과 등록 - PM 결과 입력 POPUP Check List Grid 조회 쿼리
--SelectPmResultCheckList
--Param : SITEID, LINEID, PMSYSID, PMDEFINITIONID
------------------------------------------
SELECT B.PMSYSID,
       A.PMITEMID,
	   A.PMITEMNAME,
       A.PMTYPE,
	   B.PMRESULT,
	   B.PMDETAILRESULT,
	   IIF(B.UNITID IS NULL, A.UNITID, B.UNITID) AS UNITID,
	   B.DESCRIPTION,
	   B.UNITID
  FROM CST_PMITEM A WITH(NOLOCK)
  LEFT JOIN CST_PMITEMRESULT B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.PMITEMID = B.PMITEMID
   AND B.PMSYSID = @PMSYSID
  LEFT JOIN CST_PMDEFINITIONITEMREL D WITH(NOLOCK)
    ON A.SITEID = D.SITEID 
   AND A.PMITEMID = D.PMITEMID
  LEFT JOIN CST_PMDEFINITION E WITH(NOLOCK)
    ON D.SITEID = E.SITEID 
   AND D.PMDEFINITIONID = E.PMDEFINITIONID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.ISUSABLE = 'Usable'
   AND E.PMDEFINITIONID = @PMDEFINITIONID
 ORDER BY A.PMITEMID

------------------------------------------
--EM_PB_004_PM 결과 등록 - BM 등록 목록 조회 쿼리
--SelectPmInspBmRequestList
--Param : SITEID, PMSYSID
------------------------------------------
SELECT S1.BMID,
	   CC.CODENAME AS BMNAME,
	   S1.EQUIPMENTCLASSID,
	   S1.EQUIPMENTID,
	   S1.EQUIPMENTNAME,
	   S1.FAILURECODE,
	   S1.FAILUREDATETIME,
	   S1.DESCRIPTION
  FROM (
	SELECT A.SITEID,
		   A.BMID,
	       IIF(CHARINDEX('PM', A.BMID) = 1, 'PMNM', 'BMNM') AS BMNAMECODE,
	       C.EQUIPMENTCLASSID,
	       A.EQUIPMENTID,
	       B.EQUIPMENTNAME,
	       A.FAILURECODE,
	       A.FAILUREDATETIME,
	       A.DESCRIPTION
	  FROM CST_BMRESULT A WITH(NOLOCK)
	  LEFT JOIN CIM_EQUIPMENT B WITH(NOLOCK)
	    ON A.SITEID = B.SITEID
	   AND A.EQUIPMENTID = B.EQUIPMENTID 
	  LEFT JOIN CIM_EQUIPMENTCLASS C WITH(NOLOCK)
	    ON B.SITEID = C.SITEID 
	   AND B.EQUIPMENTCLASSID = C.EQUIPMENTCLASSID
	 WHERE A.SITEID = @SITEID
	   AND A.PMSYSID = @PMSYSID  
  ) S1
  LEFT JOIN CIM_CODE CC WITH(NOLOCK) ON S1.SITEID = CC.SITEID AND S1.BMNAMECODE = CC.CODEID AND CC.CODECLASSID = 'BMNAME'
 ORDER BY S1.BMID DESC
