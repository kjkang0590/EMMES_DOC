		ON 1=1

	   AND EQ.SITEID = PD.SITEID

-- SelectCboProcessSegment4RecipeMaster = ProcessSegment COMBO 조회

------------------------------------

-- INPUT = SITEID, PRODUCTDEFINITIONID, EQUIPMENTID

SELECT DISTINCT

	   EL.PROCESSSEGMENTID,

	   PS.PROCESSSEGMENTNAME

  FROM CIM_PRODUCTDEFINITION PD

	INNER JOIN CIM_PRODUCTPROCESSREL PL 

		ON 1=1

	   AND PL.SITEID = PD.SITEID

	   AND PL.PRODUCTDEFINITIONID = PD.PRODUCTDEFINITIONID 

	   AND PL.ISUSABLE = 'Usable'

	INNER JOIN CIM_PROCESSNODE PN 

		ON 1=1

	   AND PN.SITEID = PD.SITEID

	   AND PN.PROCESSDEFINITIONID = PL.PROCESSDEFINITIONID 

	   AND PN.ISUSABLE = 'Usable'

	INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL EL 

		ON 1=1

	   AND EL.SITEID = PD.SITEID

	   AND EL.PROCESSSEGMENTID = PN.PROCESSSEGMENTID  --  이건 ISUSABLE 전부

	INNER JOIN CIM_PROCESSSEGMENT PS

		ON 1=1

	   AND PS.SITEID = PD.SITEID

	   AND PS.PROCESSSEGMENTID = EL.PROCESSSEGMENTID

 WHERE 1=1

   AND PD.SITEID = :SITEID

   AND PD.PRODUCTDEFINITIONID = :PRODUCTDEFINITIONID

   AND EL.EQUIPMENTID = :EQUIPMENTID
