-- SelectMaterialDefWithoutFertList : Material Definition 조회

------------------------------------

-- INPUT : SITEID, ISUSABLE, MATERIALDEFINITIONID, MATERIALDEFINITIONNAME

SELECT MD.*

  FROM CIM_MATERIALDEFINITION MD

 WHERE 1=1

   AND MD.SITEID = :SITEID

   AND (:ISUSABLE IS NULL OR ISUSABLE = :ISUSABLE)

   AND (:MATERIALDEFINITIONID IS NULL OR MATERIALDEFINITIONID LIKE :MATERIALDEFINITIONID || '%')

   AND (:MATERIALDEFINITIONNAME IS NULL OR MATERIALDEFINITIONNAME LIKE :MATERIALDEFINITIONNAME || '%')

   AND MD.PRODUCTTYPE <> 'FERT'
