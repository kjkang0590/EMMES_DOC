--EM_PB_002_PM 검사 그룹 관리 - PM/BM Item Grid 조회 쿼리
--SelectInspectionItemListForComPop
--Param : SITEID, LINEID, PMITEMID, PMITEMNAME, ISUSABLE
------------------------------------------
SELECT A.PMITEMID,
       A.PMTYPE,
       A.EQUIPMENTTYPE,
       A.EQUIPMENTID,
       A.PMCATEGORYID,
	   A.PMITEMNAME,
	   A.PMCLASSID,
	   A.SCHEDULEUNIT,
	   A.ISUSABLE,
	   A.COMMENTS,
	   A.CREATOR,
	   B.USERNAME AS CREATORNAME,
	   A.CREATETIME,
	   A.MODIFIER,
	   C.USERNAME AS MODIFIERNAME,
	   A.MODIFYTIME 
  FROM CST_PMITEM A WITH(NOLOCK)
  LEFT JOIN CIM_USER B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.CREATOR = B.USERID
  LEFT JOIN CIM_USER C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.MODIFIER = C.USERID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)
   AND (@PMCATEGORYID = '' OR A.PMCATEGORYID = @PMCATEGORYID)
   AND (@PMITEMNAME = '' OR A.PMITEMNAME LIKE '%' + @PMITEMNAME + '%')
   AND A.ISUSABLE = 'Usable'
 ORDER BY A.MODIFYTIME DESC
