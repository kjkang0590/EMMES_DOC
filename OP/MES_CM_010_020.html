SELECT C.*, dbo.FN_USERNAME(C.SITEID, C.CREATOR) AS CREATOR2, dbo.FN_USERNAME(C.SITEID, C.MODIFIER) AS MODIFIER2
FROM CIM_MATERIALDEFINITION C
WHERE (:MATERIALDEFINITIONID = '' OR MATERIALDEFINITIONID LIKE :MATERIALDEFINITIONID + '%')
      AND (:MATERIALDEFINITIONNAME = '' OR MATERIALDEFINITIONNAME LIKE :MATERIALDEFINITIONNAME + '%')
      AND (:MATERIALCLASSID = '' OR MATERIALCLASSID LIKE :MATERIALCLASSID + '%')
      AND (:ISUSABLE = '' OR ISUSABLE = :ISUSABLE)
      AND SITEID = :SITEID
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
