--EM_PB_002_PM 검사 그룹 관리 - PM / BM 그룹 목록 Grid 조회 쿼리
--SelectPmInspectionGroupList
--Param : SITEID, LINEID, PMTYPE, PMDEFINITIONID, PMSYSNAME, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID, ISUSABLE
------------------------------------------   
SELECT A.PMDEFINITIONID,
	   A.PMSYSNAME,
	   A.PMTYPE,
	   A.ROOTEQUIPMENTID,
	   C.EQUIPMENTCLASSID,
	   C.EQUIPMENTCLASSNAME,
	   A.EQUIPMENTID,
	   D.PROCESSSEGMENTNAME,
	   D.PROCESSSEGMENTID,
	   A.SCHEDULEUNIT,
	   A.SCHEDULETARGET,
	   CONVERT(CHAR(19), A.VALIDSTARTTIME, 20) AS VALIDSTARTTIME,
	   CONVERT(CHAR(19), A.VALIDENDTIME, 20) AS VALIDENDTIME,
	   A.ISAUTOSCHEDULE,
	   A.ISUSABLE,
	   A.CREATOR,
	   A.CREATETIME,
	   A.MODIFIER,
	   A.MODIFYTIME
  FROM CST_PMDEFINITION A WITH(NOLOCK)
  LEFT JOIN CIM_EQUIPMENT B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.EQUIPMENTID = B.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.EQUIPMENTCLASSID = C.EQUIPMENTCLASSID
  LEFT JOIN CIM_PROCESSSEGMENT D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.PROCESSSEGMENTID = D.PROCESSSEGMENTID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND (@PMTYPE = '' OR A.PMTYPE = @PMTYPE)
   --AND (@PMDEFINITIONID = '' OR A.PMDEFINITIONID LIKE '%' + @PMDEFINITIONID + '%')
   AND (@PMSYSNAME = '' OR A.PMSYSNAME LIKE '%' + @PMSYSNAME + '%')
   AND A.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
   AND (@EQUIPMENTCLASSID = '' OR C.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%')
   AND (@ISUSABLE = '' OR A.ISUSABLE = @ISUSABLE)
 ORDER BY A.VALIDSTARTTIME, A.VALIDENDTIME

--EM_PB_002_PM 검사 그룹 관리 - PM / BM 검사 항목 목록 Grid 조회 쿼리
--SelectPmInspectionDetailList
--Param : SITEID, LINEID, PMDEFINITIONID
------------------------------------------
SELECT A.PMITEMID,
       A.PMTYPE,
       A.EQUIPMENTTYPE,
       A.EQUIPMENTID,
       A.PMCATEGORYID,
	   A.PMITEMNAME,
	   A.PMCLASSID,
	   A.SCHEDULEUNIT,
	   A.ISUSABLE,
	   A.COMMENTS,
	   A.CREATOR,
	   D.USERNAME AS CREATORNAME,
	   A.CREATETIME,
	   A.MODIFIER,
	   E.USERNAME AS MODIFIERNAME,
	   A.MODIFYTIME 
  FROM CST_PMITEM A WITH(NOLOCK)
  LEFT JOIN CST_PMDEFINITIONITEMREL B WITH(NOLOCK)
    ON A.SITEID = B.SITEID 
   AND A.PMITEMID = B.PMITEMID
  LEFT JOIN CST_PMDEFINITION C WITH(NOLOCK)
    ON B.SITEID = C.SITEID 
   AND B.PMDEFINITIONID = C.PMDEFINITIONID
  LEFT JOIN CIM_USER D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.CREATOR = D.USERID
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.MODIFIER = E.USERID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND C.PMDEFINITIONID = @PMDEFINITIONID
 ORDER BY A.PMITEMID

CREATE PROCEDURE dbo.SP_EM_PB_002_INSERTPMRESULTLIST

	@SITEID VARCHAR(40)

	, @USERID VARCHAR(40)

	, @LINEID VARCHAR(40)

	, @PMDEFINITIONID VARCHAR(40)

	, @DELETEFLAG VARCHAR(1)

	--, @RETURNVALUE VARCHAR(100) OUTPUT

AS	

BEGIN

		

	DECLARE @TARGETDAYINFO TABLE

	(

		NUMBERID 			INT

		, BASE_WEEK			INT

		, TARGET_DAY		DATE

		, WEEKOFYEAR		INT

		, WEEKDAY			INT

		, ISLASTMONTHDAY	VARCHAR(1)

		, ISFEBLASTDAY		VARCHAR(1)

		, CREATETARGET      VARCHAR(1)

	)

	

	DECLARE @VALIDSTARTDATE DATETIME

		  , @VALIDENDDATE DATETIME

		  , @TODATE DATETIME

		  , @SCHEDULEUNIT VARCHAR(4)

		  , @SCHEDULETARGET INT

		  , @SQL VARCHAR(MAX)

	;

	

	--SET @RETURNVALUE = 'SUCCESS';



	BEGIN

		-- PM 검사 그룹 관리에서 선택한 그룹의 시작일자 기준 2년치의 스케줄을 생성하기 위해 조회

		SELECT 

			@VALIDSTARTDATE = VALIDSTARTTIME,

			@VALIDENDDATE = DATEADD(YEAR, 2, VALIDSTARTTIME),		

			@TODATE = CONVERT(DATE, GETDATE()),

			@SCHEDULEUNIT = SCHEDULEUNIT,

			@SCHEDULETARGET = SCHEDULETARGET

		  FROM CST_PMDEFINITION

		 WHERE SITEID = @SITEID

		   AND PMDEFINITIONID = @PMDEFINITIONID

				

		--SELECT @VALIDSTARTDATE, @VALIDENDDATE, @TODATE, @SCHEDULEUNIT, @SCHEDULETARGET

		

		IF @SCHEDULEUNIT NOT IN ( 'D', 'W', 'M', 'Y' ) 

		BEGIN

			--PRINT '스케쥴 단위가 D,W,M,Y 에서 벗어남.'

			--SET @RETURNVALUE = 'SP_EM_PB_002_E001'

			--RETURN;

			

			SET @SQL = 'SELECT ''E'' AS errCode' 

	  		EXEC (@SQL)

		END

		

		IF @SCHEDULETARGET = 0 OR @SCHEDULETARGET IS NULL  

		BEGIN 

			--PRINT '스케쥴 주기가 없거나 0 입니다'

			--SET @RETURNVALUE = 'SP_EM_PB_002_E002'

			--RETURN;

			

			SET @SQL = 'SELECT ''E'' AS errCode' 

	  		EXEC (@SQL)

		END

		

		-- 일별 2년치 일정 저장

		INSERT INTO @TARGETDAYINFO 

		SELECT

			number AS NUMBERID								-- 순번

			, DATEPART(WK,@VALIDSTARTDATE) AS BASE_WEEK		-- 기준 WEEK (DEFINITION 정보)

			, DATEADD(d, number, CONVERT(DATE, @TODATE, 112)) AS TARGET_DAY	-- 대상 일자 

			, DATEPART(WK,DATEADD(d, number, CONVERT(DATE, @TODATE, 112))) AS WEEKOFYEAR

			, DATEPART(WEEKDAY, DATEADD(d, number, CONVERT(DATE, @TODATE, 112))) AS WEEKDAY	 -- 0 : SUNDAY

			, CASE WHEN EOMONTH ( DATEADD(d, number, CONVERT(DATE, @TODATE, 112)) ) = DATEADD(d, number, CONVERT(DATE, @TODATE, 112)) THEN 'Y' ELSE 'N' END AS ISLASTMONTHDAY

			, CASE WHEN DATEPART(MONTH, DATEADD(d, number, CONVERT(DATE, @TODATE, 112))) = 2 

				        AND CONVERT(INT, DATEPART(d, EOMONTH ( DATEADD(d, number, CONVERT(DATE, @TODATE, 112)) ) )) <= CONVERT(INT, DATEPART(d, @VALIDSTARTDATE) )

	   			        AND  EOMONTH ( DATEADD(d, number, CONVERT(DATE, @TODATE, 112)) ) = DATEADD(d, number, CONVERT(DATE, @TODATE, 112))

			         THEN 'Y' 

			       ELSE 'N' 

			  END AS ISFEBLASTDAY			

			, 'N' AS CREATETARGET			

		FROM master..spt_values A

		WHERE A.type = 'P'

		AND number <= DATEDIFF(d, @TODATE, DATEADD(YEAR, 2, CONVERT(DATE, @TODATE)))

		;		



	END

	--SELECT * FROM @TARGETDAYINFO



	-- 오늘 날짜 이후 기준 PM 결과 등록되지 않은 데이터 삭제 처리

	BEGIN

		IF @DELETEFLAG = 'Y'

		BEGIN

			--SELECT *

			DELETE

			FROM CST_PMRESULT

			WHERE SITEID = @SITEID

			AND PMDEFINITIONID = @PMDEFINITIONID

			AND SCHEDULESTARTDATE > @TODATE

			AND (STARTDATE IS NULL AND ENDDATE IS NULL)

		END

	END

	

	-- 등록 대상 선정

	BEGIN TRY

		-- 일 기준 주기일 경우

		IF @SCHEDULEUNIT = 'D'

		BEGIN

			PRINT 'D'

			UPDATE @TARGETDAYINFO

			SET CREATETARGET = 'Y'

			WHERE NUMBERID % @SCHEDULETARGET = 0

		END

		

		-- 주 기준 주기일 경우

		ELSE IF @SCHEDULEUNIT = 'W'

		BEGIN

			PRINT 'W'

			UPDATE @TARGETDAYINFO

			SET CREATETARGET = 'Y'

			WHERE NUMBERID IN 

			(

				SELECT NUMBERID	FROM @TARGETDAYINFO MA

				WHERE DATEPART(WEEKDAY, @VALIDSTARTDATE ) = WEEKDAY

				AND   (WEEKOFYEAR - BASE_WEEK) % @SCHEDULETARGET = 0

			)

 

		END 

		

		-- 월 기준 주기일 경우

		ELSE IF @SCHEDULEUNIT = 'M'

		BEGIN 

			PRINT 'M'

			

			-- 마지막 날인 경우

			IF EOMONTH(@VALIDSTARTDATE) = CONVERT(DATE, @VALIDSTARTDATE)

			BEGIN

				UPDATE @TARGETDAYINFO

				SET CREATETARGET = 'Y'

				WHERE NUMBERID IN 

					(

						SELECT NUMBERID

						FROM 

						(

							SELECT 

								NUMBERID

								, RANK() OVER(ORDER BY M.NUMBERID ASC) AS MONTHROW

							FROM @TARGETDAYINFO M

							WHERE M.ISLASTMONTHDAY = 'Y' OR M.ISFEBLASTDAY = 'Y'

						) MA

						WHERE MA.MONTHROW % @SCHEDULETARGET = 0

					)



			END

			

			-- 마지막 일이 아닌 경우

			ELSE

			BEGIN

				UPDATE @TARGETDAYINFO

				SET CREATETARGET = 'Y'

				WHERE NUMBERID IN 

				(			

					SELECT 

						NUMBERID

					FROM 

					(

						SELECT 

							ROW_NUMBER() OVER(ORDER BY NUMBERID ASC) AS MONTHROW

							, MA.NUMBERID

						FROM @TARGETDAYINFO MA

						WHERE DAY(@VALIDSTARTDATE) = DAY(TARGET_DAY) -- DEFINITION 의 기준일 중 일자와 대상 일자 비교 (년월 제외한 일자 비교)

						 

					) M

					WHERE (M.MONTHROW - 1) % @SCHEDULETARGET = 0 AND M.MONTHROW > 1

				) 

			END

		END

		ELSE IF @SCHEDULEUNIT = 'Y'

		BEGIN 

		

			UPDATE @TARGETDAYINFO

			SET CREATETARGET = 'Y'

			WHERE TARGET_DAY IN 

				(

					SELECT TARGET_DAY

					FROM

					(

						SELECT 

							number AS NUMBERID

							, DATEADD(YY, number, CONVERT(DATE, @VALIDSTARTDATE, 112)) AS TARGET_DAY

						FROM master..spt_values A

						WHERE A.type = 'P'

						AND number <= DATEDIFF(YY, @VALIDSTARTDATE, DATEADD(YEAR, 2, CONVERT(DATE, @VALIDSTARTDATE)))

					) M

					WHERE NUMBERID % @SCHEDULETARGET = 0

				)

		END	

		

	END TRY

	BEGIN CATCH

		PRINT 'CATCH'

		--SET @RETURNVALUE = 'SP_EM_PB_002_E003'

		SET @SQL = 'SELECT ''E'' AS errCode' 

	  	EXEC (@SQL)

	END CATCH

	

	INSERT INTO CST_PMRESULT

	(

		PMDEFINITIONID, 

		SITEID,

		PMDEFINITIONNAME,

		PMTYPE,

		STATE,

		ROOTEQUIPMENTID,

		PROCESSSEGMENTID, 

		EQUIPMENTID, 

		SCHEDULESTARTDATE, 

		SCHEDULEENDDATE, 

		LINEID,

		ACTIVITY,

		ISUSABLE, 

		CREATOR, 

		CREATETIME,

		MODIFIER,

		MODIFYTIME,

		LASTEVENTTIME

	)

	SELECT S1.PMDEFINITIONID,

		   S1.SITEID,

		   S1.PMDEFINITIONNAME,

		   S1.PMTYPE,

		   S1.STATE,

		   S1.ROOTEQUIPMENTID,

		   S1.PROCESSSEGMENTID,

		   S1.EQUIPMENTID,

		   S1.TARGET_DAY AS SCHEDULESTARTDATE,

		   S1.TARGET_DAY AS SCHEDULEENDDATE,

		   S1.LINEID,

		   S1.ACTIVITY,

		   S1.ISUSABLE,

		   S1.CREATOR,

		   S1.CREATETIME,

		   S1.MODIFIER,

		   S1.MODIFYTIME,

		   S1.LASTEVENTTIME 

	  FROM (

	  	SELECT A.TARGET_DAY, B.*

		  FROM @TARGETDAYINFO A

		  LEFT JOIN (

		  	SELECT A.PMDEFINITIONID,

			       @SITEID AS SITEID,

			       A.PMSYSNAME AS PMDEFINITIONNAME,

			       A.PMTYPE,

			       'Waiting' AS STATE,

			       A.ROOTEQUIPMENTID,

			       A.PROCESSSEGMENTID,

			       A.EQUIPMENTID,

			       @LINEID AS LINEID,

			       'CreatePmSchedule' AS ACTIVITY,

			       'Usable' AS ISUSABLE,

			       @USERID AS CREATOR,

			       GETDATE() AS CREATETIME,

			       @USERID AS MODIFIER,

			       GETDATE() AS MODIFYTIME,

			       GETDATE() AS LASTEVENTTIME

		  	  FROM CST_PMDEFINITION A

		  	 WHERE A.SITEID = @SITEID

			   AND A.PMDEFINITIONID = @PMDEFINITIONID

		  ) B

		    ON B.SITEID = @SITEID

		 WHERE CREATETARGET = 'Y'

	  ) S1

	  

	  SET @SQL = 'SELECT ''S'' AS errCode' 

  	  EXEC (@SQL)

	

END
