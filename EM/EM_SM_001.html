-- 설비 상태 현황 조회 : SelectEquipmentStateList

-- 일단 데이터가 나오게 해놓고 입력 PARAMETER 적용을 하는 것으로 진행하도록함

SITEID : 필수 : 단건 : 사용자가 로그인되어 있는 사업장

PLANTID : 필수 : 단건 : 사용자가 선택한 PLANT

LINEID : 옵션 : 멀티 : 사용자가 선택한 라인 목록, 미선택시 전체 적용

EQUIPMENTID : 옵션 : 멀티 : 사용자가 선택한 설비 목록, 미선택시 전체 적용

WITH DT AS (

  SELECT

    EQ.SITEID, EQ.ISUSABLE, EQ.EQUIPMENTID

  FROM CIM_EQUIPMENT EQ

  LEFT OUTER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL EL ON EL.SITEID = EQ.SITEID AND EL.EQUIPMENTID = EQ.EQUIPMENTID 

  LEFT OUTER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = EL.SITEID AND PS.PROCESSSEGMENTID = EL.PROCESSSEGMENTID 

  WHERE 1=1

    AND EQ.SITEID = :SITEID

    AND EQ.ISUSABLE = 'Usable'

    AND EQ.PLANTID = :PLANTID

    AND EQ.PARENTID IS NULL

    AND 1 = CASE

              WHEN :LINEID IS NULL THEN 1

              WHEN EQ.LINEID IN (SELECT * FROM TABLE(FN_MES_SPLIT(:LINEID, ','))) THEN 1

              ELSE 0

            END

    AND 1 = CASE

              WHEN :EQUIPMENTID IS NULL THEN 1

              WHEN EQ.EQUIPMENTID IN (SELECT * FROM TABLE(FN_MES_SPLIT(:EQUIPMENTID, ','))) THEN 1

              ELSE 0

            END

    AND 1 = CASE

              WHEN :STATE IS NULL THEN 1

              WHEN EQ.STATE IN (SELECT * FROM TABLE(FN_MES_SPLIT(:STATE, ','))) THEN 1

              ELSE 0

            END

    AND 1 = CASE

              WHEN :PROCESSSEGMENTID IS NULL THEN 1

              WHEN EL.PROCESSSEGMENTID IN (SELECT * FROM TABLE(FN_MES_SPLIT(:PROCESSSEGMENTID, ','))) THEN 1

              ELSE 0

            END

)

  SELECT 

      EQ.SITEID

    , EQ.PLANTID

    , PL.FACILITYNAME AS PLANTNANE

    , EQ.LINEID

    , LN.FACILITYNAME AS LINENAME

    , EQ.EQUIPMENTCLASSID 

    , EC.EQUIPMENTCLASSNAME 

    , EQ.EQUIPMENTID

    , EQ.EQUIPMENTNAME

    , EQ.SHORTNAME

    , EQ.STATE

    , EQ.STATETIME

    , EQ.PROCESSSTATE

    , EQ.PROCESSSTATETIME

    , EQ.CONTROLMODE

    , EQ.CONTROLMODETIME

    , EQ.OPERATIONMODE

    , EQ.OPERATIONMODETIME

    , EL.PROCESSSEGMENTID

    , PS.PROCESSSEGMENTNAME

    , EQ.BATCHID

    , EQ.RECIPEID AS RECIPEDEFINITIONID

    , EQ.ALARMID

    , EQ.DESCRIPTION

    , EQ.COMMENTS

    , FN_USERNAME(EQ.SITEID, EQ.MODIFIER) AS MODIFIERNAME

    , EQ.MODIFYTIME

  FROM DT

  INNER JOIN CIM_EQUIPMENT EQ ON EQ.SITEID = DT.SITEID AND EQ.ISUSABLE = DT.ISUSABLE

    AND (EQ.EQUIPMENTID = DT.EQUIPMENTID OR EQ.PARENTID = DT.EQUIPMENTID)

  LEFT OUTER JOIN CIM_FACILITY PL ON PL.SITEID = EQ.SITEID AND PL.FACILITYID = EQ.PLANTID

  LEFT OUTER JOIN CIM_FACILITY LN ON LN.SITEID = EQ.SITEID AND LN.FACILITYID = EQ.LINEID

  LEFT OUTER JOIN CIM_EQUIPMENTCLASS EC ON EC.SITEID = EQ.SITEID AND EC.EQUIPMENTCLASSID = EQ.EQUIPMENTCLASSID 

  LEFT OUTER JOIN CIM_USER UR ON UR.SITEID = EQ.SITEID AND UR.USERID = EQ.MODIFIER

  LEFT OUTER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL EL ON EL.SITEID = EQ.SITEID AND EL.EQUIPMENTID = EQ.EQUIPMENTID 

  LEFT OUTER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = EL.SITEID AND PS.PROCESSSEGMENTID = EL.PROCESSSEGMENTID 

  WHERE 1=1

  ORDER BY EQ.PLANTID, EQ.LINEID, EQ.EQUIPMENTID, EL.PROCESSSEGMENTID
테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

-- SelectProcessSegmentList

SELECT DISTINCT EQ.SITEID, EQ.PLANTID, EQ.LINEID
	, PC.PROCESSSEGMENTID AS CODE
	, PC.PROCESSSEGMENTNAME AS NAME
	, PC.PROCESSTYPE
	, PC.BIZRULEID
	, CASE WHEN (SELECT COUNT(*) FROM CIM_EQUIPMENT SUB WHERE SUB.PARENTID = EQ.EQUIPMENTID) > 0 THEN 'Y' ELSE 'N' END ISSUBEQP
FROM CIM_EQUIPMENT EQ
	INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER
	ON 1=1
	AND PER.EQUIPMENTID = EQ.EQUIPMENTID
	AND PER.SITEID = EQ.SITEID
	AND PER.ISUSABLE = 'Usable'
	INNER JOIN CIM_PROCESSSEGMENT PC
	ON 1=1
	AND PC.PROCESSSEGMENTID = PER.PROCESSSEGMENTID
	AND PC.SITEID = PER.SITEID
	AND PC.ISUSABLE = 'Usable'
WHERE 1=1
	AND (:PLANTID IS NULL OR EQ.PLANTID = :PLANTID)
	AND (:LINEID IS NULL OR  EQ.LINEID = :LINEID)
ORDER BY EQ.PLANTID, EQ.LINEID, PC.PROCESSSEGMENTID

-- SelectEquipmentListByProcessSegment

SELECT EQ.SITEID, EQ.PLANTID, EQ.LINEID

	, EQ.EQUIPMENTID AS CODE

	, EQ.EQUIPMENTNAME AS NAME

FROM CIM_EQUIPMENT EQ

	INNER JOIN CIM_PROCESSSEGMENTEQUIPMENTREL PER

	ON 1=1

	AND PER.EQUIPMENTID = EQ.EQUIPMENTID

	AND PER.SITEID = EQ.SITEID

	AND PER.ISUSABLE = 'Usable'

	INNER JOIN CIM_PROCESSSEGMENT PC

	ON 1=1

	AND PC.PROCESSSEGMENTID = PER.PROCESSSEGMENTID

	AND PC.SITEID = PER.SITEID

	AND PC.ISUSABLE = 'Usable'

WHERE 1=1

	AND (:PLANTID IS NULL OR EQ.PLANTID = :PLANTID)

	AND (:LINEID IS NULL OR  EQ.LINEID = :LINEID)

	AND (:PROCESSSEGMENTID IS NULL OR PC.PROCESSSEGMENTID = :PROCESSSEGMENTID)

ORDER BY PC.PROCESSSEGMENTID
