-- SelectDurableUsageHistoryList

-- 설비별 사용 치공구 사용 제품 이력 조회

------------------------------------

-- INPUT : SITEID, PLANTID, LINEID, EQUIPMENTID, DURABLEDEFINITIONID, PRODUCTDEFINITIONID

------------------------------------



SELECT

    DT.SITEID 

  , TO_CHAR(DT.MODIFYTIME, 'YYYY-MM-DD') AS REPORTDAY

  , EQ.PLANTID

  , PT.FACILITYNAME AS PLANTNAME

  , EQ.LINEID

  , LT.FACILITYNAME AS LINENAME

  , DT.EQUIPMENTID

  , EQ.EQUIPMENTNAME

  , DT.PROCESSSEGMENTID

  , PS.PROCESSSEGMENTNAME

  , DU.DURABLEDEFINITIONID

  , DD.DURABLEDEFINITIONNAME

  , DT.DURABLEID

  , DT.PRODUCTDEFINITIONID

  , PD.PRODUCTDEFINITIONNAME

  , WO.PRODUCTORDERID

  , BT.WORKORDERID

  , BT.BATCHID

  , DT.LOTID

  , NULL AS MODIFIERNAME

  , DT.MODIFIER

  , DT.MODIFYTIME

FROM CIM_LOTDURABLEREL DT

LEFT OUTER JOIN CIM_EQUIPMENT EQ ON EQ.SITEID = DT.SITEID AND EQ.EQUIPMENTID = DT.EQUIPMENTID

LEFT OUTER JOIN CIM_FACILITY PT ON PT.SITEID = EQ.SITEID AND PT.FACILITYID = EQ.PLANTID

LEFT OUTER JOIN CIM_FACILITY LT ON LT.SITEID = EQ.SITEID AND LT.FACILITYID = EQ.LINEID

LEFT OUTER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = DT.SITEID AND PS.PROCESSSEGMENTID = DT.PROCESSSEGMENTID 

LEFT OUTER JOIN CIM_PRODUCTDEFINITION PD ON PD.SITEID = DT.SITEID AND PD.PRODUCTDEFINITIONID = DT.PRODUCTDEFINITIONID 

LEFT OUTER JOIN CIM_DURABLE DU ON DU.SITEID = DT.SITEID AND DU.DURABLEID = DT.DURABLEID

LEFT OUTER JOIN CIM_DURABLEDEFINITION DD ON DD.SITEID = DU.SITEID AND DD.DURABLEDEFINITIONID = DU.DURABLEDEFINITIONID 

INNER JOIN CIM_LOTBATCHREL RL ON RL.SITEID = DT.SITEID AND RL.LOTID = DT.LOTID 

INNER JOIN CIM_BATCH BT ON BT.SITEID = RL.SITEID AND BT.BATCHID = RL.BATCHID AND BT.BATCHTYPE = PS.PROCESSTYPE

INNER JOIN CIM_WORKORDER WO ON WO.SITEID = BT.SITEID AND WO.WORKORDERID = BT.WORKORDERID

WHERE 1=1

  AND DT.SITEID               = :SITEID

  AND DT.ISUSABLE             = 'Usable'

  AND DT.EQUIPMENTID          = NVL(:EQUIPMENTID, DT.EQUIPMENTID)

  AND DT.PRODUCTDEFINITIONID  = NVL(:PRODUCTDEFINITIONID, DT.PRODUCTDEFINITIONID)

  AND DU.DURABLEDEFINITIONID  = NVL(:DURABLEDEFINITIONID, DU.DURABLEDEFINITIONID)

ORDER BY REPORTDAY, EQUIPMENTID, DURABLEDEFINITIONID, PRODUCTDEFINITIONID

-- SelectLineList

SELECT 

	FACILITYID AS CODE, -- LINEID,

	FACILITYNAME AS NAME, -- LINENAME,

	SHORTNAME, SHOP, LINEGROUP

FROM CIM_FACILITY LINE

	WHERE 1=1

	AND UPPER(LINE.FACILITYTYPE) = 'LINE'

	AND LINE.SITEID = :SITEID

	AND LINE.PARENTFACILITYID IN 

	(

		SELECT FACILITYID FROM CIM_FACILITY PLANTLINE

		WHERE 1=1

			AND (:PLANTID IS NULL OR PLANTLINE.PARENTFACILITYID = :PLANTID)

			AND UPPER(PLANTLINE.FACILITYTYPE) = 'PLANTLINE'

			AND PLANTLINE.SITEID = :SITEID

	)

	AND LINE.ISUSABLE = 'Usable'

	AND (:SHOP IS NULL OR LINE.SHOP IN (SELECT COLUMN_VALUE FROM FN_MES_SPLIT(:SHOP, ',')))

ORDER BY UISEQUENCE

CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
