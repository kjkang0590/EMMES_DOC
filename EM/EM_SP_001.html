--EM_SP_001_Spare Parts 현황 조회 쿼리
--SelectSpartList
--Param : SITEID, SPARTCLASSID, SPARTDEFINITIONID, SPARTDEFINITIONNAME
------------------------------------------
SELECT D.FACILITYNAME AS LOCATION,
          A.SPARTCLASSID,
          B.SPARTCLASSNAME,
          A.SPARTDEFINITIONID,
          A.SPARTDEFINITIONNAME,
          C.SPARTID,
          ISNULL(M.QTY, 0) AS QTY,
          A.UNITID,
          SUM(A.SAFETYSTOCK) AS SAFETYSTOCK
  FROM CIM_SPARTDEFINITION A WITH(NOLOCK)
 INNER JOIN CIM_SPARTCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.SPARTCLASSID = B.SPARTCLASSID
  LEFT JOIN CIM_SPART C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.SPARTDEFINITIONID = C.SPARTDEFINITIONID
  LEFT JOIN CIM_FACILITY D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.LOCATION = D.FACILITYID 
  LEFT JOIN (SELECT SITEID, MATERIALDEFINITIONID, SUM(QTY) AS QTY
               FROM CST_MATERIALLOT WITH(NOLOCK)
              WHERE STATE = 'Active'
              GROUP BY SITEID, MATERIALDEFINITIONID) M
    ON A.SITEID = M.SITEID
   AND A.SPARTDEFINITIONID = M.MATERIALDEFINITIONID
 WHERE A.SITEID = @SITEID
   AND A.SPARTCLASSID = @SPARTCLASSID
   AND (@SPARTDEFINITIONID = '' OR A.SPARTDEFINITIONID = @SPARTDEFINITIONID)
   AND (@SPARTDEFINITIONNAME = '' OR A.SPARTDEFINITIONNAME LIKE '%' + @SPARTDEFINITIONNAME + '%')
 GROUP BY A.SPARTCLASSID, A.SPARTDEFINITIONID, A.SPARTDEFINITIONNAME, B.SPARTCLASSNAME, C.SPARTID, D.FACILITYNAME, A.UNITID, M.QTY
 ORDER BY A.SPARTDEFINITIONID

--EM_SP_001_Spare Parts 현황 - Spare Parts 이력 조회 쿼리
--SelectSpartHistList
--Param : SITEID, SPARTCLASSID, SPARTDEFINITIONID, SPARTID
------------------------------------------
SELECT CONVERT(CHAR(19), A.LASTEVENTTIME, 20) AS LASTEVENTTIME,
       A.LOCATION,
       C.SPARTCLASSID,
       B.SPARTCLASSNAME,
       A.MATERIALDEFINITIONID AS SPARTDEFINITIONID,
       C.SPARTDEFINITIONNAME,
       A.QTY,
       A.UNITID,
       C.SAFETYSTOCK
  FROM CST_MATERIALLOTHIST A WITH(NOLOCK)
 INNER JOIN CIM_SPARTDEFINITION C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.MATERIALDEFINITIONID = C.SPARTDEFINITIONID
 INNER JOIN CIM_SPARTCLASS B WITH(NOLOCK)
    ON C.SITEID = B.SITEID
   AND C.SPARTCLASSID = B.SPARTCLASSID
 WHERE A.SITEID = @SITEID
   AND C.SPARTCLASSID = @SPARTCLASSID
   AND A.MATERIALDEFINITIONID = @SPARTDEFINITIONID
 ORDER BY A.LASTEVENTTIME DESC
