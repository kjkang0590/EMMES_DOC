--EM_PB_005_BM 관리 - BM 목록 Grid 조회 쿼리
--SelectBmManagementSearch
--Param : SITEID, USERID, LINEID, EQUIPMENTID, EQUIPMENTNAME, REQUESTUSERID, FAILURECODE, STATE
------------------------------------------
SELECT S1.BMID,
	   CC.CODENAME AS BMNAME,
	   S1.ROOTEQUIPMENTID,
	   S1.EQUIPMENTCLASSID,
	   S1.EQUIPMENTID,
	   S1.EQUIPMENTNAME,
	   S1.STATE,
	   S1.STATENAME,
	   S1.REQUESTUSERID,
	   S1.REQUESTUSERNAME,
       S1.REQUESTCODENAME,
	   S1.REQUESTTYPENAME,
	   S1.REQUESTDESCRIPTION,
	   S1.REQUESTDATETIME,
	   S1.MAINUSERID,
	   S1.MAINUSERNAME,
	   S1.SUBUSERID,
	   S1.SUBUSERNAME,
	   S1.RECEIPTDESCRIPTION,
	   S1.RECEIPTDATETIME,
	   S1.FAILURECODE,
	   S1.FAILURENAME,
	   S1.REPAIRECODE,
	   S1.REPAIRENAME,
	   S1.REPAIRESTARTDATETIME,
	   S1.REPAIREENDDATETIME,
       S1.MAINTENANCETIME,
	   S1.REPAIREDESCRIPTION,
	   S1.RESULTDATETIME
FROM (
	SELECT A.SITEID,
		   A.BMID,
		   IIF(CHARINDEX('PM', A.BMID) = 1, 'PMNM', 'BMNM') AS BMNAMECODE,
	       B.ROOTEQUIPMENTID,
		   C.EQUIPMENTCLASSID,
		   A.EQUIPMENTID,
		   B.EQUIPMENTNAME,
		   A.STATE,
		   D.STATENAME,
		   A.REQUESTUSERID,
		   E.USERNAME AS REQUESTUSERNAME,
           J.CODENAME AS REQUESTCODENAME,
		   K.CODENAME AS REQUESTTYPENAME,
		   A.REQUESTDESCRIPTION,
		   A.REQUESTDATETIME,
		   A.MAINUSERID,
		   F.USERNAME AS MAINUSERNAME,
		   A.SUBUSERID,
		   G.USERNAME AS SUBUSERNAME,
		   A.RECEIPTDESCRIPTION,
		   A.RECEIPTDATETIME,
		   A.FAILURECODE,
		   H.CODENAME AS FAILURENAME,
		   A.REPAIRECODE,
		   I.CODENAME AS REPAIRENAME,
		   A.REPAIRESTARTDATETIME,
		   A.REPAIREENDDATETIME,
           A.MAINTENANCETIME,
		   A.REPAIREDESCRIPTION,
		   A.RESULTDATETIME
  FROM CST_BMRESULT A WITH(NOLOCK)
  LEFT JOIN CIM_EQUIPMENT B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.EQUIPMENTID = B.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.EQUIPMENTCLASSID = C.EQUIPMENTCLASSID
  LEFT JOIN CIM_STATE D WITH(NOLOCK)
    ON A.SITEID = D.SITEID
   AND A.STATE = D.STATEID
   AND D.STATEMODELID = 'BmState'
  LEFT JOIN CIM_USER E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.REQUESTUSERID = E.USERID
  LEFT JOIN CIM_USER F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.MAINUSERID = F.USERID
  LEFT JOIN CIM_USER G WITH(NOLOCK)
    ON A.SITEID = G.SITEID
   AND A.SUBUSERID = G.USERID
  LEFT JOIN CIM_CODE H WITH(NOLOCK)
    ON A.SITEID = H.SITEID
   AND A.FAILURECODE = H.CODEID
   AND H.CODECLASSID = 'FAILURECODE'
  LEFT JOIN CIM_CODE I WITH(NOLOCK)
    ON A.SITEID = I.SITEID
   AND A.REPAIRECODE = I.CODEID
   AND I.CODECLASSID = 'REPAIRECODE'
  LEFT JOIN CIM_CODE J WITH(NOLOCK)
    ON A.SITEID = J.SITEID
   AND A.REQUESTCODE = J.CODEID
   AND J.CODECLASSID = 'REQUESTCODE'
  LEFT JOIN CIM_CODE K WITH(NOLOCK)
    ON A.SITEID = K.SITEID
   AND A.REQUESTTYPE = K.CODEID
   AND K.CODECLASSID = 'REQUESTTYPE'
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.REQUESTDATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND B.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
   AND (@EQUIPMENTCLASSID = '' OR C.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID LIKE '%' + @EQUIPMENTID + '%')
   AND (@REQUESTUSERID = '' OR E.USERNAME LIKE '%' + @REQUESTUSERID + '%')
   AND (@FAILURECODE = '' OR A.FAILURECODE = @FAILURECODE)
   AND (@STATE = '' OR A.STATE = @STATE)
) S1
LEFT JOIN CIM_CODE CC WITH(NOLOCK) ON S1.SITEID = CC.SITEID AND S1.BMNAMECODE = CC.CODEID AND CC.CODECLASSID = 'BMNAME'
ORDER BY S1.BMID DESC
