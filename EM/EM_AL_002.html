--EM_AL_002_설비 상태 현황 조회 쿼리
--SelectEquipmentStatusList
--Param : SITEID, LINEID, EQUIPMENTCLASSID, EQUIPMENTID, STATE
------------------------------------------
SELECT A.LINEID,
       A.EQUIPMENTCLASSID,
       B.EQUIPMENTCLASSNAME,
       A.EQUIPMENTID,
       A.EQUIPMENTNAME,
       A.EQUIPMENTTYPE,
       A.PREVSTATE,
       CONVERT(CHAR(19), A.PREVSTATETIME, 20) AS PREVSTATETIME,
       A.STATE,
       CONVERT(CHAR(19), A.STATETIME, 20) AS STATETIME
  FROM CIM_EQUIPMENT A WITH(NOLOCK)
  LEFT JOIN CIM_EQUIPMENTCLASS B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.EQUIPMENTCLASSID = B.EQUIPMENTCLASSID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.EQUIPMENTCLASSID = @EQUIPMENTCLASSID
   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)
   AND (@STATE = '' OR A.STATE = @STATE)
 ORDER BY A.EQUIPMENTCLASSID
