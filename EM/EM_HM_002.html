/* 

--------------------------------------

-- 설비 이력 조회 : SelectEquipmentHistoryList

--------------------------------------

-- 입력 PARAMETER

SITEID : 필수 : 단건 : 사용자가 로그인되어 있는 사업장

EQUIPMENTID : 필수 : 단건 : 사용자가 선택한 설비

FROMDATE: 필수 : YYYY-MM-DD

TODATE: 필수 : YYYY-MM-DD

--------------------------------------

*/

SELECT

    EQ.SITEID

  , FN_USERNAME(EQ.SITEID, EQ.MODIFIER) AS MODIFIERNAME

  , EQ.MODIFYTIME

  , EQ.CUSTOMACTIVITY 

  , EQ.PLANTID

  , PL.FACILITYNAME AS PLANTNANE

  , EQ.LINEID

  , LN.FACILITYNAME AS LINENAME

  , EQ.EQUIPMENTCLASSID 

  , EC.EQUIPMENTCLASSNAME 

  , EQ.EQUIPMENTID

  , EQ.EQUIPMENTNAME

  , EQ.STATE

  , EQ.PROCESSSTATE

  , EQ.CONTROLMODE

  , EQ.OPERATIONMODE

  , RD.PRODUCTDEFINITIONID  

  , PD.PRODUCTDEFINITIONNAME

  , BT.WORKORDERID

  , EQ.BATCHID

  , RD.PROCESSSEGMENTID

  , PS.PROCESSSEGMENTNAME

  , EQ.RECIPEID AS RECIPEDEFINITIONID

  , RD.RECIPEDEFINITIONNAME 

  , EQ.ALARMID

  , EQ.DESCRIPTION

  , EQ.COMMENTS

FROM CIM_EQUIPMENTHIST EQ

LEFT OUTER JOIN CIM_FACILITY PL ON PL.SITEID = EQ.SITEID AND PL.FACILITYID = EQ.PLANTID

LEFT OUTER JOIN CIM_FACILITY LN ON LN.SITEID = EQ.SITEID AND LN.FACILITYID = EQ.LINEID

LEFT OUTER JOIN CIM_EQUIPMENTCLASS EC ON EC.SITEID = EQ.SITEID AND EC.EQUIPMENTCLASSID = EQ.EQUIPMENTCLASSID 

---- W/O, RECIPE정보

LEFT OUTER JOIN CIM_BATCH BT ON BT.SITEID = EQ.SITEID AND BT.BATCHID = EQ.BATCHID 

LEFT OUTER JOIN CIM_RECIPEDEFINITION RD ON RD.SITEID = EQ.SITEID AND RD.RECIPEDEFINITIONID = EQ.RECIPEID  

LEFT OUTER JOIN CIM_PRODUCTDEFINITION PD ON PD.SITEID = RD.SITEID AND PD.PRODUCTDEFINITIONID = RD.PRODUCTDEFINITIONID 

LEFT OUTER JOIN CIM_PROCESSSEGMENT PS ON PS.SITEID = RD.SITEID AND PS.PROCESSSEGMENTID = RD.PROCESSSEGMENTID  AND BT.BATCHTYPE = PS.PROCESSTYPE 

WHERE 1=1

  AND EQ.SITEID = :SITEID

  AND EQ.ISUSABLE = 'Usable'  

  AND EQ.PLANTID = :PLANTID 

  AND EQ.LINEID = :LINEID  

  AND EQ.EQUIPMENTID= :EQUIPMENTID 

  AND EQ.MODIFYTIME >= TO_DATE(:FROMDATE, 'YYYY-MM-DD')

  AND EQ.MODIFYTIME <  TO_DATE(:TODATE, 'YYYY-MM-DD') +1

ORDER BY EQ.LASTEVENTSEQ DESC

CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END

	FACILITYID AS CODE, -- LINEID,

	FACILITYNAME AS NAME, -- LINENAME,

	SHORTNAME, SHOP, LINEGROUP

FROM CIM_FACILITY LINE

	WHERE 1=1

	AND UPPER(LINE.FACILITYTYPE) = 'LINE'

	AND LINE.SITEID = :SITEID

	AND LINE.PARENTFACILITYID IN 

	(

		SELECT FACILITYID FROM CIM_FACILITY PLANTLINE

		WHERE 1=1

			AND (:PLANTID IS NULL OR PLANTLINE.PARENTFACILITYID = :PLANTID)

			AND UPPER(PLANTLINE.FACILITYTYPE) = 'PLANTLINE'

			AND PLANTLINE.SITEID = :SITEID

	)

	AND LINE.ISUSABLE = 'Usable'

	AND (:SHOP IS NULL OR LINE.SHOP IN (SELECT COLUMN_VALUE FROM FN_MES_SPLIT(:SHOP, ',')))

테이블 반환 함수 FN_MES_SPLIT 정의 보기
CREATE FUNCTION [dbo].[FN_MES_SPLIT]

(

  @I_STR     NVARCHAR(MAX),

  @I_DELIM    NVARCHAR(40) = ','

)

RETURNS @RETARRAY TABLE

(

	VALUE NVARCHAR(MAX)

)

AS  

BEGIN  

	DECLARE @VALUE			VARCHAR(8000)

	DECLARE	@BCONTINUE		BIT

	DECLARE @ISTRIKE		SMALLINT

	DECLARE @IDELIMLENGTH	TINYINT

   



	SET @I_STR = LTRIM(RTRIM(@I_STR))  

	SET @IDELIMLENGTH = LEN(@I_DELIM)  

	SET @BCONTINUE = 1  



	WHILE @BCONTINUE = 1  

	BEGIN  

		--텍스트에서 구분자를 발견하면, 첫 번째 요소를 반환하고  

		--반환되는 테이블에 인덱스를 입력한다.  

		IF CHARINDEX(@I_DELIM, @I_STR)>0  

		BEGIN  

			-- 첫 번째 요소를 반환  

			SET @VALUE = SUBSTRING(@I_STR,1, CHARINDEX(@I_DELIM,@I_STR)-1)  

				

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--다음 요소와 구분자를 문자열의 앞에서부터 제거하고,  

			--INDEX를 증가시키고 반복작업(LOOP)을 이어간다.  

			SET @ISTRIKE = LEN(@VALUE) + @IDELIMLENGTH  

			SET @I_STR = LTRIM(RIGHT(@I_STR,LEN(@I_STR) - @ISTRIKE))  

		END

		ELSE  

		BEGIN  

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END

			--만약 텍스트 안에서 구분자를 더 이상 찾을 수 없게 되면,   

			--@I_STR가 @RETARRAY의 마지막 값이다.  

			SET @VALUE = @I_STR  

			-- 마지막 값도 구분자로 나누어져 있으뎐  

			INSERT @RETARRAY (VALUE) VALUES (@VALUE)  



			--WHILE 루프를 빠져 나온다.  

			SET @BCONTINUE = 0  

		END  

	END 

	

	RETURN  

END
