마스터/명칭 조회용 조인 제외(기준 테이블 제외): CIM_USER, CIM_EQUIPMENT, CIM_CODE

--EM_AL_003_설비 이력 - 설비 수리 이력 조회 쿼리
--SelectEquipmentHistList
--Param : SITEID, LINEID, EQUIPMENTCLASSID, EQUIPMENTID, TAGID
------------------------------------------
SELECT S1.*,
	   T1.CODENAME AS ISSPCONSUME
  FROM (
	SELECT A.BMID, 
	       A.REQUESTDATETIME,
	       A.REPAIREENDDATETIME,
	       E.EQUIPMENTCLASSID,
	       A.EQUIPMENTID,
	       D.EQUIPMENTNAME,
	       A.REQUESTCODE,
	       F.CODENAME AS REQUESTCODENAME,
	       A.FAILURECODE,
	       G.CODENAME AS FAILURECODENAME,
	       C.DEPTNAME AS DEPTNAME,
	       A.REPAIRECODE,
	       H.CODENAME AS REPAIRECODENAME,
	       CASE WHEN B.QTY > 0 THEN 'Y' ELSE 'N' END ISSPCONSUMECODE,
	       A.MAINUSERID,
	       C.USERNAME AS MAINUSERNAME,
	       A.REPAIREDESCRIPTION,
	       A.SITEID
	  FROM CST_BMRESULT A WITH(NOLOCK)
	  LEFT JOIN CST_SPARTCONSUME B WITH(NOLOCK)
	    ON A.SITEID = B.SITEID
	   AND A.EQUIPMENTID = B.EQUIPMENTID
	  LEFT JOIN CIM_USER C WITH(NOLOCK)
	    ON A.SITEID = C.SITEID 
	   AND A.MAINUSERID = C.USERID
	  LEFT JOIN CIM_EQUIPMENT D WITH(NOLOCK)
	    ON A.SITEID = D.SITEID
	   AND A.LINEID = D.LINEID
	   AND A.EQUIPMENTID = D.EQUIPMENTID
	  LEFT JOIN CIM_EQUIPMENTCLASS E WITH(NOLOCK)
	    ON D.SITEID = E.SITEID
	   AND D.EQUIPMENTCLASSID = E.EQUIPMENTCLASSID
	  LEFT JOIN CIM_CODE F WITH(NOLOCK)
	    ON A.SITEID = F.SITEID
	   AND A.REQUESTCODE = F.CODEID
	   AND F.CODECLASSID = 'REQUESTCODE'
	  LEFT JOIN CIM_CODE G WITH(NOLOCK)
	    ON A.SITEID = G.SITEID
	   AND A.FAILURECODE = G.CODEID
	   AND G.CODECLASSID = 'FAILURECODE'
	  LEFT JOIN CIM_CODE H WITH(NOLOCK)
	    ON A.SITEID = H.SITEID
	   AND A.REPAIRECODE = H.CODEID
	   AND H.CODECLASSID = 'REPAIRECODE'
	 WHERE A.SITEID = @SITEID
	   AND A.LINEID = @LINEID
	   AND A.REPAIREENDDATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
	   AND D.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
	   AND (@EQUIPMENTCLASSID = '' OR E.EQUIPMENTCLASSID = @EQUIPMENTCLASSID)
	   AND (@EQUIPMENTID = '' OR A.EQUIPMENTID = @EQUIPMENTID)
) S1
  LEFT JOIN CIM_CODE T1 WITH(NOLOCK)
    ON S1.SITEID = T1.SITEID
   AND S1.ISSPCONSUMECODE = T1.CODEID
   AND T1.CODECLASSID = 'ISEXIST'
 ORDER BY S1.REQUESTDATETIME
