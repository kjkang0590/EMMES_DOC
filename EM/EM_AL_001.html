--EM_AL_001_설비 알람 현황 - 설비 Alarm 조회 쿼리
--SelectAlarmList
--Param : SITEID, FROMDATE, TODATE, ROOTEQUIPMENTID, EQUIPMENTCLASSID, EQUIPMENTID
------------------------------------------
SELECT A.CREATETIME,
       C.ROOTEQUIPMENTID,
	   D.EQUIPMENTCLASSNAME,
       C.EQUIPMENTID,
	   C.EQUIPMENTNAME,
	   B.DCSTAGID,
	   A.ALARMDATA,
	   A.ALARMTAG
  FROM CIM_ALARM A WITH(NOLOCK)
 INNER JOIN CST_OPCTAGINFO B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.ALARMTAG = B.TAGID
 INNER JOIN CIM_EQUIPMENT C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.EQUIPMENTID = C.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS D WITH(NOLOCK)
    ON C.SITEID = D.SITEID
   AND C.EQUIPMENTCLASSID = D.EQUIPMENTCLASSID
 WHERE A.SITEID = @SITEID
   AND A.CREATETIME BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND C.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
   AND C.EQUIPMENTCLASSID = @EQUIPMENTCLASSID
   AND C.EQUIPMENTID = @EQUIPMENTID
 ORDER BY A.LASTEVENTTIME DESC

--SelectCboEquipmentClassList
--Param : SITEID, ROOTEQUIPMENTID
------------------------------------------	
SELECT
	EC.EQUIPMENTCLASSID AS CODE
	, EC.EQUIPMENTCLASSNAME AS NAME
FROM CIM_EQUIPMENTCLASS EC WITH(NOLOCK),
(
	SELECT
		E.EQUIPMENTCLASSID
		, E.SITEID
	FROM CIM_EQUIPMENT E WITH(NOLOCK), CST_PUBLICREL R WITH(NOLOCK)
	WHERE E.EQUIPMENTID = R.PARENTID
	AND   E.SITEID = R.SITEID
	AND   E.ROOTEQUIPMENTID = @ROOTEQUIPMENTID
	GROUP BY E.EQUIPMENTCLASSID, E.SITEID
) E
WHERE EC.EQUIPMENTCLASSID = E.EQUIPMENTCLASSID
AND   EC.SITEID = E.SITEID
