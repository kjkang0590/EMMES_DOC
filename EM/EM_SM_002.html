----------- SelectAllCarrierList CARRIER 재고관리 조회
SELECT
	DL.CARRIERID
  , DL.ISUSABLE
  , DD.SITEID
  , DL.CARRIERNAME 
  , DD.CARRIERDEFINITIONID
  , DD.CARRIERDEFINITIONNAME 
  , DD.CARRIERCLASSID 
  , DC.CARRIERCLASSNAME
  , DL.MATERIALTYPE 
  , DL.EQUIPMENTID 
  , DL.PORTID
  , DL.LOADSTATE
  , DL.SLOTQTY
  , DL.ASSIGNQTY 
  , DL.MAXCAPACITY 
  , DL.VENDORID 
  , CV2.VENDORNAME 
  , DL.STATE 
  , DL.PREVSTATE 
  , DL.LOCATION 
  , DL.PREVLOCATION
  , DD.USAGETYPE
  , DD.USAGECOUNTLIMIT
  , DL.USAGECOUNT
  , DL.USAGEPERIODUNIT
  , DL.USAGEPERIODLIMIT
  , DL.USAGEPERIOD 
  , DL.CLEANTYPE 
  , DL.CLEANCOUNTLIMIT 
  , DL.CLEANCOUNT 
  , DL.CLEANPERIODUNIT 
  , DL.CLEANPERIODLIMIT 
  , DL.CLEANPERIOD 
  , DL.LASTCLEANTIME 
  , DL.NEXTCLEANTIME 
  , DL.REPAIRTYPE 
  , DL.REPAIRCOUNTLIMIT 
  , DL.REPAIRCOUNT 
  , DL.REPAIRPERIODUNIT 
  , DL.REPAIRPERIODLIMIT 
  , DL.REPAIRPERIOD 
  , DL.LASTREPAIRTIME 
  , DL.NEXTREPAIRTIME 
  , DL.ACTIVITY 
  , DL.PREVACTIVITY 
  , DL.CUSTOMACTIVITY 
  , DL.PREVCUSTOMACTIVITY 
  , DL.DESCRIPTION 
  , DL.REASONCODE 
  , DL.COMMENTS
  , DL.CREATOR 
  , DL.CREATETIME 
  , DL.MODIFIER 
  , DL.MODIFYTIME 
  , DL.LASTEVENTTIME 
  , DL.TID 
  , FN_USERNAME(DL.SITEID, DL.MODIFIER) AS MODIFERNAME
FROM CIM_CARRIERCLASS DC
INNER JOIN CIM_CARRIERDEFINITION DD ON DD.SITEID = DC.SITEID AND DD.CARRIERCLASSID = DC.CARRIERCLASSID 
  AND DD.CARRIERCLASSID = NVL(:CARRIERCLASSID, DD.CARRIERCLASSID)
INNER JOIN CIM_CARRIER DL ON DL.SITEID = DD.SITEID AND DL.CARRIERDEFINITIONID = DD.CARRIERDEFINITIONID 
LEFT OUTER JOIN CIM_CODE US ON US.SITEID = DL.SITEID AND US.CODECLASSID = 'IsUsable' AND US.CODEID = DL.ISUSABLE 
LEFT OUTER JOIN CIM_VENDOR CV2 ON CV2.SITEID = DL.SITEID AND CV2.ISUSABLE = 'Usable' AND CV2.VENDORID = DL.VENDORID 
WHERE 1=1
  AND DL.SITEID = :SITEID
  AND DL.CARRIERDEFINITIONID = NVL(:CARRIERDEFINITIONID, DL.CARRIERDEFINITIONID)
ORDER BY DL.CARRIERCLASSID, DL.CARRIERDEFINITIONID, DL.CARRIERID
스칼라 함수 FN_USERNAME 정의 보기
CREATE FUNCTION [dbo].[FN_USERNAME](

	@P_SITEID NVARCHAR(500),

	@P_USERID NVARCHAR(500)

) 

RETURNS NVARCHAR(2000)

AS

BEGIN

	DECLARE @V_RETURN NVARCHAR(100);

 

	SELECT @V_RETURN = ISNULL(U.USERNAME, '') + '(' + ISNULL(U.USERID, '') + ')' FROM CIM_USER U

	WHERE U.SITEID  = @P_SITEID

		AND U.USERID = @P_USERID

	RETURN @V_RETURN

END
