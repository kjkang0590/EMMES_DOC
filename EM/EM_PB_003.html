--EM_PB_003_PM 검사 일정 관리/EM_PB_004_PM 결과 등록 - PM 스케줄 목록 조회 쿼리
--SelectPmInspScheduleList
--Param : SITEID, LINEID, FROMDATE, TODATE, PMDEFINITIONID, PMSYSNAME, SCHEDULEUNIT, MAINUSERID, SUBUSERID, STATE, ISUSABLE
------------------------------------------
SELECT A.SCHEDULESTARTDATE,
       A.PMSYSID,
	   A.PMDEFINITIONID,
	   B.PMSYSNAME,
       B.PMTYPE,
	   A.STATE,
            H.STATENAME,
	   A.PROCESSSEGMENTID,
	   C.PROCESSSEGMENTNAME,
	   E.EQUIPMENTCLASSID,
	   B.EQUIPMENTID,
	   D.EQUIPMENTNAME,
	   A.STARTDATE,
	   A.ENDDATE,
	   A.MAINUSERID,
	   F.USERNAME AS MAINTAINER,
	   A.ISUSABLE,
            G.CODENAME AS ISUSABLENAME,
	   A.COMMENTS,
	   A.DESCRIPTION,
       IIF(A.STARTDATE IS NULL, 'A', 'U') AS ACTIONFLAG
  FROM CST_PMRESULT A WITH(NOLOCK)
 INNER JOIN CST_PMDEFINITION B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.PMDEFINITIONID = B.PMDEFINITIONID
  LEFT JOIN CIM_PROCESSSEGMENT C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.PROCESSSEGMENTID = C.PROCESSSEGMENTID
  LEFT JOIN CIM_EQUIPMENT D WITH(NOLOCK)
    ON B.SITEID = D.SITEID
   AND B.EQUIPMENTID = D.EQUIPMENTID
  LEFT JOIN CIM_EQUIPMENTCLASS E WITH(NOLOCK)
    ON D.SITEID = E.SITEID
   AND D.EQUIPMENTCLASSID = E.EQUIPMENTCLASSID
  LEFT JOIN CIM_USER F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.MAINUSERID = F.USERNO
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON A.SITEID = G.SITEID
   AND A.ISUSABLE = G.CODEID
   AND G.CODECLASSID = 'ISUSABLE'
  LEFT JOIN CIM_STATE H WITH(NOLOCK)
    ON A.SITEID = H.SITEID
   AND A.STATE = H.STATEID
   AND H.STATEMODELID = 'PmState'
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.SCHEDULESTARTDATE BETWEEN @FROMDATE AND DATEADD(DAY, 1, @TODATE)
   AND (@EQUIPMENTID = '' OR B.EQUIPMENTID = @EQUIPMENTID)
   AND B.PMSYSNAME LIKE '%' + @PMSYSNAME + '%'
   AND (@SCHEDULEUNIT = '' OR B.SCHEDULEUNIT = @SCHEDULEUNIT)
   AND (@MAINUSERID = '' OR F.USERNO LIKE '%' + @MAINUSERID + '%') 
--  AND (@SUBUSERID = '' OR G.USERNAME LIKE '%' + @SUBUSERID + '%')  
   AND (@STATE = '' OR A.STATE = @STATE)
   AND (@ISUSABLE = '' OR B.ISUSABLE = @ISUSABLE)
 ORDER BY A.SCHEDULESTARTDATE

--EM_PB_003_PM 검사 일정 관리 - PM 스케줄 이력 조회 쿼리
--SelectPmInspScheduleHistList
--Param : SITEID, LINEID, PMSYSID, PMDEFINITIONID
------------------------------------------
SELECT A.LASTEVENTTIME,
	   A.SCHEDULESTARTDATE,
	   A.PMDEFINITIONID,
	   B.PMSYSNAME,
	   A.STATE,
	   E.STATENAME,
	   A.STARTDATE,
	   A.ENDDATE,
	   A.MAINUSERID,
	   F.USERNAME AS MAINTAINER,
	   A.COMMENTS,
	   A.DESCRIPTION,
	   G.CODENAME AS EQUIPMENTTYPE,
	   B.EQUIPMENTID,
	   H.CODENAME AS PMCATEGORYID,
	   I.CODENAME AS PMCLASSID
  FROM CST_PMRESULTHIST A WITH(NOLOCK)
 INNER JOIN CST_PMDEFINITION B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.PMDEFINITIONID = B.PMDEFINITIONID
 INNER JOIN CST_PMDEFINITIONITEMREL C WITH(NOLOCK)
    ON B.SITEID = C.SITEID
   AND B.PMDEFINITIONID = C.PMDEFINITIONID
 INNER JOIN CST_PMITEM D WITH(NOLOCK)
    ON C.SITEID = D.SITEID
   AND C.PMITEMID = D.PMITEMID
  LEFT JOIN CIM_STATE E WITH(NOLOCK)
    ON A.SITEID = E.SITEID
   AND A.STATE = E.STATEID
   AND E.STATEMODELID = 'PmState'
  LEFT JOIN CIM_USER F WITH(NOLOCK)
    ON A.SITEID = F.SITEID
   AND A.MAINUSERID = F.USERID
   AND F.DEPTID = 'EM0091'
  LEFT JOIN CIM_CODE G WITH(NOLOCK)
    ON D.SITEID = G.SITEID
   AND D.EQUIPMENTTYPE = G.CODEID
   AND G.CODECLASSID = 'EQUIPMENTTYPE'
  LEFT JOIN CIM_CODE H WITH(NOLOCK)
    ON D.SITEID = H.SITEID
   AND D.PMCATEGORYID = H.CODEID
   AND H.CODECLASSID = 'PMCATEGORY'
  LEFT JOIN CIM_CODE I WITH(NOLOCK)
    ON D.SITEID = I.SITEID
   AND D.PMCLASSID = I.CODEID
   AND I.CODECLASSID = 'PMCLASS'
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND A.PMDEFINITIONID = @PMDEFINITIONID
 ORDER BY A.LASTEVENTSEQ
