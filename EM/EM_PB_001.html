--EM_PB_001_PM 검사 항목 관리 - PM 검사 항목 목록 Grid 조회 쿼리
--SelectPmInspectionItemList
--Param : SITEID, USERID, LINEID, PMTYPE, PMCATEGORYID, PMITEMNAME, PMCLASSID, SCHEDULEUNIT, ISUSABLE
------------------------------------------
SELECT A.PMITEMID,
       A.PMTYPE,
       A.EQUIPMENTTYPE,
       A.EQUIPMENTID,
       A.PMCATEGORYID,
	   A.PMITEMNAME,
	   A.PMCLASSID,
	   A.SCHEDULEUNIT,
	   A.ISUSABLE,
	   A.COMMENTS,
	   A.CREATOR,
	   B.USERNAME AS CREATORNAME,
	   A.CREATETIME,
	   A.MODIFIER,
	   C.USERNAME AS MODIFIERNAME,
	   A.MODIFYTIME  
  FROM CST_PMITEM A WITH(NOLOCK)
  LEFT JOIN CIM_USER B WITH(NOLOCK)
    ON A.SITEID = B.SITEID
   AND A.CREATOR = B.USERID
  LEFT JOIN CIM_USER C WITH(NOLOCK)
    ON A.SITEID = C.SITEID
   AND A.MODIFIER = C.USERID
 WHERE A.SITEID = @SITEID
   AND A.LINEID = @LINEID
   AND (@PMTYPE = '' OR A.PMTYPE = @PMTYPE)
   AND (@PMCATEGORYID = '' OR A.PMCATEGORYID = @PMCATEGORYID)
   AND (@PMITEMNAME = '' OR A.PMITEMNAME LIKE '%' + @PMITEMNAME + '%')
   AND (@PMCLASSID = '' OR A.PMCLASSID = @PMCLASSID)
   AND (@SCHEDULEUNIT = '' OR A.SCHEDULEUNIT = @SCHEDULEUNIT)
   AND (@ISUSABLE = '' OR A.ISUSABLE = @ISUSABLE)
 ORDER BY A.PMITEMID
